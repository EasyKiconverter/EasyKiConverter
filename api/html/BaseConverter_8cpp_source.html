44bool BaseConverter::runExport(const QStringList& componentIds) {
45 printMessage(QCoreApplication::translate("CliConverter", "找到 %1 个元器件").arg(componentIds.size()));
46
47 // 创建当前 CLI 请求的导出选项。
48 ExportOptions options = context()->createExportOptions();
49 // 获取负责预加载和导出的共享服务。
50 ParallelExportService* exportService = context()->exportService();
51 exportService->setOptions(options);
52 exportService->setOutputPath(context()->parser().outputDir());
53
54 connect(exportService, &ParallelExportService::preloadCompleted, this, &BaseConverter::onPreloadCompleted);
55 connect(exportService, &ParallelExportService::progressChanged, this, &BaseConverter::onProgressChanged);
56 connect(exportService, &ParallelExportService::completed, this, &BaseConverter::onExportCompleted);
57 connect(exportService, &ParallelExportService::failed, this, &BaseConverter::onExportFailed);
58
59 CliExportWaiter::waitForPreload(exportService, componentIds);
60 if (errorMessage().isEmpty()) {
61 printMessage(QCoreApplication::translate("CliConverter", "预加载完成，开始导出..."));
62 CliExportWaiter::waitForExport(exportService);
63 }
64
65 printMessage(QCoreApplication::translate("CliConverter", "\n转换完成: 成功 %1, 失败 %2")
66 .arg(m_successCount)
67 .arg(m_failedCount));
68 return m_exportSuccess;
69}
