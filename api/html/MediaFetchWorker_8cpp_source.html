29void MediaFetchWorker::run() {
30 if (m_isAborted.loadAcquire()) {
31 return;
32 }
33
34 m_previewImageDataList.clear();
35 m_datasheetData.clear();
36 m_diagnostics.clear();
37 m_pendingOperations = 0;
38
39 ComponentCacheService* cache = ComponentCacheService::instance(); // 当前磁盘缓存服务。
40 if (!cache) {
41 qWarning() << "MediaFetchWorker: ComponentCacheService is null";
42 emit fetchCompleted(m_componentId, m_previewImageDataList, m_datasheetData, m_diagnostics);
43 return;
44 }
45
46 // 如果预览图为空，直接开始下载手册
47 if (m_previewImageUrls.isEmpty()) {
48 startDatasheetDownload(cache);
49 return;
50 }
51
52 // 开始第一个预览图下载
53 startPreviewDownload(0, cache);
54}

61void MediaFetchWorker::startPreviewDownload(int index, ComponentCacheService* cache) {
62 if (m_isAborted.loadAcquire()) {
63 finishAllDownloads();
64 return;
65 }
66
67 if (index >= m_previewImageUrls.size()) {
68 // 所有预览图下载完成，开始下载手册
69 startDatasheetDownload(cache);
70 return;
71 }
72
73 m_pendingOperations++;
74 const QString& url = m_previewImageUrls[index]; // 当前预览图的远程地址。
75
76 if (url.isEmpty()) {
77 // URL为空，尝试直接从缓存加载
78 QElapsedTimer timer; // 统计本地缓存读取耗时的计时器。
79 timer.start();
80 QByteArray imageData = cache->loadPreviewImage(m_componentId, index); // 从缓存读取的预览图数据。
81
82 ComponentExportStatus::NetworkDiagnostics diag; // 本次缓存读取的诊断信息。
83 diag.url = QString();
84 diag.latencyMs = timer.elapsed();
85 if (!imageData.isEmpty()) {
86 diag.statusCode = 200;
87 diag.errorString = "";
88 diag.wasRateLimited = false;
89 m_previewImageDataList.append(imageData);
90 } else {
91 diag.statusCode = 0;
92 diag.errorString = "Not found in cache";
93 diag.wasRateLimited = false;
94 }
95 diag.retryCount = 0;
96 m_diagnostics.append(diag);
97
98 m_pendingOperations--;
99 // 继续下一个预览图
100 startPreviewDownload(index + 1, cache);
101 return;
102 }
103
104 // 使用异步下载
105 CacheRepository::instance()->fetchPreviewImageAsync(
106 m_componentId,
107 url,
108 index,
109 &m_isAborted,
110 ConfigService::instance()->getWeakNetworkSupport(),
111 [this, index, cache](const QByteArray& imageData, const ComponentExportStatus::NetworkDiagnostics& diag) {
112 if (m_isAborted.loadRelaxed()) {
113 m_pendingOperations--;
114 if (m_pendingOperations <= 0) {
115 finishAllDownloads();
116 }
117 return;
118 }
119
120 if (!imageData.isEmpty()) {
121 m_previewImageDataList.append(imageData);
122 }
123 // 创建诊断信息的副本
124 ComponentExportStatus::NetworkDiagnostics diagCopy = diag; // 异步回调使用的诊断信息副本。
125 diagCopy.latencyMs = diag.latencyMs;
126 m_diagnostics.append(diagCopy);
127
128 m_pendingOperations--;
129 // 继续下一个预览图
130 startPreviewDownload(index + 1, cache);
131 });
132}

138void MediaFetchWorker::startDatasheetDownload(ComponentCacheService* cache) {
139 if (m_isAborted.loadAcquire()) {
140 finishAllDownloads();
141 return;
142 }
143
144 if (m_datasheetUrl.isEmpty()) {
145 // URL为空，尝试直接从缓存加载
146 QElapsedTimer timer; // 统计本地数据手册读取耗时的计时器。
147 timer.start();
148 m_datasheetData = cache->loadDatasheet(m_componentId);
149
150 ComponentExportStatus::NetworkDiagnostics diag; // 本次缓存读取的诊断信息。
151 diag.url = QString();
152 diag.latencyMs = timer.elapsed();
153 if (!m_datasheetData.isEmpty()) {
154 diag.statusCode = 200;
155 diag.errorString = "";
156 diag.wasRateLimited = false;
157 } else {
158 diag.statusCode = 0;
159 diag.errorString = "Not found in cache";
160 diag.wasRateLimited = false;
161 }
162 diag.retryCount = 0;
163 m_diagnostics.append(diag);
164
165 finishAllDownloads();
166 return;
167 }
168
169 // 使用异步下载
170 m_pendingOperations++;
171 CacheRepository::instance()->fetchDatasheetAsync(
172 m_componentId,
173 m_datasheetUrl,
174 &m_isAborted,
175 ConfigService::instance()->getWeakNetworkSupport(),
176 [this](const QByteArray& data, const ComponentExportStatus::NetworkDiagnostics& diag) {
177 if (m_isAborted.loadRelaxed()) {
178 m_pendingOperations--;
179 if (m_pendingOperations <= 0) {
180 finishAllDownloads();
181 }
182 return;
183 }
184
185 m_datasheetData = data;
186 ComponentExportStatus::NetworkDiagnostics diagCopy = diag; // 异步回调使用的诊断信息副本。
187 diagCopy.latencyMs = diag.latencyMs;
188 m_diagnostics.append(diagCopy);
189
190 m_pendingOperations--;
191 finishAllDownloads();
192 });
