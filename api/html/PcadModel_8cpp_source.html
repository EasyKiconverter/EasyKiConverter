82double number(const QString& text,
83 ParseDiagnostics* diagnostics,
84 const QString& field,
85 int line,
86 LengthUnit defaultUnit = LengthUnit::Mil) {
87 QString value = text.trimmed(); // 去除长度文本两侧空白。
88 LengthUnit unit = defaultUnit; // 当前长度使用的单位。
89 const QString lower = value.toLower(); // 用于不区分大小写的单位匹配。
90 if (lower.endsWith(QStringLiteral("mm"))) {
91 unit = LengthUnit::Millimeter;
92 value.chop(2);
93 } else if (lower.endsWith(QStringLiteral("mil"))) {
94 value.chop(3);
95 } else if (lower.endsWith(QStringLiteral("in"))) {
96 unit = LengthUnit::Inch;
97 value.chop(2);
98 }
99 const double parsed = StrictNumberParser::parseDouble(value, diagnostics, field, line); // 解析数值部分。
100 return UnitConverter::toMillimeters(parsed, unit);
101}

111QPointF point(const Node& node, ParseDiagnostics* diagnostics, const QString& field, LengthUnit defaultUnit) {
112 const QStringList values = scalarValues(node); // 坐标节点的 X、Y 标量参数。
113 if (values.size() < 2) {
114 if (diagnostics)
115 diagnostics->add(
116 ParseSeverity::Error, ParseScope::Field, QStringLiteral("P-CAD 坐标缺少分量"), field, node.line);
117 return {};
118 }
119 return QPointF(number(values.at(0), diagnostics, field + QStringLiteral(".X"), node.line, defaultUnit),
120 -number(values.at(1), diagnostics, field + QStringLiteral(".Y"), node.line, defaultUnit));
121}

129PcadPadShape padShape(const Node& node, ParseDiagnostics* diagnostics) {
130 const QString type = firstScalar(node).toUpper(); // 规范化后的形状关键字。
131 if (type == QStringLiteral("ROUND") || type == QStringLiteral("CIRCLE"))
132 return PcadPadShape::Round;
133 if (type == QStringLiteral("RECT") || type == QStringLiteral("RECTANGLE"))
134 return PcadPadShape::Rectangle;
135 if (type == QStringLiteral("OVAL"))
136 return PcadPadShape::Oval;
137 if (type == QStringLiteral("RNDRECT") || type == QStringLiteral("ROUNDRECT"))
138 return PcadPadShape::RoundRectangle;
139 if (diagnostics)
140 diagnostics->add(ParseSeverity::Warning,
141 ParseScope::Footprint,
142 QStringLiteral("P-CAD 未知焊盘形状，无法映射到统一 IR"),
143 type,
144 node.line);
145 return PcadPadShape::Unknown;
146}

155PcadPadStyle parsePadStyle(const Node& node, ParseDiagnostics* diagnostics, LengthUnit defaultUnit) {
156 PcadPadStyle style; // 保存当前 Pad Style 定义。
157 style.name = firstScalar(node);
158 const Node shape = firstList(node, QStringLiteral("PADSHAPE")); // Pad 形状节点。
159 if (!shape.atom.isEmpty()) {
160 style.shape = padShape(shape, diagnostics);
161 const Node shapeWidth = firstList(shape, QStringLiteral("SHAPEWIDTH")); // Pad 宽度节点。
162 const Node shapeHeight = firstList(shape, QStringLiteral("SHAPEHEIGHT")); // Pad 高度节点。
163 style.width =
164 number(firstScalar(shapeWidth), diagnostics, QStringLiteral("PADSTYLE.WIDTH"), shape.line, defaultUnit);
165 style.height =
166 number(firstScalar(shapeHeight), diagnostics, QStringLiteral("PADSTYLE.HEIGHT"), shape.line, defaultUnit);
167 }
168 const Node hole = firstList(node, QStringLiteral("HOLEDIAM")); // 孔径节点。
169 if (!hole.atom.isEmpty())
170 style.holeDiameter =
171 number(firstScalar(hole), diagnostics, QStringLiteral("PADSTYLE.HOLEDIAM"), hole.line, defaultUnit);
172 const Node plated = firstList(node, QStringLiteral("ISHOLEPLATED")); // 镀孔标志节点。
173 if (!plated.atom.isEmpty())
174 style.holePlated = firstScalar(plated).compare(QStringLiteral("FALSE"), Qt::CaseInsensitive) != 0;
175 if (style.name.isEmpty())
176 diagnostics->add(
177 ParseSeverity::Error, ParseScope::Field, QStringLiteral("P-CAD Pad Style 缺少名称"), {}, node.line);
178 return style;
179}

189 PcadPatternPad pad; // 保存当前 Pattern 焊盘实例。
190 const Node numberNode = firstList(node, QStringLiteral("PADNUM")); // 焊盘编号节点。
191 pad.number = firstScalar(numberNode);
192 const Node styleNode = firstList(node, QStringLiteral("PADSTYLEREF")); // Pad Style 引用节点。
193 pad.padStyleName = firstScalar(styleNode);
194 const Node position = firstList(node, QStringLiteral("PT")); // 焊盘位置节点。
195 if (!position.atom.isEmpty())
196 pad.position = point(position, diagnostics, QStringLiteral("PATTERN.PAD.PT"), defaultUnit);
197 const Node rotation = firstList(node, QStringLiteral("ROTATION")); // 焊盘旋转节点。
198 if (!rotation.atom.isEmpty())
199 pad.rotation = StrictNumberParser::parseDouble(

202 if (pad.number.isEmpty())
203 diagnostics->add(
204 ParseSeverity::Error, ParseScope::Footprint, QStringLiteral("P-CAD Pattern 焊盘缺少编号"), {}, node.line);
205 if (pad.padStyleName.isEmpty())
206 diagnostics->add(ParseSeverity::Error,

208 QStringLiteral("P-CAD Pattern 焊盘缺少 Pad Style 引用"),
209 pad.number,
210 node.line);

221PcadGraphic parseGraphic(const Node& node, ParseDiagnostics* diagnostics, LengthUnit defaultUnit) {
222 PcadGraphic graphic; // 保存当前图元的类型、几何和文本属性。
223 const QString kind = nameOf(node); // 图元关键字。
224 if (kind == QStringLiteral("LINE"))
225 graphic.type = PcadGraphicType::Line;
226 else if (kind == QStringLiteral("ARC") || kind == QStringLiteral("TRIPLEPOINTARC"))
227 graphic.type = PcadGraphicType::Arc;
228 else if (kind == QStringLiteral("CIRCLE"))
229 graphic.type = PcadGraphicType::Circle;
230 else if (kind == QStringLiteral("TEXT"))
231 graphic.type = PcadGraphicType::Text;
232 else if (kind == QStringLiteral("POLY") || kind == QStringLiteral("POLYGON") || kind == QStringLiteral("PCBPOLY"))
233 graphic.type = PcadGraphicType::Polygon;
234 else {
235 graphic.type = PcadGraphicType::Unknown;
236 diagnostics->add(
237 ParseSeverity::Skipped, ParseScope::Footprint, QStringLiteral("跳过不支持的 P-CAD 图元"), kind, node.line);
238 return graphic;
239 }
240 const Node layer = firstList(node, QStringLiteral("LAYERNUMREF")); // 图元层引用节点。
241 if (!layer.atom.isEmpty())
242 graphic.layerNumber = StrictNumberParser::parseInteger(
243 firstScalar(layer), diagnostics, QStringLiteral("GRAPHIC.LAYER"), layer.line);
244 const Node width = firstList(node, QStringLiteral("WIDTH")); // 图元线宽节点。
245 if (!width.atom.isEmpty())
246 graphic.width =
247 number(firstScalar(width), diagnostics, QStringLiteral("GRAPHIC.WIDTH"), width.line, defaultUnit);
248 for (const Node& child : node.children) {
249 if (!child.isList()) {
250 if (graphic.type == PcadGraphicType::Text && graphic.text.isEmpty())
251 graphic.text = child.atom;
252 continue;
253 }
254 const QString childName = nameOf(child); // 当前图元属性名称。
255 if (childName == QStringLiteral("PT")) {
256 if (graphic.type == PcadGraphicType::Arc || graphic.type == PcadGraphicType::Circle ||
257 graphic.type == PcadGraphicType::Text)
258 graphic.center = point(child, diagnostics, QStringLiteral("GRAPHIC.CENTER"), defaultUnit);
259 else
260 graphic.points.append(point(child, diagnostics, QStringLiteral("GRAPHIC.POINT"), defaultUnit));
261 } else if (childName == QStringLiteral("RADIUS")) {
262 graphic.radius =
263 number(firstScalar(child), diagnostics, QStringLiteral("GRAPHIC.RADIUS"), child.line, defaultUnit);
264 } else if (childName == QStringLiteral("STARTANGLE")) {
265 graphic.startAngle =
266 StrictNumberParser::parseDouble(
267 firstScalar(child), diagnostics, QStringLiteral("GRAPHIC.STARTANGLE"), child.line) /
268 10.0;
269 } else if (childName == QStringLiteral("SWEEPANGLE")) {
270 graphic.sweepAngle =
271 StrictNumberParser::parseDouble(
272 firstScalar(child), diagnostics, QStringLiteral("GRAPHIC.SWEEPANGLE"), child.line) /
273 10.0;
274 } else if (childName == QStringLiteral("HEIGHT") && graphic.type == PcadGraphicType::Text) {
275 graphic.textHeight =
276 number(firstScalar(child), diagnostics, QStringLiteral("GRAPHIC.TEXT.HEIGHT"), child.line, defaultUnit);
277 } else if (childName == QStringLiteral("ROTATION")) {
278 graphic.rotation = StrictNumberParser::parseDouble(
279 firstScalar(child), diagnostics, QStringLiteral("GRAPHIC.ROTATION"), child.line) /
280 10.0;
281 } else if (childName == QStringLiteral("WIDTH")) {
282 graphic.width =
283 number(firstScalar(child), diagnostics, QStringLiteral("GRAPHIC.WIDTH"), child.line, defaultUnit);
284 }
285 }
286 return graphic;
287}

296PcadPattern parsePattern(const Node& node, ParseDiagnostics* diagnostics, LengthUnit defaultUnit) {
297 PcadPattern pattern; // 保存当前 Pattern 的焊盘和图形。
298 pattern.name = firstScalar(node);
299 if (pattern.name.isEmpty())
300 diagnostics->add(
301 ParseSeverity::Error, ParseScope::Footprint, QStringLiteral("P-CAD Pattern 缺少名称"), {}, node.line);
302 for (const Node& child : node.children) {
303 if (!child.isList())
304 continue;
305 const QString kind = nameOf(child); // 当前 Pattern 子节点类型。
306 if (kind == QStringLiteral("PAD"))
307 pattern.pads.append(parsePatternPad(child, diagnostics, defaultUnit));
308 // 标准 P-CAD Pattern 可能把焊盘放在 MULTILAYER 容器中。
309 else if (kind == QStringLiteral("MULTILAYER")) {
310 for (const Node& padNode : child.children) {
311 if (padNode.isList() && nameOf(padNode) == QStringLiteral("PAD"))
312 pattern.pads.append(parsePatternPad(padNode, diagnostics, defaultUnit));
313 }
314 }
315 // Pattern 图形共用同一套图元解析和单位转换逻辑。
316 else if (kind == QStringLiteral("PATTERNGRAPHICS") || kind == QStringLiteral("GRAPHICS")) {
317 for (const Node& graphicNode : child.children) {
318 if (graphicNode.isList())
319 pattern.graphics.append(parseGraphic(graphicNode, diagnostics, defaultUnit));
320 }
321 } else if (kind == QStringLiteral("LAYERCONTENTS")) {
322 const Node layerNode = firstList(child, QStringLiteral("LAYERNUMREF")); // 容器层引用节点。
323 const int layerNumber = // 容器默认层号。
324 layerNode.atom.isEmpty()
325 ? 0
326 : StrictNumberParser::parseInteger(
327 firstScalar(layerNode), diagnostics, QStringLiteral("GRAPHICS.LAYER"), layerNode.line);
328 for (const Node& graphicNode : child.children) {
329 if (!graphicNode.isList() || nameOf(graphicNode) == QStringLiteral("LAYERNUMREF"))
330 continue;
331 PcadGraphic graphic = parseGraphic(graphicNode, diagnostics, defaultUnit); // 当前图元。
332 if (graphic.layerNumber == 0)
333 graphic.layerNumber = layerNumber;
334 pattern.graphics.append(graphic);
335 }
336 }
337 }
338 return pattern;
339}

351 const QString& name,
352 ParseDiagnostics* diagnostics,
353 const QString& type,
354 int line) {
355 if (name.isEmpty() || !diagnostics)
356 return;
357 int matches = 0; // 统计同名定义数量。
358 for (const Definition& definition : definitions) {

366 name,

377PcadPlacement parsePlacement(const Node& node, ParseDiagnostics* diagnostics, LengthUnit defaultUnit) {
378 PcadPlacement placement; // 保存当前器件放置及其关联。
379 const Node pattern = firstList(node, QStringLiteral("PATTERNREF")); // Pattern 引用节点。
380 placement.patternName = firstScalar(pattern);
381 const Node reference = firstList(node, QStringLiteral("REFDESREF")); // 参考标号节点。
382 placement.reference = firstScalar(reference);
383 const Node position = firstList(node, QStringLiteral("PT")); // 放置坐标节点。
384 if (!position.atom.isEmpty())
385 placement.position = point(position, diagnostics, QStringLiteral("PLACEMENT.PT"), defaultUnit);
386 const Node rotation = firstList(node, QStringLiteral("ROTATION")); // 放置旋转节点。
387 if (!rotation.atom.isEmpty())
388 placement.rotation =
389 StrictNumberParser::parseDouble(
390 firstScalar(rotation), diagnostics, QStringLiteral("PLACEMENT.ROTATION"), rotation.line) /
391 10.0;
392 const Node flipped = firstList(node, QStringLiteral("ISFLIPPED")); // 镜像标志节点。
393 placement.flipped = firstScalar(flipped).compare(QStringLiteral("TRUE"), Qt::CaseInsensitive) == 0;
394 if (placement.patternName.isEmpty())
395 diagnostics->add(ParseSeverity::Error,
396 ParseScope::Component,
397 QStringLiteral("P-CAD 器件放置缺少 Pattern 引用"),
398 placement.reference,
399 node.line);

401 diagnostics->add(ParseSeverity::Warning,
402 ParseScope::Component,
403 QStringLiteral("P-CAD 器件放置缺少参考标识"),
404 placement.patternName,

425PcadBoard PcadParser::parse(const QString& content, const QString& filePath) {
426 PcadBoard board; // 保存解析得到的 P-CAD 板模型和诊断。
427 board.diagnostics.setFilePath(filePath);
428 const QList<SExpressionNode> roots = SExpressionParser::parse(content, &board.diagnostics); // 根节点列表。
429 for (const Node& root : roots) {
430 if (nameOf(root) != QStringLiteral("ACCEL_ASCII"))
431 continue;
432 board.name = firstScalar(root);
433 // 先读取文件级单位，确保后续库定义不受节点顺序影响。
434 for (const Node& rootChild : root.children) {
435 if (rootChild.isList() &&
436 (nameOf(rootChild) == QStringLiteral("UNITS") || nameOf(rootChild) == QStringLiteral("ASCIIHEADER"))) {
437 const Node unitNode = nameOf(rootChild) == QStringLiteral("ASCIIHEADER") // 文件级单位节点。
438 ? firstList(rootChild, QStringLiteral("FILEUNITS"))
439 : rootChild;
440 board.unit = UnitConverter::parseUnit(firstScalar(unitNode), &board.diagnostics);
441 break;
442 }
443 }
444 for (const Node& child : root.children) {
445 if (!child.isList())
446 continue;
447 const QString kind = nameOf(child); // 当前根子节点类型。
448 if (kind == QStringLiteral("UNITS")) {
449 board.unit = UnitConverter::parseUnit(firstScalar(child), &board.diagnostics);
450 } else if (kind == QStringLiteral("ASCIIHEADER")) {
451 const Node unitNode = firstList(child, QStringLiteral("FILEUNITS")); // 文件单位节点。
452 if (!unitNode.atom.isEmpty())
453 board.unit = UnitConverter::parseUnit(firstScalar(unitNode), &board.diagnostics);
454 } else if (kind == QStringLiteral("LAYER") || kind == QStringLiteral("LAYERDEF")) {
455 const QStringList values = scalarValues(child); // 层定义的标量参数。
456 PcadLayer layer; // 当前层定义。
457 if (!values.isEmpty())
458 layer.number = static_cast<int>(StrictNumberParser::parseInteger(
459 values.first(), &board.diagnostics, QStringLiteral("LAYER.NUMBER"), child.line));
460 if (values.size() > 1)
461 layer.name = values.at(1);
462 if (values.isEmpty())
463 board.diagnostics.add(ParseSeverity::Error,
464 ParseScope::Field,
465 QStringLiteral("P-CAD 层定义缺少编号"),
466 {},
467 child.line);
468 board.layers.append(layer);
469 } else if (kind == QStringLiteral("LIBRARY")) {
470 for (const Node& definition : child.children) {
471 if (!definition.isList())
472 continue;
473 const QString definitionName = nameOf(definition); // 库定义类型。
474 if (definitionName == QStringLiteral("PADSTYLEDEF")) {
475 board.padStyles.append(parsePadStyle(definition, &board.diagnostics, board.unit));
476 reportDuplicateDefinition(board.padStyles,
477 board.padStyles.last().name,
478 &board.diagnostics,
479 QStringLiteral("Pad Style"),
480 definition.line);
481 } else if (definitionName == QStringLiteral("PATTERNDEF") ||
482 definitionName == QStringLiteral("PATTERNDEFEXTENDED")) {
483 board.patterns.append(parsePattern(definition, &board.diagnostics, board.unit));
484 reportDuplicateDefinition(board.patterns,
485 board.patterns.last().name,
486 &board.diagnostics,
487 QStringLiteral("Pattern"),
488 definition.line);
489 }
490 }
491 } else if (kind == QStringLiteral("PCBDESIGN")) {
492 for (const Node& design : child.children) {
493 if (!design.isList())
494 continue;
495 if (nameOf(design) == QStringLiteral("MULTILAYER")) {
496 for (const Node& placement : design.children) {
497 if (placement.isList() && nameOf(placement) == QStringLiteral("PATTERN"))
498 board.placements.append(parsePlacement(placement, &board.diagnostics, board.unit));
499 }
500 } else if (nameOf(design) == QStringLiteral("LAYERCONTENTS")) {
501 for (const Node& graphic : design.children) {
502 if (graphic.isList())
503 board.graphics.append(parseGraphic(graphic, &board.diagnostics, board.unit));
504 }
505 }
506 }
507 }
508 }
509 break;
510 }
511 if (roots.isEmpty() || !board.isRecognized())
512 board.diagnostics.add(
513 ParseSeverity::Error, ParseScope::File, QStringLiteral("未识别到有效的 P-CAD ASCII PCB 定义"), filePath);
514 for (int index = 0; index < board.padStyles.size(); ++index) {
515 const PcadPadStyle& style = board.padStyles.at(index); // 当前待检查的 Pad Style。
516 for (int otherIndex = index + 1; otherIndex < board.padStyles.size(); ++otherIndex) {
517 if (style.name == board.padStyles.at(otherIndex).name && !style.name.isEmpty()) {
518 board.diagnostics.add(ParseSeverity::Error,
519 ParseScope::Footprint,
520 QStringLiteral("P-CAD Pad Style 名称重复"),
521 style.name);
522 break;
523 }
524 }
525 }
526 for (int index = 0; index < board.patterns.size(); ++index) {
527 const PcadPattern& pattern = board.patterns.at(index); // 当前待检查的 Pattern。
528 for (int otherIndex = index + 1; otherIndex < board.patterns.size(); ++otherIndex) {
529 if (pattern.name == board.patterns.at(otherIndex).name && !pattern.name.isEmpty()) {
530 board.diagnostics.add(ParseSeverity::Error,
531 ParseScope::Footprint,
532 QStringLiteral("P-CAD Pattern 名称重复"),
533 pattern.name);
534 break;
535 }
536 }
537 }
538 return board;
539}
