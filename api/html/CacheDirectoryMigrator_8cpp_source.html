118bool collectComponentEntries(const QString& sourceDir,
119 const QString& targetDir,
120 QList<MigrationEntry>* entries,
121 QString* error) {
122 const QDir source(sourceDir); // 组件缓存源目录对象。
123 const QFileInfoList sourceEntries = // 源目录中的实际条目。
124 source.entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot);
125 for (const QFileInfo& entry : sourceEntries) {
126 if (!entry.isFile() || entry.isSymLink() || !knownComponentFiles().contains(entry.fileName())) {
127 if (error)
128 *error = QStringLiteral("缓存迁移遇到无法验证的组件内容：%1").arg(entry.absoluteFilePath());
129 return false;
130 }
131 const QString targetPath = QDir(targetDir).filePath(entry.fileName()); // 当前条目的目标路径。
132 if (QFileInfo::exists(targetPath)) {
133 if (error)
134 *error = QStringLiteral("缓存迁移目标已存在同名文件：%1").arg(targetPath);
135 return false;
136 }
137 entries->append({entry.absoluteFilePath(), targetPath});
138 }
139 return true;

173bool moveEntriesTransactional(const QList<MigrationEntry>& entries, QString* error) {
174 QList<MigrationEntry> moved; // 已成功移动、失败时需要回滚的条目。
175 const auto rollback = [&moved]() {
176 for (auto it = moved.crbegin(); it != moved.crend(); ++it) {
177 if (!QFile::rename(it->targetPath, it->sourcePath)) {
178 LOG_ERROR(LogModule::Core,
179 "Failed to roll back cache migration entry: {} -> {}",
180 it->targetPath,
181 it->sourcePath);
182 }
183 }
184 };
185 for (const MigrationEntry& entry : entries) {
186 QDir targetParent(QFileInfo(entry.targetPath).absolutePath()); // 当前目标文件的父目录。
187 if (!targetParent.exists() && !targetParent.mkpath(QStringLiteral("."))) {
188 LOG_WARN(LogModule::Core, "Failed to create cache migration parent directory: {}", targetParent.path());
189 if (error)
190 *error = QStringLiteral("无法创建缓存迁移目录：%1").arg(targetParent.path());
191 rollback();
192 return false;
193 }
194 if (!QFile::rename(entry.sourcePath, entry.targetPath)) {
195 LOG_WARN(LogModule::Core, "Failed to migrate cache file: {} -> {}", entry.sourcePath, entry.targetPath);
196 if (error)
197 *error = QStringLiteral("无法迁移缓存文件：%1").arg(entry.sourcePath);
198 rollback();
199 return false;
200 }
201 moved.append(entry);
202 }
203 return true;
204}

215bool CacheDirectoryMigrator::migrate(const QString& oldCacheDir, const QString& newCacheDir, QString* error) {
216 if (oldCacheDir.isEmpty() || newCacheDir.isEmpty() || oldCacheDir == newCacheDir) {
217 return true;
218 }
219
220 QDir source(oldCacheDir);
221 if (!source.exists()) {
222 return true;
223 }
224 const bool legacySource = !CacheSafety::isOwnedRoot(oldCacheDir); // 源目录是否为旧版未标记缓存。
225 if (legacySource && !CacheSafety::canAdoptLegacyRoot(oldCacheDir) &&
226 !source.entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot).isEmpty()) {
227 if (error)
228 *error = QStringLiteral("旧缓存目录包含无法验证归属的数据，源目录已保留：%1").arg(oldCacheDir);
229 LOG_WARN(
230 LogModule::Core, "Skipped legacy cache migration because source ownership is ambiguous: {}", oldCacheDir);
231 return false;
232 }
233 if (!CacheSafety::isOwnedRoot(newCacheDir)) {
234 if (error)
235 *error = QStringLiteral("缓存迁移目标没有有效所有权标记：%1").arg(newCacheDir);
236 LOG_WARN(
237 LogModule::Core, "Skipped cache migration because target ownership cannot be verified: {}", newCacheDir);
238 return false;
239 }
240
241 QDir target; // 缓存迁移目标目录对象。
242 if (!target.exists(newCacheDir) && !target.mkpath(newCacheDir)) {
243 if (error)
244 *error = QStringLiteral("无法创建缓存迁移目标目录：%1").arg(newCacheDir);
245 LOG_WARN(LogModule::Core, "Failed to create cache migration target directory: {}", newCacheDir);
246 return false;
247 }
248
249 QList<MigrationEntry> migrationEntries; // 预检通过后待执行的全部迁移条目。
250 const QStringList componentDirectories = // 待迁移的组件缓存目录列表。
251 legacySource ? CacheSafety::legacyComponentDirectories(oldCacheDir)
252 : CacheSafety::ownedComponentDirectories(oldCacheDir);
253 for (const QString& sourcePath : componentDirectories) {
254 const QString targetPath = // 当前组件目录的迁移目标路径。
255 QDir(newCacheDir).filePath(QFileInfo(sourcePath).fileName());
256 if (QDir(targetPath).exists() && !CacheSafety::isOwnedComponentDirectory(newCacheDir, targetPath) &&
257 !QDir(targetPath).entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot).isEmpty()) {
258 if (error)
259 *error = QStringLiteral("缓存迁移目标包含未托管目录：%1").arg(targetPath);
260 LOG_WARN(LogModule::Core, "Refusing cache migration into an unowned target directory: {}", targetPath);
261 return false;
262 }
263 QString componentError; // 当前组件内容预检错误。
264 if (!collectComponentEntries(sourcePath, targetPath, &migrationEntries, &componentError)) {
265 if (error)
266 *error = componentError;
267 LOG_WARN(LogModule::Core,
268 "Refusing cache migration because source contains unknown data: {} ({})",
269 sourcePath,
270 componentError);
271 return false;
272 }
273 }
274 QStringList modelFiles = CacheSafety::ownedModel3DFiles(oldCacheDir); // 已验证的三维模型文件列表。
275 if (legacySource) {
276 const QDir modelRoot(QDir(oldCacheDir).filePath(QStringLiteral("model3d")));
277 for (const QFileInfo& info : modelRoot.entryInfoList(QDir::Files | QDir::NoDotAndDotDot, QDir::Name)) {
278 if (isLegacyModelFile(info))
279 modelFiles.append(info.absoluteFilePath());
280 }
281 }
282 QString preflightError; // 三维模型目标预检错误。
283 if (!collectModelEntries(modelFiles, newCacheDir, &migrationEntries, &preflightError)) {
284 if (error)
285 *error = preflightError;
286 LOG_WARN(LogModule::Core, "{}", preflightError);
287 return false;
288 }
289
290 QList<MetadataSnapshot> metadataSnapshots; // 旧版元数据快照，用于事务失败回滚。
291 if (legacySource) {
292 QStringList metadataPaths; // 待升级的旧版元数据路径列表。
293 for (const QString& sourcePath : componentDirectories) {
294 metadataPaths.append(QDir(sourcePath).filePath(QStringLiteral("component.json")));
295 }
296 if (!snapshotMetadata(metadataPaths, &metadataSnapshots, error))
297 return false;
298 for (const QString& metadataPath : metadataPaths) {
299 if (!upgradeMetadata(metadataPath)) {
300 if (error)
301 *error = QStringLiteral("无法升级旧缓存元数据：%1").arg(metadataPath);
302 LOG_WARN(LogModule::Core, "Refusing cache migration because metadata is invalid: {}", metadataPath);
303 restoreMetadata(metadataSnapshots);
304 return false;
305 }
306 }
307 }
308 if (!moveEntriesTransactional(migrationEntries, error)) {
309 restoreMetadata(metadataSnapshots);
310 return false;
311 }
312
313 for (const QString& sourcePath : componentDirectories) {
314 if (QDir(sourcePath).entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot).isEmpty())
315 QDir().rmdir(sourcePath);
316 }
317 LOG_DEBUG(LogModule::Core, "Migrated owned cache entries from {} to {}", oldCacheDir, newCacheDir);
318 return true;
319}

330bool CacheDirectoryMigrator::moveDirectoryContents(const QString& sourceRoot,
331 const QString& sourceDir,
332 const QString& targetRoot,
333 const QString& targetDir,
334 bool legacySource) {
335 QDir source(sourceDir); // 源组件目录对象。
336 if (!source.exists()) {
337 return true;
338 }
339
340 QDir target; // 目标组件目录对象。
341 if (target.exists(targetDir) && !CacheSafety::isOwnedComponentDirectory(targetRoot, targetDir) &&
342 !QDir(targetDir).entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot).isEmpty()) {
343 LOG_WARN(LogModule::Core, "Refusing cache migration into an unowned target directory: {}", targetDir);
344 return false;
345 }
346 if (!target.exists(targetDir) && !target.mkpath(targetDir)) {
347 LOG_WARN(LogModule::Core, "Failed to create cache migration directory: {}", targetDir);
348 return false;
349 }
350
351 // 组件目录的元数据会在迁移第一个文件后暂时离开源目录，因此必须在循环前完成一次所有权确认。
352 const bool sourceOwned = legacySource || CacheSafety::isOwnedComponentDirectory(sourceRoot, sourceDir);
353 if (!sourceOwned) {
354 LOG_WARN(LogModule::Core, "Refusing cache migration from an unowned component directory: {}", sourceDir);
355 return false;
356 }
357
358 QList<MigrationEntry> entries; // 当前组件目录的待迁移条目。
359 if (!collectComponentEntries(sourceDir, targetDir, &entries, nullptr))
360 return false;
361 const QString metadataPath = QDir(sourceDir).filePath(QStringLiteral("component.json")); // 当前组件元数据路径。
362 QFile metadataFile(metadataPath); // 当前组件元数据文件。
363 if (!metadataFile.open(QIODevice::ReadOnly))
364 return false;
365 const MetadataSnapshot metadataSnapshot{metadataPath, metadataFile.readAll()}; // 当前组件元数据快照。
366 if (!upgradeMetadata(metadataPath))
367 return false;
368 if (!moveEntriesTransactional(entries, nullptr)) {
369 restoreMetadata({metadataSnapshot});
370 return false;
371 }
372 const QString targetMetadataPath = // 迁移后目标目录中的元数据路径。
373 QDir(targetDir).filePath(QStringLiteral("component.json"));
374 return QFileInfo::exists(targetMetadataPath);
375}

383bool CacheDirectoryMigrator::moveCacheEntry(const QString& sourcePath, const QString& targetPath) {
384 QFileInfo sourceInfo(sourcePath); // 源条目文件信息。
385 if (!sourceInfo.exists()) {
386 return true;
387 }
388
389 QFileInfo targetInfo(targetPath); // 目标条目文件信息。
390 if (targetInfo.exists()) {
391 LOG_WARN(LogModule::Core, "Skipping cache migration entry because target already exists: {}", targetPath);
392 return false;
393 }
394
395 QDir targetParent(targetInfo.absolutePath()); // 目标条目父目录。
396 if (!targetParent.exists() && !targetParent.mkpath(QStringLiteral("."))) {
397 LOG_WARN(LogModule::Core, "Failed to create cache migration parent directory: {}", targetInfo.absolutePath());
398 return false;
399 }
400
401 if (sourceInfo.isDir()) {
402 LOG_WARN(LogModule::Core, "Refusing to migrate an unexpected cache directory entry: {}", sourcePath);
403 return false;
404 }
405
406 if (QFile::rename(sourcePath, targetPath))
407 return true;
408 LOG_WARN(LogModule::Core, "Failed to migrate cache file: {} -> {}", sourcePath, targetPath);
409 return false;
410}
