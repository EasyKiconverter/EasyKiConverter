19 for (const ComponentListItemData* item : items) {

21 continue;
22 }
23
24 // 获取当前元件的验证阶段。
25 const QString phase = item->validationPhase();
26 if (phase == QStringLiteral("validating")) {
27 ++validatingCount;
28 } else if (phase == QStringLiteral("completed") || phase == QStringLiteral("fetching_preview")) {
29 ++validCount;
30 } else if (phase == QStringLiteral("failed")) {
31 ++invalidCount;
32 if (item->retryable()) {
33 ++retryableInvalidCount;
34 }
35 }
36 }

40 m_invalidCount = invalidCount;
