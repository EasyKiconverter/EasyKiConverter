27bool CacheDirectoryCoordinator::setDirectory(const QString& cacheDir, bool migrateExistingCache, QString* error) {
28 const auto fail = [error](const QString& reason) {
29 if (error)
30 *error = reason;
31 return false;
32 };
33 QString newCacheDir; // 通过安全校验后的新缓存根目录。
34 QString validationError; // 目录选择或所有权校验的具体错误。
35 if (!CacheSafety::validateSelection(cacheDir, &newCacheDir, &validationError) ||
36 !CacheSafety::ensureOwnedRoot(newCacheDir, &validationError)) {
37 emit m_owner.cacheMaintenanceWarning(validationError);
38 return fail(validationError);
39 }
40 QString oldCacheDir; // 当前生效的旧缓存根目录。
41 {
42 QMutexLocker locker(&m_owner.m_cacheDirMutex);
43 oldCacheDir = m_owner.m_cacheDir;
44 }
45
46 const bool cacheDirChanged = oldCacheDir != newCacheDir; // 是否需要执行目录切换和一级缓存失效。
47 {
48 QMutexLocker diskLocker(&m_owner.m_diskWriteMutex);
49 if (cacheDirChanged) {
50 // 先使切换前排队的异步写入失效，再进行目录迁移，避免旧请求污染新目录。
51 m_owner.m_cacheGeneration.fetch_add(1);
52 m_owner.m_tombstones.reset();
53 }
54
55 // 迁移期间持有磁盘写锁，避免异步写入与目录迁移交错。
56 if (migrateExistingCache && !oldCacheDir.isEmpty() && cacheDirChanged) {
57 QString model3dError; // 新目录三维模型子目录所有权错误。
58 if (!CacheSafety::ensureOwnedModel3DDirectory(newCacheDir, &model3dError)) {
59 emit m_owner.cacheMaintenanceWarning(model3dError);
60 return fail(model3dError);
61 }
62 QString migrationError; // 目录迁移失败的具体原因。
63 if (!CacheDirectoryMigrator::migrate(oldCacheDir, newCacheDir, &migrationError)) {
64 // 优先保留迁移器提供的具体错误，避免覆盖可操作的失败原因。
65 const QString message = migrationError.isEmpty()
66 ? QStringLiteral("缓存目录迁移失败，源目录已保留：%1").arg(oldCacheDir)
67 : migrationError;
68 emit m_owner.cacheMaintenanceWarning(message);
69 return fail(message);
70 }
71 }
72
73 // 目录创建也在锁内，确保切换期间读写使用完整的目录结构。
74 QDir dir;
75 if (!dir.exists(newCacheDir)) {
76 dir.mkpath(newCacheDir);
77 }
78 QString model3dError; // 新目录三维模型子目录所有权错误。
79 if (!CacheSafety::ensureOwnedModel3DDirectory(newCacheDir, &model3dError)) {
80 emit m_owner.cacheMaintenanceWarning(model3dError);
81 return fail(model3dError);
82 }
83
84 // 原子切换缓存目录指针。
85 {
86 QMutexLocker locker(&m_owner.m_cacheDirMutex);
87 m_owner.m_cacheDir = newCacheDir;
88 }
89 if (cacheDirChanged) {
90 // L1 数据没有目录归属信息，切换 L2 目录后必须全部失效，避免
91 // 同一元件 ID 从旧目录泄漏到新目录。
92 m_owner.m_memoryCache.clear();
93 }
94 }
95 if (cacheDirChanged) {
96 emit m_owner.memoryCacheSizeChanged(0);
97 }
98
99 m_owner.selfHealCache();
100 LOG_DEBUG(LogModule::Core, "Cache directory set to: {}", newCacheDir);
101 return true;
102}
