24QString safeName(const QString& value) {
25 QString result;
26 for (const QChar character : value) {
27 const ushort code = character.unicode();
28 if ((code >= 'A' && code <= 'Z') || (code >= 'a' && code <= 'z') || (code >= '0' && code <= '9') ||
29 character == QChar('_') || character == QChar('-')) {
30 result.append(character);
31 } else {
32 result.append(QChar('_'));
33 }
34 }
35 return result.left(40);
36}

79QChar partTypePinCode(IR::PinElectricalType type) {
80 // PADS 使用固定的单字符类型，未知或无源语义统一保留为未定义类型。
81 switch (type) {
82 case IR::PinElectricalType::Input:
83 return QChar('L');
84 case IR::PinElectricalType::Output:
85 return QChar('S');
86 case IR::PinElectricalType::Bidirectional:
87 return QChar('B');
88 case IR::PinElectricalType::Power:
89 return QChar('P');
90 case IR::PinElectricalType::OpenCollector:
91 return QChar('C');
92 case IR::PinElectricalType::OpenEmitter:
93 return QChar('O');
94 case IR::PinElectricalType::Passive:
95 case IR::PinElectricalType::Unspecified:
96 default:
97 return QChar('U');
98 }

108QString uniqueName(const QString& value, QSet<QString>& usedNames, QStringList& diagnostics) {
109 const QString base = safeName(value);
110 if (base.isEmpty()) {
111 diagnostics.append(QStringLiteral("PADS: 名称清洗后为空: %1").arg(value));
112 return {};
113 }
114 QString candidate = base.left(40);
115 int suffix = 2;
116 while (usedNames.contains(candidate.toLower())) {
117 const QString suffixText = QStringLiteral("_%1").arg(suffix++);
118 candidate = base.left(40 - suffixText.size()) + suffixText;
119 }
120 usedNames.insert(candidate.toLower());
121 return candidate;
122}

132bool writePiece(QTextStream& stream, const QString& type, const QList<QPointF>& points, double width) {
133 if (points.isEmpty() || !finiteWidth(width))
134 return false;
135 stream << type << ' ' << points.size() << ' ' << number(width) << " 0 -1\n";
136 for (const QPointF& point : points)
137 stream << number(point.x()) << ' ' << number(-point.y()) << '\n';
138 return stream.status() == QTextStream::Ok;

148bool validatePart(const IR::SymbolComponentIR& symbol, int part, QStringList& diagnostics) {
149 if (safeName(symbol.name).isEmpty()) {
150 diagnostics.append(QStringLiteral("PADS: 符号名称清洗后为空: %1").arg(symbol.name));
151 return false;
152 }
153 if (part < 0 || part >= qMax(1, symbol.partCount)) {
154 diagnostics.append(QStringLiteral("PADS: 符号 %1 的部件索引无效: %2").arg(symbol.name).arg(part));
155 return false;
156 }
157 if (!symbol.arcs.isEmpty() || !symbol.ellipses.isEmpty() || !symbol.pies.isEmpty() ||
158 !symbol.ellipticalArcs.isEmpty() || !symbol.paths.isEmpty() || !symbol.beziers.isEmpty() ||
159 !symbol.images.isEmpty() || !symbol.textFrames.isEmpty()) {
160 diagnostics.append(
161 QStringLiteral("PADS: 符号 %1 含当前 Schematic Decal writer 无法无损表达的图元").arg(symbol.name));
162 return false;
163 }
164 for (const IR::SymbolPinIR& pin : symbol.pins) {
165 if (pin.partIndex != part && !pin.commonToAllParts)
166 continue;
167 if (pin.designator.isEmpty() || pin.designator.contains(QRegularExpression(QStringLiteral("[\\s\\\"]"))) ||
168 !finitePoint(pin.position) || !std::isfinite(pin.length) || pin.length < 0.0) {
169 diagnostics.append(
170 QStringLiteral("PADS: 符号 %1 的引脚编号、坐标或长度无效: %2").arg(symbol.name).arg(pin.designator));
171 return false;
172 }
173 if (pin.name.contains(QRegularExpression(QStringLiteral("[\\s\\\"]")))) {
174 diagnostics.append(
175 QStringLiteral("PADS: 符号 %1 的引脚名称不能包含空格或引号: %2").arg(symbol.name).arg(pin.name));
176 return false;
177 }
178 }
179 for (const IR::SymbolTextIR& text : symbol.texts) {
180 if (text.partIndex != part || !text.visible)
181 continue;
182 if (!finitePoint(text.position) || !std::isfinite(text.rotation) || !std::isfinite(text.fontSizeMm) ||
183 text.fontSizeMm < 0.0) {
184 diagnostics.append(QStringLiteral("PADS: 符号 %1 的文本几何无效").arg(symbol.name));
185 return false;
186 }
187 }
188 return true;
189}

197int pieceCount(const IR::SymbolComponentIR& symbol, int part) {
198 int count = 0;
199 for (const auto& rectangle : symbol.rectangles)
200 count += rectangle.partIndex == part ? 1 : 0;
201 for (const auto& circle : symbol.circles)
202 count += circle.partIndex == part ? 1 : 0;
203 for (const auto& polyline : symbol.polylines)
204 count += polyline.partIndex == part && polyline.points.size() >= 2 ? 1 : 0;
205 for (const auto& polygon : symbol.polygons)
206 count += polygon.partIndex == part && polygon.points.size() >= 3 ? 1 : 0;
207 return count;
208}

245bool writePart(QTextStream& stream,
246 const IR::SymbolComponentIR& symbol,
247 const QString& baseName,
248 int part,
249 QStringList& diagnostics) {
250 const int pieces = pieceCount(symbol, part);
251 const int texts = textCount(symbol, part);
252 const int pins = pinCount(symbol, part);
253 const QString partName = part == 0 ? baseName : baseName + QStringLiteral("_P%1").arg(part + 1);
254 stream << partName << " 0 0 50 5 50 5 2 " << pieces << ' ' << texts << ' ' << pins << " 0\n";
255 stream << "TIMESTAMP " << QDateTime::currentDateTimeUtc().toString(QStringLiteral("yyyy.MM.dd.hh.mm.ss")) << "\n";
256 stream << "Default Font\nDefault Font\n";
257 stream << "0 0 0 12 50 5 Default Font\nREF-DES\n";
258 stream << "0 0 0 12 50 5 Default Font\nPARTTYPE\n";
259
260 for (const auto& rectangle : symbol.rectangles) {
261 if (rectangle.partIndex != part)
262 continue;
263 if (!finiteWidth(rectangle.strokeWidth) || !std::isfinite(rectangle.x0) || !std::isfinite(rectangle.y0) ||
264 !std::isfinite(rectangle.x1) || !std::isfinite(rectangle.y1) || rectangle.cornerRadiusX > 0.0 ||
265 rectangle.cornerRadiusY > 0.0) {
266 diagnostics.append(QStringLiteral("PADS: 符号 %1 的矩形包含非法值或圆角，无法无损写入").arg(symbol.name));
267 return false;
268 }
269 const QList<QPointF> points = {{rectangle.x0, rectangle.y0},
270 {rectangle.x1, rectangle.y0},
271 {rectangle.x1, rectangle.y1},
272 {rectangle.x0, rectangle.y1},
273 {rectangle.x0, rectangle.y0}};
274 if (!writePiece(stream, QStringLiteral("CLOSED"), points, rectangle.strokeWidth))
275 return false;
276 }
277 for (const auto& circle : symbol.circles) {
278 if (circle.partIndex != part || !finitePoint(circle.center) || !std::isfinite(circle.radius) ||
279 circle.radius <= 0.0 || !finiteWidth(circle.strokeWidth)) {
280 if (circle.partIndex == part)
281 diagnostics.append(QStringLiteral("PADS: 符号 %1 的圆图元无效").arg(symbol.name));
282 if (circle.partIndex == part)
283 return false;
284 continue;
285 }
286 const QList<QPointF> points =
287 {{circle.center.x() - circle.radius, circle.center.y() - circle.radius},
288 {circle.center.x() + circle.radius, circle.center.y() + circle.radius}};
289 if (!writePiece(stream, QStringLiteral("CIRCLE"), points, circle.strokeWidth))
290 return false;
291 }
292 for (const auto& polyline : symbol.polylines) {
293 if (polyline.partIndex != part)
294 continue;
295 if (polyline.points.size() < 2 || !finiteWidth(polyline.strokeWidth) ||
296 !std::all_of(polyline.points.cbegin(), polyline.points.cend(), finitePoint)) {
297 diagnostics.append(QStringLiteral("PADS: 符号 %1 的折线图元无效").arg(symbol.name));
298 return false;
299 }
300 if (!writePiece(stream, QStringLiteral("OPEN"), polyline.points, polyline.strokeWidth))
301 return false;
302 }
303 for (const auto& polygon : symbol.polygons) {
304 if (polygon.partIndex != part)
305 continue;
306 if (polygon.points.size() < 3 || !finiteWidth(polygon.strokeWidth) ||
307 !std::all_of(polygon.points.cbegin(), polygon.points.cend(), finitePoint)) {
308 diagnostics.append(QStringLiteral("PADS: 符号 %1 的多边形图元无效").arg(symbol.name));
309 return false;
310 }
311 QList<QPointF> points = polygon.points;
312 points.append(points.first());
313 if (!writePiece(stream, QStringLiteral("CLOSED"), points, polygon.strokeWidth))
314 return false;
315 }
316 for (const auto& text : symbol.texts) {
317 if (text.partIndex != part || !text.visible || text.text.isEmpty())
318 continue;
319 const double font = text.fontSizeMm > 0.0 ? text.fontSizeMm : 1.27;
320 stream << number(text.position.x()) << ' ' << number(-text.position.y()) << ' '
321 << QString::number(text.rotation) << " 0 " << number(font) << " " << number(0.127)
322 << " 0 12 0 0 Default Font\n"
323 << textValue(text.text) << '\n';
324 }
325 for (const auto& pin : symbol.pins) {
326 if (pin.partIndex != part && !pin.commonToAllParts)
327 continue;
328 stream << "T " << number(pin.position.x()) << ' ' << number(-pin.position.y()) << " 0 0 0 0 0 12 0 0 0 12 PIN\n"
329 << "P 0 0 0 12 0 0 0 12 192\n";
330 }
331 return stream.status() == QTextStream::Ok;
332}

341QByteArray buildPartTypeFile(const QList<IR::SymbolComponentIR>& symbols,
342 const QList<QString>& symbolNames,
343 QStringList& diagnostics) {
344 QByteArray bytes;
345 QBuffer buffer(&bytes);
346 if (!buffer.open(QIODevice::WriteOnly)) {
347 diagnostics.append(QStringLiteral("PADS: 无法创建 Part Type 缓冲区"));
348 return {};
349 }
350 QTextStream stream(&buffer);
351 stream.setEncoding(QStringConverter::Utf8);
352 stream << "*PADS-LIBRARY-PART-TYPES-V9*\n";
353 for (int index = 0; index < symbols.size(); ++index) {
354 const auto& symbol = symbols.at(index);
355 const QString& symbolName = symbolNames.at(index);
356 const QString footprintName = safeName(symbol.footprintName);
357 if (footprintName.isEmpty()) {
358 diagnostics.append(QStringLiteral("PADS: 符号 %1 缺少有效封装关联，无法生成 Part Type").arg(symbol.name));
359 return {};
360 }
361 const int partCount = qMax(1, symbol.partCount);
362 if (partCount > 702) {
363 diagnostics.append(QStringLiteral("PADS: 符号 %1 的部件数量超过 Part Type 限制").arg(symbol.name));
364 return {};
365 }
366 stream << symbolName << ' ' << footprintName << " I STD 0 " << partCount << " 0 0 0\n";
367 stream << "TIMESTAMP " << QDateTime::currentDateTimeUtc().toString(QStringLiteral("yyyy.MM.dd.hh.mm.ss"))
368 << "\n";
369 for (int part = 0; part < partCount; ++part) {
370 const QString decalName =
371 part == 0 ? symbolName : symbolName + QStringLiteral("_P%1").arg(part + 1);
372 QList<const IR::SymbolPinIR*> pins;
373 QSet<QString> pinNumbers;
374 for (const auto& pin : symbol.pins) {
375 if (pin.partIndex != part && !pin.commonToAllParts)
376 continue;
377 if (pin.designator.isEmpty() || pin.designator.size() > 7 || pin.designator.contains(' ')) {
378 diagnostics.append(
379 QStringLiteral("PADS: 符号 %1 的 Part Type 引脚编号无效: %2").arg(symbol.name, pin.designator));
380 return {};
381 }
382 if (pinNumbers.contains(pin.designator.toLower())) {
383 diagnostics.append(QStringLiteral("PADS: 符号 %1 的部件 %2 存在重复引脚编号: %3")
384 .arg(symbol.name)
385 .arg(part + 1)
386 .arg(pin.designator));
387 return {};
388 }
389 if (pin.name.contains(QRegularExpression(QStringLiteral("[\\s\\\"]")))) {
390 diagnostics.append(
391 QStringLiteral("PADS: 符号 %1 的 Part Type 引脚名称无效: %2").arg(symbol.name, pin.name));
392 return {};
393 }
394 pinNumbers.insert(pin.designator.toLower());
395 pins.append(&pin);
396 }
397 stream << "GATE 1 " << pins.size() << " 0\n" << decalName << '\n';
398 for (const auto* pin : pins) {
399 stream << pin->designator << " 0 " << partTypePinCode(pin->electricalType) << ' '
400 << (pin->name.isEmpty() ? QStringLiteral("UNNAMED") : pin->name) << '\n';
401 }
402 }
403 }
404 stream << "*END*\n";
405 stream.flush();
406 if (stream.status() != QTextStream::Ok) {
407 diagnostics.append(QStringLiteral("PADS: Part Type 文件生成失败"));
408 return {};
409 }
410 return bytes;
411}

441bool ExporterPadsSymbol::exportSymbolLibrary(const QList<IR::SymbolComponentIR>& symbols,
442 const QString&,
443 const QString& filePath,
444 bool appendMode,
445 bool updateMode,
446 const QString&) {
447 m_diagnostics.clear();
448 m_companionFiles.clear();
449 if (appendMode || updateMode) {
450 m_diagnostics.append(QStringLiteral("PADS Schematic Decal 当前只支持完整重写，不支持追加或更新模式"));
451 return false;
452 }
453 if (symbols.isEmpty()) {
454 m_diagnostics.append(QStringLiteral("PADS: 没有可导出的符号"));
455 return false;
456 }
457 QFile file(filePath);
458 if (!file.open(QIODevice::WriteOnly | QIODevice::Truncate | QIODevice::Text)) {
459 m_diagnostics.append(QStringLiteral("PADS: 无法写入符号库 %1").arg(filePath));
460 return false;
461 }
462 QTextStream stream(&file);
463 stream.setEncoding(QStringConverter::Utf8);
464 stream << "*PADS-LIBRARY-SCH-DECALS-V9*\n";
465 QSet<QString> usedNames;
466 QList<QString> symbolNames;
467 symbolNames.reserve(symbols.size());
468 for (const auto& symbol : symbols) {
469 const QString outputName = uniqueName(symbol.name, usedNames, m_diagnostics);
470 if (outputName.isEmpty())
471 return false;
472 symbolNames.append(outputName);
473 const int count = qMax(1, symbol.partCount);
474 for (int part = 0; part < count; ++part) {
475 if (!validatePart(symbol, part, m_diagnostics) ||
476 !writePart(stream, symbol, outputName, part, m_diagnostics))
477 return false;
478 }
479 }
480 stream << "*END*\n";
481 stream.flush();
482 if (stream.status() != QTextStream::Ok) {
483 m_diagnostics.append(QStringLiteral("PADS: 写入符号库时发生 I/O 错误"));
484 return false;
485 }
486 const QByteArray partTypeBytes = buildPartTypeFile(symbols, symbolNames, m_diagnostics);
487 if (partTypeBytes.isEmpty())
488 return false;
489 m_companionFiles.insert(QFileInfo(filePath).completeBaseName() + QStringLiteral(".p"), partTypeBytes);
490 m_diagnostics.append(
491 QStringLiteral("PADS: 已为 %1 个符号输出 Schematic Decal 与 Part Type 封装关联").arg(symbols.size()));
492 return true;
493}
