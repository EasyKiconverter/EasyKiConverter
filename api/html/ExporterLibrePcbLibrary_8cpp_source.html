156QString uuidFor(const QString& kind, const QString& name, int index = -1) {
157 // 将元素类型、名称和可选索引组合为稳定 UUID 的输入串。
158 const QByteArray input = (QStringLiteral("EasyKiConverter/LibrePCB/2/") + kind + QLatin1Char('/') + name +
159 (index >= 0 ? QStringLiteral("/") + QString::number(index) : QString()))
160 .toUtf8();
161 // LibrePCB 2.x 只接受 RFC 4122 的随机 UUID；通过哈希生成稳定值后重写版本位。
162 QByteArray digest = QCryptographicHash::hash(input, QCryptographicHash::Sha1).left(16);
163 digest[6] = char((uchar(digest.at(6)) & 0x0f) | 0x40);
164 digest[8] = char((uchar(digest.at(8)) & 0x3f) | 0x80);
165 const auto hex = [](uchar byte) { return QStringLiteral("%1").arg(byte, 2, 16, QLatin1Char('0')); };
166 QString result;
167 for (int i = 0; i < digest.size(); ++i) {
168 result += hex(uchar(digest.at(i)));
169 if (i == 3 || i == 5 || i == 7 || i == 9)
170 result += QLatin1Char('-');
171 }
172 return result;
173}

314bool alignSymbolPinToGrid(const IR::SymbolPinIR& pin, QPointF& position, double& length) {
315 constexpr double kGridMm = 2.54;
316 constexpr double kMaxExtensionMm = kGridMm / 2.0;
317 position = pin.position;
318 length = pin.length;
319
320 double snappedX = position.x();
321 double snappedY = position.y();
322 if (!snapToLibrePcbGrid(position.x(), snappedX) || !snapToLibrePcbGrid(position.y(), snappedY))
323 return false;
324
325 // 按引脚方向选择连接端所在的目标栅格边界。
326 switch (pin.direction) {
327 case IR::PinDirection::Right:
328 snappedX = qCeil(position.x() / kGridMm - 1e-9) * kGridMm;
329 if (snappedX - position.x() > kMaxExtensionMm + 1e-6)
330 return false;
331 length += snappedX - position.x();
332 break;
333 case IR::PinDirection::Left:
334 snappedX = qFloor(position.x() / kGridMm + 1e-9) * kGridMm;
335 if (position.x() - snappedX > kMaxExtensionMm + 1e-6)
336 return false;
337 length += position.x() - snappedX;
338 break;
339 case IR::PinDirection::Up:
340 snappedY = qCeil(position.y() / kGridMm - 1e-9) * kGridMm;
341 if (snappedY - position.y() > kMaxExtensionMm + 1e-6)
342 return false;
343 length += snappedY - position.y();
344 break;
345 case IR::PinDirection::Down:
346 snappedY = qFloor(position.y() / kGridMm + 1e-9) * kGridMm;
347 if (position.y() - snappedY > kMaxExtensionMm + 1e-6)
348 return false;
349 length += position.y() - snappedY;
350 break;
351 }
352 position = QPointF(snappedX, snappedY);
353 return true;
354}

551QList<QPointF> rectanglePoints(const QRectF& bounds, double rotation) {
552 const QPointF center = bounds.center();
553 const double radians = qDegreesToRadians(rotation);
554 const double c = std::cos(radians);
555 const double s = std::sin(radians);
556 QList<QPointF> result;
557 for (const QPointF point : {bounds.topLeft(), bounds.topRight(), bounds.bottomRight(), bounds.bottomLeft()}) {
558 const QPointF offset = point - center;
559 result.append(center + QPointF(offset.x() * c - offset.y() * s, offset.x() * s + offset.y() * c));
560 }
561 result.append(result.first());
562 return result;
563}

573QList<QPointF> sampleArc(const QPointF& center, double radius, double startAngle, double endAngle) {
574 constexpr int kSegments = 32;
575 double sweep = endAngle - startAngle;
576 if (qFuzzyIsNull(sweep))
577 sweep = 360.0;
578 const int segments = qMax(2, qCeil(qAbs(sweep) / 360.0 * kSegments));
579 QList<QPointF> points;
580 for (int i = 0; i <= segments; ++i) {
581 const double angle = qDegreesToRadians(startAngle + sweep * i / segments);
582 points.append(center + QPointF(radius * std::cos(angle), radius * std::sin(angle)));
583 }
584 return points;
585}

638QRectF footprintBounds(const IR::FootprintComponentIR& footprint, bool& valid) {
639 QRectF bounds;
640 valid = false; // 标记是否已纳入有效图元。
641 const auto include = [&bounds, &valid](const QPointF& point) {
642 if (!valid) {
643 bounds = QRectF(point, QSizeF());
644 valid = true;
645 } else {
646 const double left = qMin(bounds.left(), point.x());
647 const double top = qMin(bounds.top(), point.y());
648 const double right = qMax(bounds.right(), point.x());
649 const double bottom = qMax(bounds.bottom(), point.y());
650 bounds = QRectF(QPointF(left, top), QPointF(right, bottom));
651 }
652 };
653 for (const auto& pad : footprint.pads) {
654 include(pad.position + QPointF(-pad.size.width() / 2.0, -pad.size.height() / 2.0));
655 include(pad.position + QPointF(pad.size.width() / 2.0, pad.size.height() / 2.0));
656 }
657 for (const auto& rectangle : footprint.rectangles) {
658 for (const QPointF& point : rectanglePoints(rectangle.bounds, rectangle.rotation))
659 include(point);
660 }
661 for (const auto& circle : footprint.circles) {
662 include(circle.center - QPointF(circle.radius, circle.radius));
663 include(circle.center + QPointF(circle.radius, circle.radius));
664 }
665 for (const auto& outline : footprint.outlines) {
666 for (const QPointF& point : outline.points)
667 include(point);
668 }
669 for (const auto& track : footprint.tracks) {
670 for (const QPointF& point : track.points)
671 include(point);
672 }
673 for (const auto& arc : footprint.arcs) {
674 for (const QPointF& point : sampleArc(arc.center, arc.radius, arc.startAngle, arc.endAngle))
675 include(point);
676 }
677 for (const auto& region : footprint.regions) {
678 for (const QPointF& point : region.vertices)
679 include(point);
680 }
681 for (const auto& hole : footprint.holes) {
682 include(hole.center - QPointF(hole.radius, hole.radius));
683 include(hole.center + QPointF(hole.radius, hole.radius));
684 }
685 return bounds;

911bool writeSymbol(const IR::SymbolComponentIR& symbol, const QString& rootPath, QStringList& diagnostics) {
912 if (symbol.name.trimmed().isEmpty()) {
913 diagnostics.append(QStringLiteral("LibrePCB: 符号名称为空"));
914 return false;
915 }
916 SExpr root(QStringLiteral("librepcb_symbol"));
917 appendCommon(root,
918 uuidFor(QStringLiteral("symbol"), symbol.name),
919 sanitizeName(symbol.name),
920 symbol.description,
921 QStringLiteral("librepcb_symbol"));
922 root.list(QStringLiteral("category")).token(componentCategoryUuid()).close();
923 root.list(QStringLiteral("grid_interval")).token(QStringLiteral("0.01")).close();
924 if (symbol.partCount != 1) {
925 diagnostics.append(QStringLiteral("LibrePCB: 多部件符号当前未实现安全拆分：%1").arg(symbol.name));
926 return false;
927 }
928 if (!symbol.aliases.isEmpty())
929 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的别名未写入 LibrePCB 原生库").arg(symbol.name));
930 if (!symbol.graphicOrder.isEmpty())
931 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的原始图元顺序未保留").arg(symbol.name));
932 if (!symbol.models.isEmpty()) {
933 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 包含暂不支持的模型关联").arg(symbol.name));
934 return false;
935 }
936 QSet<QString> designators;
937 for (int i = 0; i < symbol.pins.size(); ++i) {
938 const auto& pin = symbol.pins.at(i);
939 const QString designator = pin.designator.trimmed();
940 if (designator.isEmpty()) {
941 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 存在空引脚编号").arg(symbol.name));
942 return false;
943 }
944 if (designators.contains(designator)) {
945 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 存在重复引脚编号：%2").arg(symbol.name, designator));
946 return false;
947 }
948 QPointF pinPosition;
949 double pinLength = 0.0;
950 if (!alignSymbolPinToGrid(pin, pinPosition, pinLength)) {
951 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的引脚 %2 不在 2.54 mm 栅格上（坐标 %3, %4）")
952 .arg(symbol.name, designator)
953 .arg(number(pin.position.x()), number(pin.position.y())));
954 return false;
955 }
956 if (!isOnLibrePcbGrid(pin.position.x()) || !isOnLibrePcbGrid(pin.position.y())) {
957 diagnostics.append(
958 QStringLiteral("LibrePCB: 符号 %1 的引脚 %2 已自动对齐到 2.54 mm 栅格").arg(symbol.name, designator));
959 }
960 designators.insert(designator);
961 SExpr& node = root.list(QStringLiteral("pin"));
962 node.token(uuidFor(QStringLiteral("symbol-pin"), symbol.name, i));
963 node.list(QStringLiteral("name")).atom(pin.name).close();
964 node.list(QStringLiteral("position")).token(number(pinPosition.x())).token(number(pinPosition.y())).close();

966 node.list(QStringLiteral("rotation")).token(number(rotation)).close();
967 node.list(QStringLiteral("length")).token(number(pinLength)).close();
968 node.list(QStringLiteral("name_position"))
969 .token(number(symbolPinNamePosition(pin, pinLength, rotation).x()))
970 .token(number(symbolPinNamePosition(pin, pinLength, rotation).y()))
971 .close();
972 node.list(QStringLiteral("name_rotation")).token(QStringLiteral("0")).close();
973 node.list(QStringLiteral("name_height"))
974 .token(number(pin.nameFontSizeMm > 0 ? pin.nameFontSizeMm : 1.5))
975 .close();
976 node.list(QStringLiteral("name_align")).token(QStringLiteral("left")).token(QStringLiteral("center")).close();
977 node.close();
978 }
979 int geometryIndex = 0;
980 for (const auto& rectangle : symbol.rectangles) {
981 appendPolygon(
982 root,
983 uuidFor(QStringLiteral("symbol-rectangle"), symbol.name, geometryIndex++),
984 QStringLiteral("sym_outlines"),

986 rectangle.strokeWidth,
987 rectangle.isFilled);
988 }
989 for (const auto& polygon : symbol.polygons) {
990 appendPolygon(root,
991 uuidFor(QStringLiteral("symbol-polygon"), symbol.name, geometryIndex++),
992 QStringLiteral("sym_outlines"),
993 polygon.points,
994 polygon.strokeWidth,
995 polygon.isFilled);
996 }
997 for (const auto& polyline : symbol.polylines) {
998 appendPolygon(root,
999 uuidFor(QStringLiteral("symbol-polyline"), symbol.name, geometryIndex++),
1000 QStringLiteral("sym_outlines"),
1001 polyline.points,
1002 polyline.strokeWidth,
1003 false);
1004 }
1005 for (const auto& arc : symbol.arcs) {
1006 appendPolygon(root,
1007 uuidFor(QStringLiteral("symbol-arc"), symbol.name, geometryIndex++),
1008 QStringLiteral("sym_outlines"),
1009 {arc.startPoint, arc.midPoint, arc.endPoint},
1010 arc.strokeWidth,
1011 arc.isFilled);
1012 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的三点圆弧已离散为折线").arg(symbol.name));
1013 }
1014 for (const auto& ellipse : symbol.ellipses) {
1015 appendPolygon(root,
1016 uuidFor(QStringLiteral("symbol-ellipse"), symbol.name, geometryIndex++),
1017 QStringLiteral("sym_outlines"),
1018 sampleEllipse(ellipse.center, ellipse.radiusX, ellipse.radiusY, 0, 360),
1019 ellipse.strokeWidth,
1020 ellipse.isFilled);
1021 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的椭圆已离散为多边形").arg(symbol.name));
1022 }
1023 for (const auto& pie : symbol.pies) {
1024 QList<QPointF> points = {pie.center};
1025 points.append(sampleArc(pie.center, pie.radius, pie.startAngle, pie.endAngle));
1026 appendPolygon(root,
1027 uuidFor(QStringLiteral("symbol-pie"), symbol.name, geometryIndex++),
1028 QStringLiteral("sym_outlines"),
1029 points,
1030 pie.strokeWidth,
1031 pie.isFilled);
1032 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的扇形已离散为多边形").arg(symbol.name));
1033 }
1034 for (const auto& arc : symbol.ellipticalArcs) {
1035 appendPolygon(root,
1036 uuidFor(QStringLiteral("symbol-elliptical-arc"), symbol.name, geometryIndex++),
1037 QStringLiteral("sym_outlines"),
1038 sampleEllipse(arc.center, arc.radiusX, arc.radiusY, arc.startAngle, arc.endAngle),
1039 arc.strokeWidth,
1040 arc.isFilled);
1041 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的椭圆弧已离散为多边形").arg(symbol.name));
1042 }
1043 for (const auto& path : symbol.paths) {
1044 QList<QPointF> points = path.points;
1045 if (points.isEmpty() && !path.segments.isEmpty()) {
1046 for (const auto& segment : path.segments) {
1047 if (segment.type == IR::SymbolPathSegmentIR::Type::CubicBezier)
1048 points.append(sampleBezier({segment.start, segment.control1, segment.control2, segment.end}));
1049 else
1050 points.append({segment.start, segment.end});
1051 }
1052 }
1053 if (points.size() < 2) {
1054 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的路径没有可表达的顶点").arg(symbol.name));
1055 return false;
1056 }
1057 appendPolygon(root,
1058 uuidFor(QStringLiteral("symbol-path"), symbol.name, geometryIndex++),
1059 QStringLiteral("sym_outlines"),
1060 points,
1061 path.strokeWidth,
1062 path.isFilled);
1063 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的路径曲线已离散为折线").arg(symbol.name));
1064 }
1065 for (const auto& bezier : symbol.beziers) {
1066 const QList<QPointF> points = sampleBezier(bezier.controlPoints);
1067 if (points.size() < 2) {
1068 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的 Bézier 曲线控制点不足").arg(symbol.name));
1069 return false;
1070 }
1071 appendPolygon(root,
1072 uuidFor(QStringLiteral("symbol-bezier"), symbol.name, geometryIndex++),
1073 QStringLiteral("sym_outlines"),
1074 points,
1075 bezier.strokeWidth,
1076 false);
1077 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的 Bézier 曲线已离散为折线").arg(symbol.name));
1078 }
1079 if (!symbol.ieeeSymbols.isEmpty() || !symbol.images.isEmpty() || !symbol.textFrames.isEmpty()) {
1080 diagnostics.append(
1081 QStringLiteral("LibrePCB: 符号 %1 包含当前适配器尚未支持的 IEEE、图片或文本框图元").arg(symbol.name));
1082 return false;
1083 }
1084 for (const auto& circle : symbol.circles) {
1085 SExpr& node = root.list(QStringLiteral("circle"));
1086 node.token(uuidFor(QStringLiteral("symbol-circle"), symbol.name, geometryIndex++));
1087 node.list(QStringLiteral("layer")).token(QStringLiteral("sym_outlines")).close();
1088 node.list(QStringLiteral("width")).token(number(circle.strokeWidth)).close();
1089 node.list(QStringLiteral("fill")).token(boolValue(circle.isFilled)).close();
1090 node.list(QStringLiteral("grab_area")).token(QStringLiteral("false")).close();
1091 node.list(QStringLiteral("diameter")).token(number(circle.radius * 2.0)).close();
1092 node.list(QStringLiteral("position")).token(number(circle.center.x())).token(number(circle.center.y())).close();
1093 node.close();
1094 }
1095 const QRectF bounds = symbolBounds(symbol);
1096 const QRectF visibleTextArea = bounds.adjusted(-25.4, -25.4, 25.4, 25.4);
1097 int relocatedTextIndex = 0;
1098 for (const auto& text : symbol.texts) {
1099 if (!text.visible)
1100 continue;
1101 QPointF textPosition = text.position;
1102 double textRotation = text.rotation;
1103 if (!visibleTextArea.contains(text.position)) {
1104 textPosition = QPointF(0.0, qMin(bounds.top(), bounds.bottom()) - 4.0 - relocatedTextIndex * 2.0);
1105 textRotation = 0.0;
1106 ++relocatedTextIndex;
1107 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的文本“%2”超出图形边界，已移动到图形下方")
1108 .arg(symbol.name, text.text));
1109 }
1110 SExpr& node = root.list(QStringLiteral("text"));
1111 node.token(uuidFor(QStringLiteral("symbol-text"), symbol.name, geometryIndex++));
1112 node.list(QStringLiteral("layer")).token(QStringLiteral("sym_values")).close();
1113 node.list(QStringLiteral("height")).token(number(text.fontSizeMm > 0 ? text.fontSizeMm : 1.5)).close();
1114 node.list(QStringLiteral("align")).token(QStringLiteral("center")).token(QStringLiteral("center")).close();
1115 node.list(QStringLiteral("position")).token(number(textPosition.x())).token(number(textPosition.y())).close();
1116 node.list(QStringLiteral("rotation")).token(number(textRotation)).close();
1117 node.list(QStringLiteral("lock")).token(QStringLiteral("false")).close();
1118 node.list(QStringLiteral("value")).atom(text.text).close();
1119 node.close();
1120 }
1121 for (const auto& parameter : symbol.parameters) {
1122 if (!parameter.visible) {
1123 diagnostics.append(
1124 QStringLiteral("LibrePCB: 符号 %1 的隐藏参数未写入原生库：%2").arg(symbol.name, parameter.name));
1125 continue;
1126 }
1127 QPointF parameterPosition = parameter.position;
1128 double parameterRotation = parameter.rotation;
1129 if (!visibleTextArea.contains(parameter.position)) {
1130 parameterPosition = QPointF(0.0, qMin(bounds.top(), bounds.bottom()) - 4.0 - relocatedTextIndex * 2.0);
1131 parameterRotation = 0.0;
1132 ++relocatedTextIndex;
1133 diagnostics.append(QStringLiteral("LibrePCB: 符号 %1 的参数“%2”超出图形边界，已移动到图形下方")
1134 .arg(symbol.name, parameter.name));
1135 }
1136 SExpr& node = root.list(QStringLiteral("text"));
1137 node.token(uuidFor(QStringLiteral("symbol-parameter"), symbol.name, geometryIndex++));
1138 node.list(QStringLiteral("layer")).token(QStringLiteral("sym_values")).close();
1139 node.list(QStringLiteral("height"))
1140 .token(number(parameter.fontSizeMm > 0 ? parameter.fontSizeMm : 1.5))
1141 .close();
1142 node.list(QStringLiteral("align")).token(QStringLiteral("center")).token(QStringLiteral("center")).close();
1143 node.list(QStringLiteral("position"))
1144 .token(number(parameterPosition.x()))
1145 .token(number(parameterPosition.y()))
1146 .close();
1147 node.list(QStringLiteral("rotation")).token(number(parameterRotation)).close();
1148 node.list(QStringLiteral("lock")).token(boolValue(parameter.readOnly)).close();
1149 node.list(QStringLiteral("value")).atom(parameter.value).close();
1150 node.close();
1151 }
1152 bool hasNameText = false;
1153 bool hasValueText = false;
1154 for (const auto& text : symbol.texts) {
1155 hasNameText = hasNameText || text.text == QStringLiteral("{{NAME}}");
1156 hasValueText = hasValueText || text.text == QStringLiteral("{{VALUE}}");
1157 }
1158 // LibrePCB 将器件名称和值作为符号必需字段；源数据缺失时仍生成可编辑的默认占位文本。
1159 // QRectF 的 top/bottom 语义容易与 EDA 坐标系的视觉上下方向混淆，显式取最小 Y，
1160 // 确保默认字段始终位于符号图形的视觉下方，而不是落到图形顶部。
1161 const double fieldBaseY = qMin(bounds.top(), bounds.bottom()) - 2.54;
1162 const auto appendDefaultText = [&root, &geometryIndex, &symbol](
1163 const QString& layer, const QString& value, double y) {
1164 SExpr& node = root.list(QStringLiteral("text"));
1165 node.token(uuidFor(QStringLiteral("symbol-default-text"), symbol.name, geometryIndex++));
1166 node.list(QStringLiteral("layer")).token(layer).close();
1167 node.list(QStringLiteral("height")).token(QStringLiteral("1.5")).close();
1168 node.list(QStringLiteral("align")).token(QStringLiteral("center")).token(QStringLiteral("center")).close();
1169 node.list(QStringLiteral("position")).token(QStringLiteral("0")).token(number(y)).close();
1170 node.list(QStringLiteral("rotation")).token(QStringLiteral("0")).close();
1171 node.list(QStringLiteral("lock")).token(QStringLiteral("false")).close();
1172 node.list(QStringLiteral("value")).atom(value).close();
1173 node.close();
1174 };
1175 if (!hasNameText)
1176 appendDefaultText(QStringLiteral("sym_names"), QStringLiteral("{{NAME}}"), fieldBaseY);
1177 if (!hasValueText)
1178 appendDefaultText(QStringLiteral("sym_values"), QStringLiteral("{{VALUE}}"), fieldBaseY - 2.0);
1179 root.close();
1180 return writeElement(rootPath,
1181 QStringLiteral("sym"),
1182 QStringLiteral("symbol"),
1183 uuidFor(QStringLiteral("symbol"), symbol.name),
1184 root.finish(),
1185 diagnostics);
1186}

1368 .list(QStringLiteral("angle"))
1369 .token(QStringLiteral("0"))

1371 .close();

1374 .list(QStringLiteral("position"))
1375 .token(number(pad.holeLength / 2.0))
1376 .token(QStringLiteral("0"))
1377 .close()
1378 .list(QStringLiteral("angle"))
1379 .token(QStringLiteral("0"))
1380 .close()
1381 .close();

1406 false);

1416 if (outline.isLocked)

1420 layer,
1421 localPoints(outline.points),
1422 outline.strokeWidth,
1423 false);

1427 if (layer.isEmpty()) {
1428 diagnostics.append(QStringLiteral("LibrePCB: 圆形图元使用无法映射的图层"));
1429 return false;

1431 if (isAssemblyLayer(circle.layer))
1432 diagnostics.append(QStringLiteral("LibrePCB: 装配层圆形已映射到 Documentation 层"));

1435 QStringLiteral("LibrePCB: 封装 %1 的用户定义圆形已降级到 Documentation 层").arg(footprint.name));

1437 node.token(uuidFor(QStringLiteral("footprint-circle"), footprint.name, index++));

1441 node.list(QStringLiteral("grab_area")).token(QStringLiteral("false")).close();
1442 node.list(QStringLiteral("diameter")).token(number(circle.radius * 2)).close();

1444 node.list(QStringLiteral("position"))
1445 .token(number(circlePosition.x()))
1446 .token(number(circlePosition.y()))
1447 .close();
1448 node.close();

1451 const auto& track = footprint.tracks.at(trackIndex);

1460 localPoints(track.points),
1461 track.width,
1462 false);

1469 const auto& arc = footprint.arcs.at(arcIndex);
1470 const QString layer = layerForFootprint(arc.layer);
1471 if (layer.isEmpty() || arc.radius <= 0) {
1472 diagnostics.append(QStringLiteral("LibrePCB: 封装圆弧无法映射：%1").arg(footprint.name));
1473 return false;

1476 uuidFor(QStringLiteral("footprint-arc"), footprint.name, arcIndex),
1477 layer,
1478 localPoints(sampleArc(arc.center, arc.radius, arc.startAngle, arc.endAngle)),
1479 arc.width,
1480 false);

1483 diagnostics.append(QStringLiteral("LibrePCB: 封装圆弧网络名称不会写入库图元：%1").arg(arc.netName));

1490 if (layer.isEmpty() || region.vertices.size() < 3) {

1500 localPoints(region.vertices),
1501 0,

1504 diagnostics.append(QStringLiteral("LibrePCB: 封装区域锁定属性不会写入库定义：%1").arg(footprint.name));

1550 }
1551 for (const auto& rectangle : footprint.rectangles) {
1552 hasPackageOutline = hasPackageOutline || rectangle.layer == IR::LayerType::EdgeCuts;
1553 }
1554 for (const auto& outline : footprint.outlines) {
1555 hasPackageOutline = hasPackageOutline || outline.layer == IR::LayerType::EdgeCuts;

1562 QStringLiteral("top_package_outlines"),
1563 localPoints(rectanglePoints(bounds, 0)),
1564 0.1,
1565 false);

1567 .arg(footprint.name));
1568 }

1570 appendPolygon(fp,
1571 uuidFor(QStringLiteral("fallback-courtyard"), footprint.name),
1572 QStringLiteral("top_courtyard"),
1573 localPoints(rectanglePoints(bounds.adjusted(-0.25, -0.25, 0.25, 0.25), 0)),
1574 0.05,
1575 false);
1576 diagnostics.append(
1577 QStringLiteral("LibrePCB: 封装 %1 缺少 courtyard，已根据 IR 图元生成回退").arg(footprint.name));
1578 }
1579 if (boundsValid && !hasNameText)
1580 appendStrokeText(fp,
1581 uuidFor(QStringLiteral("fallback-name"), footprint.name),
1582 QStringLiteral("top_names"),
1583 QStringLiteral("{{NAME}}"),
1584 localPoint(QPointF(bounds.center().x(), bounds.bottom() + 1.0)));
1585 if (boundsValid && !hasValueText)
1586 appendStrokeText(fp,
1587 uuidFor(QStringLiteral("fallback-value"), footprint.name),
1588 QStringLiteral("top_values"),
1589 QStringLiteral("{{VALUE}}"),
1590 localPoint(QPointF(bounds.center().x(), bounds.top() - 1.0)));
1591 fp.close();
1592 root.close();
1593 if (!writeElement(
1594 rootPath, QStringLiteral("pkg"), QStringLiteral("package"), packageUuid, root.finish(), diagnostics))

1598 if (!writeFile(QDir(packageDirectory).filePath(model.fileName), model.data, diagnostics))
1599 return false;
1600 }

1611bool writeComponentAndDevice(const IR::ComponentIR& component, const QString& rootPath, QStringList& diagnostics) {
1612 if (!component.hasSymbol() || !component.hasFootprint()) {
1613 diagnostics.append(QStringLiteral("LibrePCB: 组件 %1 缺少符号或封装，无法建立完整关联").arg(component.name));
1614 return false;
1615 }
1616 if (!component.datasheet.isEmpty())
1617 diagnostics.append(QStringLiteral("LibrePCB: 组件数据手册地址未写入器件属性：%1").arg(component.datasheet));

1619 const QString packageUuid = uuidFor(QStringLiteral("package"), component.footprint.name);
1620 const QString componentUuid = uuidFor(QStringLiteral("component"), component.name);
1621 const QString deviceUuid = uuidFor(QStringLiteral("device"), component.name);
1622 SExpr cmp(QStringLiteral("librepcb_component"));
1623 appendCommon(
1624 cmp, componentUuid, sanitizeName(component.name), component.description, QStringLiteral("librepcb_component"));
1625 cmp.list(QStringLiteral("category")).token(componentCategoryUuid()).close();
1626 cmp.list(QStringLiteral("schematic_only")).token(QStringLiteral("false")).close();
1627 cmp.list(QStringLiteral("default_value")).atom(component.name).close();
1628 cmp.list(QStringLiteral("prefix"))
1629 .atom(component.prefix.isEmpty() ? QStringLiteral("U") : component.prefix)
1630 .close();
1631 int signalIndex = 0;
1632 QHash<QString, QString> signalUuids;
1633 for (const auto& pin : component.symbol.pins) {
1634 const QString designator = pin.designator.trimmed();

1636 diagnostics.append(
1637 QStringLiteral("LibrePCB: 组件 %1 存在空引脚编号，无法建立信号映射").arg(component.name));
1638 return false;

1641 diagnostics.append(QStringLiteral("LibrePCB: 组件 %1 存在重复引脚编号，无法建立唯一信号映射：%2")
1642 .arg(component.name, designator));
1643 return false;
1644 }
1645 const QString signalUuid = uuidFor(QStringLiteral("signal"), component.name, signalIndex);
1646 signalUuids.insert(designator, signalUuid);
1647 SExpr& signal = cmp.list(QStringLiteral("signal"));
1648 signal.token(signalUuid)
1649 .list(QStringLiteral("name"))
1650 .atom(pin.name.isEmpty() ? pin.designator : pin.name)
1651 .close();
1652 signal.list(QStringLiteral("role")).token(QStringLiteral("passive")).close();
1653 signal.list(QStringLiteral("required")).token(QStringLiteral("false")).close();
1654 signal.list(QStringLiteral("negated")).token(QStringLiteral("false")).close();
1655 signal.list(QStringLiteral("clock")).token(QStringLiteral("false")).close();
1656 signal.list(QStringLiteral("forced_net")).atom(QString()).close();
1657 signal.close();
1658 ++signalIndex;
1659 }
1660 const QString variantUuid = uuidFor(QStringLiteral("variant"), component.name);
1661 const QString itemUuid = uuidFor(QStringLiteral("variant-item"), component.name);
1662 SExpr& variant = cmp.list(QStringLiteral("variant"));
1663 variant.token(variantUuid).list(QStringLiteral("norm")).atom(QString()).close();

1665 appendDescription(variant, component.description);
1666 SExpr& item = variant.list(QStringLiteral("gate"));
1667 item.token(itemUuid).list(QStringLiteral("symbol")).token(symbolUuid).close();
1668 item.list(QStringLiteral("position")).token(QStringLiteral("0")).token(QStringLiteral("0")).close();
1669 item.list(QStringLiteral("rotation")).token(QStringLiteral("0")).close();
1670 item.list(QStringLiteral("required")).token(QStringLiteral("true")).close();
1671 item.list(QStringLiteral("suffix")).atom(QString()).close();

1673 const auto& pin = component.symbol.pins.at(i);
1674 const QString signalUuid = signalUuids.value(pin.designator);
1675 SExpr& map = item.list(QStringLiteral("pin"));
1676 map.token(uuidFor(QStringLiteral("symbol-pin"), component.symbol.name, i));
1677 map.list(QStringLiteral("signal")).token(signalUuid).close();
1678 map.list(QStringLiteral("text")).token(QStringLiteral("signal")).close();
1679 map.close();
1680 }
1681 item.close();
1682 variant.close();
1683 cmp.close();
1684 if (!writeElement(
1685 rootPath, QStringLiteral("cmp"), QStringLiteral("component"), componentUuid, cmp.finish(), diagnostics))
1686 return false;
1687
1688 SExpr dev(QStringLiteral("librepcb_device"));
1689 appendCommon(
1690 dev, deviceUuid, sanitizeName(component.name), component.description, QStringLiteral("librepcb_device"));
1691 dev.list(QStringLiteral("category")).token(componentCategoryUuid()).close();
1692 dev.list(QStringLiteral("component")).token(componentUuid).close();
1693 dev.list(QStringLiteral("package")).token(packageUuid).close();
1694 if (!component.manufacturerPart.trimmed().isEmpty()) {
1695 dev.list(QStringLiteral("part"))
1696 .atom(component.manufacturerPart)
1697 .list(QStringLiteral("manufacturer"))
1698 .atom(component.manufacturer)
1699 .close()
1700 .close();
1701 }
1702 for (int i = 0; i < component.footprint.pads.size(); ++i) {
1703 const auto& pad = component.footprint.pads.at(i);
1704 SExpr& map = dev.list(QStringLiteral("pad"));
1705 map.token(uuidFor(QStringLiteral("package-pad"), component.footprint.name, i));
1706 map.list(QStringLiteral("optional")).token(QStringLiteral("false")).close();
1707 const QString padNumber = pad.number.trimmed();
1708 if (signalUuids.contains(padNumber))
1709 map.list(QStringLiteral("signal")).token(signalUuids.value(padNumber)).close();
1710 else
1711 map.list(QStringLiteral("signal")).token(QStringLiteral("none")).close();
1712 map.close();
1713 }
1714 dev.close();
1715 return writeElement(
1716 rootPath, QStringLiteral("dev"), QStringLiteral("device"), deviceUuid, dev.finish(), diagnostics);
1717}
