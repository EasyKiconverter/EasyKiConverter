127void ComponentDataCache::flushToDisk(const QString& componentId) {
128 QMutexLocker locker(&m_cacheMutex);
129
130 if (m_cacheDir.isEmpty()) {
131 qWarning() << "ComponentDataCache: Cache dir not set";
132 return;
133 }
134
135 QStringList idsToFlush;
136 if (componentId.isEmpty()) {
137 idsToFlush = m_cache.keys();
138 } else if (m_cache.contains(componentId)) {
139 idsToFlush.append(componentId);
140 } else {
141 return;
142 }
143
144 for (const QString& id : idsToFlush) {
145 const auto& data = m_cache.value(id);
146 if (!data)
147 continue;
148
149 // 构建元器件数据文件路径
150 QString dataFilePath =
151 m_cacheDir + QStringLiteral("/") + id + QStringLiteral(".json");
152
153 QJsonObject json;
154 json[QStringLiteral("componentId")] = id;
155
156 // 序列化符号数据
157 if (data->symbolData()) {
158 QJsonObject symbolJson = SymbolDataSerializer::toJson(*data->symbolData());
159 json[QStringLiteral("symbolData")] = symbolJson;
160 }
161
162 // 序列化封装数据
163 if (data->footprintData()) {
164 QJsonObject footprintJson =
165 FootprintDataSerializer::toJson(*data->footprintData());
166 json[QStringLiteral("footprintData")] = footprintJson;
167 }
168
169 // 写入JSON文件
170 QFile file(dataFilePath);
171 if (file.open(QIODevice::WriteOnly)) {
172 file.write(QJsonDocument(json).toJson());
173 file.close();
174 } else {
175 qWarning() << "ComponentDataCache: Failed to write cache file" << dataFilePath;
176 }
177 }
178}

185bool ComponentDataCache::loadFromDisk(const QString& componentId) {
186 if (m_cacheDir.isEmpty()) {
187 return false;
188 }
189
190 QString dataFilePath =
191 m_cacheDir + QStringLiteral("/") + componentId + QStringLiteral(".json");
192 QFile file(dataFilePath);
193
194 if (!file.exists()) {
195 return false;
196 }
197
198 if (!file.open(QIODevice::ReadOnly)) {
199 return false;
200 }
201
202 QByteArray jsonData = file.readAll();
203 file.close();
204
205 QJsonParseError error;
206 QJsonDocument doc = QJsonDocument::fromJson(jsonData, &error);
207 if (error.error != QJsonParseError::NoError) {
208 qWarning() << "ComponentDataCache: Failed to parse cache file" << error.errorString();
209 return false;
210 }
211
212 // 反序列化为ComponentData
213 QJsonObject json = doc.object();
214 auto data = QSharedPointer<ComponentData>::create();
215
216 if (json.contains(QStringLiteral("symbolData"))) {
217 QJsonObject symbolJson = json[QStringLiteral("symbolData")].toObject();
218 SymbolData symbolData;
219 if (SymbolDataSerializer::fromJson(symbolData, symbolJson)) {
220 data->setSymbolData(QSharedPointer<SymbolData>::create(symbolData));
221 }
222 }
223
224 if (json.contains(QStringLiteral("footprintData"))) {
225 QJsonObject footprintJson = json[QStringLiteral("footprintData")].toObject();
226 FootprintData footprintData;
227 if (FootprintDataSerializer::fromJson(footprintData, footprintJson)) {
228 data->setFootprintData(QSharedPointer<FootprintData>::create(footprintData));
229 }
230 }
231
232 put(componentId, data);
233 return true;
234}
