124void Logger::log(LogLevel level,
125 LogModule module,
126 const QString& message,
127 const char* file,
128 const char* function,
129 int line) {
130 // 快速检查是否应该记录（无锁快速路径）
131 if (!shouldLog(level, module)) {
132 return;
133 }
134
135 // 构建日志记录
136 LogRecord record;
137 record.level = level;
138 record.module = module;
139 record.message = message;
140 record.timestamp = QDateTime::currentMSecsSinceEpoch();
141 record.threadId = QThread::currentThreadId();
142
143 if (file) {
144 record.fileName = QString::fromUtf8(file);
145 }
146 if (function) {
147 record.functionName = QString::fromUtf8(function);
148 }
149 record.line = line;
150
151 // 获取线程名称
152 if (s_threadNames.hasLocalData()) {
153 record.threadName = s_threadNames.localData();
154 }
155
156 // 发射信号
157 emit logRecord(record);
158
159 // 分发到各 Appender
160 QMutexLocker locker(&m_appendersMutex);
161 for (auto& appender : m_appenders) {
162 QString formatted;
163 if (appender->formatter()) {
164 formatted = appender->formatter()->format(record);
165 } else {
166 formatted = record.message;
167 }
168 appender->append(record, formatted);
169 }
170}

177bool Logger::shouldLog(LogLevel level, LogModule module) const {
178 // 快速路径：先检查全局级别（使用原子变量，无锁）
179 int globalLvl = m_globalLevel.loadRelaxed();
180 if (static_cast<int>(level) < globalLvl) {
181 // 如果全局级别已经过滤掉了，直接返回
182 // 但还需要检查模块级别（可能模块级别更低）
183
184 // 检查模块级别（需要读锁）
185 QReadLocker locker(&m_moduleLevelsLock);
186 if (m_moduleLevels.isEmpty()) {
187 return false; // 没有模块级别设置，全局过滤生效
188 }
189
190 auto it = m_moduleLevels.find(module);
191 if (it == m_moduleLevels.end()) {
192 return false; // 该模块没有特定设置，全局过滤生效
193 }
194
195 // 模块有特定设置，使用模块级别
196 return static_cast<int>(level) >= static_cast<int>(it.value());
197 }
198
199 // 全局级别检查通过
200 // 但还需要检查模块是否有更严格的级别设置
201 QReadLocker locker(&m_moduleLevelsLock);
202 auto it = m_moduleLevels.find(module);
203 if (it != m_moduleLevels.end()) {
204 return static_cast<int>(level) >= static_cast<int>(it.value());
205 }
206
207 return true; // 模块没有特定设置，使用全局结果
208}

263void LogUtils::replaceFirstPlaceholder(QString& result, const QString& replacement) {
264 // 安全检查：避免对包含格式化字符的输入进行处理
265 // 仅替换第一个 {} 占位符，不处理其他格式化字符
266 int pos = result.indexOf(QStringLiteral("{}"));
267 if (pos >= 0) {
268 // 对替换内容进行安全处理，转义可能的格式化字符
269 QString safeReplacement = replacement;
270 // 不对内容进行额外的格式化处理，直接替换
271 result.replace(pos, 2, safeReplacement);
272 }
273}
