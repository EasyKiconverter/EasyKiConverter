56qsizetype findEndOfCentralDirectory(const QByteArray& data) {
57 if (data.size() < kEndOfCentralDirectoryMinSize)
58 return -1;
59 const qsizetype begin = qMax<qsizetype>(0, data.size() - kMaxEndOfCentralDirectorySearch);
60 for (qsizetype offset = data.size() - kEndOfCentralDirectoryMinSize; offset >= begin; --offset) {
61 if (read32(data, offset) == kEndOfCentralDirectory)
62 return offset;
63 }
64 return -1;
65}

72bool isUnsafePath(const QString& path) {
73 QString normalized = path;
74 normalized.replace(QChar('\\'), QChar('/'));
75 if (normalized.isEmpty() || normalized.contains(QChar('\0')) || normalized.startsWith(QChar('/')))
76 return true;
77 if (normalized.size() >= 2 && normalized.at(1) == QChar(':'))
78 return true;
79 const QStringList parts = normalized.split(QChar('/'), Qt::KeepEmptyParts);
80 for (const QString& part : parts) {
81 if (part == QStringLiteral(".."))
82 return true;
83 }
84 return false;
85}

105ArchiveInspectionResult ArchiveInspector::inspect(const QByteArray& data,
106 const ArchiveLimits& limits,
107 const QString& filePath) {
108 ArchiveInspectionResult result;
109 result.diagnostics.setFilePath(filePath);
110 const qsizetype eocd = findEndOfCentralDirectory(data);
111 if (eocd < 0 || !hasBytes(data, eocd, kEndOfCentralDirectoryMinSize)) {
112 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 缺少中央目录结束标记"));
113 return result;
114 }
115 const quint16 disk = read16(data, eocd + 4);
116 const quint16 centralDirectoryDisk = read16(data, eocd + 6);
117 const quint16 entryCount = read16(data, eocd + 10);
118 const quint32 centralDirectorySize = read32(data, eocd + 12);
119 const quint32 centralDirectoryOffset = read32(data, eocd + 16);
120 if (disk != 0 || centralDirectoryDisk != 0 || entryCount == 0xFFFF || centralDirectorySize == 0xFFFFFFFF ||
121 centralDirectoryOffset == 0xFFFFFFFF) {
122 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("不支持多磁盘 ZIP 归档"));
123 return result;
124 }
125 if (entryCount > limits.maxEntries) {
126 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目数量超过安全限制"));
127 return result;
128 }
129 if (centralDirectoryOffset > static_cast<quint32>(data.size()) ||
130 centralDirectorySize > static_cast<quint32>(data.size() - centralDirectoryOffset) ||
131 !hasBytes(data, centralDirectoryOffset, centralDirectorySize)) {
132 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 中央目录范围无效"));
133 return result;
134 }
135 QSet<QString> paths;
136 qsizetype cursor = static_cast<qsizetype>(centralDirectoryOffset);
137 const qsizetype centralDirectoryEnd =
138 cursor + static_cast<qsizetype>(centralDirectorySize);
139 for (quint16 index = 0; index < entryCount; ++index) {
140 if (cursor > centralDirectoryEnd || centralDirectoryEnd - cursor < 46 ||
141 read32(data, cursor) != kCentralDirectoryEntry) {
142 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 中央目录条目损坏"));
143 return result;
144 }
145 const quint16 flags = read16(data, cursor + 8);
146 const quint16 method = read16(data, cursor + 10);
147 const quint32 compressedSize = read32(data, cursor + 20);
148 const quint32 uncompressedSize = read32(data, cursor + 24);
149 const quint16 nameLength = read16(data, cursor + 28);
150 const quint16 extraLength = read16(data, cursor + 30);
151 const quint16 commentLength = read16(data, cursor + 32);
152 const quint32 externalAttributes = read32(data, cursor + 38);
153 const quint32 localHeaderOffset = read32(data, cursor + 42);
154 const qsizetype recordSize = 46 + nameLength + extraLength + commentLength;
155 if (recordSize > centralDirectoryEnd - cursor) {
156 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目长度超出中央目录"));
157 return result;
158 }
159 const QString path = QString::fromUtf8(data.constData() + cursor + 46, nameLength);
160 if (localHeaderOffset > static_cast<quint32>(data.size()) ||
161 !hasBytes(data, static_cast<qsizetype>(localHeaderOffset), 30) ||
162 read32(data, static_cast<qsizetype>(localHeaderOffset)) != kLocalFileEntry) {
163 result.diagnostics.add(
164 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目本地文件头无效"), path);
165 return result;
166 }
167 const qsizetype localOffset = static_cast<qsizetype>(localHeaderOffset);
168 const quint16 localNameLength = read16(data, localOffset + 26);
169 const quint16 localExtraLength = read16(data, localOffset + 28);
170 const qsizetype localRecordSize = 30 + localNameLength + localExtraLength;
171 if (!hasBytes(data, localOffset, localRecordSize)) {
172 result.diagnostics.add(
173 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目本地文件头范围无效"), path);
174 return result;
175 }
176 const QString localPath =
177 QString::fromUtf8(data.constData() + localOffset + 30, localNameLength);
178 if (localPath != path) {
179 result.diagnostics.add(
180 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目名称与本地文件头不一致"), path);
181 return result;
182 }
183 const qsizetype dataOffset = localOffset + localRecordSize;
184 if (!hasBytes(data, dataOffset, compressedSize)) {
185 result.diagnostics.add(
186 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目压缩数据范围无效"), path);
187 return result;
188 }
189 const bool directory = path.endsWith(QChar('/'));
190 if (isUnsafePath(path) || isUnixSymlink(externalAttributes)) {
191 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目路径不安全"), path);
192 return result;
193 }
194 if (paths.contains(path)) {
195 result.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目路径重复"), path);
196 return result;
197 }
198 paths.insert(path);
199 if ((flags & kEncryptedFlag) != 0 || (method != 0 && method != kDeflate)) {
200 result.diagnostics.add(
201 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目压缩或加密方式不受支持"), path);
202 return result;
203 }
204 if (static_cast<qint64>(uncompressedSize) > limits.maxEntryUncompressedBytes) {
205 result.diagnostics.add(
206 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 单个条目展开后超过大小限制"), path);
207 return result;
208 }
209 if (result.totalUncompressedBytes > limits.maxTotalUncompressedBytes - uncompressedSize) {
210 result.diagnostics.add(
211 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 总展开大小超过限制"), path);
212 return result;
213 }
214 if (compressedSize == 0 && uncompressedSize > 0) {
215 result.diagnostics.add(
216 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 条目压缩大小无效"), path);
217 return result;
218 }
219 if (compressedSize > 0 &&
220 static_cast<double>(uncompressedSize) / static_cast<double>(compressedSize) > limits.maxCompressionRatio) {
221 result.diagnostics.add(
222 ParseSeverity::Error, ParseScope::File, QStringLiteral("ZIP 压缩比超过安全限制"), path);
223 return result;
224 }
225 result.totalUncompressedBytes += uncompressedSize;
226 result.entries.append({path, compressedSize, uncompressedSize, method, directory});
227 cursor += recordSize;
228 }
229 result.safe = true;
230 return result;
231}
