50ExportCompletionTotals ExportProgressAggregator::countCompletedComponents(
51 const QStringList& componentIds,
52 const QMap<QString, ExportTypeProgress>& typeProgress) {
53 ExportCompletionTotals totals;
54 for (const QString& componentId : componentIds) {
55 bool anyFailed = false;
56 bool allDone = true;
57
58 for (auto it = typeProgress.cbegin(); it != typeProgress.cend(); ++it) {
59 const ExportItemStatus itemStatus =
60 it.value().itemStatus.value(componentId, ExportItemStatus{ExportItemStatus::Status::Pending});
61 if (itemStatus.status == ExportItemStatus::Status::Failed) {
62 anyFailed = true;
63 }
64 if (itemStatus.status != ExportItemStatus::Status::Success &&
65 itemStatus.status != ExportItemStatus::Status::Failed &&
66 itemStatus.status != ExportItemStatus::Status::Skipped) {
67 allDone = false;
68 }
69 }
70
71 if (!allDone) {
72 continue;
73 }
74 if (anyFailed) {
75 totals.failedCount++;
76 } else {
77 totals.successCount++;
78 }
79 }
80 return totals;
81}
