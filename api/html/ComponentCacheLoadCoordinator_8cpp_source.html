25void ComponentCacheLoadCoordinator::load(ComponentService& owner,
26 const QString& normalizedId,
27 bool fetch3DModel,
28 ComponentCacheService* cache) {
29 // 从请求状态读取创建时的缓存代次，防止取消后旧任务污染新请求。
30 uint64_t generation = 0; // 本次请求捕获的缓存代次。
31 {
32 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
33 const auto it = owner.m_fetchingComponents.find(normalizedId); // 当前请求状态。
34 if (it != owner.m_fetchingComponents.end()) {
35 generation = it->cacheGeneration;
36 }
37 }
38 if (generation == 0) {
39 qWarning() << "ComponentCacheLoadCoordinator: No fetching state for" << normalizedId << ", aborting";
40 return;
41 }
42
43 // 缓存加载属于 I/O 密集型工作，放到后台线程避免阻塞界面线程。
44 const QFuture<ComponentCacheLoadResult> future =
45 QtConcurrent::run([normalizedId, fetch3DModel, cache]() { // 后台加载任务。
46 return ComponentCacheLoadWorker::load(normalizedId, fetch3DModel, cache);
47 });
48 auto* watcher = new QFutureWatcher<ComponentCacheLoadResult>(&owner); // 缓存加载完成观察器。
49 QObject::connect(
50 watcher,
51 &QFutureWatcher<ComponentCacheLoadResult>::finished,
52 &owner,
53 [&owner, watcher, normalizedId, fetch3DModel, generation]() {
54 ComponentCacheLoadResult result = watcher->result(); // 后台缓存加载结果。
55 watcher->deleteLater();
56
57 // 取消和重试会替换代次，旧缓存结果只能被丢弃。
58 {
59 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
60 const auto it = owner.m_fetchingComponents.find(normalizedId); // 当前请求状态。
61 if (it == owner.m_fetchingComponents.end() || it->cacheGeneration != generation) {
62 qDebug() << "ComponentService: Discarding stale cache result for" << normalizedId;
63 return;
64 }
65 }
66
67 if (!result.success || !result.cachedData) {
68 qWarning() << "ComponentService: Failed to load cache for" << normalizedId
69 << ", falling back to network fetch";
70 uint64_t retryGeneration = 0; // 网络重试使用的新缓存代次。
71 {
72 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
73 const auto it = owner.m_fetchingComponents.find(normalizedId); // 当前请求状态。
74 if (it == owner.m_fetchingComponents.end() || it->cacheGeneration != generation) {
75 qDebug() << "ComponentService: Discarding stale cache retry for" << normalizedId;
76 return;
77 }
78 ComponentService::FetchingComponent& fetchingComponent = *it; // 当前元器件请求状态。
79 owner.initializeFetchingComponent(fetchingComponent, normalizedId, fetch3DModel);
80 retryGeneration = fetchingComponent.cacheGeneration;
81 }
82 if (retryGeneration == 0) {
83 qWarning() << "ComponentCacheLoadCoordinator: No retry generation for" << normalizedId;
84 return;
85 }
86
87 const auto retryFuture = // CAD 网络重试任务。
88 QtConcurrent::run([normalizedId]() { return CadDataLoader::fetchAndParseCadData(normalizedId); });
89 auto* retryWatcher = new QFutureWatcher<CadFetchTaskResult>(&owner); // 网络重试观察器。
90 QObject::connect(retryWatcher,
91 &QFutureWatcher<CadFetchTaskResult>::finished,
92 &owner,
93 [&owner, retryWatcher, retryGeneration]() {
94 const CadFetchTaskResult retryResult = retryWatcher->result(); // 网络重试结果。
95 retryWatcher->deleteLater();
96 owner.handleCadFetchResult(retryResult, retryGeneration);
97 });
98 retryWatcher->setFuture(retryFuture);
99 return;
100 }
101
102 // 将后台解析得到的符号、封装和三维模型关联回缓存数据。
103 if (result.symbolData) {
104 result.cachedData->setSymbolData(result.symbolData);
105 }
106 if (result.footprintData) {
107 result.cachedData->setFootprintData(result.footprintData);
108 ComponentCacheLoadWorker::restoreModel3DFromFootprint(*result.cachedData, result.footprintData);
109 }
110
111 {
112 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
113 const auto it = owner.m_fetchingComponents.find(normalizedId); // 当前请求状态。
114 if (it == owner.m_fetchingComponents.end() || it->cacheGeneration != generation) {
115 qDebug() << "ComponentService: Discarding stale cached component for" << normalizedId;
116 return;
117 }
118 ComponentService::FetchingComponent& fetchingComponent = *it; // 当前元器件请求状态。
119 fetchingComponent.componentId = normalizedId;
120 fetchingComponent.data = *result.cachedData;
121 fetchingComponent.fetch3DModel = fetch3DModel;
122 fetchingComponent.hasCadData =
123 result.cachedData->symbolData() != nullptr && result.cachedData->footprintData() != nullptr;
124 fetchingComponent.requestActive = false;
125 }
126
127 qDebug() << "ComponentService: Emitting cadDataReady for" << normalizedId
128 << "with symbolData:" << (result.cachedData->symbolData() != nullptr)
129 << "footprintData:" << (result.cachedData->footprintData() != nullptr);
130 owner.updateComponentCache(normalizedId, *result.cachedData);
131 emit owner.cadDataReady(normalizedId, *result.cachedData);
132
133 // 并行获取场景需要同步释放队列中的当前请求槽位。
134 if (owner.m_parallelContext != nullptr) {
135 owner.handleParallelDataCollected(normalizedId, *result.cachedData);
136 }
137
138 if (!result.encodedPreviewImages.isEmpty()) {
139 const QStringList encodedImages = result.encodedPreviewImages; // 待派发的 Base64 预览图。
140 QTimer::singleShot(0, &owner, [&owner, normalizedId, encodedImages]() {
141 emit owner.previewImagesReady(normalizedId, encodedImages);
142 });
143 }
144 if (!result.datasheetData.isEmpty()) {
145 emit owner.datasheetReady(normalizedId, result.datasheetData);
146 }
147 qDebug() << "ComponentService: Cache loaded successfully for" << normalizedId;
148 });
149 watcher->setFuture(future);
150}
