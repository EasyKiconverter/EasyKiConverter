22QString safeName(const QString& value) {
23 QString result; // 清洗后的名称。
24 for (const QChar character : value) { // 逐字符检查原始名称。
25 const ushort code = character.unicode(); // 当前字符的 Unicode 编码。
26 if ((code >= 'A' && code <= 'Z') || (code >= 'a' && code <= 'z') || (code >= '0' && code <= '9') ||
27 character == QChar('_') || character == QChar('-'))
28 result.append(character);
29 else
30 result.append(QChar('_'));
31 }
32 return result.left(64);
33}

98bool writeArc(QXmlStreamWriter& xml, const IR::FootprintArcIR& arc, QStringList& diagnostics) {
99 if (!std::isfinite(arc.center.x()) || !std::isfinite(arc.center.y()) || !std::isfinite(arc.radius) ||
100 !std::isfinite(arc.startAngle) || !std::isfinite(arc.endAngle) || !std::isfinite(arc.width) ||
101 arc.radius <= 0.0 || arc.width < 0.0) {
102 diagnostics.append(QStringLiteral("Eagle: 圆弧包含非法圆心、半径、角度或线宽"));
103 return false;
104 }
105 const auto layer = layerNumber(arc.layer); // Eagle 目标图层编号。
106 if (!layer.has_value()) {
107 diagnostics.append(QStringLiteral("Eagle: 圆弧使用无法映射的图层"));
108 return false;
109 }
110
111 double sweep = arc.endAngle - arc.startAngle; // 圆弧扫过角度。
112 while (sweep <= -360.0)
113 sweep += 360.0;
114 while (sweep > 360.0)
115 sweep -= 360.0;
116 if (qFuzzyIsNull(sweep)) {
117 diagnostics.append(QStringLiteral("Eagle: 圆弧起止角度相同，无法区分零弧和整圆"));
118 return false;
119 }
120
121 const double startRadians = qDegreesToRadians(arc.startAngle); // 起始角度（弧度）。
122 const double endRadians = qDegreesToRadians(arc.startAngle + sweep); // 终止角度（弧度）。
123 const QPointF start = // 圆弧起点。
124 arc.center + QPointF(arc.radius * std::cos(startRadians), arc.radius * std::sin(startRadians));
125 const QPointF end = // 圆弧终点。
126 arc.center + QPointF(arc.radius * std::cos(endRadians), arc.radius * std::sin(endRadians));
127
128 xml.writeEmptyElement(QStringLiteral("wire"));
129 xml.writeAttribute(QStringLiteral("x1"), number(start.x()));
130 xml.writeAttribute(QStringLiteral("y1"), number(start.y()));
131 xml.writeAttribute(QStringLiteral("x2"), number(end.x()));
132 xml.writeAttribute(QStringLiteral("y2"), number(end.y()));
133 xml.writeAttribute(QStringLiteral("width"), number(arc.width));
134 xml.writeAttribute(QStringLiteral("layer"), QString::number(*layer));
135 xml.writeAttribute(QStringLiteral("curve"), number(sweep));
136 return true;
137}

163 diagnostics.append(QStringLiteral("Eagle: 焊盘 %1 的异形或槽孔语义当前无法无损表达").arg(pad.number));

224 const QPointF offset = point - center; // 顶点相对旋转中心的偏移。

402bool writeSymbolArc(QXmlStreamWriter& xml, const IR::SymbolArcIR& arc, QStringList& diagnostics) {
403 if (arc.isFilled || arc.strokeStyle != IR::StrokeStyle::Solid) {
404 diagnostics.append(QStringLiteral("Eagle: 符号圆弧的填充或非实线样式无法由 XML wire 无损表达"));
405 return false;
406 }
407 const QPointF& start = arc.startPoint; // 圆弧起点。
408 const QPointF& middle = arc.midPoint; // 圆弧中间点。
409 const QPointF& end = arc.endPoint; // 圆弧终点。
410 // 使用三点坐标计算圆拟合的行列式。
411 const double determinant = 2.0 * (start.x() * (middle.y() - end.y()) + middle.x() * (end.y() - start.y()) +
412 end.x() * (start.y() - middle.y())); // 三点拟合圆的行列式。
413 if (!std::isfinite(determinant) || qFuzzyIsNull(determinant) || !std::isfinite(arc.strokeWidth) ||
414 arc.strokeWidth < 0.0) {
415 diagnostics.append(QStringLiteral("Eagle: 符号圆弧三点共线或包含非法线宽"));
416 return false;
417 }
418
419 const double startSquare = QPointF::dotProduct(start, start); // 起点坐标平方和。
420 const double middleSquare = QPointF::dotProduct(middle, middle); // 中点坐标平方和。
421 const double endSquare = QPointF::dotProduct(end, end); // 终点坐标平方和。
422 const QPointF center( // 三点拟合得到的圆心。
423 (startSquare * (middle.y() - end.y()) + middleSquare * (end.y() - start.y()) +
424 endSquare * (start.y() - middle.y())) /
425 determinant,
426 (startSquare * (end.x() - middle.x()) + middleSquare * (start.x() - end.x()) +
427 endSquare * (middle.x() - start.x())) /
428 determinant);
429 const double radius = QLineF(center, start).length(); // 圆心到起点的半径。
430 const double middleRadius = QLineF(center, middle).length(); // 圆心到中点的半径。
431 if (!std::isfinite(radius) || !std::isfinite(middleRadius) || radius <= 0.0 ||
432 qAbs(radius - middleRadius) > qMax(1e-6, radius * 1e-6)) {
433 diagnostics.append(QStringLiteral("Eagle: 符号圆弧三点无法构成有效圆"));
434 return false;
435 }
436
437 const auto angle = [&center](const QPointF& point) {
438 return qRadiansToDegrees(std::atan2(point.y() - center.y(), point.x() - center.x()));
439 };
440 const double startAngle = angle(start); // 起点极角。
441 const double middleAngle = angle(middle); // 中点极角。
442 const double endAngle = angle(end); // 终点极角。
443 double counterClockwise = std::fmod(endAngle - startAngle + 360.0, 360.0); // 逆时针扫掠角。
444 if (qFuzzyIsNull(counterClockwise))
445 counterClockwise = 360.0;
446 const double middleSweep = std::fmod(middleAngle - startAngle + 360.0, 360.0); // 中点扫掠角。
447 const double sweep = middleSweep <= counterClockwise ? counterClockwise : counterClockwise - 360.0; // 最终扫掠角。
448
449 xml.writeEmptyElement(QStringLiteral("wire"));
450 xml.writeAttribute(QStringLiteral("x1"), number(start.x()));
451 xml.writeAttribute(QStringLiteral("y1"), number(start.y()));
452 xml.writeAttribute(QStringLiteral("x2"), number(end.x()));
453 xml.writeAttribute(QStringLiteral("y2"), number(end.y()));
454 xml.writeAttribute(QStringLiteral("width"), number(arc.strokeWidth));
455 xml.writeAttribute(QStringLiteral("layer"), QStringLiteral("94"));
456 xml.writeAttribute(QStringLiteral("curve"), number(sweep));
457 return true;
458}

541 for (const IR::SymbolPathIR& path : symbol.paths) {
542 if (path.partIndex == partIndex && !path.segments.isEmpty()) {
543 diagnostics.append(
544 QStringLiteral("Eagle: 符号 %1 含路径曲线，当前 XML writer 无法无损表达").arg(symbol.name));
545 return false;
546 }
547 if (path.partIndex == partIndex && (path.isFilled || path.strokeStyle != IR::StrokeStyle::Solid)) {
548 diagnostics.append(QStringLiteral("Eagle: 符号路径的填充或非实线样式无法由 XML wire 无损表达"));
549 return false;
550 }
551 if (path.partIndex == partIndex)
552 writePointList(path.points, path.strokeWidth);
553 }
554 for (const IR::SymbolArcIR& arc : symbol.arcs) {
555 if (arc.partIndex == partIndex) {
556 if (!writeSymbolArc(xml, arc, diagnostics))
557 return false;
558 }
559 }
560 for (const IR::SymbolPinIR& pin : symbol.pins) {
561 if (pin.partIndex != partIndex && !pin.commonToAllParts)
562 continue;
563 if (pin.name.isEmpty() || pin.designator.isEmpty()) {
564 diagnostics.append(QStringLiteral("Eagle: 符号 %1 存在空引脚名称或编号").arg(symbol.name));
565 return false;
566 }
567 xml.writeEmptyElement(QStringLiteral("pin"));
568 xml.writeAttribute(QStringLiteral("name"), pin.name);
569 xml.writeAttribute(QStringLiteral("x"), number(pin.position.x()));
570 xml.writeAttribute(QStringLiteral("y"), number(pin.position.y()));
571 xml.writeAttribute(QStringLiteral("length"), number(pin.length > 0.0 ? pin.length : 2.54));
572 xml.writeAttribute(QStringLiteral("rot"), symbolPinRotation(pin.direction));
573 }
574 for (const IR::SymbolTextIR& text : symbol.texts) {
575 if (!text.visible || text.partIndex != partIndex || text.text.isEmpty())
576 continue;
577 xml.writeStartElement(QStringLiteral("text"));
578 xml.writeAttribute(QStringLiteral("x"), number(text.position.x()));
579 xml.writeAttribute(QStringLiteral("y"), number(text.position.y()));
580 xml.writeAttribute(QStringLiteral("size"), number(text.fontSizeMm > 0.0 ? text.fontSizeMm : 1.27));
581 xml.writeAttribute(QStringLiteral("layer"), QStringLiteral("94"));
582 xml.writeAttribute(QStringLiteral("rot"), rotation(text.rotation));
583 xml.writeCharacters(text.text);
584 xml.writeEndElement();
585 }
586 xml.writeEndElement();
587 return !xml.hasError();
588}

597bool writeComponentLibrary(QXmlStreamWriter& xml, const QList<IR::ComponentIR>& components, QStringList& diagnostics) {
598 QSet<QString> symbolNames; // 已使用的符号名称。
599 QSet<QString> packageNames; // 已使用的封装名称。
600 QSet<QString> deviceNames; // 已使用的器件名称。
601 for (const IR::ComponentIR& component : components) {
602 const QString baseSymbolName = safeName(component.symbol.name); // 基础符号名称。
603 const QString packageName = safeName(component.footprint.name); // 封装名称。
604 const QString deviceName =
605 safeName(component.name.isEmpty() ? component.symbol.name : component.name); // 器件名称。
606 if (baseSymbolName.isEmpty() || packageName.isEmpty() || deviceName.isEmpty() ||
607 packageNames.contains(packageName) || deviceNames.contains(deviceName)) {
608 diagnostics.append(QStringLiteral("Eagle: 符号或封装名称为空或清洗后冲突：%1").arg(component.name));
609 return false;
610 }
611 if (component.symbol.partCount < 1) {
612 diagnostics.append(QStringLiteral("Eagle: 组件 %1 的符号部件数非法").arg(component.name));
613 return false;
614 }
615 if (!component.hasSymbol() || !component.hasFootprint()) {
616 diagnostics.append(QStringLiteral("Eagle: 组件 %1 缺少符号或封装，无法建立 DeviceSet").arg(component.name));
617 return false;
618 }
619 for (int partIndex = 0; partIndex < component.symbol.partCount; ++partIndex) {
620 const QString symbolName =
621 partIndex == 0 ? baseSymbolName : baseSymbolName + QStringLiteral("_P%1").arg(partIndex + 1);
622 if (symbolNames.contains(symbolName)) {
623 diagnostics.append(QStringLiteral("Eagle: 符号名称冲突：%1").arg(symbolName));
624 return false;
625 }
626 symbolNames.insert(symbolName);
627 }
628 packageNames.insert(packageName);
629 deviceNames.insert(deviceName);
630 }
631
632 xml.writeStartElement(QStringLiteral("symbols"));
633 for (const IR::ComponentIR& component : components) {
634 for (int partIndex = 0; partIndex < component.symbol.partCount; ++partIndex) {
635 const QString symbolName =
636 partIndex == 0 ? safeName(component.symbol.name)
637 : safeName(component.symbol.name) + QStringLiteral("_P%1").arg(partIndex + 1);
638 if (!writeSymbol(xml, component.symbol, partIndex, symbolName, diagnostics))
639 return false;
640 }
641 }
642 xml.writeEndElement();
643
644 xml.writeStartElement(QStringLiteral("devicesets"));
645 for (const IR::ComponentIR& component : components) {
646 const QString deviceName =
647 safeName(component.name.isEmpty() ? component.symbol.name : component.name); // 器件名称。
648 xml.writeStartElement(QStringLiteral("deviceset"));
649 xml.writeAttribute(QStringLiteral("name"), deviceName);
650 xml.writeAttribute(
651 QStringLiteral("prefix"),
652 component.symbol.designatorPrefix.isEmpty() ? QStringLiteral("U") : component.symbol.designatorPrefix);
653 xml.writeStartElement(QStringLiteral("gates"));
654 for (int partIndex = 0; partIndex < component.symbol.partCount; ++partIndex) {
655 const QString gateName = QStringLiteral("G$%1").arg(partIndex + 1); // 当前门名称。
656 const QString symbolName = // 当前门引用的符号名称。
657 partIndex == 0 ? safeName(component.symbol.name)
658 : safeName(component.symbol.name) + QStringLiteral("_P%1").arg(partIndex + 1);
659 xml.writeEmptyElement(QStringLiteral("gate"));
660 xml.writeAttribute(QStringLiteral("name"), gateName);
661 xml.writeAttribute(QStringLiteral("symbol"), symbolName);
662 xml.writeAttribute(QStringLiteral("x"), QStringLiteral("0"));
663 xml.writeAttribute(QStringLiteral("y"), QStringLiteral("0"));
664 }
665 xml.writeEndElement();
666 xml.writeStartElement(QStringLiteral("devices"));
667 xml.writeStartElement(QStringLiteral("device"));
668 xml.writeAttribute(QStringLiteral("name"), QString());
669 xml.writeAttribute(QStringLiteral("package"), safeName(component.footprint.name));
670 xml.writeStartElement(QStringLiteral("connects"));
671 for (int partIndex = 0; partIndex < component.symbol.partCount; ++partIndex) {
672 const QString gateName = QStringLiteral("G$%1").arg(partIndex + 1); // 当前门名称。
673 for (const IR::SymbolPinIR& pin : component.symbol.pins) {
674 if (pin.partIndex != partIndex && !pin.commonToAllParts)
675 continue;
676 bool padExists = false; // 是否找到对应封装焊盘。
677 for (const IR::FootprintPadIR& pad : component.footprint.pads) {
678 if (pad.number == pin.designator) {
679 padExists = true;
680 break;
681 }
682 }
683 if (!padExists) {
684 diagnostics.append(
685 QStringLiteral("Eagle: 组件 %1 的引脚 %2 找不到对应焊盘").arg(component.name, pin.designator));
686 return false;
687 }
688 xml.writeEmptyElement(QStringLiteral("connect"));
689 xml.writeAttribute(QStringLiteral("gate"), gateName);
690 xml.writeAttribute(QStringLiteral("pin"), pin.name);
691 xml.writeAttribute(QStringLiteral("pad"), pin.designator);
692 }
693 }
694 xml.writeEndElement();
695 xml.writeStartElement(QStringLiteral("technologies"));
696 xml.writeEmptyElement(QStringLiteral("technology"));
697 xml.writeAttribute(QStringLiteral("name"), QString());
698 xml.writeEndElement();
699 xml.writeEndElement();
700 xml.writeEndElement();
701 xml.writeEndElement();
702 xml.writeEndElement();
703 }
704 xml.writeEndElement();
705 return !xml.hasError();
706}
