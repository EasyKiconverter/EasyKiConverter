83QJsonObject _loadMetadataFromPath(const QString& metaPath) {
84 QFile metaFile(metaPath);
85 if (!metaFile.open(QIODevice::ReadOnly)) {
86 return QJsonObject();
87 }
88 QByteArray data = metaFile.readAll(); // 元数据文件内容。
89 metaFile.close();
90 QJsonParseError error; // JSON 解析错误。
91 QJsonDocument doc = QJsonDocument::fromJson(data, &error); // 解析后的 JSON 文档。
92 if (error.error != QJsonParseError::NoError || !doc.isObject()) {
93 return QJsonObject();
94 }
95 return doc.object();
96}

104bool _saveMetadataToPath(const QString& metaPath, const QJsonObject& metadata) {
105 QFile metaFile(metaPath);
106 if (!metaFile.open(QIODevice::WriteOnly)) {
107 return false;
108 }
109 QByteArray data = QJsonDocument(metadata).toJson(QJsonDocument::Compact); // 紧凑 JSON 数据。
110 qint64 written = metaFile.write(data); // 实际写入字节数。
111 metaFile.close();
112 return written > 0;

120bool _hasValidModel3DMetadata(const QJsonObject& metadata) {
121 if (!metadata.contains(QStringLiteral("model3duuid")))
122 return true;
123
124 const QJsonValue uuid = metadata.value(QStringLiteral("model3duuid")); // 三维模型标识字段。

154int CacheHealthManager::healAll() {
155 QDir rootDir(m_cacheRoot); // 缓存根目录对象。
156 if (!rootDir.exists()) {
157 return 0;
158 }
159 if (!CacheSafety::isOwnedRoot(m_cacheRoot)) {
160 qWarning().noquote() << "Skipped cache self-heal because ownership cannot be verified:" << m_cacheRoot;
161 return 0;
162 }
163
164 int repairedComponents = 0; // 修复并保留的组件数量。
165 int removedComponents = 0; // 判定无效的组件数量。
166
167 const QStringList componentDirs = // 缓存根目录下的组件目录。
168 rootDir.entryList(QDir::Dirs | QDir::NoDotAndDotDot, QDir::Name);
169 for (const QString& entry : componentDirs) {
170 if (entry == "model3d" || entry == CacheSafety::ownershipMarkerName()) {
171 continue;
172 }
173 if (!CacheSafety::isOwnedComponentDirectory(m_cacheRoot, rootDir.filePath(entry))) {
174 qWarning().noquote() << "Preserved cache entry with unverifiable ownership:" << rootDir.filePath(entry);
175 continue;
176 }
177 if (repairComponentCache(entry)) {
178 ++repairedComponents;
179 } else {
180 ++removedComponents;
181 }
182 }
183
184 repairModel3DCache();
185
186 if (repairedComponents > 0 || removedComponents > 0) {
187 qInfo().noquote()
188 << QString("Cache self-heal completed, retained %1 component cache dirs and removed %2 invalid dirs")
189 .arg(repairedComponents)
190 .arg(removedComponents);
191 }
192
193 return repairedComponents;
194}

268bool CacheHealthManager::repairComponentCache(const QString& lcscId) {
269 const QString dirPath = componentCacheDir(lcscId); // 组件缓存目录路径。
270 QDir componentDir(dirPath); // 组件缓存目录对象。
271 if (!componentDir.exists()) {
272 return false;
273 }
274
275 const QString metaPath = metadataPath(lcscId); // 元数据文件路径。
276 if (!QFileInfo::exists(metaPath)) {
277 qWarning().noquote() << "Preserved invalid cache dir without verifiable metadata:" << dirPath;
278 return false;
279 }
280
281 QJsonObject metadata = loadMetadata(lcscId); // 当前组件元数据。
282 if (metadata.isEmpty()) {
283 qWarning().noquote() << "Preserved cache dir with unreadable metadata:" << dirPath;
284 return false;
285 }
286
287 const QJsonValue metadataId = metadata.value(QStringLiteral("lcscId")); // 元数据中的组件编号。
288 if (metadata.contains(QStringLiteral("lcscId")) &&
289 (!metadataId.isString() ||
290 (!metadataId.toString().isEmpty() && metadataId.toString().compare(lcscId, Qt::CaseInsensitive) != 0))) {
291 qWarning().noquote() << "Preserved cache dir with mismatched component ID:" << dirPath;
292 return false;
293 }
294
295 if (!_hasValidModel3DMetadata(metadata)) {
296 qWarning().noquote() << "Preserved cache dir with invalid 3D metadata:" << dirPath;
297 return false;
298 }
299
300 bool metadataChanged = false; // 是否需要重新保存元数据。
301 if (metadata.value("lcscId").toString().isEmpty()) {
302 metadata["lcscId"] = lcscId;
303 metadataChanged = true;
304 }
305 if (!metadata.contains("cachedAt")) {
306 metadata["cachedAt"] = QDateTime::currentDateTime().toString(Qt::ISODate);
307 metadataChanged = true;
308 }
309
310 const QString cadPath = componentDir.filePath("cad_data.json"); // CAD JSON 文件路径。
311 bool hasValidCad = false; // CAD JSON 是否有效。
312 if (QFileInfo::exists(cadPath)) {
313 QFile cadFile(cadPath);
314 if (cadFile.open(QIODevice::ReadOnly)) {
315 const QByteArray cadData = cadFile.readAll(); // CAD JSON 文件内容。
316 cadFile.close();
317
318 QJsonParseError parseError; // CAD JSON 解析错误。
319 const QJsonDocument cadDoc = QJsonDocument::fromJson(cadData, &parseError); // 解析后的文档。
320 hasValidCad = !cadData.isEmpty() && parseError.error == QJsonParseError::NoError && cadDoc.isObject();
321 }
322
323 if (!hasValidCad) {
324 QString error; // 回收站操作的错误文本。
325 if (!CacheSafety::moveToTrash(cadPath, &error, m_trash))
326 qWarning().noquote() << "Preserved broken cad_data.json:" << cadPath << error;
327 }
328 }
329
330 for (int index = 0; index < 3; ++index) {
331 const QString previewPath = previewImagePath(lcscId, index); // 当前预览图路径。
332 const QFileInfo previewInfo(previewPath); // 预览图文件信息。
333 if (previewInfo.exists()) {
334 QFile previewFile(previewPath);
335 const bool valid = previewFile.open(QIODevice::ReadOnly) && // 预览图内容是否有效。
336 ComponentCacheService::isValidPreviewImageData(previewFile.readAll());
337 previewFile.close();
338 if (!valid) {
339 QString error; // 回收站操作的错误文本。
340 if (!CacheSafety::moveToTrash(previewPath, &error, m_trash))
341 qWarning().noquote() << "Preserved invalid preview cache file:" << previewPath << error;
342 }
343 }
344 }
345
346 const QString legacyDatasheetPath = datasheetPath(lcscId); // 遗留数据手册路径。
347 if (QFileInfo::exists(legacyDatasheetPath)) {
348 QFile legacyDatasheet(legacyDatasheetPath);
349 if (legacyDatasheet.open(QIODevice::ReadOnly)) {
350 const QByteArray legacyData = legacyDatasheet.readAll(); // 遗留数据手册内容。
351 legacyDatasheet.close();
352
353 const QString format = // 根据文件头推断的手册格式。
354 legacyData.startsWith("%PDF-") ? QStringLiteral("pdf") : QStringLiteral("html");
355 if (!ComponentCacheService::isValidDatasheetData(legacyData, format)) {
356 QString error; // 回收站操作的错误文本。
357 if (!CacheSafety::moveToTrash(legacyDatasheetPath, &error, m_trash))
358 qWarning().noquote() << "Preserved invalid legacy datasheet cache file:" << legacyDatasheetPath
359 << error;
360 } else {
361 const QString targetPath = resolveDatasheetPath(lcscId, format, true); // 新格式目标路径。
362 if (QFile::exists(targetPath)) {
363 qWarning().noquote() << "Preserved legacy datasheet because target already exists:" << targetPath;
364 } else if (legacyDatasheet.rename(legacyDatasheetPath, targetPath)) {
365 metadata["datasheetFormat"] = format;
366 metadataChanged = true;
367 qInfo().noquote() << "Migrated legacy datasheet cache file:" << targetPath;
368 } else {
369 qWarning().noquote() << "Preserved legacy datasheet that could not be migrated:"
370 << legacyDatasheetPath;
371 }
372 }
373 } else {
374 qWarning().noquote() << "Preserved unreadable legacy datasheet cache file:" << legacyDatasheetPath;
375 }
376 }
377
378 const QString datasheetFormat = metadata.value("datasheetFormat").toString(); // 元数据声明的手册格式。
379 const QString resolvedDatasheetPath = // 当前格式数据手册路径。
380 resolveDatasheetPath(lcscId, datasheetFormat, false);
381 if (QFileInfo::exists(resolvedDatasheetPath)) {
382 const QFileInfo datasheetInfo(resolvedDatasheetPath); // 数据手册文件信息。
383 QFile datasheetFile(resolvedDatasheetPath); // 待读取的数据手册文件。
384 // 根据文件扩展名确定数据手册格式。
385 const QString format = resolvedDatasheetPath.endsWith(QStringLiteral(".pdf"), Qt::CaseInsensitive)
386 ? QStringLiteral("pdf")
387 : QStringLiteral("html");
388 const bool valid = datasheetFile.open(QIODevice::ReadOnly) && // 数据手册内容是否有效。
389 ComponentCacheService::isValidDatasheetData(datasheetFile.readAll(), format);
390 datasheetFile.close();
391 if (datasheetInfo.size() <= 0 || !valid) {
392 QString error; // 回收站操作的错误文本。
393 if (!CacheSafety::moveToTrash(resolvedDatasheetPath, &error, m_trash))
394 qWarning().noquote() << "Preserved invalid datasheet cache file:" << resolvedDatasheetPath << error;
395 } else if (datasheetFormat.isEmpty()) {
396 metadata["datasheetFormat"] =
397 resolvedDatasheetPath.endsWith(".html") ? QStringLiteral("html") : QStringLiteral("pdf");
398 metadataChanged = true;
399 }
400 }
401
402 const bool hasBasicIdentity = !metadata.value("name").toString().isEmpty(); // 是否有基本身份信息。
403 const bool hasPreviewUrls = !metadata.value("previewImages").toArray().isEmpty(); // 是否有预览图链接。
404 const bool hasDatasheetUrl = !metadata.value("datasheet").toString().isEmpty(); // 是否有手册链接。
405 const bool hasModel3DUuid = !metadata.value("model3duuid").toString().isEmpty(); // 是否有关联三维模型。
406
407 if (!hasValidCad && !hasBasicIdentity && !hasPreviewUrls && !hasDatasheetUrl && !hasModel3DUuid) {
408 qWarning().noquote() << "Preserved unusable cache dir because recursive deletion is disabled:" << dirPath;
409 return false;
410 }
411
412 if (metadataChanged) {
413 saveMetadata(lcscId, metadata);
414 }
415
416 return true;
417}

422void CacheHealthManager::repairModel3DCache() {
423 const QString modelCacheDir = m_cacheRoot + "/model3d"; // 三维模型缓存目录。
424 QDir dir(modelCacheDir); // 三维模型缓存目录对象。
425 if (!dir.exists()) {
426 return;
427 }
428 const QFileInfo modelCacheInfo(modelCacheDir); // 三维模型缓存目录信息。
429 if (modelCacheInfo.isSymLink() || !CacheSafety::isOwnedRoot(m_cacheRoot) ||
430 !QFileInfo::exists(dir.filePath(CacheSafety::model3DMarkerName()))) {
431 qWarning().noquote() << "Skipped 3D cache self-heal because ownership cannot be verified:" << modelCacheDir;
432 return;
433 }
434
435 const QSet<QString> validSuffixes = // 支持的三维模型扩展名。
436 {QStringLiteral("obj"), QStringLiteral("step"), QStringLiteral("wrl")};
437 const QFileInfoList files = // 三维模型目录中的文件。
438 dir.entryInfoList(QDir::Files | QDir::NoDotAndDotDot, QDir::Name);
439 for (const QFileInfo& fileInfo : files) {
440 if (!CacheSafety::isOwnedModel3DFile(m_cacheRoot, fileInfo.absoluteFilePath()))
441 continue;
442 const QString suffix = fileInfo.suffix().toLower(); // 当前文件扩展名。
443 bool validContent = false; // 文件内容是否通过格式校验。
444 if (validSuffixes.contains(suffix) && fileInfo.size() > 0) {
445 QFile modelFile(fileInfo.absoluteFilePath());
446 if (modelFile.open(QIODevice::ReadOnly)) {
447 const QByteArray data = modelFile.readAll(); // 三维模型文件内容。
448 if (suffix == QStringLiteral("obj")) {
449 validContent = Exporter3DModel::hasUsableObjGeometry(data);
450 } else if (suffix == QStringLiteral("step")) {
451 validContent = Exporter3DModel::hasUsableStepData(data);
452 } else {
453 validContent = Exporter3DModel::hasUsableWrlGeometry(data);
454 }
455 }
456 }
457 const bool removable = // 当前文件是否应移入回收站。
458 !validContent || !validSuffixes.contains(suffix) || fileInfo.fileName().startsWith(".tmp");
459 if (removable) {
460 QString error; // 回收站操作的错误文本。
461 if (!CacheSafety::moveToTrash(fileInfo.absoluteFilePath(), &error, m_trash))
462 qWarning().noquote() << "Preserved broken 3D cache file:" << fileInfo.absoluteFilePath() << error;
463 }
464 }
465}
