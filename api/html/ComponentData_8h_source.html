185 void addPreviewImageData(const QByteArray& data, int imageIndex = -1) {
186 qDebug() << "ComponentData::addPreviewImageData called - data size:" << data.size()
187 << "bytes, index:" << imageIndex << "current count:" << m_previewImageData.size();
188
189 if (data.isEmpty()) {
190 qDebug() << " Skipping: data is empty";
191 return;
192 }
193
194 // 指定索引时先扩容到目标位置，保证乱序回调不会改变图片与索引的对应关系。
195 if (imageIndex >= 0) {
196 if (imageIndex >= m_previewImageData.size()) {
197 m_previewImageData.resize(imageIndex + 1);
198 }
199
200 // 如果该索引已经有数据，跳过（避免重复添加）
201 if (!m_previewImageData[imageIndex].isEmpty()) {
202 qDebug() << " Skipping: index" << imageIndex << "already has data ("
203 << m_previewImageData[imageIndex].size() << "bytes)";
204 return;
205 }
206 // 替换该索引的数据
207 qDebug() << " Replacing data at index:" << imageIndex;
208 m_previewImageData[imageIndex] = data;
209 } else {
210 // 没有指定索引或索引超出范围，直接追加
211 qDebug() << " Appending data, new count:" << (m_previewImageData.size() + 1);
212 m_previewImageData.append(data);
213 }
214
215 qDebug() << " Final count:" << m_previewImageData.size();
216 }
