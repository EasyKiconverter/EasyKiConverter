25void AltiumSchComponentStorageWriter::write(OLECompoundWriter& ole,
26 const AltiumSchComponent& component,
27 const QString& sectionKey) {
28 // 创建组件存储区，并准备对应的 Data 流。
29 ole.addStorage(sectionKey);
30 QByteArray data; // 当前组件 Data 流的序列化缓冲区。
31 AltiumBinaryWriter writer(data);
32 AltiumSchGraphicOrderWriter graphicOrderWriter(m_owner);
33
34 // Altium 对图元和二进制引脚使用同一个从 0 开始的内容记录计数器。
35 // 首条内容记录隐含索引 0，文本记录因此省略 IndexInSheet=0。
36 m_owner.m_nextIndexInSheet = 0;
37 m_owner.writeComponentRecord(writer, component);
38
39 const bool useGraphicOrder =
40 !component.graphicOrder.isEmpty() && AltiumSchGraphicOrderValidator::validate(component);
41 const bool hasImageOrder = std::any_of(
42 component.graphicOrder.cbegin(), component.graphicOrder.cend(), [](const AltiumSchGraphicOrder& order) {
43 return order.type == QStringLiteral("I");
44 });
45 if (!component.graphicOrder.isEmpty() && !useGraphicOrder) {
46 m_owner.m_diagnostics.append(
47 QStringLiteral("符号 %1 的 graphicOrder 不完整或包含无效引用，已回退到默认图元顺序").arg(component.name));
48 }
49
50 if (useGraphicOrder) {
51 // 优先按照源文件顺序写入图元。
52 for (const AltiumSchGraphicOrder& order : component.graphicOrder)
53 graphicOrderWriter.write(writer, component, order);
54 // 引脚名称和编号是由引脚派生出的文本，不在源 shape 顺序中。
55 for (const AltiumSchText& text : component.texts) {
56 if (text.isPinLabel)
57 m_owner.writeTextRecord(writer, text);
58 }
59 for (const AltiumSchText& text : component.texts) {
60 if (!text.isPinLabel && text.sourceGraphicIndex < 0)
61 m_owner.writeTextRecord(writer, text);
62 }
63 // 兼容没有来源顺序引用的扩展图元，例如手工构造的 Bézier。
64 for (const AltiumSchLine& line : component.lines)
65 m_owner.writeLineRecord(writer, line);
66 for (const AltiumSchPie& pie : component.pies)
67 m_owner.writePieRecord(writer, pie);
68 for (const AltiumSchEllipticalArc& arc : component.ellipticalArcs) {
69 if (arc.sourceGraphicType.isEmpty())
70 m_owner.writeEllipticalArcRecord(writer, arc);
71 }
72 for (const AltiumSchIeee& ieee : component.ieeeSymbols)
73 m_owner.writeIeeeRecord(writer, ieee);
74 for (const AltiumSchBezier& bezier : component.beziers) {
75 if (bezier.sourceGraphicType.isEmpty())
76 m_owner.writeBezierRecord(writer, bezier);
77 }
78 for (const AltiumSchRectangle& rect : component.rectangles) {
79 if (rect.sourceGraphicIndex < 0)
80 m_owner.writeRectangleRecord(writer, rect);
81 }
82 for (const AltiumSchRoundRectangle& rect : component.roundRectangles) {
83 if (rect.sourceGraphicIndex < 0)
84 m_owner.writeRoundRectangleRecord(writer, rect);
85 }
86 for (const AltiumSchPolygon& polygon : component.polygons) {
87 if (polygon.sourceGraphicIndex < 0)
88 m_owner.writePolygonRecord(writer, polygon);
89 }
90 for (const AltiumSchEllipse& ellipse : component.ellipses) {
91 if (ellipse.sourceGraphicType.isEmpty())
92 m_owner.writeEllipseRecord(writer, ellipse);
93 }
94 for (const AltiumSchArc& arc : component.arcs) {
95 if (arc.sourceGraphicType.isEmpty())
96 m_owner.writeArcRecord(writer, arc);
97 }
98 for (const AltiumSchPolyline& polyline : component.polylines) {
99 if (polyline.sourceGraphicIndex < 0)
100 m_owner.writePolylineRecord(writer, polyline);
101 }
102 for (const AltiumSchPath& path : component.paths) {
103 if (path.sourceGraphicType.isEmpty())
104 m_owner.writePathRecord(writer, path);
105 }
106 } else {
107 // 没有可用来源顺序时，保持兼容格式的默认记录顺序。
108 for (const AltiumSchPin& pin : component.pins)
109 m_owner.writePinRecord(writer, pin);
110 for (const AltiumSchRectangle& rect : component.rectangles)
111 m_owner.writeRectangleRecord(writer, rect);
112 for (const AltiumSchRoundRectangle& rect : component.roundRectangles)
113 m_owner.writeRoundRectangleRecord(writer, rect);
114 for (const AltiumSchLine& line : component.lines)
115 m_owner.writeLineRecord(writer, line);
116 for (const AltiumSchArc& arc : component.arcs)
117 m_owner.writeArcRecord(writer, arc);
118 for (const AltiumSchPolygon& polygon : component.polygons)
119 m_owner.writePolygonRecord(writer, polygon);
120 for (const AltiumSchEllipse& ellipse : component.ellipses)
121 m_owner.writeEllipseRecord(writer, ellipse);
122 for (const AltiumSchPie& pie : component.pies)
123 m_owner.writePieRecord(writer, pie);
124 for (const AltiumSchEllipticalArc& arc : component.ellipticalArcs)
125 m_owner.writeEllipticalArcRecord(writer, arc);
126 for (const AltiumSchPolyline& polyline : component.polylines)
127 m_owner.writePolylineRecord(writer, polyline);
128 for (const AltiumSchPath& path : component.paths)
129 m_owner.writePathRecord(writer, path);
130 for (const AltiumSchBezier& bezier : component.beziers)
131 m_owner.writeBezierRecord(writer, bezier);
132 for (const AltiumSchIeee& ieee : component.ieeeSymbols)
133 m_owner.writeIeeeRecord(writer, ieee);
134 for (const AltiumSchText& text : component.texts)
135 m_owner.writeTextRecord(writer, text);
136 }
137
138 // 这些记录不参与来源图元排序，统一放在图元之后。
139 for (const AltiumSchTextFrame& frame : component.textFrames)
140 m_owner.writeTextFrameRecord(writer, frame);
141 for (const AltiumSchImage& image : component.images) {
142 if (!useGraphicOrder || !hasImageOrder || image.sourceGraphicIndex < 0)
143 m_owner.writeImageRecord(writer, image);
144 }
145 m_owner.writeComponentParameterRecords(writer, component);
146 m_owner.writeImplementationRecords(writer, component);
147
148 // 将完整 Data 流写入组件存储区。
149 ole.writeStream(sectionKey, "Data", data);
150}
