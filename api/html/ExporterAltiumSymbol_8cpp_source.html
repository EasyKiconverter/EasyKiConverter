39bool ExporterAltiumSymbol::exportSymbol(const IR::SymbolComponentIR& symbol, const QString& filePath) {
40 m_diagnostics.clear();
41 QList<AltiumSchComponent> components; // 待写入单个符号文件的组件列表。

62 const QString& libName,
63 const QString& filePath,
64 bool appendMode,
65 bool updateMode,
66 const QString& libraryDescription) {
67 m_diagnostics.clear();

81 m_diagnostics.append(m_writer.diagnostics());
82 if (!ok) {
83 qWarning() << "ExporterAltiumSymbol: Failed to write symbol library to" << filePath;
84 }
85 return ok;

103 const auto sourceIndexForPart = [](const auto& values, int index, int partIndex) {
104 int localIndex = 0; // 当前部件内已遍历图元数量。
105 for (int i = 0; i < index; ++i) {
106 if (values.at(i).partIndex == partIndex)
107 ++localIndex;

111 const auto isFinitePoint = [](const QPointF& point) {
112 return std::isfinite(point.x()) && std::isfinite(point.y());
113 };
114 const auto hasFinitePoints = [&](const QList<QPointF>& points, int minimum) {
115 if (points.size() < minimum)
116 return false;
117 return std::all_of(points.cbegin(), points.cend(), isFinitePoint);
118 };
119 const auto isValidStrokeWidth = [](double width) { return std::isfinite(width) && width >= 0.0; };
120 const auto isValidBounds = [](double x0, double y0, double x1, double y1) {
121 return std::isfinite(x0) && std::isfinite(y0) && std::isfinite(x1) && std::isfinite(y1) && x0 != x1 && y0 != y1;
122 };
123 const auto normalizeTextAnchor = [this, &data](const QString& anchor, const QString& context) {
124 const QString normalized = anchor.trimmed().toLower(); // 规范化后的文本锚点。
125 if (normalized.isEmpty())
126 return QStringLiteral("middle");
127 if (!QStringList{QStringLiteral("start"), QStringLiteral("middle"), QStringLiteral("end")}.contains(
128 normalized)) {
129 m_diagnostics.append(QStringLiteral("符号 %1 %2 的对齐锚点无效: %3，已回退为 middle")
130 .arg(data.name)
131 .arg(context)
132 .arg(anchor));
133 return QStringLiteral("middle");
134 }
135 return normalized;
