62void WriteWorker::run() {
63 if (!m_status) {
64 qWarning() << "WriteWorker: null export status, skip run";
65 return;
66 }
67
68 // 检查是否已被取消 (v3.0.5+)
69 if (m_isAborted.loadRelaxed()) {
70 m_status->writeSuccess = false;
71 m_status->writeMessage = "Export cancelled";
72 m_status->isCancelled = true;
73 m_status->addDebugLog(QString("WriteWorker cancelled for component: %1").arg(m_status->componentId));
74 emit writeCompleted(m_status);
75 return;
76 }
77
78 // 记录当前元件写入阶段的耗时。
79 QElapsedTimer writeTimer;
80 writeTimer.start();
81
82 m_status->addDebugLog(QString("WriteWorker started for component: %1").arg(m_status->componentId));
83
84 // 初始化所有写入状态
85 m_status->symbolWritten = false;
86 m_status->footprintWritten = false;
87 m_status->model3DWritten = false;
88 m_status->previewImageWritten = false;
89 m_status->datasheetWritten = false;
90
91 if (!m_status->fetchSuccess || !m_status->processSuccess) {
92 m_status->writeDurationMs = 0;
93 m_status->writeSuccess = false;
94 m_status->writeMessage = "Skipped writing due to previous stage failure";
95 m_status->addDebugLog("Skipping write stage because fetch or process failed.");
96 emit writeCompleted(m_status);
97 return;
98 }
99
100 if ((m_exportSymbol && (!m_status->symbolData || m_status->symbolData->info().name.isEmpty())) &&
101 (m_exportFootprint && (!m_status->footprintData || m_status->footprintData->info().name.isEmpty()))) {
102 m_status->writeDurationMs = writeTimer.elapsed();
103 m_status->writeSuccess = false;
104 m_status->writeMessage = "No valid symbol or footprint data to write";
105 m_status->addDebugLog("ERROR: Symbol and Footprint data are empty or invalid.");
106 emit writeCompleted(m_status);
107 return;
108 }
109
110 if (!createOutputDirectory(m_outputPath)) {
111 m_status->writeDurationMs = writeTimer.elapsed();
112 m_status->writeSuccess = false;
113 m_status->writeMessage = "Failed to create output directory";
114 m_status->addDebugLog(
115 QString("ERROR: Failed to create output directory, Duration: %1ms").arg(m_status->writeDurationMs));
116 emit writeCompleted(m_status);
117 return;
118 }
119
120 // 并行写入文件
121 // 使用 QtConcurrent::run() + QFutureWatcher 实现并行执行
122 // 每项写完后立即发出 itemWriteCompleted 信号，实现实时进度更新
123 // 注意：QtConcurrent 不支持工作线程抛出异常，所有 lambda 必须捕获异常
124 QList<QFutureWatcher<bool>*> watchers;
125
126 // 符号写入任务
127 if (m_exportSymbol && m_status->symbolData) {
128 QFuture<bool> future = QtConcurrent::run([this]() {
129 try {
130 QMutexLocker locker(&m_symbolMutex);
131 return writeSymbolFile(*m_status);
132 } catch (const std::bad_alloc& e) {
133 qCritical() << "Memory allocation failed in symbol write:" << e.what();
134 return false;
135 } catch (const std::exception& e) {
136 qCritical() << "Exception in symbol write:" << e.what();
137 return false;
138 }
139 });
140 // 监视符号写入任务的完成状态。
141 QFutureWatcher<bool>* watcher = new QFutureWatcher<bool>();
142 connect(watcher, &QFutureWatcher<bool>::finished, this, [this, watcher]() {
143 // 读取符号写入任务的结果。
144 bool result = watcher->result();
145 m_status->symbolWritten = result;
146 emit itemWriteCompleted(m_status->componentId, static_cast<int>(ExportItemType::Symbol), result);
147 // deleteLater removed - watchers are cleaned up by qDeleteAll below
148 });
149 watcher->setFuture(future);
150 watchers.append(watcher);
151 }
152
153 // 封装写入任务
154 if (m_exportFootprint && m_status->footprintData) {
155 QFuture<bool> future = QtConcurrent::run([this]() {
156 try {
157 QMutexLocker locker(&m_footprintMutex);
158 return writeFootprintFile(*m_status);
159 } catch (const std::bad_alloc& e) {
160 qCritical() << "Memory allocation failed in footprint write:" << e.what();
161 return false;
162 } catch (const std::exception& e) {
163 qCritical() << "Exception in footprint write:" << e.what();
164 return false;
165 }
166 });
167 // 监视封装写入任务的完成状态。
168 QFutureWatcher<bool>* watcher = new QFutureWatcher<bool>();
169 connect(watcher, &QFutureWatcher<bool>::finished, this, [this, watcher]() {
170 // 读取封装写入任务的结果。
171 bool result = watcher->result();
172 m_status->footprintWritten = result;
173 emit itemWriteCompleted(m_status->componentId, static_cast<int>(ExportItemType::Footprint), result);
174 // deleteLater removed - watchers are cleaned up by qDeleteAll below
175 });
176 watcher->setFuture(future);
177 watchers.append(watcher);
178 }
179
180 // 3D模型写入任务
181 if (m_exportModel3D && m_status->model3DData &&
182 (!m_status->model3DData->rawObj().isEmpty() || !m_status->cachedModel3DWrlPath.isEmpty())) {
183 QFuture<bool> future = QtConcurrent::run([this]() {
184 try {
185 QMutexLocker locker(&m_model3DMutex);
186 return write3DModelFile(*m_status);
187 } catch (const std::bad_alloc& e) {
188 qCritical() << "Memory allocation failed in 3D model write:" << e.what();
189 return false;
190 } catch (const std::exception& e) {
191 qCritical() << "Exception in 3D model write:" << e.what();
192 return false;
193 }
194 });
195 // 监视三维模型写入任务的完成状态。
196 QFutureWatcher<bool>* watcher = new QFutureWatcher<bool>();
197 connect(watcher, &QFutureWatcher<bool>::finished, this, [this, watcher]() {
198 // 读取三维模型写入任务的结果。
199 bool result = watcher->result();
200 m_status->model3DWritten = result;
201 emit itemWriteCompleted(m_status->componentId, static_cast<int>(ExportItemType::Model3D), result);
202 // deleteLater removed - watchers are cleaned up by qDeleteAll below
203 });
204 watcher->setFuture(future);
205 watchers.append(watcher);
206 }
207
208 // 预览图写入任务
209 if (m_exportPreviewImages && !m_status->previewImageDataList.isEmpty()) {
210 QFuture<bool> future = QtConcurrent::run([this]() {
211 try {
212 return writePreviewImageFile(*m_status);
213 } catch (const std::bad_alloc& e) {
214 qCritical() << "Memory allocation failed in preview image write:" << e.what();
215 return false;
216 } catch (const std::exception& e) {
217 qCritical() << "Exception in preview image write:" << e.what();
218 return false;
219 }
220 });
221 // 监视预览图写入任务的完成状态。
222 QFutureWatcher<bool>* watcher = new QFutureWatcher<bool>();
223 connect(watcher, &QFutureWatcher<bool>::finished, this, [this, watcher]() {
224 // 读取预览图写入任务的结果。
225 bool result = watcher->result();
226 m_status->previewImageWritten = result;
227 emit itemWriteCompleted(m_status->componentId, static_cast<int>(ExportItemType::PreviewImage), result);
228 // deleteLater removed - watchers are cleaned up by qDeleteAll below
229 });
230 watcher->setFuture(future);
231 watchers.append(watcher);
232 }
233
234 // 手册写入任务
235 if (m_exportDatasheet && !m_status->datasheetData.isEmpty()) {
236 QFuture<bool> future = QtConcurrent::run([this]() {
237 try {
238 return writeDatasheetFile(*m_status);
239 } catch (const std::bad_alloc& e) {
240 qCritical() << "Memory allocation failed in datasheet write:" << e.what();
241 return false;
242 } catch (const std::exception& e) {
243 qCritical() << "Exception in datasheet write:" << e.what();
244 return false;
245 }
246 });
247 // 监视数据手册写入任务的完成状态。
248 QFutureWatcher<bool>* watcher = new QFutureWatcher<bool>();
249 connect(watcher, &QFutureWatcher<bool>::finished, this, [this, watcher]() {
250 // 读取数据手册写入任务的结果。
251 bool result = watcher->result();
252 m_status->datasheetWritten = result;
253 emit itemWriteCompleted(m_status->componentId, static_cast<int>(ExportItemType::Datasheet), result);
254 // deleteLater removed - watchers are cleaned up by qDeleteAll below
255 });
256 watcher->setFuture(future);
257 watchers.append(watcher);
258 }
259
260 // 等待所有写入任务完成
261 for (QFutureWatcher<bool>* watcher : watchers) {
262 watcher->waitForFinished();
263 }
264
265 // 清理 watchers
266 qDeleteAll(watchers);
267 watchers.clear();
268
269 // 检查是否被取消
270 if (m_isAborted.loadRelaxed()) {
271 m_status->writeDurationMs = writeTimer.elapsed();
272 m_status->writeSuccess = false;
273 m_status->writeMessage = "Export cancelled";
274 m_status->isCancelled = true;
275 m_status->addDebugLog(QString("WriteWorker cancelled for component: %1").arg(m_status->componentId));
276 emit writeCompleted(m_status);
277 return;
278 }
279
280 // 根据用户请求的导出项和实际完成情况，决定最终的成功状态
281 // 计算符号写入是否满足用户请求。
282 bool symbolSuccess = (!m_exportSymbol) || m_status->symbolWritten;
283 // 计算封装写入是否满足用户请求。
284 bool footprintSuccess = (!m_exportFootprint) || m_status->footprintWritten;
285 // 计算三维模型写入是否满足用户请求。
286 bool model3DSuccess = (!m_exportModel3D) || m_status->model3DWritten;
287 // 计算预览图写入是否满足用户请求。
288 bool previewImageSuccess = (!m_exportPreviewImages) || m_status->previewImageWritten;
289 // 计算数据手册写入是否满足用户请求。
290 bool datasheetSuccess = (!m_exportDatasheet) || m_status->datasheetWritten;
291
292 // 汇总所有已请求导出项的成功状态。
293 bool overallSuccess =
294 symbolSuccess && footprintSuccess && model3DSuccess && previewImageSuccess && datasheetSuccess;
295
296 // 收集成功和失败的导出项缩写，供诊断日志使用。
297 QStringList successItems;
298 // 保存失败导出项的缩写。
299 QStringList failedItems;
300 if (m_exportSymbol) {
301 symbolSuccess ? successItems.append("S") : failedItems.append("S");
302 }
303 if (m_exportFootprint) {
304 footprintSuccess ? successItems.append("F") : failedItems.append("F");
305 }
306 if (m_exportModel3D) {
307 model3DSuccess ? successItems.append("3D") : failedItems.append("3D");
308 }
309 if (m_exportPreviewImages) {
310 previewImageSuccess ? successItems.append("P") : failedItems.append("P");
311 }
312 if (m_exportDatasheet) {
313 datasheetSuccess ? successItems.append("D") : failedItems.append("D");
314 }
315
316 m_status->writeSuccess = overallSuccess;
317 if (overallSuccess) {
318 m_status->writeMessage = "Write completed successfully";
319 } else {
320 m_status->writeMessage = QString("Export completed with failures: %1").arg(failedItems.join(", "));
321 }
322
323 m_status->addDebugLog(QString("WriteWorker completed - Symbol:%1 Footprint:%2 3D:%3 Preview:%4 Datasheet:%5")
324 .arg(symbolSuccess ? "OK" : "FAIL")
325 .arg(footprintSuccess ? "OK" : "FAIL")
326 .arg(model3DSuccess ? "OK" : "FAIL")
327 .arg(previewImageSuccess ? "OK" : "FAIL")
328 .arg(datasheetSuccess ? "OK" : "FAIL"));
329
330 if (m_debugMode) {
331 exportDebugData(*m_status);
332 }
333
334 m_status->writeDurationMs = writeTimer.elapsed();
335 m_status->addDebugLog(QString("WriteWorker completed for component: %1, Success: %2, Duration: %3ms")
336 .arg(m_status->componentId)
337 .arg(m_status->writeSuccess)
338 .arg(m_status->writeDurationMs));
339
340 emit writeCompleted(m_status);

349 if (!status.symbolData) {
350 return true;
351 }
352
353 if (!QDir(m_tempDir).exists()) {
354 status.addDebugLog(QString("ERROR: Temp directory does not exist: %1").arg(m_tempDir));

358 status.addDebugLog(QString("SECURITY ERROR: Temp dir outside output path: %1").arg(m_tempDir));

371 });
372
373 if (success) {
374 status.addDebugLog(QString("Symbol file written atomically: %1").arg(finalFilePath));
375 status.symbolWritten = true;
376 return true;
377 } else {
378 status.addDebugLog(QString("ERROR: Failed to write symbol file"));
379 return false;

388bool WriteWorker::writeFootprintFile(ComponentExportStatus& status) {
389 if (!status.footprintData) {
390 return true;
391 }
392
393 if (!QDir(m_tempDir).exists()) {
394 status.addDebugLog(QString("ERROR: Temp directory does not exist: %1").arg(m_tempDir));
395 return false;
396 }
397 if (!PathSecurity::isSafePath(m_tempDir, m_outputPath)) {
398 status.addDebugLog(QString("SECURITY ERROR: Temp dir outside output path: %1").arg(m_tempDir));

401

404 if (!createOutputDirectory(footprintLibPath)) {
405 status.addDebugLog(QString("ERROR: Failed to create footprint library directory: %1").arg(footprintLibPath));
406 return false;
407 }
408
409 // 构造三维模型输出目录路径。
410 QString modelsDirPath = QString("%1/%2.3dmodels").arg(m_outputPath, m_libName);
411 if (m_exportModel3D) {
412 if (!createOutputDirectory(modelsDirPath)) {
413 status.addDebugLog(QString("WARNING: Failed to create 3D models directory: %1").arg(modelsDirPath));
414 }
415 }
416
417 // 生成安全的封装名称和目标文件路径。
418 QString footprintName = PathSecurity::sanitizeFilename(status.footprintData->info().name);
419 // 构造封装文件的完整输出路径。
420 QString filePath = QString("%1/%2.kicad_mod").arg(footprintLibPath, footprintName);
421
422 // 保存封装文件中引用的 WRL 和 STEP 相对路径。
423 QString model3DWrlPath;
424 // 保存封装中引用的 STEP 模型相对路径。
425 QString model3DStepPath;
426 if (m_exportModel3D && status.model3DData && !status.model3DData->uuid().isEmpty()) {
427 model3DWrlPath = QString("../%1.3dmodels/%2.wrl").arg(m_libName, footprintName);
428 // 检查是否有STEP数据（无论是原始数据还是缓存路径）
429 if (!status.model3DStepRaw.isEmpty() || !status.cachedModel3DStepPath.isEmpty()) {
430 model3DStepPath = QString("../%1.3dmodels/%2.step").arg(m_libName, footprintName);
431 }
432 }
433
434 // 通过临时文件写入并原子替换目标封装文件。
435 bool success = AtomicFileWriter::writeAtomically(
436 m_tempDir,
437 filePath,
438 ".kicad_mod.tmp",
439 [this, &status, &model3DWrlPath, &model3DStepPath](const QString& tempPath) -> bool {
440 // 将组件模型转换为封装 IR。
441 IR::FootprintComponentIR footprintIR = IR::toFootprintIR(*status.footprintData);
442 if (!model3DStepPath.isEmpty()) {
443 return m_footprintExporter.exportFootprint(footprintIR, tempPath, model3DWrlPath, model3DStepPath);
444 } else if (!model3DWrlPath.isEmpty()) {
445 return m_footprintExporter.exportFootprint(footprintIR, tempPath, model3DWrlPath);
446 } else {
447 return m_footprintExporter.exportFootprint(footprintIR, tempPath);
448 }
449 });
450
451 if (success) {
452 status.addDebugLog(QString("Footprint file written atomically: %1").arg(filePath));
453 status.footprintWritten = true;
454 return true;
455 } else {
456 status.addDebugLog(QString("ERROR: Failed to write footprint file"));
457 return false;
458 }

466bool WriteWorker::write3DModelFile(ComponentExportStatus& status) {
467 // 如果既没有原始数据也没有缓存路径，则跳过
468 const bool hasWrlData =
469 status.model3DData && (!status.model3DData->rawObj().isEmpty() || !status.cachedModel3DWrlPath.isEmpty());
470 // 判断是否存在可写入的 STEP 模型数据。
471 const bool hasStepData = !status.model3DStepRaw.isEmpty() || !status.cachedModel3DStepPath.isEmpty();
472 if (!hasWrlData && !hasStepData) {
473 return true;
474 }
475
476 if (!QDir(m_tempDir).exists()) {
477 status.addDebugLog(QString("ERROR: Temp directory does not exist: %1").arg(m_tempDir));
478 return false;
479 }
480 if (!PathSecurity::isSafePath(m_tempDir, m_outputPath)) {
481 status.addDebugLog(
482 QString("SECURITY ERROR: Temp dir %1 is outside output path %2").arg(m_tempDir, m_outputPath));
483 return false;
484 }
485
486 // 构造三维模型输出目录路径。
487 QString modelsDirPath = QString("%1/%2.3dmodels").arg(m_outputPath, m_libName);
488 if (!createOutputDirectory(modelsDirPath)) {
489 status.addDebugLog(QString("ERROR: Failed to create 3D models directory: %1").arg(modelsDirPath));
490 return false;
491 }
492
493 // 使用封装名称作为模型文件名，缺少封装时回退到元器件编号。
494 QString footprintName = status.footprintData ? status.footprintData->info().name : status.componentId;
495 footprintName = PathSecurity::sanitizeFilename(footprintName);
496
497 // 记录 WRL 模型写入结果。
498 bool wrlSuccess = false;
499
500 // WRL 是 OBJ 派生文件；如果有 OBJ 原始数据，始终重新生成，避免旧 WRL 缓存携带过期坐标归一化。
501 if (status.model3DData && !status.model3DData->rawObj().isEmpty()) {
502 // 构造 WRL 模型目标路径。
503 QString wrlFilePath = QString("%1/%2.wrl").arg(modelsDirPath, footprintName);
504 wrlSuccess = AtomicFileWriter::writeAtomically(
505 m_tempDir, wrlFilePath, ".wrl.tmp", [this, &status](const QString& tempPath) -> bool {
506 // 将模型数据转换为统一的三维模型 IR。
507 IR::Model3DIR modelIR = IR::toModel3DIR(*status.model3DData);
508 return m_model3DExporter.exportToWrl(modelIR, tempPath);
509 });
510 if (wrlSuccess) {
511 status.addDebugLog(QString("3D model WRL file written atomically: %1").arg(wrlFilePath));
512 } else {
513 status.addDebugLog(QString("ERROR: Failed to write WRL file"));
514 }
515 } else if (hasWrlData && !status.cachedModel3DWrlPath.isEmpty()) {
516 // 构造缓存 WRL 模型的目标路径。
517 QString wrlFilePath = QString("%1/%2.wrl").arg(modelsDirPath, footprintName);
518 wrlSuccess = AtomicFileWriter::copyAtomically(status.cachedModel3DWrlPath, wrlFilePath, m_tempDir);
519 if (wrlSuccess) {
520 status.addDebugLog(QString("3D model WRL file copied from cache: %1").arg(status.cachedModel3DWrlPath));
521 } else {
522 status.addDebugLog(QString("ERROR: Failed to copy WRL file from cache"));
523 }
524 }
525
526 // STEP 文件保持服务器原始坐标系，避免破坏它与 WRL 模型原本一致的装配关系。
527 // 记录 STEP 模型写入结果。
528 bool stepSuccess = false;
529 if (!status.cachedModel3DStepPath.isEmpty()) {
530 // 构造缓存 STEP 模型的目标路径。
531 QString stepFilePath = QString("%1/%2.step").arg(modelsDirPath, footprintName);
532 stepSuccess = AtomicFileWriter::copyAtomically(status.cachedModel3DStepPath, stepFilePath, m_tempDir);
533 if (stepSuccess) {
534 status.addDebugLog(QString("3D model STEP file copied from cache: %1").arg(status.cachedModel3DStepPath));
535 } else {
536 status.addDebugLog(QString("WARNING: Failed to copy STEP file from cache: %1").arg(stepFilePath));
537 }
538 } else if (!status.model3DStepRaw.isEmpty()) {
539 // 构造原始 STEP 模型的目标路径。
540 QString stepFilePath = QString("%1/%2.step").arg(modelsDirPath, footprintName);
541 stepSuccess = AtomicFileWriter::writeAtomically(
542 m_tempDir, stepFilePath, ".step.tmp", [&status](const QString& tempPath) -> bool {
543 // 打开原子写入阶段使用的临时 STEP 文件。
544 QFile file(tempPath);
545 if (!file.open(QIODevice::WriteOnly)) {
546 return false;
547 }
548 // 记录实际写入的 STEP 字节数。
549 qint64 written = file.write(status.model3DStepRaw);
550 file.close();
551 return written == status.model3DStepRaw.size();
552 });
553
554 if (stepSuccess) {
555 status.addDebugLog(QString("3D model STEP file written"));
556 } else {
557 status.addDebugLog(QString("WARNING: Failed to write STEP file: %1").arg(stepFilePath));
558 }
559 status.clearStepData();
560 }
561
562 status.model3DWritten = wrlSuccess || stepSuccess;
563 return status.model3DWritten;
564}

597bool WriteWorker::writePreviewImageFile(ComponentExportStatus& status) {
598 // 定位当前导出的根目录。
599 QDir outputDir(m_outputPath);
600 // 构造预览图输出目录路径。
601 QString imagesDir = outputDir.filePath("images");
602 if (!createOutputDirectory(imagesDir)) {
603 status.addDebugLog(QString("ERROR: Failed to create images directory: %1").arg(imagesDir));
604 return false;
605 }
606
607 // 将元器件编号转换为安全文件名。
608 QString safeName = status.componentId;
609 safeName.replace(INVALID_FILENAME_CHARS, "_");
610
611 // 记录所有预览图是否均写入成功。
612 bool allSuccess = true;
613 // 统计实际写入的预览图数量。
614 int exportedCount = 0;
615
616 for (int i = 0; i < status.previewImageDataList.size(); ++i) {
617 if (status.previewImageDataList[i].isEmpty()) {
618 continue;
619 }
620
621 // 构造当前预览图的文件名。
622 QString filename = QString("%1_%2.jpg").arg(safeName).arg(i);
623 // 构造当前预览图的完整输出路径。
624 QString filePath = QDir(imagesDir).filePath(filename);
625
626 QFile file(filePath);
627 if (file.open(QIODevice::WriteOnly)) {
628 file.write(status.previewImageDataList[i]);
629 file.close();
630 exportedCount++;
631 } else {
632 allSuccess = false;
633 status.addDebugLog(QString("ERROR: Failed to write preview image: %1").arg(filePath));
634 }
635 }
636
637 if (exportedCount > 0) {
638 status.addDebugLog(QString("Preview images written: %1 files to %2").arg(exportedCount).arg(imagesDir));
639 }
640 status.previewImageWritten = allSuccess;
641 return allSuccess;
642}

649bool WriteWorker::writeDatasheetFile(ComponentExportStatus& status) {
650 if (status.datasheetData.isEmpty()) {
651 return true;
652 }
653
654 // 定位当前导出的根目录。
655 QDir outputDir(m_outputPath);
656 // 构造数据手册输出目录路径。
657 QString datasheetsDir = outputDir.filePath("datasheets");
658 if (!createOutputDirectory(datasheetsDir)) {
659 status.addDebugLog(QString("ERROR: Failed to create datasheets directory: %1").arg(datasheetsDir));
660 return false;
661 }
662
663 // 将元器件编号转换为安全文件名。
664 QString safeName = status.componentId;
665 safeName.replace(INVALID_FILENAME_CHARS, "_");
666
667 // 根据数据内容判断格式
668 // 默认按 PDF 保存，检测到 HTML 时切换扩展名。
669 QString extension = "pdf";
670 if (status.datasheetData.startsWith("<!DOCTYPE html") || status.datasheetData.startsWith("<html")) {
671 extension = "html";
672 }
673
674 // 构造数据手册文件名。
675 QString filename = QString("%1.%2").arg(safeName).arg(extension);
676 // 构造数据手册的完整输出路径。
677 QString filePath = QDir(datasheetsDir).filePath(filename);
678
679 QFile file(filePath);
680 if (file.open(QIODevice::WriteOnly)) {
681 file.write(status.datasheetData);
682 file.close();
683 status.addDebugLog(QString("Datasheet written: %1").arg(filePath));
684 status.datasheetWritten = true;
685 return true;
686 }
687
688 status.addDebugLog(QString("ERROR: Failed to write datasheet: %1").arg(filePath));
689 return false;
690}
