20void AltiumSchPinRecordWriter::write(AltiumBinaryWriter& writer, const AltiumSchPin& pin) {
21 writer.beginBlock(AltiumConstants::SCH_BLOCK_FLAG_BINARY_PIN);
22
23 writer.writeInt32(2); // Record type = 2
24 writer.writeUInt8(0); // Unknown
25 writer.writeInt16(static_cast<int16_t>(m_owner.normalizeOwnerPartId(pin.ownerPartId, QStringLiteral("引脚"))));
26 // OwnerPartId 为 -1 表示公共 Part Zero；当前符号只有一个显示模式。
27 writer.writeUInt8(1); // OwnerPartDisplayMode
28
29 // Symbol edges / IEEE 装饰必须位于描述字符串之前。
30 writer.writeUInt8(pin.symbolInnerEdge);
31 writer.writeUInt8(pin.symbolOuterEdge);
32 writer.writeUInt8(pin.symbolInside);
33 writer.writeUInt8(pin.symbolOutside);
34
35 // Description 使用空 Pascal 短字符串。
36 writer.writePascalShortString("");
37
38 // FormalType=1 才会让 Altium 将记录作为可连接的 Pin 对象处理。
39 writer.writeUInt8(1); // FormalType: normal pin
40 writer.writeUInt8(static_cast<uint8_t>(pin.electricalType)); // ElectricalType
41
42 // PinConglomerate：低两位为方向，其余位表示隐藏、显示名称和显示编号。
43 uint8_t conglomerate = static_cast<uint8_t>(pin.orientation);
44 if (pin.isHidden)
45 conglomerate |= 0x04;
46 if (pin.showName)
47 conglomerate |= 0x08;
48 if (pin.showDesignator)
49 conglomerate |= 0x10;
50 writer.writeUInt8(conglomerate);
51
52 writer.writeInt16(AltiumCoord::toDxpInt(pin.length));
53 writer.writeInt16(AltiumCoord::toDxpInt(pin.locationX));
54 writer.writeInt16(AltiumCoord::toDxpInt(pin.locationY));
55 writer.writeUInt32(pin.color);
56
57 writer.writePascalShortString(pin.name);
58 writer.writePascalShortString(pin.designator);
59 writer.writePascalShortString(""); // SwapIdGroup
60 writer.writePascalShortString(""); // PartAndSequence
61 writer.writePascalShortString(""); // DefaultValue
62
63 writer.endBlock();
64 // 二进制引脚没有文本形式的 IndexInSheet，但仍占用共享内容记录序号。
65 ++m_owner.m_nextIndexInSheet;
66}
