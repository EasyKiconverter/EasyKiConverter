123void AltiumSchPrimitiveRecordWriter::writePolygon(AltiumBinaryWriter& writer, const AltiumSchPolygon& polygon) {
124 QMap<QString, QString> params; // 当前多边形的 RECORD=7 参数。
125 params["RECORD"] = "7";
126 m_owner.addOwnerParams(params, polygon.ownerPartId);
127 params["LineWidth"] = QString::number(polygon.lineWidth);
128 if (polygon.lineStyle != 0)
129 params["LineStyle"] = QString::number(polygon.lineStyle);
130 m_owner.addColorParam(params, "Color", polygon.color);
131 if (polygon.areaColor != 0xFFFFFF)
132 params["AreaColor"] = QString::number(polygon.areaColor);
133 if (polygon.isSolid)
134 params["IsSolid"] = "T";
135 params["LocationCount"] = QString::number(polygon.vertices.size());
136 for (int i = 0; i < polygon.vertices.size(); ++i) {
137 const int index = i + 1; // Altium 多边形顶点编号。
138 const int32_t x = AltiumCoord::toSchematicUnits(static_cast<int>(polygon.vertices[i].x())); // 顶点 X 坐标。
139 const int32_t y = AltiumCoord::toSchematicUnits(static_cast<int>(polygon.vertices[i].y())); // 顶点 Y 坐标。
140 if (x != 0)
141 params[QString("X%1").arg(index)] = QString::number(x);
142 if (y != 0)
143 params[QString("Y%1").arg(index)] = QString::number(y);
144 }
145 m_owner.addUniqueID(params);
146 writer.writeCStringParameterBlock(params);
