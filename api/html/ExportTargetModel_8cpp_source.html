137void ExportTargetModel::loadPlugins(const QString& configPath) {
138 QFile file(configPath); // 插件配置文件。
139 if (!file.open(QIODevice::ReadOnly)) {
140 qWarning() << "ExportTargetModel: Failed to open plugin config:" << configPath;
141 return;
142 }
143
144 QByteArray data = file.readAll(); // 插件配置文件内容。
145 file.close();
146
147 QJsonDocument doc = QJsonDocument::fromJson(data); // 解析后的 JSON 文档。
148 if (doc.isNull() || !doc.isObject()) {
149 qWarning() << "ExportTargetModel: Invalid JSON in plugin config:" << configPath;
150 return;
151 }
152
153 QJsonObject root = doc.object(); // 插件配置根对象。
154 QJsonArray plugins = root["plugins"].toArray(); // 插件配置数组。
155
156 m_targets.clear();
157 for (const QJsonValue& val : plugins) {
158 QJsonObject obj = val.toObject(); // 当前插件配置对象。
159 TargetInfo info; // 当前导出目标元数据。
160 info.id = obj["id"].toString();
161 info.displayName = obj["displayName"].toString();
162 info.icon = obj["icon"].toString();
163 info.optionsComponent = obj["optionsComponent"].toString();
164 if (isUserVisibleTarget(info.id) && !info.displayName.isEmpty()) {
165 m_targets.append(info);
166 }
167 }
168
169 // 下拉索引会直接映射到 TargetEdaFormat，必须与枚举顺序保持一致，不能依赖配置文件顺序。
170 std::sort(m_targets.begin(), m_targets.end(), [](const TargetInfo& left, const TargetInfo& right) {
171 const auto targetOrder = [](const QString& id) {
172 if (id == QStringLiteral("kicad"))
173 return 0;
174 if (id == QStringLiteral("altium"))
175 return 1;
176 if (id == QStringLiteral("horizon"))
177 return 2;
178 if (id == QStringLiteral("librepcb"))
179 return 3;
180 return 4;
181 };
182 return targetOrder(left.id) < targetOrder(right.id);
183 });
184
185 // 重建缓存
186 m_availableTargetsCache.clear();
187 for (const TargetInfo& info : m_targets) {
188 QVariantMap map; // 提供给 QML 的当前目标元数据。
189 map["id"] = info.id;
190 map["displayName"] = info.displayName;
191 map["icon"] = info.icon;
192 map["optionsComponent"] = info.optionsComponent;
193 m_availableTargetsCache.append(map);
194 }
195
196 // 确保索引有效
197 bool indexChanged = false; // 当前选择索引是否因目标列表变化而调整。
198 if (m_currentIndex >= m_targets.size()) {
199 m_currentIndex = 0;
200 indexChanged = true;
201 }
202
203 emit availableTargetsChanged();
204 if (indexChanged) {
205 emit currentTargetChanged();
206 }
207}
