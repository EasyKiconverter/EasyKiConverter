78bool KiCadLibraryTableManager::writeImportGuide(const QString& outputDir) {
79 QDir dir(outputDir);
80 if (!dir.exists() && !dir.mkpath(QStringLiteral("."))) {
81 qWarning() << "KiCadLibraryTableManager: Failed to create output directory for import guide:" << outputDir;
82 return false;
83 }
84
85 const QString guidePath = dir.filePath(QStringLiteral("KiCad_IMPORT.md"));
86 QSaveFile file(guidePath);
87 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
88 qWarning() << "KiCadLibraryTableManager: Failed to open import guide:" << guidePath;
89 return false;
90 }
91
92 QTextStream out(&file);
93 out.setEncoding(QStringConverter::Utf8);
94
95 out << "# KiCad Library Import\n\n";
96 out << "EasyKiConverter did not find a KiCad project (`*.kicad_pro`) in this output path or its parent "
97 "directories, so it did not modify any project library table.\n\n";
98 out << "To show library descriptions in KiCad, add the exported libraries to the target project's library "
99 "tables or to KiCad's global library tables.\n\n";
100
101 const QString symTablePath = dir.filePath(QStringLiteral("sym-lib-table"));
102 if (QFileInfo::exists(symTablePath)) {
103 out << "## Symbol Library Entry\n\n";
104 out << "Use the entry from:\n\n";
105 out << "```text\n" << symTablePath << "\n```\n\n";
106 }
107
108 const QString fpTablePath = dir.filePath(QStringLiteral("fp-lib-table"));
109 if (QFileInfo::exists(fpTablePath)) {
110 out << "## Footprint Library Entry\n\n";
111 out << "Use the entry from:\n\n";
112 out << "```text\n" << fpTablePath << "\n```\n\n";
113 }
114
115 out << "Project-specific KiCad library tables are named `sym-lib-table` and `fp-lib-table` and live next to the "
116 "project's `.kicad_pro` file.\n";
117
118 if (!file.commit()) {
119 qWarning() << "KiCadLibraryTableManager: Failed to commit import guide:" << guidePath;
120 return false;
121 }
122
123 return true;
124}

131QString KiCadLibraryTableManager::findKiCadProjectRoot(const QString& outputDir) {
132 QDir dir(QFileInfo(outputDir).absoluteFilePath());
133 if (!dir.exists()) {
134 return QString();
135 }
136
137 while (true) {
138 const QStringList projectFiles =
139 dir.entryList(QStringList(QStringLiteral("*.kicad_pro")), QDir::Files);
140 if (!projectFiles.isEmpty()) {
141 return dir.absolutePath();
142 }
143
144 if (!dir.cdUp()) {
145 break;
146 }
147 }
148
149 return QString();
150}

162bool KiCadLibraryTableManager::updateProjectLibraryTable(const QString& projectRoot,
163 const QString& tableFileName,
164 const QString& tableRootName,
165 const QString& libName,
166 const QString& uri,
167 const QString& libraryDescription) {
168 QDir projectDir(projectRoot);
169 const QString tablePath = projectDir.filePath(tableFileName);
170
171 QString content;
172 {
173 QFile existingFile(tablePath);
174 if (existingFile.exists()) {
175 if (!existingFile.open(QIODevice::ReadOnly | QIODevice::Text)) {
176 qWarning() << "KiCadLibraryTableManager: Failed to read library table:" << tablePath;
177 return false;
178 }
179 QTextStream in(&existingFile);
180 in.setEncoding(QStringConverter::Utf8);
181 content = in.readAll();
182 }
183 }
184
185 if (content.trimmed().isEmpty()) {
186 content = QStringLiteral("(%1\n (version 7)\n)\n").arg(tableRootName);
187 }
188
189 QStringList lines = content.split('\n');
190 const QRegularExpression entryRegex( // 匹配同名库条目的正则表达式。
191 QStringLiteral(R"(^\s*\‍(lib\s+\‍(name\s+"?%1"?\))").arg(QRegularExpression::escape(libName)));
192
193 QStringList filteredLines;
194 filteredLines.reserve(lines.size() + 1);
195 for (const QString& line : lines) { // 当前库表文本行。
196 if (!entryRegex.match(line).hasMatch()) {
197 filteredLines.append(line);
198 }
199 }
200
201 int insertIndex = filteredLines.size();
202 while (insertIndex > 0 && filteredLines.at(insertIndex - 1).trimmed().isEmpty()) {
203 --insertIndex;
204 }
205 if (insertIndex > 0 && filteredLines.at(insertIndex - 1).trimmed() == QStringLiteral(")")) {
206 --insertIndex;
207 }
208
209 filteredLines.insert(insertIndex, makeLibraryEntry(libName, uri, libraryDescription));
210 QString updatedContent = filteredLines.join('\n');
211 if (!updatedContent.endsWith('\n')) {
212 updatedContent += '\n';
213 }
214
215 QSaveFile file(tablePath);
216 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
217 qWarning() << "KiCadLibraryTableManager: Failed to write library table:" << tablePath;
218 return false;
219 }
220
221 QTextStream out(&file);
222 out.setEncoding(QStringConverter::Utf8);
223 out << updatedContent;
224 if (!file.commit()) {
225 qWarning() << "KiCadLibraryTableManager: Failed to commit library table:" << tablePath;
226 return false;
227 }
228
229 qDebug() << "KiCadLibraryTableManager: Updated project library table:" << tablePath;
230 return true;
231}

240QString KiCadLibraryTableManager::makeLibraryEntry(const QString& libName,
241 const QString& uri,
242 const QString& libraryDescription) {
243 // 库表中待写入的单条记录文本。
244 QString entry = QStringLiteral(" (lib (name \"%1\")(type \"KiCad\")(uri \"%2\")(options \"\")")
245 .arg(KiCadTableUtils::escapeTableValue(libName), KiCadTableUtils::escapeTableValue(uri));
246 if (!libraryDescription.isEmpty()) {
247 entry += QStringLiteral("(descr \"%1\")").arg(KiCadTableUtils::escapeTableValue(libraryDescription));
248 } else {
249 entry += QStringLiteral("(descr \"\")");
250 }
251 entry += QStringLiteral(")");
252 return entry;
253}
