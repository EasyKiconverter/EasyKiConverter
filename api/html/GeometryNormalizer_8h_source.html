55inline QList<QPointF> parseFlatPointString(const QString& pointsStr, double scaleFactor = EASYEDA_PX_TO_MM) {
56 QList<QPointF> points; // 解析得到的缩放后坐标点列表。
57 // 将输入字符串按连续空白拆分为数值字段。
58 const QStringList parts = pointsStr.trimmed().split(QRegularExpression(QStringLiteral("\\s+")),
59 Qt::SkipEmptyParts); // 空格分隔的数值字段。
60 for (int i = 0; i + 1 < parts.size(); i += 2) {
61 bool ok1 = false, ok2 = false; // X/Y 字符串是否解析成功。
62 const double x = parts[i].toDouble(&ok1); // 原始 X 坐标。
63 const double y = parts[i + 1].toDouble(&ok2); // 原始 Y 坐标。
64 if (ok1 && ok2) {
65 points.append(QPointF(x * scaleFactor, y * scaleFactor));
66 }
67 }
68 return points;
69}

77inline QList<QPointF> parseCommaSeparatedPoints(const QString& pointsStr, double scaleFactor = EASYEDA_PX_TO_MM) {
78 QList<QPointF> points; // 解析得到的缩放后坐标点列表。
79 const QStringList pairs =
80 pointsStr.trimmed().split(QRegularExpression(QStringLiteral("\\s+")), Qt::SkipEmptyParts); // 坐标对字段。
81 for (const auto& pair : pairs) {
82 const QStringList xy = pair.split(','); // 当前坐标对的 X/Y 字段。
83 if (xy.size() >= 2) {
84 bool ok1 = false, ok2 = false; // X/Y 字符串是否解析成功。
85 const double x = xy[0].toDouble(&ok1); // 原始 X 坐标。
86 const double y = xy[1].toDouble(&ok2); // 原始 Y 坐标。
87 if (ok1 && ok2) {
88 points.append(QPointF(x * scaleFactor, y * scaleFactor));
89 }
90 }
91 }
92 return points;
93}
