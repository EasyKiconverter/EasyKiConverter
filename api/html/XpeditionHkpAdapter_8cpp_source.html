81void applyPadShape(const Parser::XpeditionPadDefinition& source, IR::FootprintPadIR& target) {
82 target.size = source.size;
83 // 统一形状枚举，无法表达的来源形状降级为矩形并由上层保留诊断上下文。
84 switch (source.shape) {
85 case Parser::XpeditionPadShape::Round:
86 target.shape = IR::PadShape::Ellipse;
87 break;
88 case Parser::XpeditionPadShape::Oblong:
89 target.shape = IR::PadShape::Oval;
90 break;
91 case Parser::XpeditionPadShape::Polygon:
92 target.shape = IR::PadShape::Polygon;
93 target.customShapePoints = source.polygon;
94 break;
95 case Parser::XpeditionPadShape::Rectangle:
96 case Parser::XpeditionPadShape::Square:
97 target.shape = IR::PadShape::Rect;
98 break;
99 case Parser::XpeditionPadShape::Unknown:
100 target.shape = IR::PadShape::Rect;
101 break;
102 }

112IR::LayerType outlineLayer(const QString& sourceLayer, Parser::ParseDiagnostics* diagnostics, const QString& cellName) {
113 const QString layer = sourceLayer.toUpper(); // 规范化后的来源层名称。
114 if (layer.contains(QStringLiteral("SILK")) || layer.contains(QStringLiteral("OVERLAY")))
115 return IR::LayerType::TopSilk;
116 if (layer.contains(QStringLiteral("ASSEMBLY")))
117 return IR::LayerType::TopAssembly;
118 if (layer.contains(QStringLiteral("COURTYARD")))
119 return IR::LayerType::Mechanical1;
120 if (diagnostics)
121 diagnostics->add(Parser::ParseSeverity::Warning,
122 Parser::ParseScope::Footprint,
123 QStringLiteral("Xpedition 轮廓层无法精确映射，已降级为用户层：%1").arg(sourceLayer),
124 cellName);
125 return IR::LayerType::UserDefined;
126}

170IR::FootprintComponentIR XpeditionHkpAdapter::toFootprint(const Parser::XpeditionHkpModel& model,
171 const Parser::XpeditionCellDefinition& cell,
172 Parser::ParseDiagnostics* diagnostics) {
173 IR::FootprintComponentIR result; // 输出封装组件 IR。
174 result.name = cell.name;
175 for (const Parser::XpeditionCellPin& sourcePin : cell.pins) {
176 if (model.isPadstackAmbiguous(sourcePin.padstack)) {
177 if (diagnostics)
178 diagnostics->add(Parser::ParseSeverity::Error,
179 Parser::ParseScope::Footprint,
180 QStringLiteral("Xpedition 引脚引用的 Padstack 存在歧义：%1").arg(sourcePin.padstack),
181 cell.name);
182 continue;
183 }
184 const Parser::XpeditionPadstackDefinition* padstack =
185 model.findPadstack(sourcePin.padstack); // 来源 Padstack。
186 if (padstack == nullptr) {
187 if (diagnostics)
188 diagnostics->add(Parser::ParseSeverity::Error,
189 Parser::ParseScope::Footprint,
190 QStringLiteral("Xpedition 引脚引用了不存在的 Padstack：%1").arg(sourcePin.padstack),
191 cell.name);
192 continue;
193 }
194 const bool hasTop = !padstack->topPad.isEmpty(); // 是否存在顶层 Pad。
195 const bool hasBottom = !padstack->bottomPad.isEmpty(); // 是否存在底层 Pad。
196 const QString padName = hasTop ? padstack->topPad : padstack->bottomPad; // 实际使用的 Pad 名称。
197 const Parser::XpeditionPadDefinition* sourcePad =
198 findPad(model, padName, diagnostics, cell.name); // 来源 Pad。
199 if (sourcePad == nullptr)
200 continue;
201
202 IR::FootprintPadIR pad; // 当前焊盘 IR。
203 pad.number = sourcePin.number;
204 pad.position = sourcePin.position + sourcePad->offset;
205 pad.rotation = sourcePin.rotation;
206 pad.layer = hasTop && hasBottom ? IR::LayerType::MultiLayer
207 : (hasBottom ? IR::LayerType::BottomCopper : IR::LayerType::TopCopper);
208 applyPadShape(*sourcePad, pad);
209 if (!padstack->holeName.isEmpty()) {
210 const Parser::XpeditionHoleDefinition* hole =
211 findHole(model, padstack->holeName, diagnostics, cell.name); // 来源孔定义。
212 if (hole != nullptr) {
213 pad.padType = IR::PadType::ThroughHole;
214 pad.layer = IR::LayerType::MultiLayer;
215 pad.isPlated = hole->plated;
216 pad.holeSize = hole->size.width();
217 pad.holeLength = qMax(hole->size.width(), hole->size.height());
218 }
219 }
220 result.pads.append(pad);
221 }
222 for (const Parser::XpeditionCellOutline& sourceOutline : cell.outlines) {
223 if (sourceOutline.points.size() < 2)
224 continue;
225 IR::FootprintOutlineIR outline; // 当前封装轮廓 IR。
226 outline.points = sourceOutline.points;
227 outline.layer = outlineLayer(sourceOutline.layer, diagnostics, cell.name);
228 result.outlines.append(outline);
229 }
230 return result;
231}

240XpeditionHkpConversionResult XpeditionHkpAdapter::toIR(const Parser::XpeditionHkpModel& model,
241 const QMap<QString, IR::SymbolComponentIR>& symbols,
242 Parser::ParseDiagnostics* diagnostics) {
243 XpeditionHkpConversionResult result; // 输出转换结果。
244 QMap<QString, IR::FootprintComponentIR> footprints; // 按 Cell 名称索引的封装。
245 for (const Parser::XpeditionCellDefinition& cell : model.cells) {
246 IR::FootprintComponentIR footprint = toFootprint(model, cell, diagnostics); // 当前 Cell 的封装 IR。
247 footprints.insert(cell.name, footprint);
248 result.footprints.append(footprint);
249 }
250 for (const Parser::XpeditionPartDefinition& part : model.parts) {
251 const bool hasFallbackCell =
252 !part.bottomCell.isEmpty() && part.bottomCell != part.topCell; // 是否需要底层 Cell 回退。
253 const Parser::XpeditionCellDefinition* cell =
254 findCell(model, part.topCell, diagnostics, part.number, !hasFallbackCell);
255 if (cell == nullptr && !part.bottomCell.isEmpty())
256 cell = findCell(model, part.bottomCell, diagnostics, part.number, true);
257 if (cell == nullptr)
258 continue;
259 if (!symbols.contains(part.symbol)) {
260 if (diagnostics)
261 diagnostics->add(Parser::ParseSeverity::Error,
262 Parser::ParseScope::Symbol,
263 QStringLiteral("Xpedition 器件引用了不存在的符号：%1").arg(part.symbol),
264 part.number);
265 continue;
266 }
267 IR::ComponentIR component; // 当前器件 IR。
268 component.name = part.name.isEmpty() ? part.number : part.name;
269 component.description = part.description;
270 component.prefix = part.referencePrefix;
271 component.package = cell->name;
272 component.symbol = symbols.value(part.symbol);
273 component.footprint = footprints.value(cell->name);
274 component.sourceMetadata.insert(QStringLiteral("xpeditionPartNumber"), part.number);
275 component.sourceMetadata.insert(QStringLiteral("xpeditionSymbol"), part.symbol);
276 component.sourceMetadata.insert(QStringLiteral("xpeditionTopCell"), part.topCell);
277 component.sourceMetadata.insert(QStringLiteral("xpeditionBottomCell"), part.bottomCell);
278 for (auto iterator = part.properties.cbegin(); iterator != part.properties.cend(); ++iterator)
279 component.sourceMetadata.insert(iterator.key(), iterator.value());
280 result.components.append(component);
281 }
282 return result;
283}

292XpeditionHkpConversionResult XpeditionHkpAdapter::toIR(const Parser::XpeditionHkpDocument& document,
293 const QList<Parser::XpeditionSymbolDocument>& symbolDocuments,
294 Parser::ParseDiagnostics* diagnostics) {
295 if (diagnostics)
296 diagnostics->append(document.diagnostics);
297
298 QMap<QString, IR::SymbolComponentIR> symbols; // 按符号名称索引的符号 IR。
299 for (const Parser::XpeditionSymbolDocument& symbolDocument : symbolDocuments) {
300 if (diagnostics)
301 diagnostics->append(symbolDocument.diagnostics);
302 if (!symbolDocument.isRecognized())
303 continue;
304 IR::SymbolComponentIR symbol = XpeditionSymbolAdapter::toIR(symbolDocument, diagnostics); // 当前符号 IR。
305 if (symbol.name.isEmpty())
306 continue;
307 if (symbols.contains(symbol.name)) {
308 if (diagnostics)
309 diagnostics->add(Parser::ParseSeverity::Error,
310 Parser::ParseScope::Symbol,
311 QStringLiteral("Xpedition 符号跨文件重名：%1").arg(symbol.name),
312 symbol.name);
313 continue;
314 }
315 symbols.insert(symbol.name, symbol);
316 }
317 return toIR(document.model, symbols, diagnostics);
318}
