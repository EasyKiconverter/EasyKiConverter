69QPair<int, int> pinVector(IR::PinDirection direction, double length) {
70 // Capture 的 hot point 位于引脚末端，因此这里把统一 IR 的方向和长度转换为 mil 偏移。
71 switch (direction) {
72 case IR::PinDirection::Left:
73 return {-qRound64(length * kMilPerMm), 0};
74 case IR::PinDirection::Up:
75 return {0, qRound64(length * kMilPerMm)};
76 case IR::PinDirection::Down:
77 return {0, -qRound64(length * kMilPerMm)};
78 case IR::PinDirection::Right:
79 return {qRound64(length * kMilPerMm), 0};
80 }
81 return {0, 0};
82}

98QRectF symbolBounds(const IR::SymbolComponentIR& symbol) {
99 QRectF bounds;
100 bool initialized = false;
101 const auto include = [&](const QPointF& point) {
102 if (!finitePoint(point))
103 return;
104 if (!initialized) {
105 bounds = QRectF(point, QSizeF(0, 0));
106 initialized = true;
107 } else {
108 bounds = bounds.united(QRectF(point, QSizeF(0, 0)));
109 }
110 };
111 for (const auto& rectangle : symbol.rectangles) {
112 include({rectangle.x0, rectangle.y0});
113 include({rectangle.x1, rectangle.y1});
114 }
115 for (const auto& pin : symbol.pins) {
116 include(pin.position);
117 const auto vector = pinVector(pin.direction, pin.length);
118 include({pin.position.x() + vector.first / kMilPerMm, pin.position.y() + vector.second / kMilPerMm});
119 }
120 for (const auto& polyline : symbol.polylines)
121 for (const QPointF& point : polyline.points)
122 include(point);
123 for (const auto& polygon : symbol.polygons)
124 for (const QPointF& point : polygon.points)
125 include(point);
126 return initialized ? bounds : QRectF(-1.27, -1.27, 2.54, 2.54);
127}

222void writePin(QXmlStreamWriter& xml, const IR::SymbolPinIR& pin, int position) {
223 const auto vector = pinVector(pin.direction, pin.length);
224 const qint64 startX = qRound64(pin.position.x() * kMilPerMm);
225 const qint64 startY = qRound64(pin.position.y() * kMilPerMm);
226 const qint64 hotX = startX + vector.first;
227 const qint64 hotY = startY + vector.second;
228 xml.writeStartElement(QStringLiteral("SymbolPinScalar"));
229 xml.writeEmptyElement(QStringLiteral("Defn"));
230 xml.writeAttribute(QStringLiteral("hotptX"), QString::number(hotX));
231 xml.writeAttribute(QStringLiteral("hotptY"), QString::number(hotY));
232 xml.writeAttribute(QStringLiteral("name"), pin.name);
233 xml.writeAttribute(QStringLiteral("position"), QString::number(position));
234 xml.writeAttribute(QStringLiteral("startX"), QString::number(startX));
235 xml.writeAttribute(QStringLiteral("startY"), QString::number(startY));
236 xml.writeAttribute(QStringLiteral("type"), QString::number(orcadPinType(pin.electricalType)));
237 xml.writeAttribute(QStringLiteral("visible"), pin.display.showName ? QStringLiteral("1") : QStringLiteral("0"));
238 xml.writeEndElement();
239 const auto flag = [&](const QString& name, bool value) {
240 xml.writeStartElement(name);
241 xml.writeEmptyElement(QStringLiteral("Defn"));
242 xml.writeAttribute(QStringLiteral("val"), value ? QStringLiteral("1") : QStringLiteral("0"));
243 xml.writeEndElement();
244 };
245 flag(QStringLiteral("IsLong"), false);
246 flag(QStringLiteral("IsClock"), pin.style.clock || pin.hasClock);
247 flag(QStringLiteral("IsDot"), pin.style.inverted || pin.hasDot);
248 flag(QStringLiteral("IsLeftPointing"), pin.direction == IR::PinDirection::Left);
249 flag(QStringLiteral("IsRightPointing"), pin.direction == IR::PinDirection::Right);
250 flag(QStringLiteral("IsNetStyle"), false);
251 flag(QStringLiteral("IsNoConnect"), pin.style.decoration == IR::PinDecoration::NoConnect);
252 flag(QStringLiteral("IsGlobal"), false);
253 flag(QStringLiteral("IsNumberVisible"), pin.display.showDesignator);
254}

286bool ExporterOrcadSymbol::exportSymbolLibrary(const QList<IR::SymbolComponentIR>& symbols,
287 const QString& libName,
288 const QString& filePath,
289 bool appendMode,
290 bool updateMode,
291 const QString& libraryDescription) {
292 m_diagnostics.clear();
293 // XML writer 没有安全的原生 OLB 合并语义，任何追加或更新请求都必须显式拒绝。
294 if (appendMode || updateMode) {
295 m_diagnostics.append(
296 QStringLiteral("OrCAD Capture XML 当前只支持完整重写，不支持追加或更新：%1").arg(filePath));
297 return false;
298 }
299 if (symbols.isEmpty()) {
300 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 没有可导出的符号"));
301 return false;
302 }
303
304 QSet<QString> names;
305 for (const auto& symbol : symbols) {
306 const QString name = definitionName(symbol.name);
307 if (name.isEmpty()) {
308 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 符号名称为空"));
309 return false;
310 }
311 const QString key = name.toCaseFolded();
312 if (names.contains(key)) {
313 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 符号名称冲突：%1").arg(name));
314 return false;
315 }
316 names.insert(key);
317 for (const auto& pin : symbol.pins) {
318 if (!finitePoint(pin.position) || !std::isfinite(pin.length) || pin.length < 0.0 || pin.name.isEmpty() ||
319 pin.designator.isEmpty()) {
320 m_diagnostics.append(
321 QStringLiteral("OrCAD Capture XML 符号 %1 存在非法引脚字段：%2").arg(name, pin.name));
322 return false;
323 }
324 }
325 if (!symbol.ellipses.isEmpty() || !symbol.arcs.isEmpty() || !symbol.pies.isEmpty() ||
326 !symbol.ellipticalArcs.isEmpty() || !symbol.paths.isEmpty() || !symbol.beziers.isEmpty() ||
327 !symbol.textFrames.isEmpty() || !symbol.images.isEmpty() || !symbol.circles.isEmpty() ||
328 !symbol.texts.isEmpty()) {
329 m_diagnostics.append(
330 QStringLiteral("OrCAD Capture XML 符号 %1 含当前 writer 未实现的图元，已拒绝静默丢失").arg(name));
331 return false;
332 }
333 for (const auto& polyline : symbol.polylines) {
334 if (polyline.points.size() < 2) {
335 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 符号 %1 的折线点数不足").arg(name));
336 return false;
337 }
338 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 符号 %1 的折线当前降级为线段").arg(name));
339 }
340 if (!symbol.polygons.isEmpty()) {
341 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 暂不写入多边形填充，符号 %1 已拒绝导出").arg(name));
342 return false;
343 }
344 }
345
346 QFile file(filePath);
347 if (!file.open(QIODevice::WriteOnly | QIODevice::Truncate)) {
348 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 无法写入文件：%1").arg(filePath));
349 return false;
350 }
351 QXmlStreamWriter xml(&file);
352 xml.setAutoFormatting(true);
353 xml.writeStartDocument(QStringLiteral("1.0"), true);
354 xml.writeStartElement(QStringLiteral("Lib"));
355 xml.writeAttribute(QStringLiteral("xmlns:xsd"), QStringLiteral("http://www.w3.org/2001/XMLSchema"));
356 xml.writeAttribute(QStringLiteral("xmlns:xsi"), QStringLiteral("http://www.w3.org/2001/XMLSchema-instance"));
357 xml.writeAttribute(QStringLiteral("xsi:noNamespaceSchemaLocation"), QStringLiteral("olb.xsd"));
358 xml.writeStartElement(QStringLiteral("Defn"));
359 xml.writeAttribute(QStringLiteral("name"), libraryDescription.isEmpty() ? libName : libraryDescription);
360 xml.writeEndElement();
361 writeDefaultValues(xml);
362
363 for (const auto& symbol : symbols) {
364 const QString name = definitionName(symbol.name);
365 const QRectF bounds = symbolBounds(symbol);
366 xml.writeStartElement(QStringLiteral("Package"));
367 xml.writeStartElement(QStringLiteral("Defn"));
368 xml.writeAttribute(QStringLiteral("alphabeticNumbering"), QStringLiteral("1"));
369 xml.writeAttribute(QStringLiteral("isHomogeneous"),
370 symbol.partCount > 1 ? QStringLiteral("0") : QStringLiteral("1"));
371 xml.writeAttribute(QStringLiteral("name"), name);
372 xml.writeAttribute(QStringLiteral("pcbFootprint"), symbol.footprintName);
373 xml.writeAttribute(QStringLiteral("pcbLib"), QString());
374 xml.writeAttribute(QStringLiteral("refdesPrefix"), symbol.designatorPrefix);
375 xml.writeEndElement();
376 xml.writeStartElement(QStringLiteral("LibPart"));
377 xml.writeStartElement(QStringLiteral("Defn"));
378 xml.writeAttribute(QStringLiteral("CellName"), name);
379 xml.writeEndElement();
380 xml.writeStartElement(QStringLiteral("NormalView"));
381 xml.writeStartElement(QStringLiteral("Defn"));
382 xml.writeAttribute(QStringLiteral("suffix"), QStringLiteral(".Normal"));
383 xml.writeEndElement();
384 writeDisplayProperty(xml,
385 QStringLiteral("Part Reference"),
386 qRound64(bounds.center().x() * kMilPerMm),
387 qRound64((bounds.bottom() + 2.54) * kMilPerMm));
388 writeDisplayProperty(xml,
389 QStringLiteral("Value"),
390 qRound64(bounds.center().x() * kMilPerMm),
391 qRound64((bounds.top() - 2.54) * kMilPerMm));
392 xml.writeStartElement(QStringLiteral("SymbolBBox"));
393 xml.writeEmptyElement(QStringLiteral("Defn"));
394 xml.writeAttribute(QStringLiteral("x1"), mil(bounds.left()));
395 xml.writeAttribute(QStringLiteral("x2"), mil(bounds.right()));
396 xml.writeAttribute(QStringLiteral("y1"), mil(bounds.top()));
397 xml.writeAttribute(QStringLiteral("y2"), mil(bounds.bottom()));
398 xml.writeEndElement();
399 xml.writeStartElement(QStringLiteral("IsPinNumbersVisible"));
400 xml.writeEmptyElement(QStringLiteral("Defn"));
401 xml.writeAttribute(QStringLiteral("val"), QStringLiteral("1"));
402 xml.writeEndElement();
403 xml.writeStartElement(QStringLiteral("IsPinNamesVisible"));
404 xml.writeEmptyElement(QStringLiteral("Defn"));
405 xml.writeAttribute(QStringLiteral("val"), QStringLiteral("1"));
406 xml.writeEndElement();
407 xml.writeStartElement(QStringLiteral("PartValue"));
408 xml.writeEmptyElement(QStringLiteral("Defn"));
409 xml.writeAttribute(QStringLiteral("name"), name);
410 xml.writeEndElement();
411 xml.writeStartElement(QStringLiteral("Reference"));
412 xml.writeEmptyElement(QStringLiteral("Defn"));
413 xml.writeAttribute(QStringLiteral("name"), symbol.designatorPrefix);
414 xml.writeEndElement();
415 for (const auto& rectangle : symbol.rectangles)
416 writeRectangle(xml, rectangle);
417 for (const auto& polyline : symbol.polylines) {
418 for (int i = 1; i < polyline.points.size(); ++i) {
419 const QPointF& first = polyline.points.at(i - 1);
420 const QPointF& second = polyline.points.at(i);
421 xml.writeStartElement(QStringLiteral("Line"));
422 xml.writeEmptyElement(QStringLiteral("Defn"));
423 xml.writeAttribute(QStringLiteral("lineStyle"), QStringLiteral("0"));
424 xml.writeAttribute(QStringLiteral("lineWidth"), QStringLiteral("0"));
425 xml.writeAttribute(QStringLiteral("x1"), mil(first.x()));
426 xml.writeAttribute(QStringLiteral("x2"), mil(second.x()));
427 xml.writeAttribute(QStringLiteral("y1"), mil(first.y()));
428 xml.writeAttribute(QStringLiteral("y2"), mil(second.y()));
429 xml.writeEndElement();
430 }
431 }
432 for (int i = 0; i < symbol.pins.size(); ++i)
433 writePin(xml, symbol.pins.at(i), i);
434 xml.writeEndElement();
435 xml.writeStartElement(QStringLiteral("PhysicalPart"));
436 xml.writeEmptyElement(QStringLiteral("Defn"));
437 for (int i = 0; i < symbol.pins.size(); ++i) {
438 xml.writeStartElement(QStringLiteral("PinNumber"));
439 xml.writeEmptyElement(QStringLiteral("Defn"));
440 xml.writeAttribute(QStringLiteral("number"), symbol.pins.at(i).designator);
441 xml.writeAttribute(QStringLiteral("position"), QString::number(i));
442 xml.writeEndElement();
443 }
444 xml.writeEndElement();
445 xml.writeEndElement();
446 xml.writeEndElement();
447 }
448 xml.writeEndElement();
449 xml.writeEndDocument();
450 file.close();
451 if (xml.hasError()) {
452 m_diagnostics.append(QStringLiteral("OrCAD Capture XML 写入过程中发生错误：%1").arg(filePath));
453 return false;
454 }
455 return true;
456}
