90void ComponentListMutationCoordinator::remove(ComponentListViewModel& owner, int index) {
91 owner.clearAttentionHints();
92
93 // 保存当前列表项数量。
94 int listCount;
95 {
96 QMutexLocker locker(&owner.m_listMutex);
97 listCount = owner.m_componentList.count();
98 if (index < 0 || index >= listCount) {
99 return;
100 }
101 }
102
103 // 删除最后一个元件时复用完整清空流程，确保缓存和服务状态同步清理。
104 if (listCount == 1) {
105 clear(owner);
106 return;
107 }
108
109 // 保存待删除的列表项。
110 ComponentListItemData* item = nullptr;
111 // 保存被删除元件的编号。
112 QString removedId;
113 {
114 QMutexLocker locker(&owner.m_listMutex);
115 item = owner.m_componentList.takeAt(index);
116 removedId = item->componentId();
117 owner.m_componentIdIndex.remove(removedId);
118 owner.m_validatedComponentIds.removeAll(removedId);
119 }
120
121 if (owner.m_service) {
122 owner.m_service->cancelRequestForComponent(removedId);
123 }
124
125 // 记录被删除元件是否正在验证。
126 const bool wasInFlight = owner.m_validationQueue.isInFlight(removedId);
127 owner.m_validationQueue.remove(removedId);
128
129 if (item->isFetching()) {
130 owner.m_pendingValidationCount--;
131 if (wasInFlight) {
132 owner.m_validationPendingCount--;
133 }
134 owner.m_validationStateManager->cancelValidation(1);
135 }
136
137 owner.beginRemoveRows(QModelIndex(), index, index);
138 delete item;
139 owner.endRemoveRows();
140
141 owner.rebuildComponentIdIndex();
142 owner.updateHasInvalidComponents();
143 owner.scheduleListUpdate();
144 emit owner.componentRemoved(removedId);
145}

177 // 保存清理前的列表项数量。

184 if (listCount <= 0) {

186 }
187
188 if (owner.m_service) {
189 // 先取消所有进行中的请求，防止异步响应访问已删除的列表项。
190 owner.m_service->cancelAllPendingRequests();
191 owner.m_service->clearCache();
192 }
193
194 owner.beginResetModel();
195 {
196 QMutexLocker locker(&owner.m_listMutex);
197 qDeleteAll(owner.m_componentList);
198 owner.m_componentList.clear();
199 owner.m_componentIdIndex.clear();
200 }
201 owner.endResetModel();
202
203 owner.m_validationStateManager->reset();
204 owner.recomputeStateCounters();
205 owner.updateHasInvalidComponents();
206 owner.scheduleListUpdate();
207 emit owner.listCleared();
208}
