23void AltiumSchImageStorageWriter::prepareNames(const QList<AltiumSchComponent>& components) {
24 QSet<QString> usedNames; // 已分配的 Storage 文件名集合。
25 for (const AltiumSchComponent& component : components) { // 遍历所有组件图片。
26 for (int imageIndex = 0; imageIndex < component.images.size(); ++imageIndex) {
27 const AltiumSchImage& image = component.images.at(imageIndex); // 当前待处理的图片。
28 if (!image.embedImage) {
29 if (image.fileName.trimmed().isEmpty()) {
30 const QString diagnostic = // 外部图片引用缺少文件名时的诊断文本。
31 QStringLiteral("组件 %1 图片 %2 的外部文件名为空，已跳过文件引用")
32 .arg(component.name)
33 .arg(imageIndex);
34 m_owner.m_diagnostics.append(diagnostic);
35 qWarning() << "AltiumSchLibWriter:" << diagnostic;
36 }
37 continue;
38 }
39
40 QString sourceName = image.fileName; // 图片来源文件名副本。
41 sourceName.replace('\\', '/');
42 const QString embeddedName = QFileInfo(sourceName).fileName(); // 去除目录后的嵌入文件名。
43 if (image.data.isEmpty()) {
44 const QString diagnostic = // 嵌入数据缺失时的诊断文本。
45 QStringLiteral("组件 %1 图片 %2 的嵌入数据为空，已跳过 Storage")
46 .arg(component.name)
47 .arg(imageIndex);
48 m_owner.m_diagnostics.append(diagnostic);
49 qWarning() << "AltiumSchLibWriter:" << diagnostic;
50 continue;
51 }
52 if (embeddedName.toLocal8Bit().size() > 255) {
53 const QString diagnostic = // 嵌入文件名过长时的诊断文本。
54 QStringLiteral("组件 %1 图片 %2 的嵌入文件名超过 255 字节，已跳过 Storage")
55 .arg(component.name)
56 .arg(imageIndex);
57 m_owner.m_diagnostics.append(diagnostic);
58 qWarning() << "AltiumSchLibWriter:" << diagnostic;
59 continue;
60 }
61 if (!AltiumWriterUtils::isValidImageStorageName(embeddedName)) {
62 const QString diagnostic = // 嵌入文件名不符合格式时的诊断文本。
63 QStringLiteral("组件 %1 图片 %2 的嵌入文件名无效: %3，已跳过 Storage")
64 .arg(component.name)
65 .arg(imageIndex)
66 .arg(image.fileName);
67 m_owner.m_diagnostics.append(diagnostic);
68 qWarning() << "AltiumSchLibWriter:" << diagnostic;
69 continue;
70 }
71
72 const QFileInfo fileInfo(embeddedName); // 当前嵌入文件名的信息对象。
73 const QString suffix = fileInfo.suffix(); // 文件扩展名。
74 const QString baseName = // 不含扩展名的基础文件名。
75 suffix.isEmpty() ? embeddedName : embeddedName.left(embeddedName.size() - suffix.size() - 1);
76 QString candidate = embeddedName; // 当前候选的唯一 Storage 文件名。
77 int duplicateIndex = 1; // 重名文件的递增后缀。
78 while (usedNames.contains(candidate.toCaseFolded())) {
79 ++duplicateIndex;
80 const QString suffixText = suffix.isEmpty() ? QString() : QStringLiteral(".") + suffix; // 扩展名文本。
81 const QString marker = QStringLiteral("_%1").arg(duplicateIndex); // 重名后缀。
82 QString trimmedBase = baseName; // 为长度限制准备的基础文件名副本。
83 while (!trimmedBase.isEmpty() && (trimmedBase + marker + suffixText).toLocal8Bit().size() > 255)
84 trimmedBase.chop(1);
85 candidate = trimmedBase + marker + suffixText;
86 }
87 usedNames.insert(candidate.toCaseFolded());
88 m_owner.m_embeddedImageNames.insert(&image, candidate);
89 if (candidate != embeddedName) {
90 const QString diagnostic = // 重名文件改名时的诊断文本。
91 QStringLiteral("组件 %1 图片 %2 的嵌入文件名 %3 重复，已改为 %4")
92 .arg(component.name)
93 .arg(imageIndex)
94 .arg(embeddedName)
95 .arg(candidate);
96 m_owner.m_diagnostics.append(diagnostic);
97 qWarning() << "AltiumSchLibWriter:" << diagnostic;
98 }
99 }
100 }
