151double Exporter3DModel::calculateObjMinZ(const QByteArray& objData) {
152 if (objData.isEmpty()) {
153 return std::numeric_limits<double>::max();
154 }
155
156 double minZ = std::numeric_limits<double>::max();
157 int start = 0;
158 while (start < objData.size()) {
159 int end = objData.indexOf('\n', start);
160 if (end < 0) {
161 end = objData.size();
162 }
163
164 // 检查行是否以 "v " 开头
165 if (end - start >= 2 && objData.at(start) == 'v' && objData.at(start + 1) == ' ') {
166 // 跳过 "v " 前缀，解析三个坐标分量
167 int pos = start + 2;
168 // 跳过前两个分量 (x, y)
169 int component = 0;
170 while (pos < end && component < 2) {
171 if (objData.at(pos) == ' ') {
172 component++;
173 // 跳过连续空格
174 while (pos < end && objData.at(pos) == ' ') {
175 pos++;
176 }
177 } else {
178 pos++;
179 }
180 }
181
182 // 解析 Z 分量
183 if (pos < end) {
184 bool ok = false;
185 double z = QByteArray(objData.constData() + pos, end - pos).toDouble(&ok);
186 if (ok) {
187 minZ = qMin(minZ, z / WRL_UNIT_TO_MM);
188 }
189 }
190 }
191
192 start = end + 1;
193 }
194 if (minZ == std::numeric_limits<double>::max()) {
195 return minZ;
196 }
197
198 return minZ;
199}

205bool Exporter3DModel::hasUsableObjGeometry(const QByteArray& objData) {
206 if (objData.isEmpty()) {
207 return false;
208 }
209
210 bool hasVertex = false;
211 bool hasFace = false;
212 const QStringList lines = QString::fromUtf8(objData).split('\n');
213 for (const QString& line : lines) {
214 const QStringList parts = line.trimmed().split(' ', Qt::SkipEmptyParts);
215 if (parts.isEmpty()) {
216 continue;
217 }
218
219 if (parts[0] == QStringLiteral("v") && parts.size() >= 4) {
220 bool validVertex = true;
221 for (int index = 1; index <= 3; ++index) {
222 bool ok = false;
223 const double coordinate = parts[index].toDouble(&ok);
224 validVertex = validVertex && ok && std::isfinite(coordinate);
225 }
226 hasVertex = hasVertex || validVertex;
227 } else if (parts[0] == QStringLiteral("f") && parts.size() >= 4) {
228 bool validFace = true;
229 for (int index = 1; index < parts.size(); ++index) {
230 const QString vertexIndex = parts[index].section('/', 0, 0);
231 bool ok = false;
232 const int parsedIndex = vertexIndex.toInt(&ok);
233 validFace = validFace && ok && parsedIndex != 0;
234 }
235 hasFace = hasFace || validFace;
236 }
237 }
238
239 return hasVertex && hasFace;
240}

246bool Exporter3DModel::hasUsableWrlGeometry(const QByteArray& wrlData) {
247 if (wrlData.isEmpty()) {
248 return false;
249 }
250
251 const QString text = QString::fromUtf8(wrlData);
252 if (!text.trimmed().startsWith(QStringLiteral("#VRML V2.0"))) {
253 return false;
254 }
255
256 static const QRegularExpression numberRegex(QStringLiteral("[-+]?(?:\\d+\\.?\\d*|\\.\\d+)(?:[eE][-+]?\\d+)?"));
257 const int pointStart = text.indexOf(QStringLiteral("point ["));
258 const int pointEnd = pointStart < 0 ? -1 : text.indexOf(QLatin1Char(']'), pointStart);
259 const int coordIndexStart = text.indexOf(QStringLiteral("coordIndex ["));
260 const int coordIndexEnd =
261 coordIndexStart < 0 ? -1 : text.indexOf(QLatin1Char(']'), coordIndexStart);
262 if (pointStart < 0 || pointEnd < 0 || coordIndexStart < 0 || coordIndexEnd < 0) {
263 return false;
264 }
265
266 const QString pointData =
267 text.mid(pointStart + QStringLiteral("point [").size(), pointEnd - pointStart);
268 const int coordinateCount = pointData.count(numberRegex);
269 const QString indexData =
270 text.mid(coordIndexStart + QStringLiteral("coordIndex [").size(), coordIndexEnd - coordIndexStart);
271 return coordinateCount >= 9 && indexData.contains(QStringLiteral("-1"));
272}

287double Exporter3DModel::calculateWrlDisplayMinZ(const QByteArray& wrlData) {
288 if (wrlData.isEmpty()) {
289 return std::numeric_limits<double>::max();
290 }
291
292 static const QRegularExpression numberRegex(QStringLiteral("[-+]?(?:\\d+\\.?\\d*|\\.\\d+)(?:[eE][-+]?\\d+)?"));
293
294 double minZ = std::numeric_limits<double>::max();
295 bool inPointArray = false;
296 int coordinateComponent = 0;
297 const QStringList lines = QString::fromUtf8(wrlData).split('\n');
298 for (const QString& line : lines) {
299 QString segment = line;
300 if (!inPointArray) {
301 const int pointStart = segment.indexOf(QStringLiteral("point ["));
302 if (pointStart < 0) {
303 continue;
304 }
305 inPointArray = true;
306 coordinateComponent = 0;
307 segment = segment.mid(pointStart + QStringLiteral("point [").size());
308 }
309
310 const int pointEnd = segment.indexOf(QLatin1Char(']'));
311 if (pointEnd >= 0) {
312 segment = segment.left(pointEnd);
313 }
314
315 QRegularExpressionMatchIterator matches = numberRegex.globalMatch(segment);
316 while (matches.hasNext()) {
317 const QRegularExpressionMatch match = matches.next();
318 bool ok = false;
319 const double value = match.captured(0).toDouble(&ok);
320 if (!ok) {
321 continue;
322 }
323 if (coordinateComponent % 3 == 2) {
324 minZ = qMin(minZ, value * WRL_UNIT_TO_MM);
325 }
326 coordinateComponent++;
327 }
328
329 if (pointEnd >= 0) {
330 inPointArray = false;
331 }
332 }
333
334 return minZ;
335}

349QString Exporter3DModel::generateWrlContent(const IR::Model3DIR& model, const QByteArray& objData) {
350 QString content;
351
352 // 解析 OBJ 数据
353 QJsonObject parsedData = parseObjData(objData);
354 QJsonArray vertices = parsedData["vertices"].toArray();
355 QJsonArray faces = parsedData["faces"].toArray();
356 QJsonObject materials = parsedData["materials"].toObject();
357 QJsonArray shapes = parsedData["shapes"].toArray();
358
359 // WRL 文件头部
360 content += "#VRML V2.0 utf8\n";
361 content += "# 3D model generated by EasyKiConverter (https://github.com/EasyKiconverter/EasyKiConverter)\n";
362 content += "\n";
363
364 // 如果有形状数据，使用形状数据生成 WRL
365 if (!shapes.isEmpty()) {
366 for (const QJsonValue& shapeValue : shapes) {
367 QJsonObject shape = shapeValue.toObject();
368 QString materialId = shape["materialId"].toString();
369 QJsonArray shapePoints = shape["points"].toArray();
370 QJsonArray coordIndex = shape["coordIndex"].toArray();
371
372 // 在倒数第二个位置插入最后一个点的副本
373 if (shapePoints.size() > 0) {
374 shapePoints.insert(shapePoints.size() - 1, shapePoints.last());
375 }
376
377 // 获取材质信息
378 QJsonObject material = materials[materialId].toObject();
379 QJsonArray diffuseColor = material["diffuseColor"].toArray();
380 QJsonArray specularColor = material["specularColor"].toArray();
381 QString transparency = material["transparency"].toString("0");
382
383 // 生成 Shape
384 content += "Shape {\n";
385 content += " appearance Appearance {\n";
386 content += " material Material {\n";
387 content += QString(" diffuseColor %1 %2 %3\n")
388 .arg(diffuseColor.isEmpty() ? "0.8" : diffuseColor[0].toString())
389 .arg(diffuseColor.isEmpty() ? "0.8" : diffuseColor[1].toString())
390 .arg(diffuseColor.isEmpty() ? "0.8" : diffuseColor[2].toString());
391 content += QString(" specularColor %1 %2 %3\n")
392 .arg(specularColor.isEmpty() ? "0.2" : specularColor[0].toString())
393 .arg(specularColor.isEmpty() ? "0.2" : specularColor[1].toString())
394 .arg(specularColor.isEmpty() ? "0.2" : specularColor[2].toString());
395 content += " ambientIntensity 0.2\n";
396 content += QString(" transparency %1\n").arg(transparency);
397 content += " shininess 0.5\n";
398 content += " }\n";
399 content += " }\n";
400 content += " geometry IndexedFaceSet {\n";
401 content += " ccw TRUE\n";
402 content += " solid FALSE\n";
403 content += " coord DEF co Coordinate {\n";
404 content += " point [\n";
405
406 for (const QJsonValue& pointValue : shapePoints) {
407 content += " " + pointValue.toString() + ",\n";
408 }
409
410 content += " ]\n";
411 content += " }\n";
412 content += " coordIndex [\n";
413
414 for (const QJsonValue& indexValue : coordIndex) {
415 QJsonArray faceIndices = indexValue.toArray();
416 QString indexStr;
417 for (const QJsonValue& idx : faceIndices) {
418 indexStr += QString::number(idx.toInt()) + " ";
419 }
420 content += " " + indexStr + "\n";
421 }
422
423 content += " ]\n";
424 content += " }\n";
425 content += "}\n";
426 }
427 } else {
428 // 如果没有形状数据，使用简单的 Transform 包装
429 IR::Model3DVec3 translation = model.translation();
430 IR::Model3DVec3 rotation = model.rotation();
431
432 content += "Transform {\n";
433 content += " translation ";
434 content += QString("%1 %2 %3\n").arg(translation.x).arg(translation.y).arg(translation.z);
435 content += " rotation ";
436 content += QString("%1 %2 %3 %4\n").arg(rotation.x).arg(rotation.y).arg(rotation.z).arg(1.0);
437 content += " scale 1 1 1\n";
438 content += " children [\n";
439 content += " Shape {\n";
440 content += " appearance Appearance {\n";
441 content += " material Material {\n";
442 content += " diffuseColor 0.8 0.8 0.8\n";
443 content += " specularColor 0.2 0.2 0.2\n";
444 content += " ambientIntensity 0.2\n";
445 content += " shininess 0.2\n";
446 content += " }\n";
447 content += " }\n";
448 content += " geometry IndexedFaceSet {\n";
449 content += " coord Coordinate {\n";
450 content += " point [\n";
451
452 for (const QJsonValue& vertexValue : vertices) {
453 QJsonArray vertex = vertexValue.toArray();
454 if (vertex.size() >= 3) {
455 content += QString(" %1 %2 %3\n")
456 .arg(vertex[0].toDouble())
457 .arg(vertex[1].toDouble())
458 .arg(vertex[2].toDouble());
459 }
460 }
461
462 content += " ]\n";
463 content += " }\n";
464 content += " coordIndex [\n";
465
466 for (const QJsonValue& faceValue : faces) {
467 QJsonArray face = faceValue.toArray();
468 QString faceStr;
469 for (const QJsonValue& indexValue : face) {
470 faceStr += QString::number(indexValue.toInt() - 1) + " ";
471 }
472 faceStr += "-1";
473 content += " " + faceStr + "\n";
474 }
475
476 content += " ]\n";
477 content += " }\n";
478 content += " }\n";
479 content += " ]\n";
480 content += "}\n";
481 }

577QJsonObject Exporter3DModel::parseObjData(const QByteArray& objData) {
578 QJsonObject result;
579 QJsonArray vertices;
580 QJsonArray faces;
581 QJsonObject materials;
582 QJsonArray shapes;
583
584 QString objString = QString::fromUtf8(objData);
585 QStringList lines = objString.split('\n');
586
587 // 第一遍：提取材质信息
588 QString currentMaterialId;
589 QJsonObject currentMaterial;
590 bool inMaterial = false;
591
592 for (const QString& line : lines) {
593 QString trimmedLine = line.trimmed();
594 if (trimmedLine.isEmpty() || trimmedLine.startsWith('#')) {
595 continue;
596 }
597
598 QStringList parts = trimmedLine.split(' ', Qt::SkipEmptyParts);
599 if (parts.isEmpty()) {
600 continue;
601 }
602
603 if (parts[0] == "newmtl") {
604 if (!currentMaterialId.isEmpty() && !currentMaterial.isEmpty()) {
605 materials[currentMaterialId] = currentMaterial;
606 }
607 currentMaterialId = parts[1];
608 currentMaterial = QJsonObject();
609 inMaterial = true;
610 } else if (inMaterial) {
611 if (parts[0] == "endmtl") {
612 if (!currentMaterialId.isEmpty() && !currentMaterial.isEmpty()) {
613 materials[currentMaterialId] = currentMaterial;
614 }
615 currentMaterialId.clear();
616 currentMaterial = QJsonObject();
617 inMaterial = false;
618 } else if (parts[0] == "Ka") {
619 QJsonArray ambientColor;
620 for (int i = 1; i < parts.size(); ++i) {
621 ambientColor.append(parts[i]);
622 }
623 currentMaterial["ambientColor"] = ambientColor;
624 } else if (parts[0] == "Kd") {
625 QJsonArray diffuseColor;
626 for (int i = 1; i < parts.size(); ++i) {
627 diffuseColor.append(parts[i]);
628 }
629 currentMaterial["diffuseColor"] = diffuseColor;
630 } else if (parts[0] == "Ks") {
631 QJsonArray specularColor;
632 for (int i = 1; i < parts.size(); ++i) {
633 specularColor.append(parts[i]);
634 }
635 currentMaterial["specularColor"] = specularColor;
636 } else if (parts[0] == "d") {
637 currentMaterial["transparency"] = parts[1];
638 }
639 }
640 }
641
642 if (!currentMaterialId.isEmpty() && !currentMaterial.isEmpty()) {
643 materials[currentMaterialId] = currentMaterial;
644 }
645
646 // 第二遍：提取顶点数据
647 QVector<IR::Model3DVec3> convertedVertices;
648 double minX = std::numeric_limits<double>::max();
649 double maxX = std::numeric_limits<double>::lowest();
650 double minY = std::numeric_limits<double>::max();
651 double maxY = std::numeric_limits<double>::lowest();
652 double minZ = std::numeric_limits<double>::max();
653 for (const QString& line : lines) {
654 QString trimmedLine = line.trimmed();
655 if (trimmedLine.isEmpty() || trimmedLine.startsWith('#')) {
656 continue;
657 }
658
659 QStringList parts = trimmedLine.split(' ', Qt::SkipEmptyParts);
660 if (parts.isEmpty()) {
661 continue;
662 }
663
664 if (parts[0] == 'v') {
665 if (parts.size() >= 4) {
666 double x = parts[1].toDouble() / WRL_UNIT_TO_MM;
667 double y = parts[2].toDouble() / WRL_UNIT_TO_MM;
668 double z = parts[3].toDouble() / WRL_UNIT_TO_MM;
669 convertedVertices.append(IR::Model3DVec3(x, y, z));
670 minX = qMin(minX, x);
671 maxX = qMax(maxX, x);
672 minY = qMin(minY, y);
673 maxY = qMax(maxY, y);
674 minZ = qMin(minZ, z);
675 }
676 }
677 }
678
679 const bool hasVertices = !convertedVertices.isEmpty();
680 const double normalizeX = hasVertices ? (minX + maxX) / 2.0 : 0.0;
681 const double normalizeY = hasVertices ? (minY + maxY) / 2.0 : 0.0;
682 const double normalizeZ = hasVertices && minZ > 0.0 ? minZ : 0.0;
683
684 QStringList vertexStrings;
685 for (IR::Model3DVec3 vertexData : convertedVertices) {
686 vertexData.x -= normalizeX;
687 vertexData.y -= normalizeY;
688 vertexData.z -= normalizeZ;
689
690 QString vertexStr =
691 QString("%1 %2 %3").arg(vertexData.x, 0, 'f', 4).arg(vertexData.y, 0, 'f', 4).arg(vertexData.z, 0, 'f', 4);
692 vertexStrings.append(vertexStr);
693
694 QJsonArray vertex;
695 vertex.append(vertexData.x);
696 vertex.append(vertexData.y);
697 vertex.append(vertexData.z);
698 vertices.append(vertex);
699 }
700
701 // 第三遍：按材质分割形状并提取面数
702 QString currentShapeMaterial = "default";
703 QStringList currentShapePoints;
704 QJsonArray currentShapeCoordIndex;
705 QMap<int, int> vertexIndexMap;
706 int shapeVertexCounter = 0;
707
708 for (const QString& line : lines) {
709 QString trimmedLine = line.trimmed();
710 if (trimmedLine.isEmpty() || trimmedLine.startsWith('#')) {
711 continue;
712 }
713
714 QStringList parts = trimmedLine.split(' ', Qt::SkipEmptyParts);
715 if (parts.isEmpty()) {
716 continue;
717 }
718
719 if (parts[0] == "usemtl") {
720 if (!currentShapePoints.isEmpty()) {
721 QJsonObject shape;
722 shape["materialId"] = currentShapeMaterial;
723 shape["points"] = QJsonArray::fromStringList(currentShapePoints);
724 shape["coordIndex"] = currentShapeCoordIndex;
725 shapes.append(shape);
726 }
727 currentShapeMaterial = parts[1];
728 currentShapePoints.clear();
729 currentShapeCoordIndex = QJsonArray();
730 vertexIndexMap.clear();
731 shapeVertexCounter = 0;
732 } else if (parts[0] == 'f') {
733 QJsonArray faceIndices;
734 for (int i = 1; i < parts.size(); ++i) {
735 QString vertexIndexStr = parts[i].split('/')[0];
736 int vertexIndex = vertexIndexStr.toInt() - 1;
737
738 if (!vertexIndexMap.contains(vertexIndex)) {
739 if (vertexIndex >= 0 && vertexIndex < vertexStrings.size()) {
740 currentShapePoints.append(vertexStrings[vertexIndex]);
741 } else {
742 currentShapePoints.append("0 0 0");
743 }
744 vertexIndexMap[vertexIndex] = shapeVertexCounter;
745 faceIndices.append(shapeVertexCounter);
746 shapeVertexCounter++;
747 } else {
748 faceIndices.append(vertexIndexMap[vertexIndex]);
749 }
750 }
751 faceIndices.append(-1);
752 currentShapeCoordIndex.append(QJsonArray(faceIndices));
753 }
754 }
755
756 if (!currentShapePoints.isEmpty()) {
757 QJsonObject shape;
758 shape["materialId"] = currentShapeMaterial;
759 shape["points"] = QJsonArray::fromStringList(currentShapePoints);
760 shape["coordIndex"] = currentShapeCoordIndex;
761 shapes.append(shape);
762 }
763
764 result["vertices"] = vertices;
765 result["faces"] = faces;
766 result["materials"] = materials;
767 result["shapes"] = shapes;
768
769 return result;
770}
