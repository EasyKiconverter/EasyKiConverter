20inline bool isValidImageStorageName(const QString& name) {
21 if (name.trimmed() != name || name.isEmpty() || name == QStringLiteral(".") || name == QStringLiteral("..") ||
22 name.size() > 255)
23 return false;
24 if (name.contains(QChar::Null) || name.contains(QChar('\n')) || name.contains(QChar('\r')) ||
25 name.contains(QChar('\t')))
26 return false;
27 for (const QChar ch : name) {
28 if (ch.unicode() < 0x20 || QStringLiteral("<>:\"/\\|?*").contains(ch))
29 return false;
30 }
31 if (name.endsWith(QChar(' ')) || name.endsWith(QChar('.')))
32 return false;
33 if (name.contains(QChar(0x200B)) || name.contains(QChar(0x200C)) || name.contains(QChar(0x200D)) ||
34 name.contains(QChar(0xFEFF)))
35 return false;
36
37 QString deviceName = name; // 去除扩展名后用于保留设备名检查的字符串。
38 const qsizetype extensionStart = deviceName.indexOf(QChar('.')); // 第一个扩展名分隔点。
39 if (extensionStart >= 0)
40 deviceName.truncate(extensionStart);
41 static const QStringList reservedNames = {
42 // Windows 保留设备名列表。
43 QStringLiteral("CON"), QStringLiteral("PRN"), QStringLiteral("AUX"), QStringLiteral("NUL"),
44 QStringLiteral("COM1"), QStringLiteral("COM2"), QStringLiteral("COM3"), QStringLiteral("COM4"),
45 QStringLiteral("COM5"), QStringLiteral("COM6"), QStringLiteral("COM7"), QStringLiteral("COM8"),
46 QStringLiteral("COM9"), QStringLiteral("LPT1"), QStringLiteral("LPT2"), QStringLiteral("LPT3"),
47 QStringLiteral("LPT4"), QStringLiteral("LPT5"), QStringLiteral("LPT6"), QStringLiteral("LPT7"),
48 QStringLiteral("LPT8"), QStringLiteral("LPT9")};
49 if (reservedNames.contains(deviceName.trimmed(), Qt::CaseInsensitive))
50 return false;
51
52 const QByteArray encodedName = name.toLocal8Bit(); // 当前平台本地编码后的文件名。
53 return encodedName.size() <= 255 && QString::fromLocal8Bit(encodedName) == name;
54}

70inline QString getSectionKey(const QString& name) {
71 QString key; // 生成中的 OLE 存储键。
72 const QString trimmed = name.trimmed(); // 去除首尾空白后的元件名称。
73 for (const QChar ch : trimmed) {
74 if (key.size() >= 31) {
75 break;
76 }
77 const bool asciiAlphaNumeric = ch.unicode() < 128 && ch.isLetterOrNumber(); // 是否为可保留的 ASCII 字符。
78 key += asciiAlphaNumeric || ch == '_' || ch == '-' || ch == '.' ? ch : QChar('_');
79 }
80 if (key.isEmpty()) {
81 key = "_";
82 }
83 return key;
84}

91inline QStringList makeUniqueSectionKeys(const QStringList& names) {
92 QStringList result; // 生成后的唯一存储键列表。
93 QSet<QString> used; // 已使用的大小写折叠键集合。
94 for (const QString& name : names) {
95 const QString base = getSectionKey(name); // 当前名称的基础存储键。
96 QString candidate = base; // 当前尝试使用的存储键。
97 int suffixNumber = 2; // 冲突后缀起始编号。
98 while (used.contains(candidate.toCaseFolded())) {
99 const QString suffix = QString("_%1").arg(suffixNumber++); // 当前冲突后缀。
100 candidate = base.left(qMax(1, 31 - suffix.size())) + suffix;
101 }
102 used.insert(candidate.toCaseFolded());
103 result.append(candidate);
104 }
105 return result;
106}
