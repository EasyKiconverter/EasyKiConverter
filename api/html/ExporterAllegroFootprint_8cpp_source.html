26QString safeName(const QString& value, const QString& fallback) {
27 QString name = value.trimmed();
28 name.replace(QRegularExpression(QStringLiteral("[^A-Za-z0-9._-]")), QStringLiteral("_"));
29 while (name.contains(QStringLiteral("..")))
30 name.replace(QStringLiteral(".."), QStringLiteral("_"));
31 if (name.isEmpty() || name == QStringLiteral(".") || name == QStringLiteral("_"))
32 name = fallback;
33 return name.left(120);
34}

126QString canonicalPadKey(const IR::FootprintPadIR& pad) {
127 // 规范键包含影响制造语义的全部 Pad 属性。
128 QString key = padTypeName(pad.padType) + QLatin1Char('|') + shapeName(pad.shape) + QLatin1Char('|') +
129 number(pad.size.width()) + QLatin1Char('|') + number(pad.size.height()) + QLatin1Char('|') +
130 number(pad.holeSize) + QLatin1Char('|') + number(pad.holeLength) + QLatin1Char('|') +
131 number(pad.rotation) + QLatin1Char('|') + QString::number(static_cast<int>(pad.layer)) +
132 QLatin1Char('|') + (pad.isPlated ? QStringLiteral("plated") : QStringLiteral("non-plated"));
133 for (const QPointF& point : pad.customShapePoints)
134 key += QLatin1Char('|') + number(point.x()) + QLatin1Char(',') + number(point.y());
135 return key;
136}

628bool writeAllegroSymbolFiles(const QList<IR::SymbolComponentIR>& symbols,
629 const QString& packageDir,
630 QJsonArray& symbolEntries,
631 QStringList& diagnostics) {
632 const QString symbolDir = packageDir + QDir::separator() + QStringLiteral("symbols");
633 if (!QDir().mkpath(symbolDir)) {
634 diagnostics.append(QStringLiteral("Allegro: 无法创建符号规范化目录"));
635 return false;
636 }
637 QSet<QString> names;
638 for (const IR::SymbolComponentIR& symbol : symbols) {
639 const QString name = safeName(symbol.name, QStringLiteral("symbol"));
640 if (names.contains(name)) {
641 diagnostics.append(QStringLiteral("Allegro: 符号名称清洗后冲突：%1").arg(name));
642 return false;
643 }
644 names.insert(name);
645 const QString relativePath = QStringLiteral("symbols/%1.json").arg(name);
646 if (!writeJson(packageDir + QDir::separator() + relativePath, symbolObject(symbol))) {
647 diagnostics.append(QStringLiteral("Allegro: 无法写入符号规范化数据：%1").arg(name));
648 return false;
649 }
650 symbolEntries.append(QJsonObject{{QStringLiteral("name"), name},
651 {QStringLiteral("path"), relativePath},
652 {QStringLiteral("part_count"), symbol.partCount},
653 {QStringLiteral("pin_count"), symbol.pins.size()}});
654
655 if (!symbol.paths.isEmpty() || !symbol.beziers.isEmpty() || !symbol.images.isEmpty())
656 diagnostics.append(
657 QStringLiteral("Allegro: 符号 %1 的部分高级图元仅保留图元计数，需在导入环境复核").arg(name));
658 }
659 return true;

685 QStringLiteral("Allegro: 组件 %1 存在重复符号引脚编号 %2").arg(component.name, pinNumber));

692 }

694 pinMappings.append(QJsonObject{{QStringLiteral("pin"), pinNumber}, {QStringLiteral("pad"), pinNumber}});
695 }
696 entries.append(QJsonObject{{QStringLiteral("component"), component.name},
697 {QStringLiteral("symbol"), safeName(component.symbol.name, component.name)},
698 {QStringLiteral("footprint"), safeName(component.footprint.name, component.name)},

745bool ExporterAllegroFootprint::exportFootprintLibrary(const QList<IR::FootprintComponentIR>& footprints,
746 const QString& libName,
747 const QString& filePath,
748 bool preferWrl,
749 bool exportStep,
750 const QString& libraryDescription,
751 const QString& libraryKeywords,
752 bool useAbsolutePaths,
753 const QString& model3DBaseDir) {
754 Q_UNUSED(preferWrl)
755 Q_UNUSED(useAbsolutePaths)
756 Q_UNUSED(model3DBaseDir)
757 m_diagnostics.clear();
758 if (footprints.isEmpty()) {
759 m_diagnostics.append(QStringLiteral("Allegro: 没有可导出的封装"));
760 return false;
761 }
762 if (!QDir().mkpath(filePath)) {
763 m_diagnostics.append(QStringLiteral("Allegro: 无法创建 Import Package 目录 %1").arg(filePath));
764 return false;
765 }
766 const QString normalizedDir =
767 filePath + QDir::separator() + QStringLiteral("normalized-data");
768 const QString padstackDir = filePath + QDir::separator() + QStringLiteral("padstacks");
769 const QString shapeDir = filePath + QDir::separator() + QStringLiteral("shapes");
770 const QString modelDir = filePath + QDir::separator() + QStringLiteral("models");
771 if (!QDir().mkpath(normalizedDir) || !QDir().mkpath(padstackDir) || !QDir().mkpath(shapeDir) ||
772 !QDir().mkpath(modelDir)) {
773 m_diagnostics.append(QStringLiteral("Allegro: 无法创建 Import Package 子目录"));
774 return false;
775 }
776
777 QJsonArray packages;
778 QSet<QString> packageNames;
779 QSet<QString> padstackFiles;
780 for (const IR::FootprintComponentIR& footprint : footprints) {
781 AllegroPackageModel package;
782 if (!buildPackage(footprint, package, m_diagnostics))
783 return false;
784 if (!exportStep) {
785 // 未启用 STEP 输出时不能在规范化数据中留下不存在的模型引用。
786 package.stepFiles.clear();
787 package.stepTransforms = QJsonArray();
788 }
789 if (packageNames.contains(package.name)) {
790 m_diagnostics.append(QStringLiteral("Allegro: 封装名称冲突，清理后仍为 %1").arg(package.name));
791 return false;
792 }
793 packageNames.insert(package.name);
794 const QString dataPath =
795 normalizedDir + QDir::separator() + package.name + QStringLiteral(".json");
796 if (!writeJson(dataPath, packageObject(package))) {
797 m_diagnostics.append(QStringLiteral("Allegro: 无法写入规范化封装数据 %1").arg(dataPath));
798 return false;
799 }
800 for (const AllegroPadstackModel& padstack : package.padstacks) {
801 if (padstackFiles.contains(padstack.name))
802 continue;
803 padstackFiles.insert(padstack.name);
804 if (!writeJson(padstackDir + QDir::separator() + padstack.name + QStringLiteral(".json"),
805 padstackObject(padstack))) {
806 m_diagnostics.append(QStringLiteral("Allegro: 无法写入 Padstack %1").arg(padstack.name));
807 return false;
808 }
809 if (padstack.shape == IR::PadShape::Polygon || padstack.shape == IR::PadShape::Trapezoid ||
810 padstack.shape == IR::PadShape::RoundRect) {
811 if (!writeJson(shapeDir + QDir::separator() + padstack.name + QStringLiteral(".json"),
812 QJsonObject{{QStringLiteral("padstack"), padstack.name},
813 {QStringLiteral("shape"), shapeName(padstack.shape)},
814 {QStringLiteral("points_mm"), pointArray(padstack.polygon)}})) {
815 m_diagnostics.append(QStringLiteral("Allegro: 无法写入 Shape 依赖 %1").arg(padstack.name));
816 return false;
817 }
818 }
819 }
820 int stepIndex = 0;
821 for (const IR::Model3DIR& model : footprint.models3d) {
822 if (!model.hasStepData())
823 continue;
824 if (stepIndex >= package.stepFiles.size())
825 break;
826 const QString modelPath =
827 modelDir + QDir::separator() + package.name + QLatin1Char('_') + package.stepFiles.at(stepIndex++);
828 if (exportStep && !writeText(modelPath, model.stepData())) {
829 m_diagnostics.append(QStringLiteral("Allegro: 无法写入 STEP 模型 %1").arg(modelPath));
830 return false;
831 }
832 }
833 packages.append(QJsonObject{
834 {QStringLiteral("name"), package.name},
835 {QStringLiteral("normalized_data"), QStringLiteral("normalized-data/%1.json").arg(package.name)},
836 {QStringLiteral("padstack_count"), package.padstacks.size()},
837 {QStringLiteral("pin_count"), package.pins.size()},
838 {QStringLiteral("step_count"), package.stepFiles.size()}});
839 }
840
841 const QJsonObject manifest{
842 {QStringLiteral("schema"), QStringLiteral("easykiconverter.allegro.import-package.v1")},
843 {QStringLiteral("target"), QStringLiteral("Allegro PCB Footprint")},
844 {QStringLiteral("library_name"), safeName(libName, QStringLiteral("EasyKiConverter"))},
845 {QStringLiteral("generation_mode"), QStringLiteral("import-package")},
846 {QStringLiteral("native_database_generated"), false},
847 {QStringLiteral("allegro_version"), QStringLiteral("未锁定，请使用目标环境验证")},
848 {QStringLiteral("description"), libraryDescription},
849 {QStringLiteral("keywords"), libraryKeywords},
850 {QStringLiteral("packages"), packages},
851 {QStringLiteral("required_directories"), QJsonArray{"normalized-data", "padstacks", "shapes", "models"}},
852 {QStringLiteral("entrypoint"), QStringLiteral("generator.il")},
853 {QStringLiteral("native_outputs"), QJsonArray{".dra", ".psm", ".pad"}}};
854 if (!writeJson(filePath + QDir::separator() + QStringLiteral("manifest.json"), manifest)) {
855 m_diagnostics.append(QStringLiteral("Allegro: 无法写入 manifest.json"));
856 return false;
857 }
858 const QByteArray generator =
859 ";; EasyKiConverter Allegro Import Package validator\n"
860 ";; This file does not write Cadence private databases.\n"
861 ";; Validate manifest.json, then use the documented Allegro import flow.\n"
862 "procedure(ekcImportPackage(manifestPath)\n"
863 " printf(\"EasyKiConverter package: %s\\n\" manifestPath)\n"
864 " printf(\"Use the supported Allegro environment to generate .dra/.psm/.pad.\\n\")\n"
865 ")\n";
866 if (!writeText(filePath + QDir::separator() + QStringLiteral("generator.il"), generator) ||
867 !writeText(
868 filePath + QDir::separator() + QStringLiteral("README_ALLEGRO.md"),
869 QByteArray(
870 "# Allegro Import Package\n\n"
871 "This package contains normalized footprint data. It does not contain native Cadence databases.\n\n"
872 "1. Validate manifest.json and configure PSMPATH, PADPATH and steppath.\n"
873 "2. Load generator.il in the supported Cadence Allegro environment.\n"
874 "3. Use the target Allegro import procedure to generate .dra, .psm and .pad files.\n\n"
875 "The package was not validated by a locally installed Allegro executable.\n"))) {
876 m_diagnostics.append(QStringLiteral("Allegro: 无法写入 Import Package 使用说明"));
877 return false;
878 }
879 return true;

967bool ExporterAllegroFootprint::exportComponentLibrary(const QList<IR::ComponentIR>& components,
968 const QString& libName,
969 const QString& filePath,
970 bool exportModel3D,
971 const QString& model3DBaseDir) {
972 if (components.isEmpty()) {
973 m_diagnostics = {QStringLiteral("Allegro: 没有可导出的完整组件")};
974 return false;
975 }
976 QList<IR::FootprintComponentIR> footprints;
977 QList<IR::SymbolComponentIR> symbols;
978 footprints.reserve(components.size());
979 symbols.reserve(components.size());
980 for (const IR::ComponentIR& component : components) {
981 if (!component.hasSymbol() || !component.hasFootprint()) {
982 m_diagnostics = {QStringLiteral("Allegro: 组件 %1 缺少符号或封装，无法建立完整关联").arg(component.name)};
983 return false;
984 }
985 footprints.append(component.footprint);
986 symbols.append(component.symbol);
987 }
988
989 if (!exportFootprintLibrary(
990 footprints, libName, filePath, false, exportModel3D, QString(), QString(), false, model3DBaseDir))
991 return false;
992
993 QJsonArray symbolEntries;
994 if (!writeAllegroSymbolFiles(symbols, filePath, symbolEntries, m_diagnostics))
995 return false;
996
997 QFile manifestFile(filePath + QDir::separator() + QStringLiteral("manifest.json"));
998 if (!manifestFile.open(QIODevice::ReadOnly)) {
999 m_diagnostics.append(QStringLiteral("Allegro: 无法读取已生成的 manifest.json"));
1000 return false;
1001 }
1002 QJsonParseError parseError;
1003 QJsonDocument document =
1004 QJsonDocument::fromJson(manifestFile.readAll(), &parseError);
1005 if (parseError.error != QJsonParseError::NoError || !document.isObject()) {
1006 m_diagnostics.append(QStringLiteral("Allegro: 已生成的 manifest.json 无法解析"));
1007 return false;
1008 }
1009 QJsonObject manifest = document.object();
1010 manifest.insert(QStringLiteral("target"), QStringLiteral("Allegro semantic symbol and PCB package"));
1011 manifest.insert(QStringLiteral("symbols"), symbolEntries);
1012 QJsonArray componentRelations;
1013 if (!componentEntries(components, componentRelations, m_diagnostics))
1014 return false;
1015 manifest.insert(QStringLiteral("components"), componentRelations);
1016 manifest.insert(QStringLiteral("required_directories"),
1017 QJsonArray{"normalized-data", "padstacks", "shapes", "models", "symbols"});
1018 if (!writeJson(filePath + QDir::separator() + QStringLiteral("manifest.json"), manifest) ||
1019 !writeText(filePath + QDir::separator() + QStringLiteral("README_ALLEGRO.md"),
1020 QByteArray("# Allegro semantic symbol and PCB package\n\n"
1021 "This package contains normalized symbol, footprint, pin-to-pad, and STEP data.\n"
1022 "It does not contain native Cadence schematic or PCB databases.\n"))) {
1023 m_diagnostics.append(QStringLiteral("Allegro: 无法更新包含符号关联的 Import Package 文件"));
1024 return false;
1025 }
1026 return true;
