65void Model3DExportWorker::run() {
66 if (m_cancelled.load()) {
67 emit completed(m_componentId, false, QStringLiteral("Cancelled"));
68 return;
69 }
70 const uint64_t gen = m_cacheGeneration;
71
72 if (!m_data) {
73 emit completed(m_componentId, false, QStringLiteral("No data available"));
74 return;
75 }
76
77 qDebug() << "Model3DExportWorker: Exporting" << m_componentId;
78
79 Model3DData sourceModel;
80 if (m_data->model3DData())
81 sourceModel = *m_data->model3DData();
82 if (sourceModel.uuid().isEmpty() && m_data->footprintData()) {
83 const Model3DData footprintModel = m_data->footprintData()->model3D();
84 if (!footprintModel.uuid().isEmpty())
85 sourceModel = footprintModel;
86 }
87
88 QString uuid = sourceModel.uuid();
89 if (uuid.isEmpty()) {
90 QSharedPointer<ComponentData> cachedComponent = // 从磁盘缓存读取的元器件数据。
91 ComponentCacheService::instance()->loadComponentData(m_componentId);
92 if (cachedComponent) {
93 if (cachedComponent->model3DData()) {
94 sourceModel = *cachedComponent->model3DData();
95 }
96 if (sourceModel.uuid().isEmpty() && cachedComponent->footprintData()) {
97 sourceModel = cachedComponent->footprintData()->model3D();
98 }
99 uuid = sourceModel.uuid();
100 }
101 }
102 if (uuid.isEmpty()) {
103 qDebug() << "Model3DExportWorker: 3D model UUID not available, skipping" << m_componentId;
104 emit completed(m_componentId, true, QStringLiteral("3D model UUID not available, skipped"));
105 return;
106 }
107
108 // 根据选项决定是否导出 WRL 和 STEP
109 const bool needWrl = m_options.needsModel3DWrl();
110 const bool needStep = m_options.needsModel3DStep();
111
112 qInfo() << "Model3DExportWorker::run() for" << m_componentId
113 << "exportModel3DFormat:" << m_options.exportModel3DFormat << "needWrl:" << needWrl
114 << "needStep:" << needStep;
115
116 // 如果两种格式都不需要导出，直接完成
117 if (!needWrl && !needStep) {
118 qDebug() << "Model3DExportWorker: No format selected, skipping" << m_componentId;
119 emit completed(m_componentId, true, QString());
120 return;
121 }
122
123 // [NOTE] 直接使用 Exporter3DModel 而非 ExporterFactory，因为 Worker 依赖其
124 // downloadObjDataSync/downloadStepDataSync 等 EasyEDA 特有方法，
125 // 这些方法不在 IModel3DExporter 通用接口中。
126 Exporter3DModel exporter;
127
128 // 获取模型名称用于命名文件（仅在需要时）
129 QString modelName = sourceModel.name();
130 if (modelName.isEmpty()) {
131 modelName = m_componentId;
132 } else {
133 // 使用 PathSecurity 清理文件名中的非法字符
134 modelName = PathSecurity::sanitizeFilename(modelName);
135 }
136
137 // 构建输出路径
138 QString outputDir = m_options.outputPath;
139 if (outputDir.isEmpty()) {
140 outputDir =
141 QDir(QDir(QDir::currentPath()).filePath(QStringLiteral("export"))).filePath(QStringLiteral("3dmodels"));
142 }
143
144 QDir dir;
145 if (!dir.mkpath(outputDir)) {
146 emit completed(m_componentId, false, QStringLiteral("Failed to create output directory"));
147 return;
148 }
149
150 const QDir outputDirectory(outputDir);
151 const QString wrlFinalPath =
152 needWrl ? outputDirectory.filePath(modelName + QStringLiteral(".wrl")) : QString();
153 const QString stepFinalPath =
154 needStep ? outputDirectory.filePath(modelName + QStringLiteral(".step")) : QString();
155 const QString wrlWritePath =
156 m_outputPaths.wrlTempPath.isEmpty() ? wrlFinalPath : m_outputPaths.wrlTempPath;
157 const QString stepWritePath =
158 m_outputPaths.stepTempPath.isEmpty() ? stepFinalPath : m_outputPaths.stepTempPath;
159
160 const auto ensureParentDir = [](const QString& filePath) {
161 if (filePath.isEmpty())
162 return true;
163 QFileInfo fileInfo(filePath);
164 return fileInfo.absoluteDir().exists() || QDir().mkpath(fileInfo.absolutePath());
165 };
166
167 if ((needWrl && !ensureParentDir(wrlWritePath)) || (needStep && !ensureParentDir(stepWritePath))) {
168 emit completed(m_componentId, false, QStringLiteral("Failed to create 3D model output directory"));
169 return;
170 }
171
172 if (!m_options.overwriteExistingFiles && m_outputPaths.wrlTempPath.isEmpty() &&
173 m_outputPaths.stepTempPath.isEmpty()) {
174 bool bothExist = true;
175 if (needWrl && !QFile::exists(wrlFinalPath))
176 bothExist = false;
177 if (needStep && !QFile::exists(stepFinalPath))
178 bothExist = false;
179 if (bothExist && (needWrl || needStep)) {
180 qDebug() << "Model3DExportWorker: Files already exist, skipping" << wrlFinalPath << stepFinalPath;
181 emit completed(m_componentId, true, QString());
182 return;
183 }
184 }
185
186 ComponentCacheService* cache = ComponentCacheService::instance();
187
188 // WRL 是由 OBJ 转换得到的派生文件；有 OBJ 时重新生成，避免旧 WRL 缓存携带过期归一化。
189 // 只有在没有 OBJ 且不能下载时才复用 WRL 缓存，封装阶段会按该 WRL 的实际几何计算 STEP 偏移。
190 // STEP 文件保持服务器原始坐标系，不使用派生缓存。
191
192 auto buildModelData = [&uuid, &sourceModel]() {
193 Model3DData modelData;
194 modelData.setUuid(uuid);
195 modelData.setName(sourceModel.name());
196 modelData.setTranslation(sourceModel.translation());
197 modelData.setRotation(sourceModel.rotation());
198 return modelData;
199 };
200
201 QString error;
202 // 只在需要 WRL 时导出
203 if (needWrl) {
204 QByteArray objData;
205 bool usedWrlCache = false;
206 if (m_data && !m_data->model3DObjRaw().isEmpty()) {
207 objData = m_data->model3DObjRaw();
208 }
209 if (!objData.isEmpty() && !Exporter3DModel::hasUsableObjGeometry(objData)) {
210 qWarning() << "Model3DExportWorker: Ignoring malformed OBJ data for" << m_componentId << "uuid" << uuid;
211 objData.clear();
212 }
213 if (!objData.isEmpty()) {
214 // 预加载阶段已经取得有效 OBJ 时也写入公共缓存，避免后续导出重复下载。
215 cache->saveModel3D(uuid, objData, QStringLiteral("obj"), gen);
216 }
217 if (objData.isEmpty()) {
218 QMutexLocker downloadLocker(&modelDownloadMutex());
219 objData = cache->loadModel3D(uuid, QStringLiteral("obj"));
220 if (!objData.isEmpty() && !Exporter3DModel::hasUsableObjGeometry(objData)) {
221 qWarning() << "Model3DExportWorker: Ignoring malformed cached OBJ for" << m_componentId << "uuid"
222 << uuid;
223 objData.clear();
224 }
225 const QByteArray cachedWrl = cache->loadModel3D(uuid, QStringLiteral("wrl"));
226 if (objData.isEmpty() && !cachedWrl.isEmpty() && Exporter3DModel::hasUsableWrlGeometry(cachedWrl) &&
227 cache->copyModel3DToFile(uuid, QStringLiteral("wrl"), wrlWritePath)) {
228 usedWrlCache = true;
229 qDebug() << "Model3DExportWorker: WRL cache fallback for" << uuid;
230 } else if (objData.isEmpty() && !cachedWrl.isEmpty()) {
231 qWarning() << "Model3DExportWorker: Ignoring malformed cached WRL for" << m_componentId << "uuid"
232 << uuid;
233 }
234 if (objData.isEmpty() && !usedWrlCache && !exporter.downloadObjDataSync(uuid, &objData, &error)) {
235 if (error.isEmpty()) {
236 error = QStringLiteral("Failed to download OBJ data for WRL export");
237 }
238 }
239 if (!objData.isEmpty() && !Exporter3DModel::hasUsableObjGeometry(objData)) {
240 qWarning() << "Model3DExportWorker: Rejecting invalid downloaded OBJ for" << m_componentId << "uuid"
241 << uuid;
242 objData.clear();
243 error = QStringLiteral("Downloaded OBJ data has no usable geometry");
244 }
245 if (!objData.isEmpty()) {
246 cache->saveModel3D(uuid, objData, QStringLiteral("obj"), gen);
247 }
248 }
249 if (usedWrlCache) {
250 // 已复制缓存中的 WRL，无需再次转换 OBJ。
251 } else if (objData.isEmpty()) {
252 error = QStringLiteral("OBJ data is empty for WRL export");
253 } else {
254 Model3DData modelData = buildModelData();
255 modelData.setRawObj(QString::fromUtf8(objData));
256 IR::Model3DIR modelIR = IR::toModel3DIR(modelData);
257 if (!exporter.exportToWrl(modelIR, wrlWritePath)) {
258 error = QStringLiteral("Failed to convert OBJ to WRL");
259 } else {
260 QFile file(wrlWritePath);
261 if (file.open(QIODevice::ReadOnly)) {
262 cache->saveModel3D(uuid, file.readAll(), QStringLiteral("wrl"), gen);
263 qDebug() << "Model3DExportWorker: Saved WRL to cache for" << m_componentId << "uuid" << uuid;
264 }
265 }
266 }
267 }
268 // STEP 文件保持服务器原始坐标系，优先使用磁盘缓存
269 if (error.isEmpty() && needStep && !m_cancelled.load()) {
270 QByteArray stepData = sourceModel.step();
271 if (!stepData.isEmpty() && !Exporter3DModel::hasUsableStepData(stepData)) {
272 qWarning() << "Model3DExportWorker: Ignoring malformed preloaded STEP for" << m_componentId << "uuid"
273 << uuid;
274 stepData.clear();
275 }
276 if (!stepData.isEmpty()) {
277 // 预加载阶段已经取得有效 STEP 时写入公共缓存，避免后续导出重复下载。
278 cache->saveModel3D(uuid, stepData, QStringLiteral("step"), gen);
279 }
280
281 bool stepCacheHit = false;
282 if (stepData.isEmpty()) {
283 QMutexLocker downloadLocker(&modelDownloadMutex());
284 const QByteArray cachedStep = cache->loadModel3D(uuid, QStringLiteral("step"));
285 if (!cachedStep.isEmpty() && Exporter3DModel::hasUsableStepData(cachedStep)) {
286 stepCacheHit = cache->copyModel3DToFile(uuid, QStringLiteral("step"), stepWritePath);
287 } else if (!cachedStep.isEmpty()) {
288 qWarning() << "Model3DExportWorker: Ignoring malformed cached STEP for" << m_componentId << "uuid"
289 << uuid;
290 }
291 if (!stepCacheHit && !m_cancelled.load()) {
292 if (!exporter.downloadStepDataSync(uuid, &stepData, &error) && error.isEmpty()) {
293 error = QStringLiteral("Failed to download STEP data");
294 }
295 if (!stepData.isEmpty() && !Exporter3DModel::hasUsableStepData(stepData)) {
296 qWarning() << "Model3DExportWorker: Rejecting invalid downloaded STEP for" << m_componentId
297 << "uuid" << uuid;
298 stepData.clear();
299 error = QStringLiteral("Downloaded STEP data has invalid structure");
300 }
301 if (!stepData.isEmpty()) {
302 cache->saveModel3D(uuid, stepData, QStringLiteral("step"), gen);
303 }
304 }
305 }
306
307 if (stepCacheHit) {
308 qDebug() << "Model3DExportWorker: STEP cache hit for" << m_componentId << "uuid" << uuid;
309 } else if (!stepData.isEmpty()) {
310 Model3DData modelData = buildModelData();
311 modelData.setStep(stepData);
312 IR::Model3DIR modelIR = IR::toModel3DIR(modelData);
313 if (!exporter.exportToStep(modelIR, stepWritePath)) {
314 error = QStringLiteral("Failed to write STEP file");
315 }
316 }
317 }
318
319 if (m_cancelled.load() && error.isEmpty()) {
320 error = QStringLiteral("Cancelled");
321 }
322
323 if (error.isEmpty()) {
324 qDebug() << "Model3DExportWorker: Successfully exported" << (needWrl ? "WRL" : "") << (needStep ? "STEP" : "")
325 << "for" << m_componentId;
326
327 if (m_options.debugMode) {
328 (void)QtConcurrent::run([componentId = m_componentId, data = m_data, outputPath = m_options.outputPath]() {
329 DebugExportHelper::exportDebugData(componentId, data, outputPath);
330 });
331 }
332
333 emit completed(m_componentId, true, QString());
334 return;
335 }
336
337 qCritical() << "Model3DExportWorker: Export failed for" << m_componentId << "uuid" << uuid << "error:" << error;
338 emit completed(m_componentId, false, error);
339}
