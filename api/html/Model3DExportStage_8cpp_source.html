244 return;
245 }
246
247 if (m_skippedComponents.contains(componentId)) {
248 completeSkippedItemProgress(exportWorker, componentId, QStringLiteral("No 3D model or format selected"));
249 delete exportWorker;
250 return;
251 }
252
253 const auto preflightError = m_preflightErrors.constFind(componentId);
254 if (preflightError != m_preflightErrors.cend()) {
255 completeItemProgress(exportWorker, componentId, false, preflightError.value());
256 delete exportWorker;
257 return;
258 }
259
260 exportWorker->setOptions(m_options);
261 exportWorker->setData(componentId, data, m_options);
262
263 const QString modelName =
264 m_modelFileStems.value(componentId, PathSecurity::sanitizeFilename(componentId));
265
266 const bool needWrl = m_options.needsModel3DWrl();
267 const bool needStep = m_options.needsModel3DStep();
268 QString libName =
269 m_options.libName.isEmpty() ? QStringLiteral("EasyKiConverter") : m_options.libName;
270 QString baseOutputDir = m_options.outputPath;
271 if (baseOutputDir.isEmpty()) {
272 baseOutputDir = QDir::currentPath() + QStringLiteral("/export");
273 }
274 QString outputDir =
275 baseOutputDir + QDir::separator() + libName + QStringLiteral(".3dmodels");
276
277 if (m_componentPaths.contains(componentId)) {
278 TempFilePaths& paths = m_componentPaths[componentId];
279 paths.wrlFinalPath = needWrl ? (outputDir + QDir::separator() + modelName + QStringLiteral(".wrl")) : QString();
280 paths.stepFinalPath =
281 needStep ? (outputDir + QDir::separator() + modelName + QStringLiteral(".step")) : QString();
282
283 // 临时路径存在时，Worker 无法自行判断最终文件是否已存在，因此在提交前显式执行不覆盖策略。
284 const bool wrlExists =
285 !paths.wrlFinalPath.isEmpty() && QFile::exists(paths.wrlFinalPath);
286 const bool stepExists =
287 !paths.stepFinalPath.isEmpty() && QFile::exists(paths.stepFinalPath);
288 const bool anyExisting = wrlExists || stepExists;
289 if (!m_options.overwriteExistingFiles && anyExisting) {
290 const bool allRequestedExist =
291 (!needWrl || wrlExists) && (!needStep || stepExists);
292 m_componentPaths.remove(componentId);
293 if (allRequestedExist) {
294 completeSkippedItemProgress(exportWorker, componentId, QStringLiteral("3D model file already exists"));
295 } else {
296 completeItemProgress(exportWorker,
297 componentId,
298 false,
299 QStringLiteral("3D model output partially exists and overwrite is disabled"));
300 }
301 delete exportWorker;
302 return;
303 }
304 exportWorker->setOutputPaths({paths.wrlTempPath, paths.stepTempPath});
305 }
306
307 QPointer<Model3DExportStage> stagePtr(this);
308 qInfo() << "Model3DExportStage: Connecting worker completed signal for" << componentId;
309 connect(
310 exportWorker,
311 &Model3DExportWorker::completed,
312 this,
313 [stagePtr, exportWorker, componentId](const QString&, bool success, const QString& error) {
314 qInfo() << "Model3DExportStage: Worker completed signal received for" << componentId
315 << "success:" << success;
316 if (!stagePtr) {
317 qInfo() << "Model3DExportStage: stagePtr is null!";
318 exportWorker->deleteLater();
319 return;
320 }
321
322 if (success && !stagePtr->m_cancelled.load() && stagePtr->m_componentPaths.contains(componentId)) {
323 const TempFilePaths paths = stagePtr->m_componentPaths.value(componentId);
324 const bool needWrl = stagePtr->m_options.needsModel3DWrl();
325 const bool needStep = stagePtr->m_options.needsModel3DStep();
326
327 QVector<TempFileManager::CommitItem> commitItems;
328 if (needWrl) {
329 commitItems.append({paths.wrlTempPath, paths.wrlFinalPath, false});
330 }
331 if (needStep) {
332 commitItems.append({paths.stepTempPath, paths.stepFinalPath, false});
333 }
334 if (!stagePtr->m_tempManager.commitBatch(commitItems)) {
335 success = false;
336 }
337 }
338
339 stagePtr->completeItemProgress(exportWorker, componentId, success, error);
340 exportWorker->deleteLater();
341 },
342 Qt::QueuedConnection);
343
344 m_threadPool.start(exportWorker);

350void Model3DExportStage::writeAssociationManifest() {
351 if (m_componentIds.isEmpty())
352 return;
353
354 const QString libName =
355 m_options.libName.isEmpty() ? QStringLiteral("EasyKiConverter") : m_options.libName;
356 QString outputDir = m_options.outputPath;
357 if (outputDir.isEmpty())
358 outputDir = QDir::currentPath() + QStringLiteral("/export");
359 outputDir += QDir::separator() + libName + QStringLiteral(".3dmodels");
360
361 const QString manifestPath = outputDir + QDir::separator() + QStringLiteral("manifest.json");
362 if (QFileInfo::exists(manifestPath) && !m_options.overwriteExistingFiles) {
363 QMutexLocker locker(&m_progressMutex);
364 const QString diagnostic =
365 QStringLiteral("3D 关联清单已存在且禁止覆盖：%1").arg(manifestPath);
366 if (!m_progress.diagnostics.contains(diagnostic))
367 m_progress.diagnostics.append(diagnostic);
368 const ExportTypeProgress snapshot = m_progress;
369 locker.unlock();
370 emit progressChanged(snapshot);
371 return;
372 }
373
374 QJsonObject root;
375 root.insert(QStringLiteral("format"), QStringLiteral("EasyKiConverter.3d-model-manifest"));
376 root.insert(QStringLiteral("version"), 1);
377 root.insert(QStringLiteral("library"), libName);
378 root.insert(QStringLiteral("targetFormat"), static_cast<int>(m_options.targetFormat));
379
380 QJsonArray components;
381 const ExportTypeProgress progress = getProgress();
382 const auto vectorToJson = [](const IR::Model3DVec3& vector) {
383 QJsonObject value;
384 value.insert(QStringLiteral("x"), vector.x);
385 value.insert(QStringLiteral("y"), vector.y);
386 value.insert(QStringLiteral("z"), vector.z);
387 return value;
388 };
389 for (const QString& componentId : m_componentIds) { // 当前元器件编号。
390 QJsonObject componentObject;
391 componentObject.insert(QStringLiteral("componentId"), componentId);
392 componentObject.insert(QStringLiteral("modelStem"), m_modelFileStems.value(componentId));
393
394 const auto data = m_cachedData.value(componentId);
395 if (data) {
396 if (data->symbolData())
397 componentObject.insert(QStringLiteral("symbol"), data->symbolData()->info().name);
398 if (data->footprintData())
399 componentObject.insert(QStringLiteral("footprint"), data->footprintData()->info().name);
400
401 Model3DData model;
402 if (data->model3DData())
403 model = *data->model3DData();
404 if (model.uuid().isEmpty() && data->footprintData())
405 model = data->footprintData()->model3D();
406 if (!model.uuid().isEmpty() || !model.name().isEmpty()) {
407 const IR::Model3DIR modelIr = IR::toModel3DIR(model);
408 QJsonObject modelObject;
409 modelObject.insert(QStringLiteral("name"), model.name());
410 modelObject.insert(QStringLiteral("uuid"), model.uuid());
411 modelObject.insert(QStringLiteral("translationMm"), vectorToJson(modelIr.translation()));
412 modelObject.insert(QStringLiteral("rotationDeg"), vectorToJson(modelIr.rotation()));
413 modelObject.insert(QStringLiteral("stepOffsetMm"), vectorToJson(modelIr.stepOffsetMm()));
414 componentObject.insert(QStringLiteral("model"), modelObject);
415 }
416 }
417
418 const ExportItemStatus status = progress.itemStatus.value(componentId);
419 QString statusName = QStringLiteral("pending");
420 if (status.status == ExportItemStatus::Status::Success)
421 statusName = QStringLiteral("success");
422 else if (status.status == ExportItemStatus::Status::Failed)
423 statusName = QStringLiteral("failed");
424 else if (status.status == ExportItemStatus::Status::Skipped)
425 statusName = QStringLiteral("skipped");
426 else if (status.status == ExportItemStatus::Status::InProgress)
427 statusName = QStringLiteral("in-progress");
428 componentObject.insert(QStringLiteral("status"), statusName);
429 if (!status.errorMessage.isEmpty())
430 componentObject.insert(QStringLiteral("diagnostic"), status.errorMessage);
431
432 const QString modelStem = m_modelFileStems.value(componentId);
433 QJsonObject files;
434 if (m_options.needsModel3DWrl()) {
435 const QString relativePath = modelStem + QStringLiteral(".wrl");
436 if (QFileInfo::exists(outputDir + QDir::separator() + relativePath))
437 files.insert(QStringLiteral("wrl"), relativePath);
438 }
439 if (m_options.needsModel3DStep()) {
440 const QString relativePath = modelStem + QStringLiteral(".step");
441 if (QFileInfo::exists(outputDir + QDir::separator() + relativePath))
442 files.insert(QStringLiteral("step"), relativePath);
443 }
444 componentObject.insert(QStringLiteral("files"), files);
445 components.append(componentObject);
446 }
447 root.insert(QStringLiteral("components"), components);
448
449 QSaveFile manifest(manifestPath);
450 if (!manifest.open(QIODevice::WriteOnly | QIODevice::Text) ||
451 manifest.write(QJsonDocument(root).toJson(QJsonDocument::Indented)) < 0 || !manifest.commit()) {
452 QMutexLocker locker(&m_progressMutex);
453 const QString diagnostic =
454 QStringLiteral("无法写入 3D 关联清单：%1").arg(manifestPath);
455 if (!m_progress.diagnostics.contains(diagnostic))
456 m_progress.diagnostics.append(diagnostic);
457 const ExportTypeProgress snapshot = m_progress;
458 locker.unlock();
459 emit progressChanged(snapshot);
460 }
461}
