51QString padName(const IR::FootprintComponentIR& footprint, const IR::FootprintPadIR& pad, int index) {
52 QString base = footprint.name.trimmed(); // 清理后的封装基础名称。
53 base.replace(QRegularExpression(QStringLiteral("[^A-Za-z0-9_]+")), QStringLiteral("_"));
54 if (base.isEmpty())
55 base = QStringLiteral("PACKAGE");
56 QString pin = pad.number.trimmed(); // 清理后的焊盘编号。
57 pin.replace(QRegularExpression(QStringLiteral("[^A-Za-z0-9_]+")), QStringLiteral("_"));
58 if (pin.isEmpty())
59 pin = QString::number(index + 1);
60 return QStringLiteral("%1_PAD_%2").arg(base, pin);
61}

70bool writeGraphic(QTextStream& stream, const IR::FootprintComponentIR& footprint, QStringList& diagnostics) {
71 for (const auto& rectangle : footprint.rectangles) {
72 const QRectF bounds = rectangle.bounds.normalized(); // 归一化后的矩形边界。
73 stream << "RECT " << point(bounds.topLeft()) << ' ' << point(bounds.bottomRight()) << " WIDTH "
74 << number(rectangle.strokeWidth) << '\n';
75 }
76 for (const auto& circle : footprint.circles) {
77 stream << "CIRCLE " << point(circle.center) << " RADIUS " << number(circle.radius) << " WIDTH "
78 << number(circle.strokeWidth) << '\n';
79 }
80 for (const auto& track : footprint.tracks) {
81 if (track.points.size() < 2) {
82 diagnostics.append(QStringLiteral("CADSTAR: 封装折线至少需要两个点"));
83 return false;
84 }
85 stream << "POLYLINE";
86 for (const QPointF& item : track.points)
87 stream << ' ' << point(item);
88 stream << " WIDTH " << number(track.width) << '\n';
89 }
90 for (const auto& region : footprint.regions) {
91 if (region.vertices.size() < 3) {
92 diagnostics.append(QStringLiteral("CADSTAR: 封装多边形至少需要三个点"));
93 return false;
94 }
95 stream << "POLYGON";
96 for (const QPointF& item : region.vertices)
97 stream << ' ' << point(item);
98 stream << '\n';
99 }
100 for (const auto& arc : footprint.arcs) {
101 diagnostics.append(QStringLiteral("CADSTAR: FootprintArc 暂不支持直接写入，已拒绝导出"));
102 Q_UNUSED(arc);
103 return false;
104 }
105 for (const auto& text : footprint.texts) {
106 diagnostics.append(QStringLiteral("CADSTAR: 封装文本暂不支持独立字体属性，已拒绝导出：%1").arg(text.text));
107 return false;
108 }
109 return true;
110}

120bool writePad(QTextStream& stream, const QString& name, const IR::FootprintPadIR& pad, QStringList& diagnostics) {
121 stream << "PAD " << quote(name) << '\n';
122 // 按统一 IR 焊盘形状选择 CADSTAR 可以准确表达的定义。
123 switch (pad.shape) {
124 case IR::PadShape::Ellipse:
125 if (qFuzzyCompare(pad.size.width(), pad.size.height())) {
126 stream << "SHAPE ROUND\nDIAMETER " << number(pad.size.width()) << '\n';
127 } else {
128 stream << "SHAPE OBLONG\nWIDTH " << number(pad.size.width()) << "\nHEIGHT " << number(pad.size.height())
129 << '\n';
130 }
131 break;
132 case IR::PadShape::Rect:
133 stream << "SHAPE " << (qFuzzyCompare(pad.size.width(), pad.size.height()) ? "SQUARE" : "RECTANGLE") << '\n';
134 stream << "WIDTH " << number(pad.size.width()) << "\nHEIGHT " << number(pad.size.height()) << '\n';
135 break;
136 case IR::PadShape::Oval:
137 stream << "SHAPE OBLONG\nWIDTH " << number(pad.size.width()) << "\nHEIGHT " << number(pad.size.height())
138 << '\n';
139 break;
140 case IR::PadShape::Polygon:
141 if (pad.customShapePoints.size() < 3) {
142 diagnostics.append(QStringLiteral("CADSTAR: 多边形焊盘缺少三个以上顶点：%1").arg(name));
143 return false;
144 }
145 stream << "SHAPE CUSTOM\nPOLYGON";
146 for (const QPointF& item : pad.customShapePoints)
147 stream << ' ' << point(item);
148 stream << '\n';
149 break;
150 case IR::PadShape::RoundRect:
151 case IR::PadShape::Trapezoid:
152 diagnostics.append(QStringLiteral("CADSTAR: 不支持的焊盘形状，禁止静默降级：%1").arg(name));
153 return false;
154 }
155 if (pad.isThroughHole()) {
156 if (pad.holeLength > 0.0)
157 stream << "HOLE SLOT " << number(pad.holeSize) << ' ' << number(pad.holeLength) << '\n';
158 else if (pad.holeSize > 0.0)
159 stream << "HOLE ROUND " << number(pad.holeSize) << '\n';
160 else {
161 diagnostics.append(QStringLiteral("CADSTAR: 通孔焊盘缺少孔径：%1").arg(name));
162 return false;
163 }
164 }
165 stream << "ENDPAD\n";
166 return true;
167}

177bool writePadDefinitions(QTextStream& stream,
178 const QList<IR::ComponentIR>& components,
179 QStringList& diagnostics,
180 QSet<QString>& padNames) {
181 for (const auto& component : components) {
182 for (int index = 0; index < component.footprint.pads.size(); ++index) {
183 const auto& pad = component.footprint.pads.at(index); // 当前组件焊盘。
184 const QString name = padName(component.footprint, pad, index); // 稳定焊盘定义名称。
185 if (padNames.contains(name))
186 continue;
187 if (!writePad(stream, name, pad, diagnostics))
188 return false;
189 padNames.insert(name);
190 }
191 }
192 return true;

202bool writeComponent(QTextStream& stream, const IR::SymbolComponentIR& symbol, QStringList& diagnostics) {
203 if (symbol.name.isEmpty()) {
204 diagnostics.append(QStringLiteral("CADSTAR: 符号名称为空"));
205 return false;
206 }
207 stream << "COMPONENT " << quote(symbol.name) << "\nVERSION 54\n";
208 if (!symbol.designatorPrefix.isEmpty())
209 stream << "PROPERTY REFDES " << quote(symbol.designatorPrefix) << '\n';
210 for (const auto& pin : symbol.pins) {
211 if (pin.designator.isEmpty()) {
212 diagnostics.append(QStringLiteral("CADSTAR: 符号 %1 存在空引脚编号").arg(symbol.name));
213 return false;
214 }
215 stream << "PIN " << quote(pin.designator) << '\n';
216 const QPointF end = pin.position; // 引脚终点坐标。
217 QPointF start = end; // 根据方向反推的引脚起点。
218 // 根据统一 IR 的引脚方向反推 CADSTAR 引脚起点。
219 switch (pin.direction) {
220 case IR::PinDirection::Left:
221 start.rx() += pin.length;
222 break;
223 case IR::PinDirection::Right:
224 start.rx() -= pin.length;
225 break;
226 case IR::PinDirection::Up:
227 start.ry() += pin.length;
228 break;
229 case IR::PinDirection::Down:
230 start.ry() -= pin.length;
231 break;
232 default:
233 break;
234 }
235 stream << "START " << point(start) << "\nEND " << point(end) << '\n';
236 stream << "PINTYPE " << (pin.electricalType == IR::PinElectricalType::Input ? "INPUT" : "PASSIVE") << '\n';
237 stream << "NUMBERS [" << quote(pin.designator) << ":" << quote(pin.designator) << "]\n";
238 if (!pin.name.isEmpty())
239 stream << "LABEL " << quote(pin.name) << ' ' << point(pin.hasNamePosition ? pin.namePosition : end)
240 << " 1\n";
241 stream << "ENDPIN\n";
242 }
243 for (const auto& rectangle : symbol.rectangles)
244 stream << "RECT " << point(QPointF(rectangle.x0, rectangle.y0)) << ' '
245 << point(QPointF(rectangle.x1, rectangle.y1)) << '\n';
246 for (const auto& circle : symbol.circles)
247 stream << "CIRCLE " << point(circle.center) << " RADIUS " << number(circle.radius) << '\n';
248 for (const auto& polygon : symbol.polygons) {
249 if (polygon.points.size() < 3) {
250 diagnostics.append(QStringLiteral("CADSTAR: 符号多边形至少需要三个点：%1").arg(symbol.name));
251 return false;
252 }
253 stream << "POLYGON";
254 for (const QPointF& item : polygon.points)
255 stream << ' ' << point(item);
256 stream << '\n';
257 }
258 if (!symbol.arcs.isEmpty() || !symbol.ellipses.isEmpty() || !symbol.paths.isEmpty() || !symbol.beziers.isEmpty()) {
259 diagnostics.append(
260 QStringLiteral("CADSTAR: 符号包含当前 ASCII writer 尚未表达的曲线图元：%1").arg(symbol.name));
261 return false;
262 }
263 stream << "ENDCOMPONENT\n";
264 return true;
265}

276bool writePackageAndPart(QTextStream& stream,
277 const IR::ComponentIR& component,
278 QStringList& diagnostics,
279 QSet<QString>& packageNames,
280 bool includePart) {
281 if (component.name.isEmpty() || component.footprint.name.isEmpty()) {
282 diagnostics.append(QStringLiteral("CADSTAR: 组件缺少名称或封装名称"));
283 return false;
284 }
285 QString packageName = component.footprint.name; // 当前组件封装名称。
286 if (!packageNames.contains(packageName)) {
287 stream << "PACKAGE " << quote(packageName) << '\n';
288 if (!component.footprint.description.isEmpty())
289 stream << "DESCRIPTION " << quote(component.footprint.description) << '\n';
290 for (int index = 0; index < component.footprint.pads.size(); ++index) {
291 const auto& pad = component.footprint.pads.at(index); // 当前组件焊盘。
292 const QString name = padName(component.footprint, pad, index); // 稳定焊盘定义名称。
293 stream << "PIN " << quote(pad.number) << "\nPOSITION " << point(pad.position) << "\nPAD " << quote(name)
294 << "\nENDPIN\n";
295 }
296 if (!writeGraphic(stream, component.footprint, diagnostics))
297 return false;
298 stream << "ENDPACKAGE\n";
299 packageNames.insert(packageName);
300 }
301 if (includePart) {
302 QSet<QString> padNumbers; // 当前封装中可用的焊盘编号集合。
303 for (const IR::FootprintPadIR& pad : component.footprint.pads)
304 padNumbers.insert(pad.number);
305 for (const IR::SymbolPinIR& pin : component.symbol.pins) {
306 if (pin.designator.trimmed().isEmpty() || !padNumbers.contains(pin.designator)) {
307 diagnostics.append(QStringLiteral("CADSTAR: 组件 %1 的符号引脚 %2 找不到对应封装焊盘")
308 .arg(component.name, pin.designator));
309 return false;
310 }
311 }
312 const QString symbolName =
313 component.symbol.name.isEmpty() ? component.name : component.symbol.name; // 完整组件符号名称。
314 stream << "PART " << quote(component.name) << "\nCOMPONENT " << quote(symbolName) << "\nPACKAGE "
315 << quote(packageName) << '\n';
316 if (!component.description.isEmpty())
317 stream << "DESCRIPTION " << quote(component.description) << '\n';
318 stream << "ENDPART\n";
319 }
320 return true;
321}

496bool ExporterCadstarLibrary::exportComponentLibrary(const QList<IR::ComponentIR>& components,
497 const QString& libName,
498 const QString& filePath,
499 bool exportModel3D,
500 const QString&) {
501 m_diagnostics.clear();
502 if (components.isEmpty()) {
503 m_diagnostics.append(QStringLiteral("CADSTAR: 没有可导出的完整组件"));
504 return false;
505 }
506 QFile file(filePath); // 完整组件库输出文件。
507 if (!file.open(QIODevice::WriteOnly | QIODevice::Truncate)) {
508 m_diagnostics.append(QStringLiteral("CADSTAR: 无法写入完整 ASCII 库：%1").arg(filePath));
509 return false;
510 }
511 QTextStream stream(&file); // CADSTAR ASCII 输出流。
512 stream.setEncoding(QStringConverter::Utf8);
513 stream << "UNITS MM\nPROPERTY DESCRIPTION " << quote(libName) << '\n';
514 QSet<QString> symbols; // 已写入的符号名称集合。
515 QSet<QString> padNames; // 已写入的焊盘定义名称。
516 QSet<QString> packageNames; // 已写入的封装名称。
517 if (!writePadDefinitions(stream, components, m_diagnostics, padNames))
518 return false;
519 for (const auto& component : components) {
520 const QString symbolName =
521 component.symbol.name.isEmpty() ? component.name : component.symbol.name; // 当前组件符号名称。
522 if (symbols.contains(symbolName)) {

524 return false;
525 }
526 symbols.insert(symbolName);
527 if (!writeComponent(stream, component.symbol, m_diagnostics) ||
528 !writePackageAndPart(stream, component, m_diagnostics, packageNames, true))
529 return false;
530 if (exportModel3D && component.hasModel3D())
531 m_diagnostics.append(QStringLiteral("CADSTAR: 组件 %1 的 3D 模型由独立阶段输出，ASCII Part 不写入模型关联")
