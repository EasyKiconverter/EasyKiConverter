14bool PathSecurity::isValidPathComponent(const QString& name) {
15 if (name.isEmpty() || name == QStringLiteral(".") || name == QStringLiteral(".."))
16 return false;
17
18 // 路径组件不能包含分隔符或 Windows 保留字符。
19 static const QRegularExpression illegalChars(QStringLiteral(R"([\\/:*?"<>|])"));
20 if (name.contains(illegalChars))
21 return false;
22
23 // 控制字符（包括 NUL）不能安全地出现在文件名中。
24 for (const QChar character : name) {
25 const ushort codePoint = character.unicode(); // 当前字符的 Unicode 码点。
26 if (codePoint < 0x20 || codePoint == 0x7F)
27 return false;
28 }
29
30 // Windows 会忽略文件名末尾的空格和句点，可能导致路径碰撞。
31 if (name.endsWith(QChar(' ')) || name.endsWith(QChar('.')))
32 return false;
33
34 // 这些名称即使带扩展名，在 Windows 上仍属于设备名。
35 QString deviceName = name; // 用于检测 Windows 保留设备名的名称主体。
36 const qsizetype extensionStart = deviceName.indexOf(QChar('.')); // 第一个扩展名分隔符位置。
37 if (extensionStart >= 0)
38 deviceName.truncate(extensionStart);
39 static const QStringList reservedNames = {
40 QStringLiteral("CON"), QStringLiteral("PRN"), QStringLiteral("AUX"), QStringLiteral("NUL"),
41 QStringLiteral("COM1"), QStringLiteral("COM2"), QStringLiteral("COM3"), QStringLiteral("COM4"),
42 QStringLiteral("COM5"), QStringLiteral("COM6"), QStringLiteral("COM7"), QStringLiteral("COM8"),
43 QStringLiteral("COM9"), QStringLiteral("LPT1"), QStringLiteral("LPT2"), QStringLiteral("LPT3"),
44 QStringLiteral("LPT4"), QStringLiteral("LPT5"), QStringLiteral("LPT6"), QStringLiteral("LPT7"),
45 QStringLiteral("LPT8"), QStringLiteral("LPT9")};
46 if (reservedNames.contains(deviceName, Qt::CaseInsensitive))
47 return false;
48
49 // 零宽字符可能造成显示内容与实际路径不一致。
50 if (name.contains(QChar(0x200B)) || name.contains(QChar(0x200C)) || name.contains(QChar(0x200D)) ||
51 name.contains(QChar(0xFEFF)))
52 return false;
53
54 return true;
55}

63bool PathSecurity::isSafePath(const QString& fullPath, const QString& baseDir) {
64 if (fullPath.isEmpty() || baseDir.isEmpty()) {
65 qWarning() << "isSafePath: received empty path";
66 return false;
67 }
68
69 // 1. 获取绝对路径并规范化
70 // 注意：canonicalFilePath() 在应用程序退出时可能会崩溃，因为 Qt 内部对象可能已被销毁
71 // 使用 QFile::exists() 先检查，避免调用 canonicalFilePath() 在不存在的路径上
72 QFileInfo baseInfo(baseDir);
73 QFileInfo targetInfo(fullPath);
74
75 // 安全获取绝对路径 - 避免在退出时调用 canonicalFilePath()
76 QString absBase; // 规范化后的基准目录绝对路径。
77 QString absTarget; // 规范化后的目标绝对路径。
78
79 // 检查路径是否存在且可访问
80 if (!baseInfo.exists() || !baseInfo.isReadable()) {
81 // 路径不存在时使用 cleanPath，这是安全的
82 absBase = QDir::cleanPath(baseInfo.absoluteFilePath());
83 } else {
84 // 路径存在但避免使用 canonicalFilePath，改用 absoluteFilePath + cleanPath
85 // canonicalFilePath() 会调用 Qt 内部文件引擎，在退出时可能崩溃
86 absBase = QDir::cleanPath(baseInfo.absoluteFilePath());
87 }
88
89 if (!targetInfo.exists() || !targetInfo.isReadable()) {
90 absTarget = QDir::cleanPath(targetInfo.absoluteFilePath());
91 } else {
92 absTarget = QDir::cleanPath(targetInfo.absoluteFilePath());
93 }
94
95 // 2. 统一分隔符为 '/' (Qt 默认)
96 absBase = QDir::fromNativeSeparators(absBase);
97 absTarget = QDir::fromNativeSeparators(absTarget);
98
99 // 3. 确保基准路径以 '/' 结尾，用于目录前缀匹配
100 if (!absBase.endsWith('/')) {
101 absBase += '/';
102 }
103
104 // 4. 特殊情况处理：允许目标路径等于基准路径
105 // 移除 target 可能存在的末尾斜杠
106 QString targetNoSlash = absTarget;
107 while (targetNoSlash.endsWith('/'))
108 targetNoSlash.chop(1);
109
110 // 如果 target + '/' == base，说明它们代表同一个目录
111 if (targetNoSlash + '/' == absBase) {
112 return true;
113 }
114 // 或者直接比较（忽略大小写，适应 Windows）
115 if (targetNoSlash.compare(absBase.left(absBase.length() - 1), Qt::CaseInsensitive) == 0) {
116 return true;
117 }
118
119 // 5. 前缀比较：检查 target 是否以 base 开头
120 bool isSafe = absTarget.startsWith(absBase, Qt::CaseInsensitive);
121
122 if (!isSafe) {
123 qWarning() << "isSafePath: path is outside base directory";
124 qWarning() << " Target (norm):" << absTarget;
125 qWarning() << " Base (norm):" << absBase;
126 qWarning() << " Raw Target:" << fullPath;
127 qWarning() << " Raw Base: " << baseDir;
128 }
129
130 return isSafe;
131}

139bool PathSecurity::isSafePathCanonical(const QString& fullPath, const QString& baseDir) {
140 if (fullPath.isEmpty() || baseDir.isEmpty()) {
141 qWarning() << "isSafePathCanonical: received empty path";
142 return false;
143 }
144
145 const QString canonBase = canonicalizePath(baseDir); // 解析后的基准目录路径。
146 const QString canonTarget = canonicalizePath(fullPath); // 解析后的目标路径。
147
148 if (canonBase.isEmpty() || canonTarget.isEmpty()) {
149 qWarning() << "isSafePathCanonical: failed to canonicalize paths";
150 return false;
151 }
152
153 // 确保基准路径以 '/' 结尾
154 QString normalizedBase = canonBase; // 带有目录分隔符边界的基准路径。
155 if (!normalizedBase.endsWith('/')) {
156 normalizedBase += '/';
157 }
158
159 // 允许目标等于基准
160 QString targetNoSlash = canonTarget; // 移除尾部分隔符后的目标路径。
161 while (targetNoSlash.endsWith('/'))
162 targetNoSlash.chop(1);
163
164 if (targetNoSlash + '/' == normalizedBase) {
165 return true;
166 }
167 if (targetNoSlash.compare(normalizedBase.left(normalizedBase.length() - 1), Qt::CaseInsensitive) == 0) {
168 return true;
169 }
170
171 bool isSafe = canonTarget.startsWith(normalizedBase, Qt::CaseInsensitive); // 目标路径是否通过边界检查。
172
173 if (!isSafe) {
174 qWarning() << "isSafePathCanonical: path is outside base directory";
175 qWarning() << " Target (canonical):" << canonTarget;
176 qWarning() << " Base (canonical):" << normalizedBase;
177 qWarning() << " Raw Target:" << fullPath;
178 qWarning() << " Raw Base: " << baseDir;
179 }
180
181 return isSafe;
182}

189QString PathSecurity::canonicalizePath(const QString& path) {
190 QFileInfo info(path);
191
192 // 如果路径存在，直接用 canonicalFilePath（解析符号链接）
193 if (info.exists()) {
194 QString canonical = info.canonicalFilePath(); // 已存在路径解析符号链接后的结果。
195 if (!canonical.isEmpty()) {
196 return QDir::fromNativeSeparators(canonical);
197 }
198 }
199
200 // 路径不存在：找到最近存在的父目录，对其 canonicalize，再拼接剩余部分
201 QString remaining; // 最近存在父目录之后仍需拼接的路径片段。
202 QString current = QDir::cleanPath(path); // 当前向上搜索的路径。
203
204 // 逐级向上查找存在的父目录
205 while (!QFileInfo::exists(current)) {
206 QDir parent = QDir(current); // 当前路径的父目录对象。
207 if (!parent.cdUp()) {
208 // 到达根目录仍不存在，回退到 cleanPath
209 return QDir::fromNativeSeparators(QDir::cleanPath(path));
210 }
211 QString childName = QDir(current).dirName(); // 当前路径相对于父目录的名称。
212 remaining = remaining.isEmpty() ? childName : (childName + "/" + remaining);
213 current = parent.path();
214 }
215
216 // current 现在是存在的路径，对其 canonicalize
217 QFileInfo currentInfo(current);
218 QString canonCurrent = currentInfo.canonicalFilePath(); // 最近存在目录的 canonical 路径。
219 if (canonCurrent.isEmpty()) {
220 canonCurrent = QDir::cleanPath(current);
221 }
222
223 if (remaining.isEmpty()) {
224 return QDir::fromNativeSeparators(canonCurrent);
225 }
226 return QDir::fromNativeSeparators(canonCurrent + "/" + remaining);
227}

234QString PathSecurity::sanitizeFilename(const QString& name) {
235 QString safeName = name; // 清洗过程中的文件名副本。
236
237 // 替换非法字符: \ / : * ? " < > |
238 // 使用十六进制转义确保安全: \x5C 是反斜杠, \x22 是双引号
239 static const QRegularExpression illegalChars(QStringLiteral("[\\x5C/:*?\\x22<>|]"));
240 safeName.replace(illegalChars, "_");
241
242 // 替换控制字符
243 static const QRegularExpression controlChars(R"([\x00-\x1f])");
244 safeName.replace(controlChars, "");
245
246 // 替换 Unicode 零宽字符 (可能导致显示欺骗)
247 // 使用 QChar 移除这些特殊字符
248 safeName.remove(QChar(0x200B)); // ZERO WIDTH SPACE
249 safeName.remove(QChar(0x200C)); // ZERO WIDTH NON-JOINDER
250 safeName.remove(QChar(0x200D)); // ZERO WIDTH JOINER
251 safeName.remove(QChar(0xFEFF)); // ZERO WIDTH NO-BREAK SPACE
252
253 safeName = safeName.trimmed();
254
255 if (safeName.isEmpty()) {
256 safeName = "unnamed_component";
257 }
258
259 // Windows 保留设备名，不能作为普通文件名使用。
260 static const QStringList reservedNames = {"CON", "PRN", "AUX", "NUL", "COM1", "COM2", "COM3", "COM4",
261 "COM5", "COM6", "COM7", "COM8", "COM9", "LPT1", "LPT2", "LPT3",
262 "LPT4", "LPT5", "LPT6", "LPT7", "LPT8", "LPT9"};
263
264 if (reservedNames.contains(safeName, Qt::CaseInsensitive)) {
265 safeName += "_safe";
266 }
267
268 return safeName;
269}

277bool PathSecurity::safeRemoveRecursively(const QString& path, int maxFileCount) {
278 QDir dir(path);
279 if (!dir.exists())
280 return true;
281
282 int count = countFilesRecursively(path, maxFileCount + 1); // 包含超限哨兵的目录项数量。
283 if (count > maxFileCount) {
284 qCritical() << "Security Alert: Recursive delete aborted. File count" << count << "exceeds limit of"
285 << maxFileCount << "in path:" << path;
286 return false;
287 }
288
289 qDebug() << "Safe delete: Removing directory with" << count << "items:" << path;
290 return dir.removeRecursively();
291}

300int PathSecurity::countFilesRecursively(const QString& path, int limit, int currentCount) {
301 if (currentCount >= limit)
302 return currentCount;
303
304 QDir dir(path);
305 QFileInfoList list = dir.entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot | QDir::Hidden); // 当前目录项列表。
306
307 for (const QFileInfo& info : list) {
308 currentCount++;
309 if (currentCount >= limit)
310 return limit;
311
312 if (info.isDir()) {
313 if (!info.isSymLink()) {
314 currentCount = countFilesRecursively(info.absoluteFilePath(), limit, currentCount);
315 } else {
316 // 注意：我们跳过符号链接指向的内容，不计入文件总数，也不深入检查
317 // removeRecursively 会删除符号链接本身，但不会删除目标
318 // 这是预期的安全行为
319 currentCount++;
320 }
321 if (currentCount >= limit)
322 return limit;
323 } else {
324 currentCount++;
325 }
326 }
327 return currentCount;
328}
