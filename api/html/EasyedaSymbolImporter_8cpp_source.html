348SymbolPin EasyedaSymbolImporter::importPinData(const QString& pinData) {
349 SymbolPin pin; // 待填充的符号引脚数据。
350 QList<QStringList> segments = parsePinDataString(pinData); // 按段拆分后的引脚记录。
351
352 if (segments.size() >= 7) {
353 QStringList settings = segments[0]; // 引脚主设置字段。
354 if (settings.size() >= 8) {
355 pin.settings.isDisplayed = (settings[1] == "show");
356 pin.settings.type = static_cast<PinType>(settings[2].toInt());
357 pin.settings.spicePinNumber = settings[3];
358 pin.settings.posX = settings[4].toDouble();
359 pin.settings.posY = settings[5].toDouble();
360 pin.settings.rotation = settings[6].toInt();
361 pin.settings.id = settings[7];
362 pin.settings.isLocked = (settings.size() > 8 ? EasyedaUtils::stringToBool(settings[8]) : false);
363 }
364
365 if (segments[1].size() >= 2) {
366 pin.pinDot.dotX = segments[1][0].toDouble();
367 pin.pinDot.dotY = segments[1][1].toDouble();
368 }
369
370 if (segments[2].size() >= 2) {
371 pin.pinPath.path = segments[2][0];
372 pin.pinPath.color = segments[2][1];
373 }
374
375 if (segments[3].size() >= 8) {
376 pin.name.isDisplayed = (segments[3][0] == "show");
377 pin.name.posX = segments[3][1].toDouble();
378 pin.name.posY = segments[3][2].toDouble();
379 pin.name.rotation = segments[3][3].toInt();
380 pin.name.text = segments[3][4];
381 pin.name.textAnchor = segments[3][5];
382 pin.name.font = segments[3][6];
383 pin.name.fontSize = segments[3][7].toDouble();
384 }
385
386 if (segments[4].size() >= 5) {
387 qDebug() << "Pin Segment 4 data:" << segments[4];
388 qDebug() << " Segment 4[0] (dotShow):" << segments[4][0];
389
390 pin.dot.isDisplayed = false;
391
392 pin.dot.circleX = segments[4][1].toDouble();
393 pin.dot.circleY = segments[4][2].toDouble();
394 // Segment 4 的首字段描述 Pin Dot 状态，不直接等同于 Pin Number 显示状态。
395 // 只有存在编号文本时才建立可导出的编号文本元数据。
396 pin.number.isDisplayed = false;
397 pin.number.posX = pin.dot.circleX;
398 pin.number.posY = pin.dot.circleY;
399 pin.number.rotation = pin.settings.rotation;
400 pin.number.textAnchor = QStringLiteral("middle");
401
402 qDebug() << " dot.isDisplayed set to: false (always hide)";
403
404 QString pinNumberDisplayText = segments[4][4]; // EasyEDA 提供的引脚编号显示文本。
405 if (!pinNumberDisplayText.isEmpty()) {
406 pin.settings.spicePinNumber = pinNumberDisplayText;
407 pin.number.text = pinNumberDisplayText;
408 pin.number.isDisplayed = true;
409 qDebug() << "Pin Number Display Text extracted from Segment 4:" << pinNumberDisplayText << "for pin"
410 << pin.name.text << "(replaced spicePinNumber)";
411 }
412 }
413
414 if (segments[6].size() >= 2) {
415 pin.clock.isDisplayed = (segments[6][0] == "show");
416 pin.clock.path = segments[6][1];
417 }
418 }
419
420 return pin;
421}

479SymbolArc EasyedaSymbolImporter::importArcData(const QString& arcData) {
480 SymbolArc arc; // 待填充的圆弧图元。
481 QStringList fields = EasyedaUtils::parseDataString(arcData); // 拆分后的圆弧字段。
482
483 if (fields.size() >= 8) {
484 arc.helperDots = fields[1];
485 arc.strokeColor = fields[2];
486 arc.strokeWidth = fields[3].toDouble();
487 arc.strokeStyle = fields[4];
488 arc.fillColor = EasyedaUtils::stringToBool(fields[5]);
489 arc.id = fields[6];
490 arc.isLocked = EasyedaUtils::stringToBool(fields[7]);
491
492 // 解析 helperDots 中的 SVG 弧线路径，获取完整路径点
493 // 格式如: "M 401 313 A 3 3 0 0 0 407 313"
494 if (!arc.helperDots.isEmpty()) {
495 QString pathStr = arc.helperDots.trimmed(); // 清洗后的 SVG 弧线路径字符串。
496 // 使用 SvgPathParser 解析完整的 SVG 路径（包括弧线的所有点）
497 QList<QPointF> parsedPoints = SvgPathParser::parsePath(pathStr); // 解析后的弧线路径点。
498 if (!parsedPoints.isEmpty()) {
499 arc.path = parsedPoints;
500 }
501 }
502 }
503
504 return arc;

603SymbolImage EasyedaSymbolImporter::importImageData(const QString& imageData) {
604 SymbolImage image; // 待填充的图片图元。
605 const QStringList fields = EasyedaUtils::parseDataString(imageData); // 拆分后的图片字段。
606 if (fields.size() < 7)
607 return image;
608
609 image.posX = fields[1].toDouble();
610 image.posY = fields[2].toDouble();
611 image.width = fields[3].toDouble();
612 image.height = fields[4].toDouble();
613 image.rotation = fields[5].toDouble();
614 image.source = fields[6];
615 image.isLocked = fields.size() > 7 ? EasyedaUtils::stringToBool(fields[7]) : false;
616
617 if (image.source.startsWith(QStringLiteral("data:"), Qt::CaseInsensitive)) {
618 const int comma = image.source.indexOf(','); // data URL 元数据与负载之间的分隔位置。
619 if (comma > 5) {
620 const QString metadata = image.source.mid(5, comma - 5); // data URL 的媒体类型和编码声明。
621 const QByteArray payload = image.source.mid(comma + 1).toUtf8(); // data URL 的编码负载。
622 if (metadata.contains(QStringLiteral(";base64"), Qt::CaseInsensitive))
623 image.data = QByteArray::fromBase64(payload);
624 else
625 image.data = QUrl::fromPercentEncoding(payload).toUtf8();
626
627 const QString mimeType = metadata.section(';', 0, 0).toLower(); // 图片媒体类型。
628 QString extension; // 根据媒体类型推导的文件扩展名。
629 if (mimeType == QStringLiteral("image/svg+xml"))
630 extension = QStringLiteral("svg");
631 else if (mimeType.startsWith(QStringLiteral("image/")))
632 extension = mimeType.mid(6);
633 image.fileName =
634 extension.isEmpty() ? QStringLiteral("image.bin") : QStringLiteral("image.%1").arg(extension);
635 }
636 } else {
637 image.fileName = QUrl(image.source).fileName();
638 if (image.fileName.isEmpty())
639 image.fileName = image.source;
640 }
641 return image;
642}

683QList<QStringList> EasyedaSymbolImporter::parsePinDataString(const QString& pinData) const {
684 QList<QStringList> result; // 拆分后的引脚字段列表。
685 QStringList segments; // 由双插入符分隔的引脚段列表。
686 int start = 0; // 当前字段段的起始位置。
687 int pos = 0; // 当前扫描位置。
688
689 while (pos < pinData.length()) {
690 if (pinData[pos] == '^' && pos + 1 < pinData.length() && pinData[pos + 1] == '^') {
691 segments.append(pinData.mid(start, pos - start));
692 pos += 2;
693 start = pos;
694 continue;
695 }
696
697 pos++;
698 }
699
700 if (start < pinData.length()) {
701 segments.append(pinData.mid(start));
702 }
703
704 for (const QString& segment : segments) {
705 QStringList subSegments = segment.split("~", Qt::KeepEmptyParts); // 当前引脚段的子字段列表。
706 result.append(subSegments);
707 }
708
709 return result;
710}
