22void ExportStageLaunchCoordinator::start(ParallelExportService& owner) {
23 if (owner.m_progress.currentStage == ExportOverallProgress::Stage::Exporting) {
24 qWarning() << "ParallelExportService: Export already in progress";
25 return;
26 }
27
28 if (!owner.m_preloadCompleted) {
29 qWarning() << "ParallelExportService: startExport called before preload completed";
30 emit owner.failed(QStringLiteral("Preload has not completed"));
31 return;
32 }
33
34 if (owner.m_componentIds.isEmpty()) {
35 qWarning() << "ParallelExportService: No components to export";
36 emit owner.failed(QStringLiteral("No components to export"));
37 return;
38 }
39
40 owner.cleanupExportStages();
41 owner.m_cancelRequested = false;
42 // 预加载完成，旧回调已全部处理，解除全局 tombstone。
43 ComponentCacheService::instance()->clearGlobalTombstone();
44
45 // 从 ComponentService 刷新最新的组件数据，确保用户在 UI 中的编辑生效。
46 if (owner.m_componentService) {
47 for (auto it = owner.m_cachedData.begin(); it != owner.m_cachedData.end(); ++it) {
48 ComponentData freshData =
49 owner.m_componentService->getComponentData(it.key());
50 if (freshData.symbolData() && it.value() && it.value()->symbolData()) {
51 it.value()->symbolData()->setInfo(freshData.symbolData()->info());
52 }
53 if (freshData.footprintData() && it.value() && it.value()->footprintData()) {
54 it.value()->footprintData()->setInfo(freshData.footprintData()->info());
55 }
56 }
57 }
58 const quint64 runGeneration = owner.m_activeRunGeneration;
59
60 qDebug() << "ParallelExportService: Starting export for" << owner.m_componentIds.size() << "components"
61 << (owner.m_options.retryMode ? "(retry mode)" : "");
62
63 owner.m_progress.currentStage = ExportOverallProgress::Stage::Exporting;
64 owner.m_progress.startTime = QDateTime::currentDateTime();
65 owner.m_progress.exportTypeProgress.clear();
66 owner.m_runningExportStages = 0;
67
68 // Altium PcbLib 将 STEP 直接嵌入 Library/Models，不生成 KiCad 风格的
69 // 外部 .3dmodels 目录；封装阶段仍会按需获取 STEP 并交给写入器。
70 // Model3D 统计项仍需保留，Altium 下由封装阶段镜像完成状态。
71 const ExportRunPlan plan =
72 buildExportRunPlan(owner.m_options, owner.m_componentIds, owner.m_cachedData);
73 const bool enableSymbol = plan.enableSymbol;
74 const bool enableFootprint = plan.enableFootprint;
75 const bool runExternalModel3DStage = plan.runExternalModel3DStage;
76 const bool enablePreview = plan.enablePreview;
77 const bool enableDatasheet = plan.enableDatasheet;
78 if (!plan.missingDataComponentIds.isEmpty()) {
79 qWarning() << "ParallelExportService: Missing preloaded component data for"
80 << plan.missingDataComponentIds.size() << "components:" << plan.missingDataComponentIds;
81 }
82
83 const auto initTypeProgress = [&owner](const QString& typeName) {
84 ExportTypeProgress progress;
85 progress.typeName = typeName;
86 progress.totalCount = owner.m_componentIds.size();
87 owner.m_progress.exportTypeProgress[typeName] = progress;
88 };
89
90 for (const QString& typeName : plan.progressTypeNames()) {
91 initTypeProgress(typeName);
92 }
93
94 const auto markMissingDataFailures = [&owner, &plan](const QString& typeName) {
95 for (const QString& componentId : plan.missingDataComponentIds) {
96 ExportItemStatus status;
97 status.status = ExportItemStatus::Status::Failed;
98 status.errorMessage = QStringLiteral("Component preload data missing");
99 owner.onExportItemStatusChanged(componentId, typeName, status);
100 }
101 };
102
103 if (!plan.missingDataComponentIds.isEmpty()) {
104 for (const QString& typeName : plan.progressTypeNames()) {
105 markMissingDataFailures(typeName);
106 }
107 }
108
109 owner.m_runningExportStages = plan.runningStageCount();
110 owner.logNetworkRuntimeStats(QStringLiteral("export-start"));
111
112 if (plan.exportableComponentIds.isEmpty()) {
113 qWarning() << "ParallelExportService: No exportable components after preload";
114 owner.m_progress.currentStage = ExportOverallProgress::Stage::Failed;
115 owner.m_progress.endTime = QDateTime::currentDateTime();
116 owner.writeExportDetailedReport(QStringLiteral("export-failed-no-exportable-components"));
117 emit owner.failed(QStringLiteral("No exportable components after preload"));
118 owner.cleanupExportStages();
119 return;
120 }
121
122 // 创建并启动各导出类型的 Stage。
123 if (enableSymbol) {
124 auto* stage = new SymbolExportStage(&owner);
125 stage->setOptions(owner.m_options);
126 owner.registerStageAndStart(stage, QStringLiteral("Symbol"), plan.exportableComponentIds, runGeneration);
127 }
128
129 if (enableFootprint) {
130 auto* stage = new FootprintExportStage(&owner);
131 stage->setOptions(owner.m_options);
132 owner.registerStageAndStart(
133 stage,
134 plan.symbolOnlyCombinedLibrary ? QStringLiteral("Symbol") : QStringLiteral("Footprint"),
135 plan.exportableComponentIds,
136 runGeneration);
137 }
138
139 if (runExternalModel3DStage) {
140 auto* stage = new Model3DExportStage(&owner);
141 stage->setOptions(owner.m_options);
142 owner.registerStageAndStart(stage, QStringLiteral("Model3D"), plan.exportableComponentIds, runGeneration);
143 }
144
145 if (enablePreview) {
146 auto* stage = new PreviewImagesExportStage(&owner);
147 stage->setOptions(owner.m_options);
148 owner.registerStageAndStart(stage, QStringLiteral("PreviewImages"), plan.exportableComponentIds, runGeneration);
149 }
150
151 if (enableDatasheet) {
152 auto* stage = new DatasheetExportStage(&owner);
153 stage->setOptions(owner.m_options);
154 owner.registerStageAndStart(stage, QStringLiteral("Datasheet"), plan.exportableComponentIds, runGeneration);
155 }
156
157 // 重试模式仅对本次导出生效，避免影响后续正常导出。
158 owner.m_options.retryMode = false;
159
160 if (owner.m_runningExportStages == 0) {
161 qWarning() << "ParallelExportService: No export types enabled";
162 owner.m_progress.currentStage = ExportOverallProgress::Stage::Completed;
163 owner.m_progress.endTime = QDateTime::currentDateTime();
164 owner.logNetworkRuntimeStats(QStringLiteral("export-no-types-enabled"));
165 owner.writeExportDetailedReport(QStringLiteral("export-no-types-enabled"));
166 emit owner.completed(0, 0);
167 }
168}
