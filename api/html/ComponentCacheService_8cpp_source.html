109QString ComponentCacheService::ensureComponentDir(const QString& lcscId) const {
110 // 注意：不再获取锁，因为：
111 // 1. componentCacheDir 只是构建路径字符串（只读操作）
112 // 2. dir.mkpath 是线程安全的
113 // 3. savePreviewImage 已经在持有 m_mutex 的情况下调用此函数
114 QString dirPath = componentCacheDir(lcscId);
115 if (dirPath.isEmpty()) {
116 return QString();
117 }
118 QDir dir(dirPath);
119 if (!dir.exists()) {
120 if (!dir.mkpath(dirPath))
121 return QString();
122 } else if (!CacheSafety::isOwnedComponentDirectory(cacheDir(), dirPath) &&
123 !dir.entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot).isEmpty()) {
124 qWarning() << "ensureComponentDir: refusing to reuse non-empty unowned directory:" << dirPath;
125 return QString();
126 }
127 return dirPath;
128}

610qint64 ComponentCacheService::calculateDirSize(const QString& dirPath) const {
611 // 累计当前目录及所有受控子目录中的文件大小。
612 qint64 size = 0;
613 QDir dir(dirPath);
614 if (!dir.exists()) {
615 return size;
616 }
617
618 for (const QFileInfo& info : dir.entryInfoList(QDir::Files)) {
619 size += info.size();
620 }
621
622 for (const QString& subDir : dir.entryList(QDir::Dirs)) {
623 if (subDir != "." && subDir != "..") {
624 size += calculateDirSize(dirPath + "/" + subDir);
625 }
626 }
627
628 return size;
629}

731QString ComponentCacheService::resolveDatasheetPath(const QString& lcscId,
732 const QString& preferredFormat,
733 bool forWrite) const {
734 // 计算元器件数据手册的无扩展名基础路径。
735 const QString basePath = datasheetPath(lcscId);
736 // 统一格式字符串，便于识别 pdf 或 html 请求。
737 const QString normalizedFormat = preferredFormat.toLower();
738
739 if (normalizedFormat.contains("pdf")) {
740 if (!forWrite && !QFileInfo::exists(basePath + ".pdf") && QFileInfo::exists(basePath + ".html")) {
741 return basePath + ".html";
742 }
743 return basePath + ".pdf";
744 }
745 if (normalizedFormat.contains("html")) {
746 if (!forWrite && !QFileInfo::exists(basePath + ".html") && QFileInfo::exists(basePath + ".pdf")) {
747 return basePath + ".pdf";
748 }
749 return basePath + ".html";
750 }
751
752 // 构造两种受支持格式对应的候选缓存路径。
753 const QString pdfPath = basePath + ".pdf";
754 // 构造 HTML 数据手册的候选缓存路径。
755 const QString htmlPath = basePath + ".html";
756
757 if (!forWrite) {
758 if (QFileInfo::exists(pdfPath)) {
759 return pdfPath;
760 }
761 if (QFileInfo::exists(htmlPath)) {
762 return htmlPath;
763 }
764 }
765
766 return pdfPath;
767}
