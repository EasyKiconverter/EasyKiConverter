172 void clearIntermediateData(bool log = true) {
173 // 注意：由于 mutex 可能被并发访问，不在这里锁定
174 // clearIntermediateData 通常在 ProcessWorker 中调用，
175 // 而 QtConcurrent 任务还在运行并可能调用 addDebugLog()
176 qint64 freed = 0; // 已释放的中间数据容量。
177
178 auto clearAndCount = [&](QByteArray& ba) {
179 qint64 size = ba.capacity(); // 当前字节数组的容量。
180 ba.clear();
181 ba.squeeze();
182 return size;
183 };
184
185 freed += clearAndCount(componentInfoRaw);
186 freed += clearAndCount(cinfoJsonRaw);
187 freed += clearAndCount(cadDataRaw);
188 freed += clearAndCount(cadJsonRaw);
189 freed += clearAndCount(advJsonRaw);
190
191 // model3DObjRaw 已经被解析到 Model3DData 中（作为 QString），可以清理
192 freed += clearAndCount(model3DObjRaw);
193
194 if (log && freed > 0) {
195 // qDebug() << "Freed" << freed << "bytes of intermediate data for" << componentId;
196 }
197 }

202 void clearStepData() {
203 QMutexLocker locker(&m_mutex);
204 model3DStepRaw.clear();

213 MemorySnapshot getMemorySnapshot() const {
214 MemorySnapshot snapshot; // 当前内存占用快照。
215 snapshot.rawDataSize = componentInfoRaw.capacity() + cinfoJsonRaw.capacity() + cadDataRaw.capacity() +
216 cadJsonRaw.capacity() + advJsonRaw.capacity() + model3DObjRaw.capacity() +
217 model3DStepRaw.capacity();
218
219 // 估算解析后的对象大小
220 if (model3DData) {
221 // QString 大小 = 字符数 * 2 + Qt 元数据开销（估算32字节）
222 snapshot.parsedModelSize += model3DData->rawObj().length() * 2 + sizeof(Model3DData) + 32;
223 }
224
225 if (symbolData) {
226 snapshot.parsedModelSize += sizeof(SymbolData);
227 snapshot.parsedModelSize += symbolData->pins().size() * sizeof(SymbolPin);
228 snapshot.parsedModelSize += symbolData->rectangles().size() * sizeof(SymbolRectangle);
229 snapshot.parsedModelSize += symbolData->circles().size() * sizeof(SymbolCircle);
230 snapshot.parsedModelSize += symbolData->arcs().size() * sizeof(SymbolArc);
231 snapshot.parsedModelSize += symbolData->polylines().size() * sizeof(SymbolPolyline);
232 }
233
234 if (footprintData) {
235 snapshot.parsedModelSize += sizeof(FootprintData);
236 snapshot.parsedModelSize += footprintData->pads().size() * sizeof(FootprintPad);
237 snapshot.parsedModelSize += footprintData->tracks().size() * sizeof(FootprintTrack);
238 snapshot.parsedModelSize += footprintData->circles().size() * sizeof(FootprintCircle);
239 snapshot.parsedModelSize += footprintData->arcs().size() * sizeof(FootprintArc);
240 snapshot.parsedModelSize += footprintData->texts().size() * sizeof(FootprintText);
241 }
242
243 snapshot.totalSize = snapshot.rawDataSize + snapshot.parsedModelSize;
244 return snapshot;
245 }
