64void ComponentCacheCadDataWriter::writeCadFile(const QString& componentId,
65 const QByteArray& data,
66 uint64_t expectedGeneration,
67 DataKind kind) {
68 if (!CacheDataValidator::isValidCadData(data))
69 return;
70
71 QMutexLocker diskLocker(&m_owner.m_diskWriteMutex); // 保护代次校验和磁盘写入。
72 if (!ComponentCacheWritePolicy::isAllowed(
73 m_owner.m_cacheGeneration.load(), expectedGeneration, m_owner.m_tombstones, componentId)) {
74 return;
75 }
76
77 QString path; // CAD 输出文件路径。
78 {
79 QMutexLocker locker(&m_owner.m_mutex); // 保护组件目录准备。
80 const QString directory = m_owner.ensureComponentDir(componentId); // 组件缓存目录。
81 if (directory.isEmpty())
82 return;
83 if (kind == DataKind::Symbol)
84 path = CacheFileLayout::symbolFile(directory);
85 else if (kind == DataKind::Footprint)
86 path = CacheFileLayout::footprintFile(directory);
87 else
88 path = CacheFileLayout::cadDataFile(directory);
89 }
90
91 const bool written = CacheMetadataStore::writeAtomically(path, data); // 原子写入结果。
92 // 日志中的数据类型名称。
93 const QString kindName = kind == DataKind::Symbol ? QStringLiteral("symbol")
94 : kind == DataKind::Footprint ? QStringLiteral("footprint")
95 : QStringLiteral("CAD data JSON");
96 if (written) {
97 LOG_DEBUG(LogModule::Core, "Saved {} to disk: {}", kindName, path);
98 m_owner.enforceDiskCacheLimit();
99 } else {
100 LOG_WARN(LogModule::Core, "Failed to write {}: {}", kindName, path);
101 }
102
103 if (kind == DataKind::Symbol)
104 m_owner.saveSymbolDataToMemory(componentId, data);
105 else if (kind == DataKind::Footprint)
106 m_owner.saveFootprintDataToMemory(componentId, data);
107}
