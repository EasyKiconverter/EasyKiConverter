24AltiumSchRectangle AltiumSymbolPrimitiveConverter::convertRectangle(const IR::SymbolRectangleIR& rect) {
25 AltiumSchRectangle altiumRect; // 转换后的 Altium 矩形记录。
26 altiumRect.locationX = AltiumCoord::mmToRaw(rect.x0);
27 altiumRect.locationY = AltiumCoord::mmToRaw(rect.y0);
28 altiumRect.cornerX = AltiumCoord::mmToRaw(rect.x1);
29 altiumRect.cornerY = AltiumCoord::mmToRaw(rect.y1);
30 altiumRect.lineWidth = AltiumCoord::lineWidthMmToIndex(rect.strokeWidth);
31 altiumRect.lineStyle = toAltiumLineStyle(rect.strokeStyle);
32 altiumRect.color = toAltiumColor(rect.strokeColor);
33 altiumRect.areaColor = rect.isFilled ? toAltiumColor(rect.fillColor) : 0xFFFFFF;
34 altiumRect.isSolid = rect.isFilled;
35 altiumRect.ownerPartId = toAltiumOwnerPartId(rect.partIndex);
36 return altiumRect;

87AltiumSchIeee AltiumSymbolPrimitiveConverter::convertIeee(const IR::SymbolIeeeIR& ieee) {
88 AltiumSchIeee altiumIeee; // 转换后的 Altium IEEE 记录。
89 altiumIeee.symbol = qBound(0, ieee.symbol, 34);
90 altiumIeee.locationX = AltiumCoord::mmToRaw(ieee.position.x());
91 altiumIeee.locationY = AltiumCoord::mmToRaw(ieee.position.y());
92 altiumIeee.scaleFactor = qMax(1, ieee.scaleFactor);
93 altiumIeee.orientation = ((ieee.orientation % 4) + 4) % 4;
94 altiumIeee.mirrored = ieee.mirrored;
95 altiumIeee.color = toAltiumColor(ieee.color);
96 altiumIeee.ownerPartId = toAltiumOwnerPartId(ieee.partIndex);
97 return altiumIeee;
98}
