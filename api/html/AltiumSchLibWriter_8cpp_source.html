81bool AltiumSchLibWriter::write(const QList<AltiumSchComponent>& components,
82 const QString& filePath,
83 const QString& libraryName) {
84 m_diagnostics.clear();
85 if (components.isEmpty()) {
86 m_diagnostics.append(QStringLiteral("Altium SchLib 输入组件为空，已拒绝写入"));
87 qWarning() << "AltiumSchLibWriter: Refusing to write an empty library";
88 return false;
89 }
90 if (filePath.trimmed().isEmpty()) {
91 m_diagnostics.append(QStringLiteral("Altium SchLib 输出路径为空，已拒绝写入"));
92 qWarning() << "AltiumSchLibWriter: Refusing to write without an output path";
93 return false;
94 }
95 if (!AltiumSchInputValidator::validate(*this, components)) {
96 return false;
97 }
98 m_fontRegistry.clear();
99 m_embeddedImageNames.clear();
100 m_uniqueIdCounter = 0;
101 m_nextIndexInSheet = 0;
102 m_libraryName = libraryName; // 当前输出库名称，用于生成实现记录中的 PcbLib 引用。
103
104 prepareImageStorageNames(components);
105
106 // 确保有默认字体
107 m_fontRegistry.getOrAdd(QStringLiteral("Times New Roman"), 10);
108 m_fontRegistry.registerTextFonts(components);
109
110 OLECompoundWriter ole; // 当前 SchLib 的 OLE 复合文档写入器。
111 if (!ole.create()) {
112 qWarning() << "AltiumSchLibWriter: Failed to create OLE compound document for" << filePath;
113 return false;
114 }
115
116 QStringList names; // 按输入顺序收集的组件名称。
117 for (const AltiumSchComponent& component : components) {
118 names.append(component.name);
119 }
120 const QStringList sectionKeys = AltiumWriterUtils::makeUniqueSectionKeys(names); // 组件唯一 OLE 存储键。
121
122 AltiumSchLibraryHeaderWriter headerWriter(*this);
123 headerWriter.writeFileHeader(ole, components);
124
125 headerWriter.writeSectionKeys(ole, components, sectionKeys);
126
127 // 写入每个元件的存储
128 for (int i = 0; i < components.size(); ++i) {
129 writeComponentStorage(ole, components[i], sectionKeys[i]);
130 }
131
132 writeImageStorage(ole, components);
133
134 if (ole.hasError()) {
135 qWarning() << "AltiumSchLibWriter: Failed to construct OLE document:" << ole.errorString();
136 return false;
137 }
138
139 return ole.saveToFile(filePath);
140}

201void AltiumSchLibWriter::writeComponentRecord(AltiumBinaryWriter& writer, const AltiumSchComponent& component) {
202 QMap<QString, QString> params; // 当前组件的 RECORD=1 参数。
203 params["RECORD"] = "1";
204 params["LibReference"] = component.name;
205
206 if (!component.description.isEmpty()) {
207 params["ComponentDescription"] = component.description;
208 }
209
210 params["PartCount"] = QString::number(qMax(1, component.partCount) + 1);
211 params["DisplayModeCount"] = "1";
212 params["IndexInSheet"] = "-1";
213 params["OwnerPartId"] = "-1";
214 params["CurrentPartId"] = "1";
215 params["LibraryPath"] = "*";
216 params["SourceLibraryName"] = "*";
217 params["SheetPartFileName"] = "*";
218 params["TargetFileName"] = "*";
219 params["ALLPINCOUNT"] = QString::number(component.pins.size());
220 if (!component.aliases.isEmpty())
221 params["Aliases"] = component.aliases.join(",");
222
223 addUniqueID(params);
224 writer.writeCStringParameterBlockUtf8(params);
225}

435int AltiumSchLibWriter::componentRecordCount(const AltiumSchComponent& component) const {
436 int validBezierCount = 0; // 具有四个控制点、可写出的 Bezier 数量。
437 for (const AltiumSchBezier& bezier : component.beziers) {
438 if (bezier.controlPoints.size() == 4)
439 ++validBezierCount;
440 }
441 const int graphics = // 所有图元记录数量。
442 component.pins.size() + component.rectangles.size() + component.roundRectangles.size() + // 所有图元记录数量。
443 component.lines.size() + component.arcs.size() + component.polygons.size() + component.ellipses.size() +
444 component.pies.size() + component.ellipticalArcs.size() + component.polylines.size() + component.paths.size() +
445 validBezierCount + component.ieeeSymbols.size() + component.texts.size() + component.textFrames.size() +
446 component.images.size();
447 // Component + graphics + 参数字段 + ImplementationList + implementation records.
448 // 每个实现包含 RECORD=45、46、48，以及每个符号引脚对应的 RECORD=47。
449 return 1 + graphics + componentParameterRecordCount(component) + 1 +
450 component.implementations.size() * (3 + component.pins.size());
451}

479int AltiumSchLibWriter::normalizeOwnerPartId(int ownerPartId, const QString& context) {
480 if (ownerPartId == 0 || ownerPartId < -1 || ownerPartId > 32767) {
481 const int normalized = ownerPartId < 0 ? -1 : qBound(1, ownerPartId, 32767); // 规范化后的部件编号。
482 const QString diagnostic =
483 QStringLiteral("Altium SchLib %1 OWNERPARTID=%2 无效，已规范化为 %3") // 部件编号修正诊断。
484 .arg(context)
485 .arg(ownerPartId)
486 .arg(normalized);
487 m_diagnostics.append(diagnostic);
488 qWarning() << "AltiumSchLibWriter:" << diagnostic;
489 return normalized;
490 }
491 return ownerPartId;
492}

502 if (std::isfinite(angle))
503 return angle;

507 m_diagnostics.append(diagnostic);
