80AsyncNetworkRequest* AsyncNetworkRequest::createFinished(const NetworkResult& result, QObject* parent) {
81 auto* request =
82 new AsyncNetworkRequest(QUrl(), nullptr, result.diagnostic.resourceType, RetryPolicy(), QByteArray(), parent);
83 if (result.wasCancelled) {
84 request->m_cancelled.storeRelease(1);
85 }
86 // 设置已完成状态（同步），供调用方立即检查 isFinished() / result()
87 {
88 QMutexLocker locker(&request->m_resultMutex);
89 request->m_result = result;
90 }
91 request->m_finished.storeRelease(1);
92
93 // 延迟发射 finished 信号，确保调用方有机会先 connect
94 // 使用 QTimer::singleShot(0, ...) 投递到当前线程的事件循环
95 // 调用方必须在有事件循环的线程中使用（主线程、QThread 子类）
96 QTimer::singleShot(0, request, [request, result]() { emit request->finished(result); });
97 return request;

253void AsyncNetworkRequest::startAttempt(int attemptNumber) {
254 if (isCancelled() || isFinished()) {
255 return;
256 }
257
258 m_currentAttemptTimedOut.storeRelease(0);
259
260 if (!m_networkManager) {
261 NetworkResult result = m_result;
262 result.error = "Network manager is null";
263 result.success = false;
264 result.wasCancelled = false;
265 result.diagnostic.errorType = NetworkErrorType::Other;
266 result.diagnostic.errorMessage = result.error;
267 completeWithResult(result);
268 return;
269 }
270
271 QNetworkRequest request(m_url);
272 request.setHeader(QNetworkRequest::UserAgentHeader, "EasyKiConverter/1.0");
273 request.setRawHeader("X-Requested-With", "XMLHttpRequest");
274 request.setAttribute(QNetworkRequest::CacheLoadControlAttribute, QNetworkRequest::AlwaysNetwork);
275 // 禁止不安全的重定向（HTTPS -> HTTP），防止降级攻击
276 request.setAttribute(QNetworkRequest::RedirectPolicyAttribute, QNetworkRequest::NoLessSafeRedirectPolicy);
277 // Note: We handle gzip manually in processResponse() since Qt doesn't auto-decompress
278
279 // Create the reply - use POST if body is provided
280 QNetworkReply* reply = nullptr;
281 if (m_isPostRequest && !m_postBody.isEmpty()) {
282 // Auto-detect Content-Type based on body format
283 // If body contains = and &, it's likely URL-encoded form data
284 if (m_postBody.contains('=') && m_postBody.contains('&')) {
285 request.setHeader(QNetworkRequest::ContentTypeHeader, "application/x-www-form-urlencoded");
286 } else {
287 request.setHeader(QNetworkRequest::ContentTypeHeader, "application/json");
288 }
289 reply = m_networkManager->post(request, m_postBody);
290 } else {
291 reply = m_networkManager->get(request);
292 }
293
294 {
295 QMutexLocker locker(&m_repliesMutex);
296 m_currentReply = reply;
297 m_pendingReplies.append(reply);
298 }
299
300 // Set up timeout (no parent - created/destroyed in same thread pool thread)
301 m_timeoutTimer = new QTimer();
302 m_timeoutTimer->setSingleShot(true);
303 connect(m_timeoutTimer, &QTimer::timeout, this, &AsyncNetworkRequest::handleTimeout, Qt::DirectConnection);
304 if (m_timeoutMs > 0) {
305 m_timeoutTimer->start(m_timeoutMs);
306 }
307
308 // Connect signals - use DirectConnection since we want immediate handling
309 // The slot will run in this object's thread (main thread where AsyncNetworkRequest lives)
310 connect(reply, &QNetworkReply::finished, this, &AsyncNetworkRequest::handleAttemptFinished, Qt::DirectConnection);
311 connect(reply,
312 &QNetworkReply::downloadProgress,
313 this,
314 &AsyncNetworkRequest::handleDownloadProgress,
315 Qt::DirectConnection);
316
317 qDebug() << "AsyncNetworkRequest: Attempt" << attemptNumber << "started for" << m_url;
318}

409void AsyncNetworkRequest::processResponse(QNetworkReply* reply) {
410 NetworkResult result;
411 {
412 QMutexLocker locker(&m_resultMutex);
413 result = m_result;
414 }
415
416 result.statusCode = reply->attribute(QNetworkRequest::HttpStatusCodeAttribute).toInt();
417 result.diagnostic.statusCode = result.statusCode;
418 result.diagnostic.retryCount = m_currentRetryCount;
419
420 // Check for error
421 if (reply->error() != QNetworkReply::NoError) {
422 QString errorString = reply->errorString();
423
424 // Check if cancelled
425 if (isCancelled()) {
426 completeWithResult(cancelledResult());
427 return;
428 }
429
430 if (m_currentAttemptTimedOut.loadAcquire() != 0) {
431 if (shouldRetryInternal(result.statusCode, QNetworkReply::TimeoutError, m_currentRetryCount)) {
432 scheduleRetry(m_currentRetryCount);
433 } else {
434 completeWithResult(timeoutResult());
435 }
436 return;
437 }
438
439 // Check if should retry
440 if (shouldRetryInternal(result.statusCode, reply->error(), m_currentRetryCount)) {
441 scheduleRetry(m_currentRetryCount);
442 return;
443 }
444
445 // Final error
446 captureErrorResponseDiagnostics(reply, result);
447 result.error =
448 result.statusCode > 0 ? QStringLiteral("HTTP %1: %2").arg(result.statusCode).arg(errorString) : errorString;
449 result.success = false;
450 result.wasCancelled = false;
451 result.diagnostic.errorMessage = result.error;
452
453 // 依据底层网络错误和 HTTP 状态码生成统一分类。
454 switch (reply->error()) {
455 case QNetworkReply::TimeoutError:
456 result.diagnostic.errorType = NetworkErrorType::Timeout;
457 break;
458 case QNetworkReply::ConnectionRefusedError:
459 result.diagnostic.errorType = NetworkErrorType::ConnectionRefused;
460 break;
461 case QNetworkReply::HostNotFoundError:
462 result.diagnostic.errorType = NetworkErrorType::HostNotFound;
463 break;
464 case QNetworkReply::SslHandshakeFailedError:
465 result.diagnostic.errorType = NetworkErrorType::SSLError;
466 break;
467 case QNetworkReply::OperationCanceledError:
468 result.diagnostic.errorType = NetworkErrorType::Canceled;
469 result.diagnostic.wasCanceled = true;
470 break;
471 default:
472 if (result.statusCode == 429) {
473 result.diagnostic.errorType = NetworkErrorType::RateLimited;
474 result.diagnostic.wasRateLimited = true;
475 } else if (result.statusCode >= 500) {
476 result.diagnostic.errorType = NetworkErrorType::ServerError;
477 } else if (result.statusCode == 404) {
478 result.diagnostic.errorType = NetworkErrorType::NotFound;
479 } else if (result.statusCode == 403) {
480 result.diagnostic.errorType = NetworkErrorType::Forbidden;
481 } else {
482 result.diagnostic.errorType = NetworkErrorType::Other;
483 }
484 break;
485 }
486 completeWithResult(result);
487 return;
488 }
489
490 // Success - read data
491 QByteArray data = reply->readAll();
492
493 // 安全兜底：检查实际读取的数据大小（Content-Length 可能缺失或不准确）
494 {
495 const RequestProfile profile = RequestProfiles::fromType(m_resourceType);
496 if (profile.maxResponseBytes > 0 && data.size() > profile.maxResponseBytes) {
497 result.error = QStringLiteral("Response too large: %1 bytes (limit %2)")
498 .arg(data.size())
499 .arg(profile.maxResponseBytes);
500 result.success = false;
501 result.wasCancelled = false;
502 result.diagnostic.errorType = NetworkErrorType::Other;
503 result.diagnostic.errorMessage = result.error;
504 completeWithResult(result);
505 return;
506 }
507 }
508
509 // 校验重定向后的最终 URL 是否仍在白名单内
510 {
511 const QUrl finalUrl = reply->url();
512 if (finalUrl != m_url) {
513 QString errorMsg;
514 if (!UrlUtils::isAllowedUrl(finalUrl, m_resourceType, &errorMsg)) {
515 result.error = QStringLiteral("Redirect to disallowed URL: %1").arg(errorMsg);
516 result.success = false;
517 result.wasCancelled = false;
518 result.diagnostic.errorType = NetworkErrorType::Other;
519 result.diagnostic.errorMessage = result.error;
520 completeWithResult(result);
521 return;
522 }
523 }
524 }
525
526 // Check for gzip encoding and decompress if needed
527 QVariant encodingVariant = reply->rawHeader("Content-Encoding");
528 QString encoding = encodingVariant.toString().toLower();
529 m_gzipUsed = encoding.contains("gzip");
530
531 if (m_gzipUsed || GzipUtils::isGzipped(data)) {
532 GzipUtils::DecompressResult decompResult = GzipUtils::decompress(data);
533 if (decompResult.success) {
534 data = decompResult.data;
535
536 // 解压后重新检查大小限制（压缩可能放大数据）
537 const RequestProfile profile = RequestProfiles::fromType(m_resourceType);
538 if (profile.maxResponseBytes > 0 && data.size() > profile.maxResponseBytes) {
539 result.error = QStringLiteral("Decompressed response too large: %1 bytes (limit %2)")
540 .arg(data.size())
541 .arg(profile.maxResponseBytes);
542 result.success = false;
543 result.wasCancelled = false;
544 result.diagnostic.errorType = NetworkErrorType::Other;
545 result.diagnostic.errorMessage = result.error;
546 completeWithResult(result);
547 return;
548 }
549 } else {
550 result.error = "Gzip decompression failed";
551 result.success = false;
552 result.wasCancelled = false;
553 result.diagnostic.errorType = NetworkErrorType::DecompressionFailed;
554 result.diagnostic.errorMessage = result.error;
555 completeWithResult(result);
556 return;
557 }
558 }
559
560 result.data = data;
561 result.success = true;
562 result.retryCount = m_currentRetryCount;
563 result.wasCancelled = false;
564 result.diagnostic.errorType = NetworkErrorType::None;
565 result.diagnostic.errorMessage.clear();
566
567 completeWithResult(result);
568}

574void AsyncNetworkRequest::scheduleRetry(int retryCount) {
575 if (isCancelled() || isFinished()) {
576 return;
577 }
578
579 int delay = calculateRetryDelay(retryCount);
580 m_currentRetryCount = retryCount + 1;
581
582 qDebug() << "AsyncNetworkRequest: Scheduling retry" << m_currentRetryCount << "for" << m_url << "after" << delay
583 << "ms";
584
585 emit retrying(m_currentRetryCount, delay);
586
587 // Use a timer to delay retry - this allows cancellation to be checked
588 QTimer::singleShot(delay, this, [this]() {
589 if (isCancelled() || isFinished()) {
590 return;
591 }
592
593 // Clean up old reply
594 cleanupAttempt();
595
596 // Start new attempt in main thread where QNetworkAccessManager lives
597 QMetaObject::invokeMethod(this, [this]() { startAttempt(m_currentRetryCount); }, Qt::QueuedConnection);
598 });
599}

606int AsyncNetworkRequest::calculateRetryDelay(int retryCount) const {
607 if (retryCount >= static_cast<int>(m_policy.delays.size())) {
608 return m_policy.delays.back();
609 }
610 int baseDelay = m_policy.delays[retryCount];
611
612 // Add jitter
613 int jitterRange = static_cast<int>(baseDelay * m_policy.jitterFactor);
614 int jitter = QRandomGenerator::global()->bounded(-jitterRange, jitterRange + 1);
615 return baseDelay + jitter;
616}

625bool AsyncNetworkRequest::shouldRetryInternal(int statusCode, QNetworkReply::NetworkError error, int retryCount) const {
626 if (retryCount >= m_policy.maxRetries) {
627 return false;
628 }
629
630 if (m_policy.retryableStatusCodes.contains(statusCode)) {
631 return true;
632 }
633
634 // 网络层瞬时错误可重试，明确的业务拒绝不重试。
635 switch (error) {
636 case QNetworkReply::TimeoutError:
637 case QNetworkReply::TemporaryNetworkFailureError:
638 case QNetworkReply::NetworkSessionFailedError:
639 case QNetworkReply::UnknownNetworkError:
640 case QNetworkReply::RemoteHostClosedError:
641 case QNetworkReply::HostNotFoundError:
642 case QNetworkReply::ConnectionRefusedError:
643 case QNetworkReply::ProxyTimeoutError:
644 return true;
645 default:
646 return false;
647 }
648}
