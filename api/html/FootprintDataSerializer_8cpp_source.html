109QJsonObject FootprintDataSerializer::toJson(const FootprintData& data) {
110 QJsonObject json; // 保存完整封装数据的 JSON 对象。
111
112 // 基本信息
113 json["info"] = toJson(data.info());
114
115 // 边界
116 json["bbox"] = toJson(data.bbox());
117
118 // 焊盘
119 QJsonArray padsArray; // 保存焊盘序列化结果。
120 for (const FootprintPad& pad : data.pads()) {
121 padsArray.append(toJson(pad));
122 }
123 json["pads"] = padsArray;
124
125 // 走线
126 QJsonArray tracksArray; // 保存走线序列化结果。
127 for (const FootprintTrack& track : data.tracks()) {
128 tracksArray.append(toJson(track));
129 }
130 json["tracks"] = tracksArray;
131
132 // 圆弧
133 QJsonArray holesArray; // 保存孔序列化结果。
134 for (const FootprintHole& hole : data.holes()) {
135 holesArray.append(toJson(hole));
136 }
137 json["holes"] = holesArray;
138
139 // 圆弧
140 QJsonArray circlesArray; // 保存圆序列化结果。
141 for (const FootprintCircle& circle : data.circles()) {
142 circlesArray.append(toJson(circle));
143 }
144 json["circles"] = circlesArray;
145
146 // 矩形
147 QJsonArray rectanglesArray; // 保存矩形序列化结果。
148 for (const FootprintRectangle& rect : data.rectangles()) {
149 rectanglesArray.append(toJson(rect));
150 }
151 json["rectangles"] = rectanglesArray;
152
153 // 圆弧
154 QJsonArray arcsArray; // 保存圆弧序列化结果。
155 for (const FootprintArc& arc : data.arcs()) {
156 arcsArray.append(toJson(arc));
157 }
158 json["arcs"] = arcsArray;
159
160 // 文本
161 QJsonArray textsArray; // 保存文本序列化结果。
162 for (const FootprintText& text : data.texts()) {
163 textsArray.append(toJson(text));
164 }
165 json["texts"] = textsArray;
166
167 // 实体填充区域
168 QJsonArray solidRegionsArray; // 保存实体填充区域序列化结果。
169 for (const FootprintSolidRegion& solidRegion : data.solidRegions()) {
170 solidRegionsArray.append(toJson(solidRegion));
171 }
172 json["solid_regions"] = solidRegionsArray;
173
174 // 外形轮廓
175 QJsonArray outlinesArray; // 保存外形轮廓序列化结果。
176 for (const FootprintOutline& outline : data.outlines()) {
177 outlinesArray.append(toJson(outline));
178 }
179 json["outlines"] = outlinesArray;
180
181 // 层定位
182 QJsonArray layersArray; // 保存层定义序列化结果。
183 for (const LayerDefinition& layer : data.layers()) {
184 layersArray.append(toJson(layer));
185 }
186 json["layers"] = layersArray;
187
188 // 对象可见
189 QJsonArray objectVisibilitiesArray; // 保存对象可见性序列化结果。
190 for (const ObjectVisibility& visibility : data.objectVisibilities()) {
191 objectVisibilitiesArray.append(toJson(visibility));
192 }
193 json["object_visibilities"] = objectVisibilitiesArray;
194 json["validation_errors"] = QJsonArray::fromStringList(data.validationErrors());
195
196 return json;
197}
