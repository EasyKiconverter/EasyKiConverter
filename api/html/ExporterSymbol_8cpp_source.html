56bool ExporterSymbol::exportSymbolLibrary(const QList<IR::SymbolComponentIR>& symbols,
57 const QString& libName,
58 const QString& filePath,
59 bool appendMode,
60 bool updateMode,
61 const QString& libraryDescription) {
62 qDebug() << "=== Export Symbol Library ===";
63 qDebug() << "Library name:" << libName;
64 qDebug() << "Output path:" << filePath;
65 qDebug() << "Symbol count:" << symbols.count();
66 qDebug() << "KiCad version: V6";
67 qDebug() << "Append mode:" << appendMode;
68 qDebug() << "Update mode:" << updateMode;
69 qDebug() << "Library description:" << libraryDescription;
70
71 // 重置检测到的版本，避免影响后续调用
72 m_detectedVersion.clear();
73
74 QFile file(filePath);
75 bool fileExists = file.exists(); // 输出库文件是否已经存在。
76
77 // 如果文件不存在，直接创建新库
78 if (!fileExists) {
79 qDebug() << "Creating new symbol library...";
80 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
81 qWarning() << "Failed to open file for writing:" << filePath;
82 return false;
83 }
84
85 QTextStream out(&file);
86 out.setEncoding(QStringConverter::Utf8);
87
88 // 生成头部
89 out << generateHeader(libName);
90
91 // 生成所有符号
92 int index = 0;
93 for (const IR::SymbolComponentIR& symbol : symbols) {
94 qDebug() << "Exporting symbol" << (++index) << "of" << symbols.count() << ":" << symbol.name;
95 out << generateSymbolContent(symbol, libName);
96 }
97
98 // 生成尾部
99 out << ")\n"; // 闭合 kicad_symbol_lib
100
101 file.close();
102 qDebug() << "Symbol library created successfully:" << filePath;
103 return true;
104 }
105
106 // 文件存在，需要处理追加或更新
107 qDebug() << "Existing library found, reading content...";
108
109 // 读取现有库内容
110 QString existingContent;
111 if (file.open(QIODevice::ReadOnly | QIODevice::Text)) {
112 existingContent = QTextStream(&file).readAll();
113 file.close();
114 qDebug() << "Read" << existingContent.length() << "bytes from existing library";
115 } else {
116 qWarning() << "Failed to open existing library for reading:" << filePath;
117 return false;
118 }
119
120 // 检测现有文件的版本
121 QRegularExpression versionRegex(R"(^\s*\‍(version\s+(\d+)\))", QRegularExpression::MultilineOption);
122 QRegularExpressionMatch versionMatch = versionRegex.match(existingContent); // 已有库版本匹配结果。
123 if (versionMatch.hasMatch()) {
124 m_detectedVersion = versionMatch.captured(1);
125 qDebug() << "Detected existing library version:" << m_detectedVersion;
126 } else {
127 m_detectedVersion = "20211014";
128 qDebug() << "Could not detect version, using default:" << m_detectedVersion;
129 }
130
131 // 提取现有符号
132 QMap<QString, QString> existingSymbols; // 符号名-> 符号内容
133 QSet<QString> subSymbolNames; // 属于分体式符号的子符号名称
134
135 // 使用栈来正确追踪符号的嵌套关系
136 QStringList lines = existingContent.split('\n'); // 已有库的逐行内容。
137 int braceCount = 0; // 当前符号定义的括号嵌套层级。
138
139 // 栈元素: (symbolName, startLine, braceCountAfterStart)
140 struct SymbolInfo {
141 QString name; // 已发现符号的名称。
142 int startLine; // 符号定义的起始行号。
143 int braceAfterStart; // 起始行之后的括号层级。
144 };
145
146 QList<SymbolInfo> symbolStack; // 正在解析的嵌套符号栈。
147
148 for (int i = 0; i < lines.size(); ++i) {
149 QString line = lines[i]; // 当前正在解析的库文件行。
150
151 // 逐字符计算括号变化
152 for (int j = 0; j < line.length(); ++j) {
153 if (line[j] == '(') {
154 braceCount++;
155 } else if (line[j] == ')') {
156 braceCount--;
157 }
158 }
159
160 // 检查是否有符号开始
161 int symbolPosInLine = line.indexOf("(symbol \""); // 当前行中的符号定义位置。
162 if (symbolPosInLine >= 0) {
163 int nameStart = line.indexOf("\"", symbolPosInLine) + 1; // 符号名称起始位置。
164 int nameEnd = line.indexOf("\"", nameStart); // 符号名称结束位置。
165 if (nameEnd > nameStart) {
166 QString symbolName = line.mid(nameStart, nameEnd - nameStart); // 已有符号名称。
167 SymbolInfo info; // 当前符号的范围信息。
168 info.name = symbolName;
169 info.startLine = i;
170 info.braceAfterStart = braceCount;
171 symbolStack.append(info);
172 qDebug() << "Found symbol start:" << symbolName << "at line" << i << "with braceCount" << braceCount;
173 }
174 }
175
176 // 检查是否有符号结束 - 当 braceCount 回到符号开始后的值时
177 while (!symbolStack.isEmpty() && braceCount < symbolStack.last().braceAfterStart) {
178 SymbolInfo info = symbolStack.last(); // 已完成符号的范围信息。
179 symbolStack.removeLast();
180
181 // 检查是否是子符号 - 如果栈不为空，说明有父符号
182 if (!symbolStack.isEmpty()) {
183 // 这是一个子符号，不应该被单独提取
184 subSymbolNames.insert(info.name);
185 qDebug() << "Found sub-symbol (skipping):" << info.name << "inside parent" << symbolStack.last().name;
186 } else {
187 // 这是一个顶层符号，提取它
188 QString symbolContent; // 已有符号的完整文本。
189 for (int k = info.startLine; k <= i; ++k) {
190 symbolContent += lines[k] + "\n";
191 }
192 existingSymbols[info.name] = symbolContent;
193 qDebug() << "Extracted top-level symbol:" << info.name << "from lines" << info.startLine << "-" << i;
194 }
195 }
196 }
197
198 qDebug() << "Existing symbols count:" << existingSymbols.count();
199 qDebug() << "Sub-symbol names:" << subSymbolNames;
200
201 // 确定要导出的符号
202 QList<IR::SymbolComponentIR> symbolsToExport; // 实际需要写入的符号集合。
203 int overwriteCount = 0; // 被覆盖的同名符号数量。
204 int appendCount = 0; // 新追加符号数量。
205 int skipCount = 0; // 被跳过的符号数量。
206
207 for (const IR::SymbolComponentIR& symbol : symbols) {
208 QString symbolName = symbol.name; // 当前待处理符号名称。
209
210 if (existingSymbols.contains(symbolName)) {
211 if (appendMode && !updateMode) {
212 // 追加模式（非更新）：跳过已存在的符号
213 qDebug() << "Symbol already exists, skipping (append mode):" << symbolName;
214 skipCount++;
215 } else {
216 // 更新模式或覆盖模式：替换已存在的符号
217 qDebug() << "Symbol already exists, overwriting (update mode):" << symbolName;
218 overwriteCount++;
219 symbolsToExport.append(symbol);
220 }
221 } else {
222 // 新符号
223 qDebug() << "New symbol, adding:" << symbolName;
224 appendCount++;
225 symbolsToExport.append(symbol);
226 }
227 }
228
229 qDebug() << "Symbols to export:" << symbolsToExport.count() << "(Overwrite:" << overwriteCount
230 << ", Append:" << appendCount << ", Skip:" << skipCount << ")";
231
232 // 如果没有符号需要导出，直接返回
233 if (symbolsToExport.isEmpty()) {
234 qDebug() << "No symbols to export, skipping";
235 return true;
236 }
237
238 // 打开文件进行写入
239 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
240 qWarning() << "Failed to open file for writing:" << filePath;
241 return false;
242 }
243
244 QTextStream out(&file);
245 out.setEncoding(QStringConverter::Utf8);
246
247 // 生成头部
248 out << generateHeader(libName);
249
250 // 生成所有符号（包括未覆盖的现有符号和新导出的符号）
251 int index = 0; // 当前导出符号序号。
252
253 // 收集要被覆盖的符号名
254 QSet<QString> overwrittenSymbolNames; // 需要替换的已有符号名称集合。
255 for (const IR::SymbolComponentIR& symbol : symbolsToExport) {
256 overwrittenSymbolNames.insert(symbol.name);
257 }
258
259 // 收集要被删除的子符号名称（属于被覆盖的父符号的子符号）
260 QSet<QString> subSymbolsToDelete; // 因覆盖操作需要删除的子符号集合。
261
262 // 首先分析现有符号，确定哪些是分体式符号（有多个子符号）
263 QMap<QString, QStringList> parentToSubSymbols; // 父符号名 -> 子符号列
264 for (const QString& subSymbolName : subSymbolNames) {
265 // 子符号格式：{parentName}_{unitNumber}_1
266 int lastUnderscore = subSymbolName.lastIndexOf('_'); // 子符号末段分隔符位置。
267 if (lastUnderscore > 0) {
268 int secondLastUnderscore = subSymbolName.lastIndexOf('_', lastUnderscore - 1); // 部件编号分隔符位置。
269 if (secondLastUnderscore > 0) {
270 QString parentName = subSymbolName.left(secondLastUnderscore); // 父符号名称。
271 parentToSubSymbols[parentName].append(subSymbolName);
272 }
273 }
274 }
275
276 // 对于每个被覆盖的父符号，收集其所有子符号
277 for (const QString& parentSymbolName : overwrittenSymbolNames) {
278 if (parentToSubSymbols.contains(parentSymbolName)) {
279 // 这是一个分体式符号，删除所有子符号
280 for (const QString& subSymbolName : parentToSubSymbols[parentSymbolName]) {
281 subSymbolsToDelete.insert(subSymbolName);
282 qDebug() << "Marking sub-symbol for deletion (multipart):" << subSymbolName
283 << "(parent:" << parentSymbolName << ")";
284 }
285 } else {
286 // 这是一个单体符号，查找其子符号
287 QString expectedSubSymbolName_v1 = parentSymbolName + "_0_1"; // V1 格式的预期子符号名称。
288 QString expectedSubSymbolName_v2 = parentSymbolName + "_1_1"; // V2 格式的预期子符号名称。
289 if (subSymbolNames.contains(expectedSubSymbolName_v1)) {
290 subSymbolsToDelete.insert(expectedSubSymbolName_v1);
291 qDebug() << "Marking sub-symbol for deletion (single part v1):" << expectedSubSymbolName_v1
292 << "(parent:" << parentSymbolName << ")";
293 } else if (subSymbolNames.contains(expectedSubSymbolName_v2)) {
294 subSymbolsToDelete.insert(expectedSubSymbolName_v2);
295 qDebug() << "Marking sub-symbol for deletion (single part v2):" << expectedSubSymbolName_v2
296 << "(parent:" << parentSymbolName << ")";
297 }
298 }
299 }
300
301 // 先导出未覆盖的现有符号（需要过滤掉被覆盖的子符号和以被覆盖符号名开头的顶层符号）
302 for (const QString& symbolName : existingSymbols.keys()) {
303 bool isOverwritten = overwrittenSymbolNames.contains(symbolName); // 当前符号是否需要覆盖。
304
305 // 检查是否是以被覆盖符号名开头的顶层符号
306 bool isOrphanedSubSymbol = false; // 当前子符号是否失去父符号。
307 if (!isOverwritten) {
308 for (const QString& parentSymbolName : overwrittenSymbolNames) {
309 if (symbolName.startsWith(parentSymbolName + "_")) {
310 isOrphanedSubSymbol = true;
311 qDebug() << "Skipping orphaned sub-symbol (top-level):" << symbolName
312 << "(parent:" << parentSymbolName << ")";
313 break;
314 }
315 }
316 }
317
318 if (!isOverwritten && !isOrphanedSubSymbol) {
319 // 导出未覆盖的符号，但需要过滤掉子符号
320 QString symbolContent = existingSymbols[symbolName]; // 已有符号的文本内容。
321 QStringList contentLines = symbolContent.split('\n'); // 已有符号的逐行内容。
322 QString filteredContent; // 过滤删除目标后的符号文本。
323 bool skipNextSymbol = false; // 是否正在跳过嵌套子符号。
324 int nestedBraceCount = 0; // 被跳过子符号的括号层级。
325
326 for (const QString& line : contentLines) {
327 QString trimmedLine = line.trimmed(); // 去除缩进后的当前行。
328
329 // 检查是否是子符号定义的开始
330 if (trimmedLine.startsWith("(symbol \"")) {
331 int nameStart = trimmedLine.indexOf("\"") + 1; // 子符号名称起始位置。
332 int nameEnd = trimmedLine.indexOf("\"", nameStart); // 子符号名称结束位置。
333 if (nameEnd > nameStart) {
334 QString subSymbolName = trimmedLine.mid(nameStart, nameEnd - nameStart); // 当前子符号名称。
335 if (subSymbolsToDelete.contains(subSymbolName)) {
336 skipNextSymbol = true;
337 nestedBraceCount = 1;
338 qDebug() << "Skipping deleted sub-symbol:" << subSymbolName;
339 continue;
340 }
341 }
342 }
343
344 if (skipNextSymbol) {
345 // 计算括号数量
346 for (int j = 0; j < line.length(); ++j) {
347 if (line[j] == '(')
348 nestedBraceCount++;
352 else if (line[j] == ')') {
353 nestedBraceCount--;
354 if (nestedBraceCount == 0) {
355 skipNextSymbol = false;
356 break;
357 }
358 }
359 }
360 continue;
361 }
362
363 filteredContent += line + "\n";
364 }
365
366 out << filteredContent;
367 qDebug() << "Keeping existing symbol:" << symbolName;
368 }
369 }
370
371 // 再导出新符号和被覆盖的符号
372 for (const IR::SymbolComponentIR& symbol : symbolsToExport) {
373 qDebug() << "Exporting symbol" << (++index) << "of" << symbolsToExport.count() << ":" << symbol.name;
374 out << generateSymbolContent(symbol, libName);
375 }
376
377 // 生成尾部
378 out << ")\n"; // 闭合 kicad_symbol_lib
379
380 file.close();
381 qDebug() << "Symbol library exported successfully:" << filePath;
382 return true;

408QString ExporterSymbol::generateSymbolContent(const IR::SymbolComponentIR& symbol, const QString& libName) const {
409 QString content; // 当前符号的序列化文本。
410
411 // V6 格式 - 主符号定义（包含属性）
412 QString cleanSymbolName = symbol.name; // 清理后的符号名称。
413 content += QString(" (symbol \"%1\"\n").arg(cleanSymbolName);
414 content += " (in_bom yes)\n";
415 content += " (on_board yes)\n";
416
422 double originX = 0.0; // 符号逻辑原点 X 坐标。
423 double originY = 0.0; // 符号逻辑原点 Y 坐标。
424 const QList<IR::SymbolPinIR>& pins = symbol.pins; // 符号引脚列表。
425 if (!symbol.preserveLogicalOrigin) {
426 double minOriginX = std::numeric_limits<double>::max(); // 几何原点包围盒最小 X。
427 double minOriginY = std::numeric_limits<double>::max(); // 几何原点包围盒最小 Y。
428 double maxOriginX = -std::numeric_limits<double>::max(); // 几何原点包围盒最大 X。
429 double maxOriginY = -std::numeric_limits<double>::max(); // 几何原点包围盒最大 Y。
430 auto includeOriginPoint = [&](double x, double y) {
431 minOriginX = qMin(minOriginX, x);
432 minOriginY = qMin(minOriginY, y);
433 maxOriginX = qMax(maxOriginX, x);
434 maxOriginY = qMax(maxOriginY, y);
435 };
436 for (const auto& pin : pins) {
437 includeOriginPoint(pin.position.x(), pin.position.y());
438 const double dx = pin.direction == IR::PinDirection::Right ? pin.length // 引脚方向投影到 X 轴的长度。
439 : pin.direction == IR::PinDirection::Left ? -pin.length
440 : 0.0;
441 const double dy = pin.direction == IR::PinDirection::Up ? pin.length // 引脚方向投影到 Y 轴的长度。
442 : pin.direction == IR::PinDirection::Down ? -pin.length
443 : 0.0;
444 includeOriginPoint(pin.position.x() + dx, pin.position.y() + dy);
445 }
446 for (const auto& rect : symbol.rectangles) {
447 includeOriginPoint(rect.x0, rect.y0);
448 includeOriginPoint(rect.x1, rect.y1);
449 }
450 for (const auto& circle : symbol.circles) {
451 includeOriginPoint(circle.center.x() - circle.radius, circle.center.y() - circle.radius);
452 includeOriginPoint(circle.center.x() + circle.radius, circle.center.y() + circle.radius);
453 }
454 for (const auto& ellipse : symbol.ellipses) {
455 includeOriginPoint(ellipse.center.x() - ellipse.radiusX, ellipse.center.y() - ellipse.radiusY);
456 includeOriginPoint(ellipse.center.x() + ellipse.radiusX, ellipse.center.y() + ellipse.radiusY);
457 }
458 for (const auto& arc : symbol.arcs) {
459 includeOriginPoint(arc.startPoint.x(), arc.startPoint.y());
460 includeOriginPoint(arc.midPoint.x(), arc.midPoint.y());
461 includeOriginPoint(arc.endPoint.x(), arc.endPoint.y());
462 }
463 for (const auto& polyline : symbol.polylines) {
464 for (const QPointF& point : polyline.points)
465 includeOriginPoint(point.x(), point.y());
466 }
467 for (const auto& polygon : symbol.polygons) {
468 for (const QPointF& point : polygon.points)
469 includeOriginPoint(point.x(), point.y());
470 }
471 for (const auto& path : symbol.paths) {
472 for (const QPointF& point : path.points)
473 includeOriginPoint(point.x(), point.y());
474 }
475 if (minOriginX <= maxOriginX) {
476 originX = (minOriginX + maxOriginX) / 2.0;
477 originY = (minOriginY + maxOriginY) / 2.0;
478 }
479 }
480
481 // 设置原点偏移，使所有图形元素相对于该点定位
482 m_graphicsGenerator.setCurrentOrigin(originX, originY);
483
484 // 计算所有图形元素的实际边界，用于文本左右居中对齐
485 // 多部件符号的属性使用默认值，不计算图形边界
486 double graphCenterOffsetX = 0.0; // 图形中心相对逻辑原点的 X 偏移。
487 double yHigh = 2.54; // 默认值：100mil
488 double yLow = -2.54; // 默认值：-100mil
489
490 if (!symbol.isMultiPart()) {
491 double minX = std::numeric_limits<double>::max(); // 图形包围盒最小 X。
492 double maxX = -std::numeric_limits<double>::max(); // 图形包围盒最大 X。
493 auto updateBounds = [&](double x, double w) {
494 minX = qMin(minX, x);
495 maxX = qMax(maxX, x + w);
496 };
497 for (const auto& rect : symbol.rectangles) {
498 minX = qMin(minX, rect.x0);
499 maxX = qMax(maxX, rect.x0);
500 minX = qMin(minX, rect.x1);
501 maxX = qMax(maxX, rect.x1);
502 }
503 for (const auto& circle : symbol.circles) {
504 double r = circle.radius; // 当前圆形图元半径。
505 minX = qMin(minX, circle.center.x() - r);
506 maxX = qMax(maxX, circle.center.x() + r);
507 }
508 for (const auto& ellipse : symbol.ellipses) {
509 minX = qMin(minX, ellipse.center.x() - ellipse.radiusX);
510 maxX = qMax(maxX, ellipse.center.x() + ellipse.radiusX);
511 }
512 for (const auto& arc : symbol.arcs) {
513 minX = qMin(minX, arc.startPoint.x());
514 maxX = qMax(maxX, arc.startPoint.x());
515 minX = qMin(minX, arc.midPoint.x());
516 maxX = qMax(maxX, arc.midPoint.x());
517 minX = qMin(minX, arc.endPoint.x());
518 maxX = qMax(maxX, arc.endPoint.x());
519 }
520 for (const auto& polyline : symbol.polylines) {
521 for (const QPointF& pt : polyline.points) {
522 minX = qMin(minX, pt.x());
523 maxX = qMax(maxX, pt.x());
524 }
525 }
526 for (const auto& polygon : symbol.polygons) {
527 for (const QPointF& pt : polygon.points) {
528 minX = qMin(minX, pt.x());
529 maxX = qMax(maxX, pt.x());
530 }
531 }
532 for (const auto& text : symbol.texts) {
533 minX = qMin(minX, text.position.x());
534 maxX = qMax(maxX, text.position.x());
535 }
536 for (const auto& pin : pins) {
537 minX = qMin(minX, pin.position.x());
538 maxX = qMax(maxX, pin.position.x());
539 }
540 // 如果没有任何图形元素，使用默认宽度
541 double graphWidth = (minX <= maxX) ? (maxX - minX) : 0.0; // 图形主体宽度。
542 if (minX <= maxX) {
543 graphCenterOffsetX = minX + graphWidth / 2.0 - originX;
544 }
545 qDebug() << "Graph bounds - minX:" << minX << "maxX:" << maxX << "width:" << graphWidth
546 << "centerOffsetX(mm):" << graphCenterOffsetX;
547
548 // 计算 y_high 和 y_low（坐标已在 KiCad 空间，无需再次翻转）
549 // 综合引脚和图形元素的 Y 坐标
550 if (!pins.isEmpty()) {
551 for (const IR::SymbolPinIR& pin : pins) {
552 double pinY = pin.position.y() - originY; // 引脚相对逻辑原点的 Y 坐标。
553 yHigh = qMax(yHigh, pinY);
554 yLow = qMin(yLow, pinY);
555 }
556 }
557 // 图形元素也参与 Y 边界计算
558 for (const auto& rect : symbol.rectangles) {
559 yHigh = qMax(yHigh, rect.y0 - originY);
560 yLow = qMin(yLow, rect.y0 - originY);
561 yHigh = qMax(yHigh, rect.y1 - originY);
562 yLow = qMin(yLow, rect.y1 - originY);
563 }
564 for (const auto& circle : symbol.circles) {
565 double r = circle.radius; // 当前圆形图元半径。
566 yHigh = qMax(yHigh, circle.center.y() + r - originY);
567 yLow = qMin(yLow, circle.center.y() - r - originY);
568 }
569 }
570
575 constexpr double propertySpacing = 2.54; // 100mil
576 const double symbolCenterY = (yHigh + yLow) / 2.0; // 符号主体中心 Y 坐标。
577 const double visibleTopY = symbolCenterY + propertySpacing / 2.0; // 可见属性顶部 Y 坐标。
578 double hiddenFieldY = yLow - propertySpacing; // 隐藏属性的起始 Y 坐标。
579 double fontSize = 1.27; // KiCad 属性文字字号，单位为毫米。
580
585 // 原点已经是完整几何包围盒中心，属性统一使用局部中心 X=0。
586 double valueX = 0.0; // 属性值文本 X 坐标。
587 double valueY = symbolCenterY - propertySpacing / 2.0; // 属性值文本 Y 坐标。
588 double valueRotation = 0.0; // 属性值文本旋转角度。
589
590 // 辅助函数：转义属性值
591 auto escapePropertyValue = [](const QString& value) -> QString {
592 QString escaped = value; // 经过 KiCad 字符串规则转义的属性值。
593 escaped.replace("\"", "\\\"");
594 escaped.replace("\n", " ");
595 escaped.replace("\t", " ");
596 return escaped.trimmed();
597 };
598
599 // Reference 属性（主体中心上方）
600 QString refPrefix = symbol.designatorPrefix; // 符号参考标号前缀。
601 refPrefix.replace("?", "");
602 content += QString(" (property\n");
603 content += QString(" \"Reference\"\n");
604 content += QString(" \"%1\"\n").arg(escapePropertyValue(refPrefix));
605 content += " (id 0)\n";
606 content += QString(" (at %1 %2 0)\n").arg(valueX, 0, 'f', 2).arg(visibleTopY, 0, 'f', 2);
607 content += QString(" (effects (font (size %1 %2) (thickness 0) ) )\n")
608 .arg(fontSize, 0, 'f', 2)
609 .arg(fontSize, 0, 'f', 2);
610 content += " )\n";
611
612 // Value 属性：位于主体中心并保持水平可读。
613 content += QString(" (property\n");
614 content += QString(" \"Value\"\n");
615 content += QString(" \"%1\"\n").arg(escapePropertyValue(symbol.name));
616 content += " (id 1)\n";
617 content +=
618 QString(" (at %1 %2 %3)\n").arg(valueX, 0, 'f', 2).arg(valueY, 0, 'f', 2).arg(valueRotation, 0, 'f', 0);
619 content += QString(" (effects (font (size %1 %2) (thickness 0) ) )\n")
620 .arg(fontSize, 0, 'f', 2)
621 .arg(fontSize, 0, 'f', 2);
622 content += " )\n";
623
624 // Footprint 属性（隐藏，同一高度）
625 if (!symbol.footprintName.isEmpty()) {
626 content += QString(" (property\n");
627 content += QString(" \"Footprint\"\n");
628 QString footprintPath = QString("%1:%2").arg(libName, symbol.footprintName); // 关联封装库路径。
629 content += QString(" \"%1\"\n").arg(escapePropertyValue(footprintPath));
630 content += " (id 2)\n";
631 content += QString(" (at %1 %2 0)\n").arg(valueX, 0, 'f', 2).arg(hiddenFieldY, 0, 'f', 2);
632 hiddenFieldY -= propertySpacing;
633 content += QString(" (effects (font (size %1 %2) (thickness 0) ) hide)\n")
634 .arg(fontSize, 0, 'f', 2)
635 .arg(fontSize, 0, 'f', 2);
636 content += " )\n";
637 }
638
639 // LCSC Part 属性（隐藏，同一高度）
640 const QString lcscId = symbol.sourceMetadata.value("lcscId"); // 源器件 LCSC 编号。
641 if (!lcscId.isEmpty()) {
642 content += QString(" (property\n");
643 content += QString(" \"LCSC Part\"\n");
644 content += QString(" \"%1\"\n").arg(escapePropertyValue(lcscId));
645 content += " (id 5)\n";
646 content += QString(" (at %1 %2 0)\n").arg(valueX, 0, 'f', 2).arg(hiddenFieldY, 0, 'f', 2);
647 hiddenFieldY -= propertySpacing;
648 content += QString(" (effects (font (size %1 %2) (thickness 0) ) hide)\n")
649 .arg(fontSize, 0, 'f', 2)
650 .arg(fontSize, 0, 'f', 2);
651 content += " )\n";
652 }
653
654 // ki_description 属性（隐藏，同一高度）
655 const QString symbolDescription = symbol.description.trimmed(); // 去除首尾空白的符号描述。
656 if (!symbolDescription.isEmpty()) {
657 content += QString(" (property\n");
658 content += QString(" \"ki_description\"\n");
659 content += QString(" \"%1\"\n").arg(escapePropertyValue(symbolDescription));
660 content += " (id 6)\n";
661 content += QString(" (at %1 %2 0)\n").arg(valueX, 0, 'f', 2).arg(hiddenFieldY, 0, 'f', 2);
662 hiddenFieldY -= propertySpacing;
663 content += QString(" (effects (font (size %1 %2) (thickness 0) ) hide)\n")
664 .arg(fontSize, 0, 'f', 2)
665 .arg(fontSize, 0, 'f', 2);
666 content += " )\n";
667 }
668
669 // 检查是否为多部分符号
670 bool isMultiPart = symbol.isMultiPart(); // 是否需要生成多部件单元。
671 qDebug() << "=== Symbol Type Check ===";
672 qDebug() << "Symbol name:" << cleanSymbolName;
673 qDebug() << "Is multi-part:" << isMultiPart;
674 qDebug() << "Part count:" << symbol.partCount;
675
676 if (isMultiPart) {
677 // 多部分符号：为每个 part 生成子符号
678 qDebug() << "Exporting multi-part symbol with" << symbol.partCount << "parts";
679
680 for (int partIdx = 0; partIdx < symbol.partCount; ++partIdx) {
681 qDebug() << "Generating sub-symbol for part" << partIdx + 1;
682
683 // 子符号名称：_{partNumber}_1（partNumber 从 1 开始）
684 content += QString(" (symbol \"%1_%2_1\"\n").arg(cleanSymbolName).arg(partIdx + 1);
685
686 // 收集属于此 part 的引脚和图形
687 QList<IR::SymbolPinIR> partPins; // 当前部件的引脚列表。
688 for (const auto& pin : pins) {
689 if (pin.partIndex == partIdx) {
690 partPins.append(pin);
691 }
692 }
693
694 // 设置原点为 part 的坐标原点（在 IR 中为0,0，因为转换器已减去 part 原点）
695 double partOriginX = 0.0; // 当前部件原点 X 坐标。
696 double partOriginY = 0.0; // 当前部件原点 Y 坐标。
697 m_graphicsGenerator.setCurrentOrigin(partOriginX, partOriginY);
698
699 // 生成属于此 part 的图形元素
700 content += generatePartDrawings(symbol, partIdx);
701
702 // 生成属于此 part 的引脚
703 content += m_graphicsGenerator.generatePins(partPins);
704
705 content += " )\n"; // 结束子符号
706 }
707 } else {
708 // 单部分符号：直接在主符号中包含图形元素，不使用子符号
709 qDebug() << "Exporting single-part symbol";
710
716 IR::SymbolComponentIR drawingSymbol = symbol; // 当前绘制阶段使用的符号副本。
717 if (!symbol.rectangles.isEmpty()) {
718 double bodyMinX = std::numeric_limits<double>::max(); // 主体包围盒最小 X。
719 double bodyMinY = std::numeric_limits<double>::max(); // 主体包围盒最小 Y。
720 double bodyMaxX = -std::numeric_limits<double>::max(); // 主体包围盒最大 X。
721 double bodyMaxY = -std::numeric_limits<double>::max(); // 主体包围盒最大 Y。
722 for (const auto& rect : symbol.rectangles) {
723 bodyMinX = qMin(bodyMinX, qMin(rect.x0, rect.x1));
724 bodyMinY = qMin(bodyMinY, qMin(rect.y0, rect.y1));
725 bodyMaxX = qMax(bodyMaxX, qMax(rect.x0, rect.x1));
726 bodyMaxY = qMax(bodyMaxY, qMax(rect.y0, rect.y1));
727 }
728 const double textCenterX = (bodyMinX + bodyMaxX) / 2.0; // 主体中心 X 坐标。
729 const double textCenterY = (bodyMinY + bodyMaxY) / 2.0; // 主体中心 Y 坐标。
730 int visibleTextOrder = 0; // 主体外文本的排列序号。
731 for (IR::SymbolTextIR& text : drawingSymbol.texts) {
732 const double localX = text.position.x(); // 文本局部 X 坐标。
733 const double localY = text.position.y(); // 文本局部 Y 坐标。
734 const bool outsideBody =
735 localX < bodyMinX || localX > bodyMaxX || localY < bodyMinY || localY > bodyMaxY;
736 if (!outsideBody)
737 continue;
738 text.position = QPointF(textCenterX, textCenterY - visibleTextOrder * 1.27);
739 text.rotation = 0.0;
740 ++visibleTextOrder;
741 }
742 }
743
744 // 生成图形元素
745 content += m_graphicsGenerator.generateDrawings(drawingSymbol);
746
747 // 生成引脚
748 content += m_graphicsGenerator.generatePins(pins);
749 }
750
751 content += " )\n"; // 结束主符号
752
753 return content;
754}
