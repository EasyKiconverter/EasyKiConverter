105void ParallelExportService::retireExportStage(ExportTypeStage* stage) {
106 if (!stage) {
107 return;
108 }
109
110 stage->setParent(this);
111 for (const QPointer<ExportTypeStage>& retiredStage : std::as_const(m_retiredExportStages)) {
112 if (retiredStage.data() == stage) {
113 return;
114 }
115 }
116 m_retiredExportStages.append(QPointer<ExportTypeStage>(stage));
117
118 QPointer<ExportTypeStage> stagePtr(stage);
119 connect(stage, &ExportTypeStage::completed, this, [this, stagePtr](int, int, int) {
120 if (!stagePtr) {
121 cleanupRetiredExportStages();
122 return;
123 }
124 discardExportStage(stagePtr->typeName(), stagePtr.data());
125 });

355void ParallelExportService::cancelExport() {
356 // 空闲或已结束时没有可取消的任务，避免退出流程产生误导性的取消信号和报告。
357 const ExportOverallProgress progressSnapshot = getProgress();
358 const bool hasTrackedProgress = !progressSnapshot.exportTypeProgress.isEmpty();
359 if (!isRunning() && !hasTrackedProgress) {
360 return;
361 }
362
363 qDebug() << "ParallelExportService: Cancelling export";
364 ++m_activeRunGeneration;
365 m_cancelRequested = true;
366
367 if (m_progress.currentStage == ExportOverallProgress::Stage::Preloading) {
368 cancelPreload();
369 return;
370 }
371
372 // 重置 ComponentService 批处理上下文
373 if (m_componentService) {
374 m_componentService->abortBatchFetch();
375 }
376
377 m_progress.currentStage = ExportOverallProgress::Stage::Cancelled;
378 m_progress.endTime = QDateTime::currentDateTime();
379 emit cancelled();
380
381 QList<QPointer<ExportTypeStage>> stagesToCancel;
382 for (auto* stage : m_exportStages) { // 当前活动阶段。
383 if (stage) {
384 disconnect(stage, nullptr, this, nullptr);
385 retireExportStage(stage);
386 stagesToCancel.append(QPointer<ExportTypeStage>(stage));
387 }
388 }
389 m_exportStages.clear();
390 m_runningExportStages = 0;
391
392 NetworkClient::instance().cancelAllRequests();
393 for (const QPointer<ExportTypeStage>& stage : std::as_const(stagesToCancel)) {
394 if (stage) {
395 stage->cancel();
396 }
397 }
398 cleanupRetiredExportStages();
399
400 QTimer::singleShot(0, this, [this]() {
401 logNetworkRuntimeStats(QStringLiteral("export-cancelled"));
402 writeExportDetailedReport(QStringLiteral("export-cancelled"));
403 });
404}

527void ParallelExportService::onExportItemStatusChanged(const QString& componentId,
528 const QString& typeName,
529 const ExportItemStatus& status) {
530 if (m_cancelRequested || m_progress.currentStage == ExportOverallProgress::Stage::Cancelled) {
531 return;
532 }
533
534 bool shouldMirrorAltiumModel3DStatus = false;
535 QMutexLocker locker(&m_progressMutex);
536
537 if (!m_progress.exportTypeProgress.contains(typeName)) {
538 m_progress.exportTypeProgress[typeName] = ExportTypeProgress();
539 m_progress.exportTypeProgress[typeName].typeName = typeName;
540 m_progress.exportTypeProgress[typeName].totalCount = m_componentIds.size();
541 }
542
543 ExportTypeProgress& typeProgress = m_progress.exportTypeProgress[typeName];
544 ExportProgressAggregator::mergeItemStatus(typeProgress, componentId, status);
545
546 // FootprintExportStage 会先发出真实的嵌入结果，再发出封装结果。
547 // 只有在 3D 尚未收到独立结果时才使用封装状态兜底，避免封装成功覆盖
548 // STEP 缺失等真实的 Model3D 失败状态。
549 if (typeName == QStringLiteral("Footprint") && m_options.exportModel3D &&
550 (m_options.targetFormat == TargetEdaFormat::Altium || m_options.targetFormat == TargetEdaFormat::Allegro ||
551 m_options.targetFormat == TargetEdaFormat::LibrePcb || m_options.targetFormat == TargetEdaFormat::Horizon)) {
552 const auto modelProgress =
553 m_progress.exportTypeProgress.constFind(QStringLiteral("Model3D"));
554 shouldMirrorAltiumModel3DStatus =
555 modelProgress == m_progress.exportTypeProgress.cend() || !modelProgress->itemStatus.contains(componentId);
556 }
557
558 locker.unlock();
559 emit itemStatusChanged(componentId, typeName, status);
560
561 // Altium、Allegro 和 LibrePCB 的 STEP 都在封装阶段写入目标库，没有独立的 Model3D stage。
562 // 将封装结果镜像到 Model3D，确保 UI 状态、成功率和总体完成判定一致。
563 if (shouldMirrorAltiumModel3DStatus) {
564 onExportItemStatusChanged(componentId, QStringLiteral("Model3D"), status);
565 }
566
567 // 更新整体进度
568 updateOverallProgress();
569}
