35bool ConfigManager::save() {
36 try {
37 // 保存导出路径
38 QJsonValue exportPathValue = m_config.value("export_path");
39 m_settings->setValue("export/export_path", exportPathValue.isUndefined() ? "" : exportPathValue.toString());
40
41 QJsonValue libNameValue = m_config.value("lib_name"); // 当前库名称配置值。
42 m_settings->setValue("export/lib_name",
43 libNameValue.isUndefined() ? "easyeda_convertlib" : libNameValue.toString());
44
45 // 保存导出选项
46 QJsonValue exportOptionsValue = m_config.value("export_options");
47 QJsonObject exportOptions =
48 exportOptionsValue.isUndefined() ? QJsonObject() : exportOptionsValue.toObject(); // 导出选项对象。
49
50 QJsonValue symbolValue = exportOptions.value("symbol"); // 符号库导出开关。
51 m_settings->setValue("export/export_symbol", symbolValue.isUndefined() ? true : symbolValue.toBool());
52
53 QJsonValue footprintValue = exportOptions.value("footprint"); // 封装库导出开关。
54 m_settings->setValue("export/export_footprint", footprintValue.isUndefined() ? true : footprintValue.toBool());
55
56 QJsonValue model3dValue = exportOptions.value("model3d"); // 三维模型导出开关。
57 m_settings->setValue("export/export_3d_model", model3dValue.isUndefined() ? true : model3dValue.toBool());
58
59 // 同步到磁
60 m_settings->sync();
61
62 qDebug() << "Configuration saved successfully";
63 return true;
64 } catch (const std::exception& e) {
65 qWarning() << "Failed to save configuration:" << e.what();
66 return false;
67 }
68}
