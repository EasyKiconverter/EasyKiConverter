239NetworkResult NetworkClient::executeRequest(const QUrl& url,
240 const QByteArray& body,
241 ResourceType resourceType,
242 const RetryPolicy& policy) {
243 // 同步路径：在创建 request 前做 URL 校验，避免白名单拒绝后死锁
244 // （QWaitCondition::wait() 不处理事件循环，queued 完成信号无法到达）
245 {
246 QString errorMsg; // URL 白名单校验错误。
247 if (!UrlUtils::isAllowedUrl(url, resourceType, &errorMsg)) {
248 qWarning() << "NetworkClient: URL rejected by whitelist (sync):" << errorMsg;
249 NetworkResult result; // URL 被拒绝时返回的失败结果。
250 populateDiagnostic(result.diagnostic, url, resourceType);
251 result.error = errorMsg;
252 result.success = false;
253 result.diagnostic.errorType = NetworkErrorType::Other;
254 result.diagnostic.errorMessage = errorMsg;
255 return result;
256 }
257 }
258
259 auto* request = enqueueAsyncRequest(url, body, resourceType, policy); // 创建的异步请求对象。
260 if (!request) {
261 NetworkResult result; // 创建请求失败时返回的失败结果。
262 populateDiagnostic(result.diagnostic, url, resourceType);
263 result.error = QStringLiteral("Failed to create async request");
264 result.success = false;
265 result.diagnostic.errorType = NetworkErrorType::Other;
266 result.diagnostic.errorMessage = result.error;
267 return result;
268 }
269
270 auto context = std::make_shared<BlockingRequestContext>(); // 同步等待异步结果的上下文。
271 QObject::connect(
272 request,
273 &AsyncNetworkRequest::finished,
274 request,
275 [context](const NetworkResult& result) { context->complete(result); },
276 Qt::DirectConnection);
277
278 const NetworkResult result = context->wait(); // 异步请求完成后的结果。
279 QMetaObject::invokeMethod(request, &QObject::deleteLater, Qt::QueuedConnection);
280 return result;
281}

291AsyncNetworkRequest* NetworkClient::enqueueAsyncRequest(const QUrl& url,
292 const QByteArray& body,
293 ResourceType resourceType,
294 const RetryPolicy& policy) {
295 if (!m_asyncNetworkManager) {
296 return nullptr;
297 }
298
299 // URL 白名单校验：拒绝非预期的 scheme 或 host
300 {
301 QString errorMsg; // URL 白名单校验错误。
302 if (!UrlUtils::isAllowedUrl(url, resourceType, &errorMsg)) {
303 qWarning() << "NetworkClient: URL rejected by whitelist:" << errorMsg;
304 // 返回已完成的失败 request，避免调用方需处理 nullptr
305 NetworkResult failResult; // URL 被拒绝时返回的失败请求结果。
306 failResult.success = false;
307 failResult.wasCancelled = false;
308 failResult.error = errorMsg;
309 failResult.diagnostic.url = url.toString();
310 failResult.diagnostic.host = url.host();
311 failResult.diagnostic.resourceType = resourceType;
312 failResult.diagnostic.profileName = RequestProfiles::fromType(resourceType).name;
313 failResult.diagnostic.errorType = NetworkErrorType::Other;
314 failResult.diagnostic.errorMessage = errorMsg;
315 return AsyncNetworkRequest::createFinished(failResult);
316 }
317 }
318
319 auto* request = // 新建的异步网络请求对象。
320 new AsyncNetworkRequest(url, m_asyncNetworkManager, resourceType, policy, body, nullptr);
321 request->moveToThread(&m_networkThread);
322
323 const RequestProfile profile = RequestProfiles::fromType(resourceType); // 当前资源的调度配置。
324 {
325 QMutexLocker locker(&m_asyncQueueMutex);
326 PendingAsyncRequest pending; // 待调度请求记录。
327 pending.request = request;
328 pending.resourceType = resourceType;
329 pending.priority = profile.priority;
330 pending.sequence = m_asyncSequence++;
331 pending.enqueuedAtMs = QDateTime::currentMSecsSinceEpoch();
332 m_pendingAsyncRequests.append(pending);
333 m_liveAsyncRequests.append(request);
334 updateStatsForEnqueuedRequest(resourceType);
335 }
336
337 QObject::connect(
338 request,
339 &AsyncNetworkRequest::finished,
340 this,
341 [this, request, resourceType](const NetworkResult& result) {
342 updateStatsForCompletedRequest(resourceType, result);
343 {
344 QMutexLocker locker(&m_asyncQueueMutex);
345 m_liveAsyncRequests.removeAll(request);
346 }
347 onAsyncRequestFinished(resourceType);
348 },
349 Qt::QueuedConnection);
350
351 scheduleAsyncPump();
352 return request;
353}

376void NetworkClient::pumpAsyncQueue() {
377 while (true) {
378 QPointer<AsyncNetworkRequest> requestToStart; // 本轮选中的异步请求。
379 ResourceType resourceType = ResourceType::Unknown; // 本轮请求的资源类型。
380
381 {
382 QMutexLocker locker(&m_asyncQueueMutex);
383 int bestIndex = -1; // 当前可启动请求在队列中的索引。
384 int bestPriority = std::numeric_limits<int>::max(); // 当前候选请求的最低优先级值。
385 quint64 bestSequence = std::numeric_limits<quint64>::max(); // 当前候选请求的最早入队序号。
386
387 for (int i = m_pendingAsyncRequests.size() - 1; i >= 0; --i) {
388 const PendingAsyncRequest& pending = // 反向清理阶段的待调度请求记录。
389 m_pendingAsyncRequests.at(i);
390 if (!pending.request || pending.request->isFinished()) {
391 m_pendingAsyncRequests.removeAt(i);
392 }
393 }
394
395 for (int i = 0; i < m_pendingAsyncRequests.size(); ++i) {
396 const PendingAsyncRequest& pending = // 当前优先级选择阶段的待调度请求记录。
397 m_pendingAsyncRequests.at(i);
398 if (!pending.request || pending.request->isFinished()) {
399 continue;
400 }
401
402 const RequestProfile profile = RequestProfiles::fromType(pending.resourceType); // 候选请求调度配置。
403 const int activeCount = // 候选请求资源类型当前活动数量。
404 m_activeAsyncRequestsByType.value(static_cast<int>(pending.resourceType), 0);
405 if (activeCount >= profile.maxConcurrent) {
406 continue;
407 }
408
409 if (pending.priority < bestPriority ||
410 (pending.priority == bestPriority && pending.sequence < bestSequence)) {
411 bestIndex = i;
412 bestPriority = pending.priority;
413 bestSequence = pending.sequence;
414 }
415 }
416
417 if (bestIndex < 0) {
418 m_asyncPumpScheduled = false;
419 break;
420 }
421
422 PendingAsyncRequest pending = m_pendingAsyncRequests.takeAt(bestIndex); // 取出的待调度请求记录。
423 requestToStart = pending.request;
424 resourceType = pending.resourceType;
425 m_activeAsyncRequestsByType[static_cast<int>(resourceType)] =
426 m_activeAsyncRequestsByType.value(static_cast<int>(resourceType), 0) + 1;
427 const qint64 queueDelayMs = // 请求从入队到启动前的等待时长。
428 qMax<qint64>(0, QDateTime::currentMSecsSinceEpoch() - pending.enqueuedAtMs);
429 updateStatsForDequeuedRequest(resourceType, queueDelayMs);
430 updateStatsForStartedRequest(resourceType);
431 }
432
433 if (!requestToStart) {
434 onAsyncRequestFinished(resourceType);
435 continue;
436 }
437
438 QMetaObject::invokeMethod(
439 requestToStart,
440 [requestToStart]() {
441 if (requestToStart) {
442 requestToStart->start();
443 }
444 },
445 Qt::QueuedConnection);
446 }
447}

452void NetworkClient::cancelAllRequests() {
453 QList<QPointer<AsyncNetworkRequest>> requestsToCancel; // 待取消的异步请求快照。
454 {
455 QMutexLocker locker(&m_asyncQueueMutex);
456 requestsToCancel = m_liveAsyncRequests;
457 m_pendingAsyncRequests.clear();
458 m_asyncPumpScheduled = false;
459 }
460
461 for (const QPointer<AsyncNetworkRequest>& request : std::as_const(requestsToCancel)) {
462 if (!request || request->isFinished()) {
463 continue;
464 }
465
466 if (request->thread() == QThread::currentThread()) {
467 request->cancel();
468 } else if (QThread::currentThread() == &m_networkThread) {
469 request->cancel();
470 } else {
471 const Qt::ConnectionType connectionType =
472 m_networkThread.isRunning() ? Qt::BlockingQueuedConnection : Qt::QueuedConnection;
473 QMetaObject::invokeMethod(request, &AsyncNetworkRequest::cancel, connectionType);
474 }
475 }

561void NetworkClient::updateStatsForCompletedRequest(ResourceType resourceType, const NetworkResult& result) {
562 QMutexLocker locker(&m_asyncQueueMutex); // 保护资源统计的互斥锁。
563 const int key = static_cast<int>(resourceType); // 资源类型对应的统计键。
564 MutableResourceStats& stats = m_resourceStats[key]; // 当前资源可变统计对象。
565 const RequestProfile profile = RequestProfiles::fromType(resourceType); // 当前资源调度配置。
566 stats.snapshot.resourceType = resourceType;
567 stats.snapshot.profileName = profile.name;
568 stats.snapshot.maxConcurrent = profile.maxConcurrent;
569 ++stats.snapshot.completedRequests;
570 stats.snapshot.retryAttempts += static_cast<quint64>(qMax(0, result.retryCount));
571 stats.snapshot.lastLatencyMs = result.elapsedMs;
572 stats.totalLatencyMs += result.elapsedMs;
573 stats.snapshot.averageLatencyMs =
574 stats.snapshot.completedRequests > 0
575 ? (stats.totalLatencyMs / static_cast<qint64>(stats.snapshot.completedRequests))
576 : 0;
577 stats.snapshot.maxLatencyMs = qMax(stats.snapshot.maxLatencyMs, result.elapsedMs);
578
579 if (result.wasCancelled || result.diagnostic.wasCanceled) {
580 ++stats.snapshot.cancelledRequests;
581 } else if (result.success) {
582 ++stats.snapshot.succeededRequests;
583 } else {
584 ++stats.snapshot.failedRequests;
585 }
586
587 if (result.diagnostic.errorType == NetworkErrorType::Timeout) {
588 ++stats.snapshot.timeoutRequests;
589 }
590 if (result.diagnostic.wasRateLimited || result.statusCode == 429) {
591 ++stats.snapshot.rateLimitedRequests;
592 }
593 refreshDynamicStatsLocked(resourceType);
594}

600void NetworkClient::refreshDynamicStatsLocked(ResourceType resourceType) {
601 const int key = static_cast<int>(resourceType); // 资源类型对应的统计键。
602 MutableResourceStats& stats = m_resourceStats[key]; // 当前资源的可变统计对象。
603 int queuedCount = 0; // 当前资源仍在等待队列中的请求数量。
604 for (const PendingAsyncRequest& pending : std::as_const(m_pendingAsyncRequests)) {
605 if (pending.request && !pending.request->isFinished() && pending.resourceType == resourceType) {
606 ++queuedCount;
607 }
608 }
609
610 stats.snapshot.queuedRequests = queuedCount;
611 stats.snapshot.activeRequests = m_activeAsyncRequestsByType.value(key, 0);
612 stats.snapshot.peakQueuedRequests = qMax(stats.snapshot.peakQueuedRequests, queuedCount);
613 stats.snapshot.peakActiveRequests = qMax(stats.snapshot.peakActiveRequests, stats.snapshot.activeRequests);
614}

622int NetworkClient::calculateRetryDelay(int retryCount, const RetryPolicy& policy) {
623 if (retryCount >= static_cast<int>(policy.delays.size())) {
624 return policy.delays.back();
625 }
626 int baseDelay = policy.delays[retryCount]; // 当前重试次数对应的基础延迟。
627
628 int jitterRange = static_cast<int>(baseDelay * policy.jitterFactor); // 随机抖动范围。
629 int jitter = QRandomGenerator::global()->bounded(-jitterRange, jitterRange + 1); // 本次随机抖动。
630 return baseDelay + jitter;
631}

641bool NetworkClient::shouldRetry(int statusCode,
642 QNetworkReply::NetworkError error,
643 int retryCount,
644 const RetryPolicy& policy) {
645 if (retryCount >= policy.maxRetries) {
646 return false;
647 }
648
649 if (policy.retryableStatusCodes.contains(statusCode)) {
650 return true;
651 }
652
653 // 网络层临时错误允许进入重试流程。
654 switch (error) {
655 case QNetworkReply::TimeoutError:
656 case QNetworkReply::TemporaryNetworkFailureError:
657 case QNetworkReply::NetworkSessionFailedError:
658 case QNetworkReply::UnknownNetworkError:
659 case QNetworkReply::RemoteHostClosedError:
660 case QNetworkReply::HostNotFoundError:
661 case QNetworkReply::ConnectionRefusedError:
662 case QNetworkReply::ProxyTimeoutError:
663 return true;
664 default:
665 return false;
666 }
667}
