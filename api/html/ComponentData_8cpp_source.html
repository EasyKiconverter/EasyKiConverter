132bool ComponentData::isValid() const {
133 // 验证 LCSC ID 格式（以 C 开头，后面跟数字）
134 if (m_lcscId.isEmpty()) {
135 return false;
136 }
137
138 if (!m_lcscId.startsWith('C', Qt::CaseInsensitive)) {
139 return false;
140 }
141
142 // 检查后面是否为数字
143 bool ok = false;
144 QString numberPart = m_lcscId.mid(1);
145 if (numberPart.length() > 0) {
146 numberPart.toInt(&ok);
147 }
148 if (!ok) {
149 return false;
150 }
151
152 // 至少要有符号或封装数据
153 if (!m_symbolData && !m_footprintData) {
154 return false;
155 }
156
157 return true;
158}

164QString ComponentData::validate() const {
165 if (m_lcscId.isEmpty()) {
166 return "LCSC ID is empty";
167 }
168
169 if (!m_lcscId.startsWith('C', Qt::CaseInsensitive)) {
170 return QString("LCSC ID must start with 'C': %1").arg(m_lcscId);
171 }
172
173 // 检查后面是否为数字
174 bool ok = false;
175 QString numberPart = m_lcscId.mid(1);
176 if (numberPart.length() > 0) {
177 numberPart.toInt(&ok);
178 }
179 if (!ok) {
180 return QString("LCSC ID must be in format 'C' followed by numbers: %1").arg(m_lcscId);
181 }
182
183 if (!m_symbolData && !m_footprintData) {
184 return "Component must have at least symbol or footprint data";
185 }
186
187 if (m_symbolData) {
188 QString symbolError = m_symbolData->validate();
189 if (!symbolError.isEmpty()) {
190 return QString("Symbol validation failed: %1").arg(symbolError);
191 }
192 }
193
194 if (m_footprintData) {
195 QString footprintError = m_footprintData->validate();
196 if (!footprintError.isEmpty()) {
197 return QString("Footprint validation failed: %1").arg(footprintError);
198 }
199 }
200
201 return QString(); // 返回空字符串表示验证通过
202}
