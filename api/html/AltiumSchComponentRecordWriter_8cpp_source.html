39QList<ParameterField> componentParameterFields(const AltiumSchComponent& component) {
40 static const QMap<QString, QString> knownNames = {
41 // 来源元数据到 Altium 字段名的映射。
42 {QStringLiteral("description"), QStringLiteral("Description")},
43 {QStringLiteral("manufacturer"), QStringLiteral("Manufacturer")},
44 {QStringLiteral("manufacturerPart"), QStringLiteral("Manufacturer Part Number")},
45 {QStringLiteral("datasheet"), QStringLiteral("Datasheet")},
46 {QStringLiteral("lcscId"), QStringLiteral("LCSC Part")},
47 {QStringLiteral("jlcId"), QStringLiteral("JLCPCB Part")},
48 {QStringLiteral("jlcpcbPartClass"), QStringLiteral("JLCPCB Part Class")},
49 {QStringLiteral("supplierPart"), QStringLiteral("Supplier Part Number")},
50 {QStringLiteral("supplier"), QStringLiteral("Supplier")},
51 {QStringLiteral("category"), QStringLiteral("Category")},
52 {QStringLiteral("source"), QStringLiteral("Source")},
53 };
54
55 QList<ParameterField> fields; // 按输出顺序收集的参数字段。
56 QSet<QString> names; // 已写入参数的去重键集合。
57 const auto appendField = [&](const QString& name,
58 const QString& value,
59 bool hasLocation = false,
60 int locationX = 0,
61 int locationY = 0,
62 bool hasVisibility = false,
63 bool isHidden = true,
64 bool readOnly = false,
65 int orientation = 0,
66 int ownerPartId = -1,
67 int fontId = 1,
68 uint32_t color = 0x000000,
69 double fontSizeMm = 0.0) {
70 const QString normalizedName = name.trimmed(); // 去除空白后的参数名称。
71 const QString deduplicationKey = QStringLiteral("%1:%2").arg(normalizedName).arg(ownerPartId); // 参数去重键。
72 if (normalizedName.isEmpty() || value.trimmed().isEmpty() || names.contains(deduplicationKey))
73 return;
74 ParameterField field; // 当前规范化后的参数字段。
75 field.name = normalizedName;
76 field.value = value.trimmed();
77 field.hasLocation = hasLocation;
78 field.locationX = locationX;
79 field.locationY = locationY;
80 field.hasVisibility = hasVisibility;
81 field.isHidden = isHidden;
82 field.readOnly = readOnly;
83 field.orientation = orientation;
84 field.ownerPartId = ownerPartId;
85 field.fontId = fontId;
86 field.color = color;
87 field.fontSizeMm = fontSizeMm;
88 fields.append(field);
89 names.insert(deduplicationKey);
90 };
91
92 QString value = component.sourceMetadata.value(QStringLiteral("value")).trimmed(); // 来源元数据中的器件值。
93 if (value.isEmpty())
94 value = component.name.trimmed();
95 if (!value.isEmpty())
96 appendField(QStringLiteral("Comment"), value);
97
98 QString description = component.sourceMetadata.value(QStringLiteral("description")).trimmed(); // 元件描述。
99 if (description.isEmpty())
100 description = component.description.trimmed();
101 if (!description.isEmpty())
102 appendField(QStringLiteral("Description"), description);
103
104 for (auto it = component.sourceMetadata.constBegin(); it != component.sourceMetadata.constEnd(); ++it) {
105 const QString metadataValue = it.value().trimmed(); // 当前元数据值。
106 if (metadataValue.isEmpty() || it.key() == QStringLiteral("value") || it.key() == QStringLiteral("description"))
107 continue;
108 appendField(knownNames.value(it.key(), it.key()), metadataValue);
109 }
110
111 if (!component.aliases.isEmpty())
112 appendField(QStringLiteral("Aliases"), component.aliases.join(QStringLiteral(", ")));
113
114 for (const AltiumSchParameter& parameter : component.parameters) {
115 QString name = parameter.name.trimmed(); // 当前参数名称。
116 if (name.compare(QStringLiteral("Value"), Qt::CaseInsensitive) == 0)
117 name = QStringLiteral("Comment");
118 appendField(name,
119 parameter.value,
120 parameter.locationX != 0 || parameter.locationY != 0,
121 parameter.locationX,
122 parameter.locationY,
123 true,
124 parameter.isHidden,
125 parameter.readOnly,
126 parameter.orientation,
127 parameter.ownerPartId,
128 parameter.fontId,
129 parameter.color,
130 parameter.fontSizeMm);
131 }
132 return fields;
133}

148void AltiumSchComponentRecordWriter::writeParameters(AltiumBinaryWriter& writer, const AltiumSchComponent& component) {
149 QString designator = component.designatorPrefix.trimmed(); // 规范化后的元件标识符。
150 if (designator.isEmpty())
151 designator = "?";
152 else if (!designator.endsWith('?'))
153 designator += '?';
154
155 QMap<QString, QString> designatorParams; // Designator RECORD=34 字段。
156 designatorParams["RECORD"] = "34";
157 designatorParams["OWNERPARTID"] = "-1";
158 designatorParams["LOCATION.X_FRAC"] = "-5";
159 designatorParams["LOCATION.Y_FRAC"] = "5";
160 designatorParams["COLOR"] = "8388608";
161 designatorParams["FONTID"] = "1";
162 designatorParams["TEXT"] = designator;
163 designatorParams["NAME"] = "Designator";
164 designatorParams["READONLYSTATE"] = "1";
165 m_owner.addUniqueID(designatorParams);
166 writer.writeCStringParameterBlockUtf8(designatorParams);
167
168 for (const ParameterField& field : componentParameterFields(component)) { // 写入每个参数字段。
169 QMap<QString, QString> parameterParams; // 当前参数的 RECORD=41 字段。
170 parameterParams["RECORD"] = "41";
171 const int ownerPartId =
172 m_owner.normalizeOwnerPartId(field.ownerPartId, QStringLiteral("参数")); // 规范化后的部件编号。
173 parameterParams["OWNERPARTID"] = QString::number(ownerPartId);
174 if (ownerPartId >= 1) {
175 parameterParams["OWNERPARTDISPLAYMODE"] = "1";
176 m_owner.addContentIndex(parameterParams);
177 }
178 if (field.hasLocation) {
179 m_owner.addCoordParam(parameterParams, "LOCATION.X", field.locationX);
180 m_owner.addCoordParam(parameterParams, "LOCATION.Y", field.locationY);
181 } else {
182 parameterParams["LOCATION.X_FRAC"] = "-5";
183 parameterParams["LOCATION.Y_FRAC"] = "-15";
184 }
185 if (field.color != 0)
186 parameterParams["COLOR"] = QString::number(field.color);
187 int fontId = field.fontId; // 当前参数使用的字体 ID。
188 if (std::isfinite(field.fontSizeMm) && field.fontSizeMm > 0.0) {
189 constexpr double MILLIMETERS_PER_POINT = 25.4 / 72.0; // 点制字体到毫米的换算系数。
190 fontId = m_owner.m_fontRegistry.getOrAdd(QStringLiteral("Times New Roman"),
191 qMax(1, qRound(field.fontSizeMm / MILLIMETERS_PER_POINT)));
192 } else if (fontId < 1 || fontId > m_owner.m_fontRegistry.size()) {
193 fontId = 1;
194 }
195 parameterParams["FONTID"] = QString::number(fontId);
196 parameterParams["TEXT"] = field.value;
197 parameterParams["NAME"] = field.name;
198 if (field.orientation != 0)
199 parameterParams["Orientation"] = QString::number(field.orientation);
200 if (field.hasVisibility && field.isHidden)
201 parameterParams["IsHidden"] = "T";
202 if (field.readOnly)
203 parameterParams["READONLYSTATE"] = "1";
204 m_owner.addUniqueID(parameterParams);
205 writer.writeCStringParameterBlockUtf8(parameterParams);
206 }

214void AltiumSchComponentRecordWriter::writeImplementations(AltiumBinaryWriter& writer,
215 const AltiumSchComponent& component) {
216 QMap<QString, QString> listParams; // RECORD=44 实现列表字段。
217 listParams["RECORD"] = "44";
218 listParams["DataFileFormatID"] = "";
219 listParams["Description"] = "";
220 listParams["FileName"] = "";
221 writer.writeCStringParameterBlockUtf8(listParams);
222
223 for (int implementationIndex = 0; implementationIndex < component.implementations.size(); ++implementationIndex) {
224 const AltiumSchComponent::Implementation& implementation =
225 component.implementations.at(implementationIndex); // 当前封装实现。
226 QMap<QString, QString> implementationParams; // RECORD=45 实现字段。
227 implementationParams["RECORD"] = "45";
228 implementationParams["DESCRIPTION"] = implementation.modelName;
229 implementationParams["MODELNAME"] = implementation.modelName;
230 implementationParams["MODELTYPE"] = implementation.modelType;
231 const QString modelType = implementation.modelType.trimmed().toUpper(); // 规范化后的模型类型。
232 const bool isPcbLibrary = modelType.isEmpty() || modelType == QStringLiteral("PCBLIB"); // 是否为 PCB 库实现。
233 const bool hasExplicitDataFileKind = // 是否提供了非默认数据文件类型。
234 !implementation.dataFileKind.trimmed().isEmpty() &&
235 (isPcbLibrary || implementation.dataFileKind.compare(QStringLiteral("PCBLib"), Qt::CaseInsensitive) != 0);
236 const bool hasDataFile = // 当前实现是否引用数据文件。
237 isPcbLibrary || hasExplicitDataFileKind || !implementation.dataFileEntity.trimmed().isEmpty();
238 implementationParams["DATAFILECOUNT"] = hasDataFile ? "1" : "0";
239 if (hasDataFile) {
240 implementationParams["MODELDATAFILEKIND1"] =
241 implementation.dataFileKind.isEmpty() ? "PCBLib" : implementation.dataFileKind;
242 implementationParams["MODELDATAFILEENTITY1"] =
243 implementation.dataFileEntity.isEmpty()
244 ? (isPcbLibrary && !m_owner.m_libraryName.isEmpty() ? m_owner.m_libraryName + ".PcbLib" : "*")
245 : implementation.dataFileEntity;
246 }
247 // Altium 只允许一个默认实现，候选封装和附加模型不能全部标记为当前。
248 if (implementationIndex == 0)
249 implementationParams["ISCURRENT"] = "T";
250 m_owner.addUniqueID(implementationParams);
251 writer.writeCStringParameterBlockUtf8(implementationParams);
252
253 QMap<QString, QString> mapListParams; // RECORD=46 引脚映射列表字段。
254 mapListParams["RECORD"] = "46";
255 writer.writeCStringParameterBlockUtf8(mapListParams);
256
257 // Altium 通过 RECORD=47 建立符号引脚与当前 PCBLib 实现之间的关联。
258 for (int pinIndex = 1; pinIndex <= component.pins.size(); ++pinIndex) {
259 QMap<QString, QString> pinMappingParams; // 当前引脚的 RECORD=47 映射字段。
260 pinMappingParams["RECORD"] = "47";
261 pinMappingParams["DESINTF"] = QString::number(pinIndex);
262 pinMappingParams["DESIMPCOUNT"] = "1";
263 pinMappingParams["DESIMP0"] =
264 implementation.pinMappings.value(QString::number(pinIndex), QString::number(pinIndex));
265 pinMappingParams["ISTRIVIAL"] = "T";
266 m_owner.addUniqueID(pinMappingParams);
267 writer.writeCStringParameterBlockUtf8(pinMappingParams);
268 }
269
270 QMap<QString, QString> parameterParams; // RECORD=48 实现参数字段。
271 parameterParams["RECORD"] = "48";
272 for (auto it = implementation.parameters.constBegin(); it != implementation.parameters.constEnd(); ++it)
273 parameterParams[it.key()] = it.value();
274 writer.writeCStringParameterBlockUtf8(parameterParams);
275 }
