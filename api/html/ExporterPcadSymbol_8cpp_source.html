40QString safeName(const QString& value) {
41 QString result; // 清洗后的名称。
42 for (const QChar character : value) { // 逐字符处理原始名称。
43 const ushort code = character.unicode(); // 当前字符编码。
44 if ((code >= 'A' && code <= 'Z') || (code >= 'a' && code <= 'z') || (code >= '0' && code <= '9') ||
45 character == QChar('_') || character == QChar('-'))
46 result.append(character);
47 else
48 result.append(QChar('_'));
49 }
50 return result.left(64);
51}

144bool validateSymbol(const IR::SymbolComponentIR& symbol, QStringList& diagnostics) {
145 if (safeName(symbol.name).isEmpty()) {
146 diagnostics.append(QStringLiteral("P-CAD: 符号名称清洗后为空：%1").arg(symbol.name));
147 return false;
148 }
149 if (!isAscii(symbol.name) || !isAscii(symbol.designatorPrefix)) {
150 diagnostics.append(QStringLiteral("P-CAD: 符号名称和位号前缀必须是 ASCII：%1").arg(symbol.name));
151 return false;
152 }
153 if (symbol.partCount < 1) {
154 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 的部件数量无效").arg(symbol.name));
155 return false;
156 }
157 if (!symbol.arcs.isEmpty() || !symbol.circles.isEmpty() || !symbol.ellipses.isEmpty() || !symbol.pies.isEmpty() ||
158 !symbol.ellipticalArcs.isEmpty() || !symbol.paths.isEmpty() || !symbol.beziers.isEmpty() ||
159 !symbol.ieeeSymbols.isEmpty() || !symbol.texts.isEmpty() || !symbol.textFrames.isEmpty() ||
160 !symbol.images.isEmpty()) {
161 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 含当前 ASCII writer 无法无损表达的图元").arg(symbol.name));
162 return false;
163 }
164 for (const IR::SymbolPinIR& pin : symbol.pins) {
165 if (pin.designator.isEmpty() || !isAscii(pin.designator) || !isAscii(pin.name) || !finitePoint(pin.position) ||
166 !std::isfinite(pin.length) || pin.length < 0.0 || pin.partIndex < 0 || pin.partIndex >= symbol.partCount) {
167 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 的引脚字段无效：%2").arg(symbol.name, pin.designator));
168 return false;
169 }
170 }
171 return true;
172}

183bool writeSymbolDef(QTextStream& stream,
184 const IR::SymbolComponentIR& symbol,
185 const QString& name,
186 int part,
187 QStringList& diagnostics) {
188 stream << " (symbolDef " << quote(name) << "\n"
189 << " (originalName " << quote(symbol.name) << ")\n";
190 for (const IR::SymbolPinIR& pin : symbol.pins) {
191 if (pin.partIndex != part && !pin.commonToAllParts)
192 continue;
193 stream << " (pin " << quote(pin.designator) << "\n"
194 << " (pt " << number(pin.position.x()) << ' ' << number(-pin.position.y()) << ")\n"
195 << " (rotation " << QString::number(pinRotation(pin.direction), 'f', 3) << ")\n"
196 << " (pinLength " << number(pin.length) << ")\n"
197 << " (pinDisplay (dispPinDes " << (pin.display.showDesignator ? "True" : "False")
198 << ") (dispPinName " << (pin.display.showName ? "True" : "False") << "))\n"
199 << " )\n";
200 }
201 for (const IR::SymbolRectangleIR& rectangle : symbol.rectangles) {
202 if (rectangle.partIndex != part)
203 continue;
204 if (!finitePoint(QPointF(rectangle.x0, rectangle.y0)) || !finitePoint(QPointF(rectangle.x1, rectangle.y1)) ||
205 !std::isfinite(rectangle.strokeWidth) || rectangle.strokeWidth < 0.0 || rectangle.cornerRadiusX != 0.0 ||
206 rectangle.cornerRadiusY != 0.0) {
207 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 的矩形包含无法无损写入的值").arg(symbol.name));
208 return false;
209 }
210 if (!writePoly(stream,
211 {{rectangle.x0, rectangle.y0},
212 {rectangle.x1, rectangle.y0},
213 {rectangle.x1, rectangle.y1},
214 {rectangle.x0, rectangle.y1},
215 {rectangle.x0, rectangle.y0}}))
216 return false;
217 }
218 for (const IR::SymbolPolylineIR& polyline : symbol.polylines) {
219 if (polyline.partIndex == part && !writePoly(stream, polyline.points)) {
220 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 的折线点数不足").arg(symbol.name));
221 return false;
222 }
223 }
224 for (const IR::SymbolPolygonIR& polygon : symbol.polygons) {
225 if (polygon.partIndex == part && (polygon.points.size() < 3 || !writePoly(stream, polygon.points))) {
226 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 的多边形点数不足").arg(symbol.name));
227 return false;
228 }
229 }
230 stream << " )\n";
231 return stream.status() == QTextStream::Ok;
232}

243bool writeComponentDef(QTextStream& stream,
244 const IR::SymbolComponentIR& symbol,
245 const QString& componentName,
246 const QList<QString>& partNames,
247 QStringList& diagnostics) {
248 QSet<QString> pinNumbers; // 已出现的引脚编号集合。
249 stream << " (compDef " << quote(componentName) << "\n"
250 << " (originalName " << quote(symbol.name) << ")\n"
251 << " (compHeader (compType Normal) (numPins " << symbol.pins.size() << ") (numParts "
252 << symbol.partCount << ") (composition " << (symbol.partCount > 1 ? "Heterogeneous" : "Homogeneous")
253 << ") (refDesPrefix "
254 << quote(symbol.designatorPrefix.isEmpty() ? QStringLiteral("U") : symbol.designatorPrefix)
255 << ") (numType Alpha))\n";
256 for (const IR::SymbolPinIR& pin : symbol.pins) {
257 if (pinNumbers.contains(pin.designator.toLower())) {
258 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 存在重复引脚编号：%2").arg(symbol.name, pin.designator));
259 return false;
260 }
261 pinNumbers.insert(pin.designator.toLower());
262 stream << " (compPin " << quote(pin.designator) << " " << quote(pin.name) << " (partNum "
263 << (pin.commonToAllParts ? 0 : pin.partIndex + 1) << ") (symPinNum " << quote(pin.designator)
264 << ") (gateEq 0) (pinEq 0) (pinType " << pinType(pin.electricalType) << "))\n";
265 }
266 for (int part = 0; part < partNames.size(); ++part)
267 stream << " (attachedSymbol (partNum " << part + 1 << ") (altType Normal) (symbolName "
268 << quote(partNames.at(part)) << "))\n";
269 const QString footprintName = safeName(symbol.footprintName); // 清洗后的封装名称。
270 if (!symbol.footprintName.isEmpty() && !footprintName.isEmpty())
271 stream << " (attachedPattern (patternNum 1) (patternName " << quote(footprintName) << "))\n";
272 else
273 diagnostics.append(QStringLiteral("P-CAD: 符号 %1 没有封装关联，已输出独立符号").arg(symbol.name));
274 stream << " )\n";
275 return stream.status() == QTextStream::Ok;
276}

308bool ExporterPcadSymbol::exportSymbolLibrary(const QList<IR::SymbolComponentIR>& symbols,
309 const QString& libName,
310 const QString& filePath,
311 bool appendMode,
312 bool updateMode,
313 const QString&) {
314 m_diagnostics.clear();
315 if (appendMode || updateMode) {
316 m_diagnostics.append(QStringLiteral("P-CAD 原理图库当前只支持完整重写，不支持追加或更新模式"));
317 return false;
318 }
319 if (symbols.isEmpty()) {
320 m_diagnostics.append(QStringLiteral("P-CAD: 没有可导出的符号"));
321 return false;
322 }
323 QFile file(filePath);
324 if (!file.open(QIODevice::WriteOnly | QIODevice::Truncate | QIODevice::Text)) {
325 m_diagnostics.append(QStringLiteral("P-CAD: 无法写入符号库：%1").arg(filePath));
326 return false;
327 }
328 QTextStream stream(&file);
329 stream.setEncoding(QStringConverter::Utf8);
330 stream << "(ACCEL_ASCII " << quote(libName.isEmpty() ? QStringLiteral("EasyKiConverter") : libName) << "\n"
331 << " (asciiHeader\n (asciiVersion 4 0)\n (fileUnits MM)\n )\n"
332 << " (library\n";
333 QSet<QString> usedNames; // 已使用的组件名称。
334 for (const IR::SymbolComponentIR& symbol : symbols) {
335 if (!validateSymbol(symbol, m_diagnostics))
336 return false;
337 const QString componentName = safeName(symbol.name); // 当前清洗后的组件名称。
338 if (usedNames.contains(componentName.toLower())) {
339 m_diagnostics.append(QStringLiteral("P-CAD: 符号名称清洗后冲突：%1").arg(symbol.name));
340 return false;
341 }
342 usedNames.insert(componentName.toLower());
343 QList<QString> partNames; // 当前组件部件名称列表。
344 for (int part = 0; part < symbol.partCount; ++part) {
345 const QString partName = // 当前部件名称。
346 symbol.partCount == 1 ? componentName : componentName + QStringLiteral("_P%1").arg(part + 1);
347 partNames.append(partName);
348 if (!writeSymbolDef(stream, symbol, partName, part, m_diagnostics))
349 return false;
350 }
351 if (!writeComponentDef(stream, symbol, componentName, partNames, m_diagnostics))
352 return false;
353 }
354 stream << " )\n)\n";
355 stream.flush();
356 if (stream.status() != QTextStream::Ok) {
357 m_diagnostics.append(QStringLiteral("P-CAD: 写入符号库时发生 I/O 错误"));
358 return false;
359 }
360 m_diagnostics.append(
361 QStringLiteral("P-CAD: 已输出 %1 个符号及其器件关联；封装和 3D 文件由独立阶段生成").arg(symbols.size()));
362 return true;
363}
