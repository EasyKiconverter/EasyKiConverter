49int ExportProgressStageCoordinator::averageTypeProgress(const ExportOverallProgress& progress,
50 const QStringList& typeNames) {
51 int sum = 0; // 已找到类型的进度总和。
52 int count = 0; // 已找到类型数量。
53 for (const QString& typeName : typeNames) { // 遍历请求的导出类型。
54 const auto it = progress.exportTypeProgress.constFind(typeName); // 当前类型的进度迭代器。
55 if (it == progress.exportTypeProgress.constEnd())
56 continue;
57 sum += it.value().percentage();
58 ++count;
59 }
60 return count == 0 ? 0 : qBound(0, sum / count, 100);
61}

85int ExportProgressStageCoordinator::weightedOverallProgress(const ExportProgressViewModel& viewModel) {
86 // 按三个阶段权重计算未裁剪的总体进度值。
87 const int weighted = (viewModel.m_fetchProgress * kFetchWeight) + (viewModel.m_processProgress * kProcessWeight) +
88 (viewModel.m_writeProgress * kWriteWeight);
89 return qBound(0, weighted / 100, 100);
90}

97void ExportProgressStageCoordinator::handlePreloadProgress(ExportProgressViewModel& viewModel,
98 const PreloadProgress& progress) {
99 QString statusText =
100 QString("Preloading... %1/%2").arg(progress.completedCount).arg(progress.totalCount); // 预加载状态文本。
101 if (!progress.currentComponentId.isEmpty())
102 statusText += QString(" (%1)").arg(progress.currentComponentId);
103 viewModel.setStatus(statusText);
104 viewModel.m_fetchProgress = qBound(0, progress.percentage(), 100);
105 viewModel.m_processProgress = 0;
106 viewModel.m_writeProgress = 0;
107 viewModel.setProgress(weightedOverallProgress(viewModel));
108 emit viewModel.stageProgressChanged();

116void ExportProgressStageCoordinator::handleProgress(ExportProgressViewModel& viewModel,
117 const ExportOverallProgress& progress) {
118 // 按当前阶段更新三段式进度，并确保后续阶段不超过前置阶段。
119 switch (progress.currentStage) {
120 case ExportOverallProgress::Stage::Preloading:
121 viewModel.setStatus("Preloading components...");
122 viewModel.m_fetchProgress = progress.preloadProgress.percentage();
123 viewModel.m_processProgress = 0;
124 viewModel.m_writeProgress = 0;
125 viewModel.setProgress(weightedOverallProgress(viewModel));
126 break;
127 case ExportOverallProgress::Stage::Exporting: {
128 viewModel.setStatus("Exporting components...");
129 viewModel.m_fetchProgress =
130 stageTypeProgress(viewModel, progress, {kPreviewType, kDatasheetType, kModel3DType});
131 const int rawProcessProgress =
132 stageTypeProgress(viewModel, progress, {kSymbolType, kFootprintType, kModel3DType});
133 const int rawWriteProgress = stageTypeProgress( // 未限制前的写入阶段进度。
134 viewModel,
135 progress,
136 {kSymbolType, kFootprintType, kModel3DType, kPreviewType, kDatasheetType});
137 viewModel.m_processProgress = qMin(rawProcessProgress, viewModel.m_fetchProgress);
138 viewModel.m_writeProgress = qMin(rawWriteProgress, viewModel.m_processProgress);
139 viewModel.setProgress(weightedOverallProgress(viewModel));
140 break;
141 }
142 case ExportOverallProgress::Stage::Completed:
143 viewModel.m_fetchProgress = 100;
144 viewModel.m_processProgress = 100;
145 viewModel.m_writeProgress = 100;
146 viewModel.setProgress(100);
147 viewModel.setStatus("Export completed");
148 break;
149 case ExportOverallProgress::Stage::Cancelled:
150 viewModel.m_fetchProgress = 0;
151 viewModel.m_processProgress = 0;
152 viewModel.m_writeProgress = 0;
153 viewModel.setProgress(0);
154 viewModel.setStatus("Export cancelled");
155 break;
156 case ExportOverallProgress::Stage::Failed:
157 viewModel.m_fetchProgress = 0;
158 viewModel.m_processProgress = 0;
159 viewModel.m_writeProgress = 0;
160 viewModel.setStatus("Export failed");
161 break;
162 default:
163 break;
164 }
165 emit viewModel.stageProgressChanged();
166}
