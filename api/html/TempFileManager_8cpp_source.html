355bool TempFileManager::recoverIncompleteTransactions() {
356 QDir backupRoot(backupRootPath());
357 if (!backupRoot.exists()) {
358 return true;
359 }
360
361 bool recoveredAll = true;
362 QList<QFileInfo> committedTransactionDirs;
363 const QFileInfoList transactionDirs =
364 backupRoot.entryInfoList(QDir::Dirs | QDir::NoDotAndDotDot, QDir::Time);
365 for (const QFileInfo& transactionDirInfo : transactionDirs) {
366 const QString manifestPath =
367 QDir(transactionDirInfo.absoluteFilePath()).filePath(QStringLiteral("manifest.json"));
368 if (!QFile::exists(manifestPath)) {
369 continue;
370 }
371
372 bool committed = false;
373 const QVector<BackupEntry> entries = readManifestEntries(manifestPath, &committed);
374 if (committed) {
375 committedTransactionDirs.append(transactionDirInfo);
376 continue;
377 }
378
379 for (const BackupEntry& entry : entries) {
380 if (!entry.tempPath.isEmpty() && pathExists(entry.tempPath, entry.isDirectory)) {
381 if (!removePath(entry.tempPath, entry.isDirectory)) {
382 qWarning() << "TempFileManager: Failed to remove orphaned temp path during recovery:"
383 << entry.tempPath;
384 recoveredAll = false;
385 }
386 }
387
388 if (entry.existedBefore && !pathExists(entry.finalPath, entry.isDirectory) &&
389 pathExists(entry.backupPath, entry.isDirectory)) {
390 if (!movePathWithFallback(entry.backupPath, entry.finalPath, entry.isDirectory)) {
391 qWarning() << "TempFileManager: Failed to restore backup during recovery:" << entry.backupPath
392 << "to" << entry.finalPath;
393 recoveredAll = false;
394 }
395 }
396 }
397 }
398
399 const QDateTime expirationCutoff =
400 QDateTime::currentDateTimeUtc().addDays(-kDefaultBackupRetentionDays);
401 for (int i = 0; i < committedTransactionDirs.size(); ++i) {
402 const QFileInfo& transactionDirInfo = committedTransactionDirs.at(i);
403 const bool exceedsMaxSets = i >= kDefaultMaxBackupSets;
404 const bool isExpired = transactionDirInfo.lastModified().toUTC() < expirationCutoff;
405 if ((exceedsMaxSets || isExpired) && !QDir(transactionDirInfo.absoluteFilePath()).removeRecursively()) {
406 qWarning() << "TempFileManager: Failed to prune committed backup set:"
407 << transactionDirInfo.absoluteFilePath();
408 recoveredAll = false;
409 }
410 }
411
412 return recoveredAll;
413}

419bool TempFileManager::commitBatchLocked(const QVector<CommitItem>& items) {
420 if (items.isEmpty()) {
421 return true;
422 }
423
424 const QString transactionId = createTransactionId();
425 const QString transactionDirPath = QDir(backupRootPath()).filePath(transactionId);
426 if (!QDir().mkpath(transactionDirPath)) {
427 qWarning() << "TempFileManager: Failed to create backup transaction directory:" << transactionDirPath;
428 return false;
429 }
430
431 QVector<BackupEntry> entries;
432 entries.reserve(items.size());
433 for (int i = 0; i < items.size(); ++i) {
434 const CommitItem& item = items[i];
435 if (item.tempPath.isEmpty() || item.finalPath.isEmpty()) {
436 qWarning() << "TempFileManager: Cannot commit empty temp/final path";
437 return false;
438 }
439 if (!pathExists(item.tempPath, item.isDirectory)) {
440 qWarning() << "TempFileManager: Temp path does not exist:" << item.tempPath;
441 m_tempFiles.remove(item.tempPath);
442 return false;
443 }
444 if (!ensureParentDirectory(item.finalPath)) {
445 qWarning() << "TempFileManager: Failed to create final parent directory:" << item.finalPath;
446 return false;
447 }
448
449 const QString backupName =
450 QStringLiteral("%1_%2").arg(i, 4, 10, QLatin1Char('0')).arg(QFileInfo(item.finalPath).fileName());
451 BackupEntry entry;
452 entry.finalPath = item.finalPath;
453 entry.backupPath = QDir(transactionDirPath).filePath(backupName);
454 entry.tempPath = item.tempPath;
455 entry.isDirectory = item.isDirectory;
456 entry.existedBefore = pathExists(item.finalPath, item.isDirectory);
457 entries.append(entry);
458 }
459
460 const QString manifestPath =
461 QDir(transactionDirPath).filePath(QStringLiteral("manifest.json"));
462 if (!writeManifest(manifestPath, transactionId, m_outputPath, entries, false)) {
463 return false;
464 }
465
466 QVector<BackupEntry> backedUpEntries;
467 for (const BackupEntry& entry : entries) {
468 if (!entry.existedBefore) {
469 backedUpEntries.append(entry);
470 continue;
471 }
472
473 if (!movePathWithFallback(entry.finalPath, entry.backupPath, entry.isDirectory)) {
474 qWarning() << "TempFileManager: Failed to back up existing target:" << entry.finalPath << "to"
475 << entry.backupPath;
476 for (auto it = backedUpEntries.crbegin(); it != backedUpEntries.crend(); ++it) {
477 if (it->existedBefore && pathExists(it->backupPath, it->isDirectory) &&
478 !pathExists(it->finalPath, it->isDirectory)) {
479 movePathWithFallback(it->backupPath, it->finalPath, it->isDirectory);
480 }
481 }
482 return false;
483 }
484
485 backedUpEntries.append(entry);
486 }
487
488 QVector<BackupEntry> promotedEntries;
489 for (const BackupEntry& entry : entries) {
490 if (!movePathWithFallback(entry.tempPath, entry.finalPath, entry.isDirectory)) {
491 qWarning() << "TempFileManager: Failed to promote temp path:" << entry.tempPath << "to" << entry.finalPath;
492
493 removePath(entry.finalPath, entry.isDirectory);
494 for (auto it = promotedEntries.crbegin(); it != promotedEntries.crend(); ++it) {
495 removePath(it->finalPath, it->isDirectory);
496 }
497 for (auto it = backedUpEntries.crbegin(); it != backedUpEntries.crend(); ++it) {
498 if (it->existedBefore && pathExists(it->backupPath, it->isDirectory)) {
499 if (!movePathWithFallback(it->backupPath, it->finalPath, it->isDirectory)) {
500 qWarning() << "TempFileManager: Failed to restore backup:" << it->backupPath << "to"
501 << it->finalPath;
502 }
503 }
504 }
505 return false;
506 }
507
508 promotedEntries.append(entry);
509 }
510
511 if (!writeManifest(manifestPath, transactionId, m_outputPath, entries, true)) {
512 qWarning() << "TempFileManager: Commit succeeded but manifest could not be finalized:" << manifestPath;
513 }
514
515 for (const BackupEntry& entry : entries) {
516 m_tempFiles.remove(entry.tempPath);
517 qDebug() << "TempFileManager: Committed with backup:" << entry.tempPath << "->" << entry.finalPath;
518 }
519 cleanupEmptyTempDirectoryLocked();
520
521 return true;
522}

555void TempFileManager::cleanupTempDirectory() {
556 QMutexLocker locker(&m_mutex);
557
558 QString tempDir = tempDirectory();
559 if (tempDir.isEmpty() || !QDir(tempDir).exists()) {
560 return;
561 }
562 // 共享临时目录仍被其他导出阶段使用时，不能清理其中的文件。
563 if (hasOtherTempDirectoryUsers(tempDir)) {
564 qDebug() << "TempFileManager: Skip cleanup while another stage uses" << tempDir;
565 return;
566 }
567
568 int deletedCount = 0;
569 QDir dir(tempDir);
570 for (const QFileInfo& info : dir.entryInfoList(QDir::Files)) {
571 if (deleteFile(info.absoluteFilePath())) {
572 deletedCount++;
573 }
574 }
575
576 if (dir.isEmpty()) {
577 QDir().rmdir(tempDir);
578 }
579
580 locker.unlock();
581 emit cleanupCompleted(deletedCount);
582}

587void TempFileManager::cleanupOrphanedTempFiles() {
588 QString tempDir = tempDirectory();
589 if (tempDir.isEmpty() || !QDir(tempDir).exists()) {
590 return;
591 }
592 // 孤立文件清理同样必须尊重共享目录引用，避免破坏并行导出任务。

594 qDebug() << "TempFileManager: Skip orphan cleanup while another stage uses" << tempDir;
595 return;
596 }
597
598 int deletedCount = 0;
599 QDir dir(tempDir);
600 for (const QFileInfo& info : dir.entryInfoList(QDir::Files)) {
601 if (info.fileName().startsWith('.')) {
602 continue; // 跳过隐藏文件
603 }
604 if (deleteFile(info.absoluteFilePath())) {
605 deletedCount++;
606 }
607 }
608
609 if (dir.isEmpty()) {
610 QDir().rmdir(tempDir);
611 }
612
613 qDebug() << "TempFileManager: Cleaned up" << deletedCount << "orphaned temp files";
