17QByteArray AltiumSchImageStorageEncoder::encode(const QList<AltiumSchComponent>& components,
18 const QHash<const AltiumSchImage*, QString>& imageNames,
19 QStringList& diagnostics) {
20 QList<const AltiumSchImage*> embeddedImages; // 已确认需要嵌入 Storage 的图片对象。
21 for (const AltiumSchComponent& component : components) {
22 for (const AltiumSchImage& image : component.images) {
23 if (image.embedImage && !image.data.isEmpty() && imageNames.contains(&image))
24 embeddedImages.append(&image);
25 }
26 }
27
28 constexpr qsizetype kMaxStorageEntrySize = 0x00FFFFFF; // Altium Storage 单条记录的最大字节数。
29 QList<QByteArray> encodedEntries; // 已压缩并通过大小检查的 Storage 条目。
30 encodedEntries.reserve(embeddedImages.size());
31 for (const AltiumSchImage* image : embeddedImages) {
32 const QByteArray compressed = qCompress(image->data, 9).mid(4); // 去除 Qt 压缩头后的图片数据。
33 const QByteArray name = imageNames.value(image).toLocal8Bit(); // Storage 条目文件名。
34 if (compressed.isEmpty() || name.isEmpty() || name.size() > 255) {
35 diagnostics.append(
36 QStringLiteral("Altium SchLib 图片 Storage 条目无效，已跳过: %1").arg(QString::fromLocal8Bit(name)));
37 continue;
38 }
39
40 QByteArray entry; // 当前图片的 Storage 条目缓冲区。
41 QDataStream entryStream(&entry, QIODevice::WriteOnly); // 条目小端序写入器。
42 entryStream.setByteOrder(QDataStream::LittleEndian);
43 entryStream << static_cast<quint8>(0xD0) << static_cast<quint8>(name.size());
44 entry.append(name);
45 entryStream.device()->seek(entry.size());
46 entryStream << static_cast<quint32>(compressed.size());
47 entry.append(compressed);
48 if (entry.size() > kMaxStorageEntrySize) {
49 diagnostics.append(
50 QStringLiteral("Altium SchLib 图片 Storage 条目过大，已跳过: %1").arg(QString::fromLocal8Bit(name)));
51 continue;
52 }
53 encodedEntries.append(entry);
54 }
55
56 QByteArray storageData; // 最终 Storage 流缓冲区。
57 AltiumBinaryWriter storageWriter(storageData); // Storage 头部参数写入器。
58 QMap<QString, QString> storageParams; // Storage 头部参数。
59 storageParams["HEADER"] = "Icon storage";
60 storageParams["Weight"] = QString::number(encodedEntries.size());
61 storageWriter.writeCStringParameterBlock(storageParams);
62
63 for (const QByteArray& entry : encodedEntries) { // 追加每个图片 Storage 条目。
64 QDataStream blockStream(&storageData, QIODevice::WriteOnly | QIODevice::Append); // 条目块写入器。
65 blockStream.setByteOrder(QDataStream::LittleEndian);
66 blockStream << static_cast<quint32>(0x01000000U | static_cast<quint32>(entry.size()));
67 blockStream.writeRawData(entry.constData(), entry.size());
68 }
69 return storageData;
70}
