20void ComponentCadFetchCoordinator::handleDataFetched(ComponentService& service,
21 const QString& componentId,
22 const QJsonObject& data) {
23 uint64_t generation = 0; // 当前请求的缓存代次。
24 {
25 QMutexLocker locker(&service.m_fetchingComponentsMutex);
26 const auto it = service.m_fetchingComponents.find(componentId.toUpper()); // 活动请求状态。
27 if (it != service.m_fetchingComponents.end()) {
28 generation = it->cacheGeneration;
29 }
30 }
31
32 if (generation == 0) {
33 // 找不到请求状态时，说明回调已经晚于请求生命周期，不能再写入缓存。
34 qDebug() << "handleCadDataFetched: No FetchingComponent for" << componentId << ", discarding stale callback";
35 return;
36 }
37
38 auto future = // 后台 CAD 解析任务。
39 QtConcurrent::run([componentId, data]() { return CadDataLoader::parseCadPayload(componentId, data); });
40 auto* watcher = new QFutureWatcher<CadParseResult>(&service); // 解析任务完成监听器。
41 QObject::connect(watcher, &QFutureWatcher<CadParseResult>::finished, &service, [&service, watcher, generation]() {
42 const CadParseResult parsed = watcher->result(); // CAD 解析结果。
43 watcher->deleteLater();
44
45 CadFetchTaskResult result; // 转交服务层的 CAD 任务结果。
46 result.componentId = parsed.componentId;
47 result.parsed = parsed;
48 result.errorMessage = parsed.errorMessage;
49 result.success = parsed.success;
50 ComponentCadFetchCoordinator::handleFetchResult(service, result, generation);
51 });
52 watcher->setFuture(future);
53}

61void ComponentCadFetchCoordinator::handleFetchResult(ComponentService& service,
62 const CadFetchTaskResult& result,
63 uint64_t expectedGeneration) {
64 {
65 QMutexLocker locker(&service.m_fetchingComponentsMutex);
66 const auto it = service.m_fetchingComponents.find(result.componentId); // 待校验的活动请求。
67 if (it == service.m_fetchingComponents.end() || it->cacheGeneration != expectedGeneration) {
68 qDebug() << "ComponentService: Discarding stale CAD result for" << result.componentId;
69 return;
70 }
71 }
72
73 if (!result.success) {
74 service.emitFetchErrorAndClearState(result.componentId, result.errorMessage, expectedGeneration);
75 return;
76 }
77
78 {
79 QMutexLocker locker(&service.m_fetchingComponentsMutex);
80 auto it = service.m_fetchingComponents.find(result.componentId); // 待更新的活动请求。
81 if (it == service.m_fetchingComponents.end() || it->cacheGeneration != expectedGeneration) {
82 qDebug() << "ComponentService: Discarding stale CAD result for" << result.componentId;
83 return;
84 }
85 it->data = result.parsed.componentData;
86 it->hasCadData = true;
87 it->requestActive = false;
88 }
89
90 service.updateComponentCache(result.componentId, result.parsed.componentData);
91 emit service.cadDataReady(result.componentId, result.parsed.componentData);
92 // 使用异步保存，不阻塞 UI。
93 ComponentCacheService::instance()->saveComponentMetadataAsync(
94 result.componentId, result.parsed.componentData, expectedGeneration, /*replaceModel3DMetadata=*/true);
95 ComponentCacheService::instance()->saveCadDataJson(
96 result.componentId, QJsonDocument(result.parsed.resultData).toJson(QJsonDocument::Compact), expectedGeneration);
97
98 if (service.m_parallelContext != nullptr) {
99 service.handleParallelDataCollected(result.componentId, result.parsed.componentData);
100 }
101}
