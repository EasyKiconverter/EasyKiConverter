23QString safeDecalName(const QString& value) {
24 QString result;
25 result.reserve(value.size());
26 for (const QChar character : value) {
27 const ushort code = character.unicode();
28 if ((code >= 'A' && code <= 'Z') || (code >= 'a' && code <= 'z') || (code >= '0' && code <= '9') ||
29 character == QChar('_') || character == QChar('-'))
30 result.append(character);
31 else
32 result.append(QChar('_'));
33 }
34 return result.left(40);
35}

75QString padShape(const IR::FootprintPadIR& pad, QStringList& diagnostics) {
76 // 只返回规范中有明确制造语义的形状代码。
77 switch (pad.shape) {
78 case IR::PadShape::Ellipse:
79 return qFuzzyCompare(pad.size.width(), pad.size.height()) ? QStringLiteral("R") : QStringLiteral("OF");
80 case IR::PadShape::Rect:
81 return qFuzzyCompare(pad.size.width(), pad.size.height()) ? QStringLiteral("S") : QStringLiteral("RF");
82 case IR::PadShape::Oval:
83 return QStringLiteral("OF");
84 case IR::PadShape::RoundRect:
85 case IR::PadShape::Polygon:
86 case IR::PadShape::Trapezoid:
87 diagnostics.append(
88 QStringLiteral("PADS: 焊盘 %1 的形状无法由当前 ASCII Pad Stack 语义无损表达").arg(pad.number));
89 return {};
90 }
91 diagnostics.append(QStringLiteral("PADS: 焊盘 %1 使用未知形状").arg(pad.number));
92 return {};
93}

116bool writeFootprint(const IR::FootprintComponentIR& footprint, const QString& path, QStringList& diagnostics) {
117 const QString name = safeDecalName(footprint.name);
118 if (name.isEmpty()) {
119 diagnostics.append(QStringLiteral("PADS: 封装名称清洗后为空"));
120 return false;
121 }
122
123 if (!footprint.holes.isEmpty()) {
124 diagnostics.append(
125 QStringLiteral("PADS: 封装 %1 包含独立安装孔，当前 Decal 导出无法保持其机械孔语义").arg(name));
126 return false;
127 }
128 if (!footprint.arcs.isEmpty()) {
129 diagnostics.append(QStringLiteral("PADS: 封装 %1 包含圆弧，当前 Decal 导出尚未实现圆弧图元").arg(name));
130 return false;
131 }
132 for (const IR::FootprintPadIR& pad : footprint.pads) {
133 if (pad.number.isEmpty() || !isAsciiText(pad.number)) {
134 diagnostics.append(QStringLiteral("PADS: 焊盘编号必须为非空 ASCII 文本：%1").arg(pad.number));
135 return false;
136 }
137 if (!std::isfinite(pad.position.x()) || !std::isfinite(pad.position.y()) || !std::isfinite(pad.size.width()) ||
138 !std::isfinite(pad.size.height())) {
139 diagnostics.append(QStringLiteral("PADS: 焊盘 %1 包含非法坐标或尺寸").arg(pad.number));
140 return false;
141 }
142 }
143 for (const IR::FootprintTextIR& text : footprint.texts) {
144 if (text.isDisplayed && !isAsciiText(text.text)) {
145 diagnostics.append(
146 QStringLiteral("PADS: 文本包含非 ASCII 字符，当前 Decal 文本编码未定义：%1").arg(text.text));
147 return false;
148 }
149 }
150 if (!footprint.models3d.isEmpty())
151 diagnostics.append(QStringLiteral("PADS: PCB Decal 不包含三维模型关联，将由独立三维阶段输出 %1 个模型")
152 .arg(footprint.models3d.size()));
153
154 QList<QString> padShapes;
155 for (const IR::FootprintPadIR& pad : footprint.pads) {
156 const QString shape = padShape(pad, diagnostics);
157 if (shape.isEmpty())
158 return false;
159 padShapes.append(shape);
160 }
161
162 // PADS 图元数量。
163 const int pieces = footprint.circles.size() + footprint.rectangles.size() +
164 std::count_if(footprint.tracks.cbegin(),
165 footprint.tracks.cend(),
166 [](const auto& track) { return track.points.size() >= 2; }) +
167 std::count_if(footprint.regions.cbegin(),
168 footprint.regions.cend(),
169 [](const auto& region) { return region.vertices.size() >= 3; }) +
170 std::count_if(footprint.outlines.cbegin(), footprint.outlines.cend(), [](const auto& outline) {
171 return outline.points.size() >= 2;
172 });
173 // 可见文本数量。
174 const int textCount = std::count_if(footprint.texts.cbegin(), footprint.texts.cend(), [](const auto& text) {
175 return text.isDisplayed && !text.text.isEmpty();
176 });
177 const QDateTime timestamp = QDateTime::currentDateTimeUtc();
178
179 QFile file(path);
180 if (!file.open(QIODevice::WriteOnly | QIODevice::Truncate | QIODevice::Text)) {
181 diagnostics.append(QStringLiteral("PADS: 无法写入文件 %1").arg(path));
182 return false;
183 }
184 QTextStream stream(&file);
185 stream.setEncoding(QStringConverter::Utf8);
186 stream << name << " I 0 0 0 0 " << pieces << ' ' << textCount << ' ' << footprint.pads.size() << ' '
187 << footprint.pads.size() << " 0\n";
188 stream << "TIMESTAMP " << timestamp.toString(QStringLiteral("yyyy.MM.dd.hh.mm.ss")) << "\n";
189
190 for (const IR::FootprintCircleIR& circle : footprint.circles) {
191 if (!writePieceHeader(stream, QStringLiteral("CIRCLE"), 2, circle.strokeWidth, 0))
192 return false;
193 stream << number(circle.center.x() - circle.radius) << ' ' << number(circle.center.y() - circle.radius) << '\n';
194 stream << number(circle.center.x() + circle.radius) << ' ' << number(circle.center.y() + circle.radius) << '\n';
195 }
196 for (const IR::FootprintRectangleIR& rectangle : footprint.rectangles) {
197 if (!writePieceHeader(stream, QStringLiteral("CLOSED"), 5, rectangle.strokeWidth, 0))
198 return false;
199 const QRectF bounds = rectangle.bounds;
200 const QList<QPointF> corners = {
201 bounds.topLeft(),
202 bounds.topRight(),
203 bounds.bottomRight(),
204 bounds.bottomLeft(),
205 bounds.topLeft()};
206 for (const QPointF& point : corners)
207 stream << number(point.x()) << ' ' << number(point.y()) << '\n';
208 }
209 for (const IR::FootprintTrackIR& track : footprint.tracks) {
210 if (track.points.size() < 2)
211 continue;
212 if (!writePieceHeader(stream, QStringLiteral("OPEN"), track.points.size(), track.width, 0))
213 return false;
214 for (const QPointF& point : track.points)
215 stream << number(point.x()) << ' ' << number(point.y()) << '\n';
216 }
217 for (const IR::FootprintRegionIR& region : footprint.regions) {
218 if (region.vertices.size() < 3)
219 continue;
220 if (!writePieceHeader(stream, QStringLiteral("CLOSED"), region.vertices.size() + 1, 0.0, 0))
221 return false;
222 for (const QPointF& point : region.vertices)
223 stream << number(point.x()) << ' ' << number(point.y()) << '\n';
224 const QPointF& first = region.vertices.first();
225 stream << number(first.x()) << ' ' << number(first.y()) << '\n';
226 }
227 for (const IR::FootprintOutlineIR& outline : footprint.outlines) {
228 if (outline.points.size() < 2)
229 continue;
230 if (!writePieceHeader(stream, QStringLiteral("OPEN"), outline.points.size(), outline.strokeWidth, 0))
231 return false;
232 for (const QPointF& point : outline.points)
233 stream << number(point.x()) << ' ' << number(point.y()) << '\n';
234 }
235 for (int index = 0; index < footprint.texts.size(); ++index) {
236 const IR::FootprintTextIR& text = footprint.texts.at(index);
237 if (!text.isDisplayed || text.text.isEmpty())
238 continue;
239 stream << number(text.position.x()) << ' ' << number(text.position.y()) << ' ' << angle(text.rotation)
240 << " 0 1 " << number(text.fontSize > 0.0 ? text.fontSize : 1.0) << ' '
241 << number(text.strokeWidth > 0.0 ? text.strokeWidth : 0.15) << " 0 0 0 0 Default Font\n";
242 QString textValue = text.text;
243 stream << textValue.replace(QChar('\n'), QChar(' ')) << '\n';
244 }
245 for (int index = 0; index < footprint.pads.size(); ++index) {
246 const IR::FootprintPadIR& pad = footprint.pads.at(index);
247 const QString pin = pad.number;
248 stream << "T " << number(pad.position.x()) << ' ' << number(pad.position.y()) << ' ' << number(pad.position.x())
249 << ' ' << number(pad.position.y()) << ' ' << pin << '\n';
250 const QString plated = pad.isPlated ? QStringLiteral("P") : QStringLiteral("N");
251 const double drill = pad.isThroughHole() ? pad.holeSize : 0.0;
252 const int layerCount = pad.isThroughHole() ? 2 : 1;
253 stream << "PAD " << pin << ' ' << layerCount << ' ' << plated << ' ' << number(drill);
254 if (pad.isThroughHole() && pad.holeLength > 0.0)
255 stream << " 0 " << number(pad.holeLength) << " 0";
256 stream << '\n';
257 const QString shape = padShapes.at(index);
258 const double width = qMax(pad.size.width(), pad.size.height());
259 stream << (pad.isThroughHole() ? "-2 " : "-2 ") << number(width) << ' ' << shape;
260 if (shape == QStringLiteral("OF") || shape == QStringLiteral("RF"))
261 stream << " 0 " << angle(pad.rotation) << ' ' << number(pad.size.height()) << " 0";
262 stream << '\n';
263 if (pad.isThroughHole()) {
264 stream << "-0 " << number(width) << ' ' << shape;
265 if (shape == QStringLiteral("OF") || shape == QStringLiteral("RF"))
266 stream << " 0 " << angle(pad.rotation) << ' ' << number(pad.size.height()) << " 0";
267 stream << '\n';
268 }
269 }
270 stream.flush();
271 if (stream.status() != QTextStream::Ok) {
272 diagnostics.append(QStringLiteral("PADS: 写入文件时发生 I/O 错误 %1").arg(path));
273 return false;
274 }
275 return true;
276}
