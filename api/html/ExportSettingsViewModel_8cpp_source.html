30QString translateCacheDirectoryError(const QString& error) {
31 const auto translate = [](const char* source) {
32 return QCoreApplication::translate("CacheDirectoryErrors", source);
33 };
34 const auto translateWithSuffix = [&error, &translate](const QString& prefix, const char* source) {
35 return translate(source).arg(error.mid(prefix.size()));
36 };
37
38 const QStringList exactMessages = {
39 QStringLiteral("缓存目录不能为空"),
40 QStringLiteral("缓存目录必须使用绝对路径"),
41 QStringLiteral("缓存目录不能包含符号链接路径分量"),
42 QStringLiteral("不能选择文件系统根目录或用户主目录"),
43 QStringLiteral("缓存路径必须是目录"),
44 QStringLiteral("缓存目录必须为空，或已经包含 EasyKiConverter 所有权标记"),

47 if (exactMessages.contains(error))
48 return translate(error.toUtf8().constData());

314bool ExportSettingsViewModel::requiresFullReplacement() const {
315 if (!m_targetModel)
316 return false;
317
318 const auto target = // 当前目标格式枚举值。
319 static_cast<TargetEdaFormat>(m_targetModel->currentTargetFormat());
320 return target == TargetEdaFormat::Xpedition || target == TargetEdaFormat::Allegro ||
321 target == TargetEdaFormat::Pads || target == TargetEdaFormat::Eagle || target == TargetEdaFormat::Pcad ||
322 target == TargetEdaFormat::Cadstar || target == TargetEdaFormat::Orcad ||
323 target == TargetEdaFormat::LibrePcb;

351void ExportSettingsViewModel::setTargetModel(ExportTargetModel* model) {
352 if (m_targetModel == model)
353 return;
354 m_targetModel = model;
355 if (m_targetModel) {
356 connect(m_targetModel, &ExportTargetModel::currentTargetChanged, this, [this]() {
357 if (requiresFullReplacement() && m_exportMode != 2)
358 setExportMode(2);
359 // 读取目标格式，以应用目标格式专属导出限制。
360 const auto targetFormat = static_cast<TargetEdaFormat>(m_targetModel->currentTargetFormat());
361 if (targetFormat == TargetEdaFormat::Allegro) {
362 setExportSymbol(false);
363 if (m_exportModel3D)
364 setExportModel3DFormat(ExportOptions::MODEL_3D_FORMAT_STEP);
365 }
366 if (targetFormat == TargetEdaFormat::Orcad)
367 setExportFootprint(false);
368 if ((targetFormat == TargetEdaFormat::Altium || targetFormat == TargetEdaFormat::LibrePcb ||
369 targetFormat == TargetEdaFormat::Horizon) &&
370 (m_exportModel3DFormat & ExportOptions::MODEL_3D_FORMAT_WRL)) {
371 // 这些原生库只接受 STEP，切换目标时移除 WRL 位。
372 setExportModel3DFormat(ExportOptions::MODEL_3D_FORMAT_STEP);
373 }
374 emit requiresFullReplacementChanged();
375 });
376 if (requiresFullReplacement() && m_exportMode != 2)
377 setExportMode(2);
378 // 初始化当前目标格式对应的导出限制。
379 const auto targetFormat = static_cast<TargetEdaFormat>(m_targetModel->currentTargetFormat());
380 if (targetFormat == TargetEdaFormat::Allegro) {
381 setExportSymbol(false);
382 if (m_exportModel3D)
383 setExportModel3DFormat(ExportOptions::MODEL_3D_FORMAT_STEP);
384 }
385 if (targetFormat == TargetEdaFormat::Orcad)
386 setExportFootprint(false);
387 if ((targetFormat == TargetEdaFormat::Altium || targetFormat == TargetEdaFormat::LibrePcb ||
388 targetFormat == TargetEdaFormat::Horizon) &&
389 (m_exportModel3DFormat & ExportOptions::MODEL_3D_FORMAT_WRL)) {
390 setExportModel3DFormat(ExportOptions::MODEL_3D_FORMAT_STEP);
391 }
392 emit requiresFullReplacementChanged();
393 }

455void ExportSettingsViewModel::setCacheDir(const QString& path) {
456 // 保存安全校验返回的规范化缓存目录路径。
457 QString normalizedPath;
458 // 保存缓存目录校验或迁移失败的原始错误信息。
459 QString error;
460 if (!CacheSafety::validateSelection(path, &normalizedPath, &error)) {
461 // 将安全校验错误转换为当前界面语言。
462 const QString localizedError = translateCacheDirectoryError(error);
463 setStatus(localizedError);
464 emit cacheDirChangeRejected(path, localizedError);
465 return;
466 }
467 if (m_cacheDir == normalizedPath)
468 return;
469
470 if (!ComponentCacheService::instance()->setCacheDir(normalizedPath, /*migrateExistingCache=*/true, &error)) {
471 // 保存迁移失败的可读原因，供状态和拒绝信号共同使用。
472 const QString reason = translateCacheDirectoryError(error.isEmpty() ? QStringLiteral("迁移失败") : error);
473 setStatus(QStringLiteral("缓存目录未切换：%1").arg(reason));
474 emit cacheDirChangeRejected(path, reason);
475 return;
476 }
477
478 m_cacheDir = normalizedPath;
479 m_configService->setCacheDir(normalizedPath);
480 emit cacheDirChanged();
481}

502void ExportSettingsViewModel::startExport(const QStringList& componentIds) {
503 qDebug() << "Starting export for" << componentIds.size() << "components";
504
505 if (m_isExporting) {
506 qWarning() << "Export already in progress";
507 return;
508 }
509
510 if (!m_exportService) {
511 qWarning() << "ParallelExportService is not available";
512 setStatus("Export service not available");
513 return;
514 }
515
516 if (componentIds.isEmpty()) {
517 qWarning() << "No components to export";
518 setStatus("No components to export");
519 return;
520 }
521
522 // Save component IDs for after preload
523 m_pendingComponentIds = componentIds;
524
525 // Build and set export options
526 buildExportOptions();
527
528 setIsExporting(true);
529 setStatus("Preloading component data...");
530
531 // Start preload first
532 m_exportService->startPreload(componentIds);
533}
