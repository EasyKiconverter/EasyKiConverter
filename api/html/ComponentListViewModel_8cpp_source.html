329void ComponentListViewModel::refreshComponentInfo(int index) {
330 // 保存待刷新的列表项。
331 ComponentListItemData* item = nullptr;
332 // 保存待刷新的元器件编号。
333 QString componentId;
334 {
335 QMutexLocker locker(&m_listMutex);
336 if (index >= 0 && index < m_componentList.count()) {
337 item = m_componentList.at(index);
338 componentId = item->componentId();
339 }
340 }
341
342 if (item) {
343 clearAttentionHints();
344 m_bomImportComplete = false;
345 item->setFetching(true);
346 item->setValid(false);
347 item->setRetryable(true);
348 item->setValidationPhase("validating");
349 item->setErrorMessage("");
350 m_service->fetchComponentData(componentId, false);
351
352 m_validationStateManager->startValidation(1);
353 emit filteredCountChanged();
354 }

570void ComponentListViewModel::scheduleListUpdate() {
571 // 防抖：防止定时器级联重启
572 // 如果已经有待处理的列表更新，直接跳过
573 if (m_listUpdatePending) {
574 return;
575 }
576 if (!m_batchListUpdateMode) {
577 m_batchListUpdateMode = true;
578 }
579 // 安全检查：如果 BOM 导入模式已启用但没有待处理组件，重置标志
580 // 这处理了批量添加计时器被外部停止或待处理列表被清空的情况
581 if (m_bomImportMode && m_pendingComponentIds.isEmpty()) {
582 m_bomImportMode = false;
583 m_bomImportPendingUpdates = 0;
584 }
585 // BOM 导入模式下使用较长的更新间隔，减少 UI 抖动
586 if (m_bomImportMode) {
587 m_listUpdatePending = true;
588 m_bomImportUpdateTimer->start();
589 } else {
590 m_listUpdatePending = true;
591 m_batchListUpdateTimer->start();
592 }
593}
