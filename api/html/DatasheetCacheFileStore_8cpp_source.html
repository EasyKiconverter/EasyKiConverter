28QByteArray DatasheetCacheFileStore::load(const ComponentCacheService& owner, const QString& componentId) {
29 QMutexLocker diskLocker(&owner.m_diskWriteMutex);
30 const QString preferredFormat =
31 CacheMetadataStore::read(owner.metadataPath(componentId)).value(QStringLiteral("datasheetFormat")).toString();
32 const QString datasheetFilePath =
33 owner.resolveDatasheetPath(componentId, preferredFormat, false); // 数据手册缓存路径。
34 if (!QFileInfo::exists(datasheetFilePath))
35 return QByteArray();
36
37 QFile file(datasheetFilePath);
38 if (!file.open(QIODevice::ReadOnly))
39 return QByteArray();
40
41 const QByteArray data = file.readAll(); // 数据手册原始字节。
42 file.close();
43 // 根据文件扩展名推断实际数据手册格式。
44 const QString format = datasheetFilePath.endsWith(QStringLiteral(".pdf"), Qt::CaseInsensitive)
45 ? QStringLiteral("pdf")
46 : QStringLiteral("html");
47 if (CacheDataValidator::isValidDatasheet(data, format))
48 return data;
49
50 QString trashError; // 回收站操作失败时的诊断信息。
51 if (CacheSafety::isOwnedComponentFile(owner.cacheDir(), datasheetFilePath))
52 CacheSafety::moveToTrash(datasheetFilePath, &trashError);
53 return QByteArray();
54}

64void DatasheetCacheFileStore::save(const QString& componentId,
65 const QByteArray& data,
66 const QString& format,
67 uint64_t expectedGeneration) {
68 QString effectiveFormat = format.toLower(); // 规范化后的数据手册格式。
69 if (effectiveFormat != QStringLiteral("pdf") && effectiveFormat != QStringLiteral("html"))
70 return;
71 if (effectiveFormat == QStringLiteral("pdf") && !data.startsWith("%PDF-"))
72 effectiveFormat = QStringLiteral("html");
73 if (!CacheDataValidator::isValidDatasheet(data, effectiveFormat))
74 return;
75
76 QMutexLocker diskLocker(&m_owner.m_diskWriteMutex);
77 if (expectedGeneration != 0 &&
78 (m_owner.m_cacheGeneration.load() != expectedGeneration || m_owner.isTombstoned(componentId)))
79 return;
80
81 QString actualPath; // 当前格式对应的数据手册缓存路径。
82 QString trashError; // 回收站操作失败时的诊断信息。
83 {
84 QMutexLocker locker(&m_owner.m_mutex);
85 if (m_owner.ensureComponentDir(componentId).isEmpty())
86 return;
87 actualPath = m_owner.resolveDatasheetPath(componentId, effectiveFormat, true);
88 const QString alternatePath = actualPath.endsWith(QStringLiteral(".pdf")) // 另一种格式的旧缓存路径。
89 ? m_owner.resolveDatasheetPath(componentId, QStringLiteral("html"), true)
90 : m_owner.resolveDatasheetPath(componentId, QStringLiteral("pdf"), true);
91 if (alternatePath != actualPath && QFile::exists(alternatePath) &&
92 CacheSafety::isOwnedComponentFile(m_owner.cacheDir(), alternatePath))
93 CacheSafety::moveToTrash(alternatePath, &trashError);
94 }
95
96 if (CacheMetadataStore::writeAtomically(actualPath, data)) {
97 LOG_DEBUG(LogModule::Core, "Saved datasheet to disk: {}", actualPath);
98 m_owner.enforceDiskCacheLimit();
99 } else {
100 LOG_WARN(LogModule::Core, "Failed to write datasheet: {}", actualPath);
101 }
102 // 数据手册体积较大，只保留二级磁盘缓存，不写入一级内存缓存。
103}
