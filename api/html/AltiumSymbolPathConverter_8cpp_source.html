18bool isFinitePoint(const QPointF& point) {

28bool hasFinitePoints(const QList<QPointF>& points, int minimum) {
29 if (points.size() < minimum)
30 return false;
31 for (const QPointF& point : points) {
32 if (!isFinitePoint(point))
33 return false;
34 }
35 return true;
36}

58int sourceIndexForPart(const Values& values, int index, int partIndex) {
59 int localIndex = 0; // 当前部件内已遍历图元的数量。
60 for (int i = 0; i < index; ++i) {

75void AltiumSymbolPathConverter::append(const IR::SymbolComponentIR& symbol,
76 AltiumSchComponent& component,
77 QStringList& diagnostics) {
78 for (int pathIndex = 0; pathIndex < symbol.paths.size(); ++pathIndex) {
79 const IR::SymbolPathIR& pathSource = symbol.paths.at(pathIndex); // 当前源路径。
80 if (!isValidStrokeWidth(pathSource.strokeWidth)) {
81 diagnostics.append(
82 QStringLiteral("符号 %1 路径图元 %2 的线宽无效，已跳过").arg(symbol.name).arg(pathIndex));
83 continue;
84 }
85
86 // 非填充路径可直接拆分为 Altium 原生线段和 Bézier 记录，避免曲线被
87 // 强制膨胀为大量折线；填充路径仍使用闭合点列以保留填充语义。
88 if (!pathSource.isFilled && !pathSource.segments.isEmpty()) {
89 int segmentIndex = 0; // 当前路径段的来源序号。
90 for (const IR::SymbolPathSegmentIR& segment : pathSource.segments) {
91 bool validSegment = isFinitePoint(segment.start) && isFinitePoint(segment.end); // 路径段参数是否有效。
92 // 按路径段类型校验所需控制点，避免将不完整曲线写入目标库。
93 switch (segment.type) {
94 case IR::SymbolPathSegmentIR::Type::QuadraticBezier:
95 validSegment = validSegment && isFinitePoint(segment.control1);
96 break;
97 case IR::SymbolPathSegmentIR::Type::CubicBezier:
98 validSegment =
99 validSegment && isFinitePoint(segment.control1) && isFinitePoint(segment.control2);
100 break;
101 case IR::SymbolPathSegmentIR::Type::CircularArc:
102 validSegment = validSegment && isFinitePoint(segment.arcMid);
103 break;
104 case IR::SymbolPathSegmentIR::Type::EllipticalArc:
105 validSegment = validSegment && isFinitePoint(segment.arcCenter) &&
106 std::isfinite(segment.radiusX) && std::isfinite(segment.radiusY) &&
107 segment.radiusX > 0.0 && segment.radiusY > 0.0 &&
108 std::isfinite(segment.arcStartAngle) && std::isfinite(segment.arcEndAngle);
109 break;
110 case IR::SymbolPathSegmentIR::Type::Line:
111 break;
112 }
113 if (!validSegment) {
114 diagnostics.append(QStringLiteral("符号 %1 路径图元 %2 的段 %3 参数无效，已跳过")
115 .arg(symbol.name)
116 .arg(pathIndex)
117 .arg(segmentIndex));
118 ++segmentIndex;
119 continue;
120 }
121 if (segment.type == IR::SymbolPathSegmentIR::Type::Line) {
122 AltiumSchPath path; // 转换后的 Altium 原生线段记录。
123 path.lineWidth = AltiumCoord::lineWidthMmToIndex(pathSource.strokeWidth);
124 path.lineStyle = AltiumSymbolConversionUtils::toAltiumLineStyle(pathSource.strokeStyle);
125 path.color = AltiumSymbolConversionUtils::toAltiumColor(pathSource.strokeColor);
126 path.ownerPartId = AltiumSymbolConversionUtils::toAltiumOwnerPartId(pathSource.partIndex);
127 path.sourceGraphicType = QStringLiteral("PT");
128 path.sourceGraphicIndex = sourceIndexForPart(symbol.paths, pathIndex, pathSource.partIndex);
129 path.sourceSegmentIndex = segmentIndex;
130 path.sourcePartIndex = pathSource.partIndex;
131 path.vertices.append(QPointF(AltiumCoord::mmToSchematicUnits(segment.start.x()),
132 AltiumCoord::mmToSchematicUnits(segment.start.y())));
133 path.vertices.append(QPointF(AltiumCoord::mmToSchematicUnits(segment.end.x()),
134 AltiumCoord::mmToSchematicUnits(segment.end.y())));
135 component.paths.append(path);
136 } else if (segment.type == IR::SymbolPathSegmentIR::Type::QuadraticBezier ||
137 segment.type == IR::SymbolPathSegmentIR::Type::CubicBezier) {
138 IR::SymbolBezierIR bezier; // 统一 IR 中间 Bézier 曲线。
139 if (segment.type == IR::SymbolPathSegmentIR::Type::QuadraticBezier) {
140 // Altium SchLib 仅提供三次 Bézier 记录；二次曲线可用
141 // C1=P0+2/3(Q-P0), C2=P1+2/3(Q-P1) 精确表示。
142 const QPointF control1 =
143 segment.start + (segment.control1 - segment.start) * (2.0 / 3.0); // 三次曲线第一控制点。
144 const QPointF control2 =
145 segment.end + (segment.control1 - segment.end) * (2.0 / 3.0); // 三次曲线第二控制点。
146 bezier.controlPoints = {segment.start, control1, control2, segment.end};
147 } else {
148 bezier.controlPoints = {segment.start, segment.control1, segment.control2, segment.end};
149 }
150 bezier.strokeColor = pathSource.strokeColor;
151 bezier.strokeWidth = pathSource.strokeWidth;
152 bezier.strokeStyle = pathSource.strokeStyle;
153 bezier.partIndex = pathSource.partIndex;
154 AltiumSchBezier altiumBezier =
155 AltiumSymbolCurveConverter::convertBezier(bezier); // 转换后的 Altium Bézier 记录。
156 altiumBezier.sourceGraphicType = QStringLiteral("PT");
157 altiumBezier.sourceGraphicIndex = sourceIndexForPart(symbol.paths, pathIndex, pathSource.partIndex);
158 altiumBezier.sourceSegmentIndex = segmentIndex;
159 altiumBezier.sourcePartIndex = pathSource.partIndex;
160 component.beziers.append(altiumBezier);
161 } else if (segment.type == IR::SymbolPathSegmentIR::Type::EllipticalArc) {
162 IR::SymbolEllipticalArcIR ellipseArc; // 统一 IR 中间椭圆弧。
163 ellipseArc.center = segment.arcCenter;
164 ellipseArc.radiusX = segment.radiusX;
165 ellipseArc.radiusY = segment.radiusY;
166 ellipseArc.startAngle = segment.arcStartAngle;
167 ellipseArc.endAngle = segment.arcEndAngle;
168 ellipseArc.strokeColor = pathSource.strokeColor;
169 ellipseArc.strokeWidth = pathSource.strokeWidth;
170 ellipseArc.strokeStyle = pathSource.strokeStyle;
171 ellipseArc.partIndex = pathSource.partIndex;
172 AltiumSchEllipticalArc altiumArc =
173 AltiumSymbolCurveConverter::convertEllipticalArc(ellipseArc); // 转换后的椭圆弧记录。
174 altiumArc.sourceGraphicType = QStringLiteral("PT");
175 altiumArc.sourceGraphicIndex = sourceIndexForPart(symbol.paths, pathIndex, pathSource.partIndex);
176 altiumArc.sourceSegmentIndex = segmentIndex;
177 altiumArc.sourcePartIndex = pathSource.partIndex;
178 if (altiumArc.radiusX <= 0 || altiumArc.radiusY <= 0) {
179 diagnostics.append(QStringLiteral("符号 %1 路径图元 %2 的段 %3 椭圆弧半径量化后无效，已跳过")
180 .arg(symbol.name)
181 .arg(pathIndex)
182 .arg(segmentIndex));
183 ++segmentIndex;
184 continue;
185 }
186 component.ellipticalArcs.append(altiumArc);
187 } else {
188 IR::SymbolArcIR arc; // 统一 IR 中间圆弧。
189 arc.startPoint = segment.start;
190 arc.midPoint = segment.arcMid;
191 arc.endPoint = segment.end;
192 arc.strokeColor = pathSource.strokeColor;
193 arc.strokeWidth = pathSource.strokeWidth;
194 arc.strokeStyle = pathSource.strokeStyle;
195 arc.partIndex = pathSource.partIndex;
196 AltiumSchArc altiumArc = AltiumSymbolCurveConverter::convertArc(arc); // 转换后的圆弧记录。
197 if (altiumArc.radius <= 0) {
198 diagnostics.append(QStringLiteral("符号 %1 路径图元 %2 的段 %3 圆弧半径量化后无效，已跳过")
199 .arg(symbol.name)
200 .arg(pathIndex)
201 .arg(segmentIndex));
202 ++segmentIndex;
203 continue;
204 }
205 altiumArc.sourceGraphicType = QStringLiteral("PT");
206 altiumArc.sourceGraphicIndex = sourceIndexForPart(symbol.paths, pathIndex, pathSource.partIndex);
207 altiumArc.sourceSegmentIndex = segmentIndex;
208 altiumArc.sourcePartIndex = pathSource.partIndex;
209 component.arcs.append(altiumArc);
210 }
211 ++segmentIndex;
212 }
213 } else {
214 const int minimumPointCount = pathSource.isFilled ? 3 : 2; // 填充路径和开放路径所需的最小点数。
215 if (!hasFinitePoints(pathSource.points, minimumPointCount)) {
216 diagnostics.append(
217 QStringLiteral("符号 %1 路径图元 %2 的点列无效，已跳过").arg(symbol.name).arg(pathIndex));
218 continue;
219 }
220 AltiumSchPath path = AltiumSymbolCurveConverter::convertPath(pathSource); // 转换后的路径记录。
221 path.sourceGraphicType = QStringLiteral("PT");
222 path.sourceGraphicIndex = sourceIndexForPart(symbol.paths, pathIndex, pathSource.partIndex);
223 path.sourcePartIndex = pathSource.partIndex;
224 component.paths.append(path);
225 }
226 }
227}
