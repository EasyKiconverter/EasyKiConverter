101 // 将后端返回的图片数据转换为 QML 图片源。
102 function previewImageSource(imageData) {
103 if (!imageData || imageData === "")
104 return "";
105 if (imageData.indexOf("data:image/") === 0)
106 return imageData;
107 if (imageData.indexOf("/9j/") === 0)
108 return "data:image/jpeg;base64," + imageData;
109 if (imageData.indexOf("iVBOR") === 0)
110 return "data:image/png;base64," + imageData;
111 if (imageData.indexOf("R0lG") === 0)
112 return "data:image/gif;base64," + imageData;
113 if (imageData.indexOf("UklGR") === 0)
114 return "data:image/webp;base64," + imageData;
115 return "data:image/png;base64," + imageData;
116 }
117 // 收集当前元件可显示的预览图源。
118 function previewImageSources() {
119 var result = [];
120 if (!itemData || !itemData.previewImages)
121 return result;
122 for (var i = 0; i < itemData.previewImages.length && result.length < 3; ++i) {
123 var source = previewImageSource(itemData.previewImages[i]);
124 if (source !== "")
125 result.push(source);
126 }
127 return result;
128 }

134 // 在搜索文本变化后更新高亮正则缓存。
135 function updateCachedRegex() {
136 if (searchText !== cachedSearchText) {
137 cachedSearchText = searchText;
138 if (!searchText || searchText.trim() === "") {
139 cachedRegex = null;
140 } else {
141 var escaped = searchText.trim().replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
142 cachedRegex = new RegExp("(" + escaped + ")", "gi");
143 }
144 }
