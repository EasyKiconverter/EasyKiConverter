25 const QList<AltiumSchComponent>& components) {

27 params["HEADER"] = "Protel for Windows - Schematic Library Editor Binary File Version 5.0";
28 int totalWeight = 0; // 所有组件 Data 流的记录权重总和。

44 const int index = i + 1; // Altium 字体表使用的 1-based 索引。
45 params[QString("FontName%1").arg(index)] = fonts[i].name;

94 QMap<QString, QString> params; // SectionKeys 参数字段。
95 int keyCount = 0; // 需要记录的名称与存储键映射数量。
96 for (int i = 0; i < components.size(); ++i) {
97 const QString& sectionKey = sectionKeys[i]; // 当前组件的 OLE 存储键。
98 if (sectionKey != components[i].name) {
99 params[QString("LibRef%1").arg(keyCount)] = components[i].name;
100 params[QString("SectionKey%1").arg(keyCount)] = sectionKey;
101 ++keyCount;
102 }
103 }
104
105 if (keyCount <= 0) {
106 return;
107 }
108 params["KeyCount"] = QString::number(keyCount);
109 QByteArray data; // SectionKeys 流缓冲区。
110 AltiumBinaryWriter writer(data); // SectionKeys 二进制写入器。
111 writer.writeCStringParameterBlockUtf8(params);
112 ole.writeStream("SectionKeys", data);
113}
