121 if (diag) {

124 diag->errorString = errorString;

130 diag->retryCount = result.retryCount;
131 diag->latencyMs = timer.elapsed();
132 diag->wasRateLimited = result.diagnostic.wasRateLimited;
133 diag->hasRateLimitHint = result.diagnostic.hasRateLimitHint;
134 }
135
136 if (errorString.isEmpty() && !data.isEmpty()) {
137 m_owner.savePreviewImage(componentId, data, imageIndex, generation);
138 } else if (!errorString.isEmpty() && errorString != QStringLiteral("Cancelled")) {
139 LOG_WARN(LogModule::Core, "Preview image download failed for {}: {}", componentId, errorString);
140 }
