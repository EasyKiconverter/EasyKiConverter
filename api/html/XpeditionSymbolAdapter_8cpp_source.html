27IR::PinDirection directionFor(const QPointF& start, const QPointF& end) {
28 const QPointF delta = end - start; // 引脚方向向量。
29 if (qAbs(delta.x()) >= qAbs(delta.y()))
30 return delta.x() < 0.0 ? IR::PinDirection::Right : IR::PinDirection::Left;
31 return delta.y() < 0.0 ? IR::PinDirection::Up : IR::PinDirection::Down;
32}

39IR::PinElectricalType electricalTypeFor(const QString& value) {
40 const QString type = value.trimmed().toUpper(); // 规范化后的类型文本。
41 if (type == QStringLiteral("IN"))
42 return IR::PinElectricalType::Input;
43 if (type == QStringLiteral("OUT"))
44 return IR::PinElectricalType::Output;
45 if (type == QStringLiteral("BI"))
46 return IR::PinElectricalType::Bidirectional;
47 if (type == QStringLiteral("PASSIVE"))
48 return IR::PinElectricalType::Passive;
49 if (type == QStringLiteral("PWR"))
50 return IR::PinElectricalType::Power;
51 if (type == QStringLiteral("OCL"))
52 return IR::PinElectricalType::OpenCollector;
53 if (type == QStringLiteral("OEM"))
54 return IR::PinElectricalType::OpenEmitter;
55 return IR::PinElectricalType::Unspecified;
56}

66IR::SymbolComponentIR XpeditionSymbolAdapter::toIR(const Parser::XpeditionSymbolDocument& document,
67 Parser::ParseDiagnostics* diagnostics) {
68 IR::SymbolComponentIR result; // 输出符号组件 IR。
69 const Parser::XpeditionSymbolModel& source = document.model; // Xpedition 符号模型。
70 result.name = source.name;
71 result.partCount = qMax(1, source.partCount);
72 result.footprintName = source.footprint;
73 result.footprintNames = source.footprint.isEmpty() ? QStringList() : QStringList{source.footprint};
74 result.designatorPrefix = source.properties.value(QStringLiteral("REFDES"));
75 result.description = source.properties.value(QStringLiteral("DESC"));
76 result.sourceMetadata.insert(QStringLiteral("xpeditionVersion"), QString::number(source.version));
77 for (auto iterator = source.properties.cbegin(); iterator != source.properties.cend(); ++iterator)
78 result.sourceMetadata.insert(iterator.key(), iterator.value());
79
80 for (const Parser::XpeditionSymbolPin& sourcePin : source.pins) {
81 IR::SymbolPinIR pin; // 当前引脚 IR。
82 pin.designator = sourcePin.numbers.join(QStringLiteral(","));
83 pin.name = sourcePin.name;
84 pin.position = toMillimeters(sourcePin.start);
85 pin.length = QLineF(sourcePin.start, sourcePin.end).length() * kThousandthInchMm;
86 pin.direction = directionFor(sourcePin.start, sourcePin.end);
87 pin.electricalType = electricalTypeFor(sourcePin.pinType);
88 pin.style.inverted = sourcePin.inverted;
89 pin.hasNamePosition = !sourcePin.name.isEmpty();
90 pin.namePosition = toMillimeters(sourcePin.namePosition);
91 pin.nameFontSizeMm = sourcePin.nameSize * kThousandthInchMm;
92 pin.nameRotation = sourcePin.nameRotation;
93 pin.display.showName = sourcePin.nameVisible;
94 pin.display.showDesignator = sourcePin.numbersVisible;
95 pin.partIndex = sourcePin.partIndex;
96 pin.commonToAllParts = sourcePin.partIndex == 0;
97 result.pins.append(pin);
98 }
99
100 for (const Parser::XpeditionSymbolRectangle& sourceRectangle : source.rectangles) {
101 IR::SymbolRectangleIR rectangle; // 当前矩形 IR。
102 const QPointF start = toMillimeters(sourceRectangle.start); // 矩形起点。
103 const QPointF end = toMillimeters(sourceRectangle.end); // 矩形终点。
104 rectangle.x0 = start.x();
105 rectangle.y0 = start.y();
106 rectangle.x1 = end.x();
107 rectangle.y1 = end.y();
108 rectangle.partIndex = sourceRectangle.partIndex;
109 result.rectangles.append(rectangle);
110 }
111 for (const Parser::XpeditionSymbolCircle& sourceCircle : source.circles) {
112 IR::SymbolCircleIR circle; // 当前圆形 IR。
113 circle.center = toMillimeters(sourceCircle.center);
114 circle.radius = sourceCircle.radius * kThousandthInchMm;
115 circle.partIndex = sourceCircle.partIndex;
116 result.circles.append(circle);
117 }
118 for (const Parser::XpeditionSymbolArc& sourceArc : source.arcs) {
119 IR::SymbolArcIR arc; // 当前圆弧 IR。
120 arc.startPoint = toMillimeters(sourceArc.start);
121 arc.midPoint = toMillimeters(sourceArc.center);
122 arc.endPoint = toMillimeters(sourceArc.end);
123 arc.partIndex = sourceArc.partIndex;
124 result.arcs.append(arc);
125 }
126 for (const Parser::XpeditionSymbolPolyline& sourcePolyline : source.polylines) {
127 QList<QPointF> points; // 折线的毫米坐标点。
128 for (const QPointF& sourcePoint : sourcePolyline.points)
129 points.append(toMillimeters(sourcePoint));
130 if (sourcePolyline.closed) {
131 IR::SymbolPolygonIR polygon; // 闭合折线对应的多边形 IR。
132 polygon.points = points;
133 polygon.partIndex = sourcePolyline.partIndex;
134 result.polygons.append(polygon);
135 } else {
136 IR::SymbolPolylineIR polyline; // 非闭合折线 IR。
137 polyline.points = points;
138 polyline.partIndex = sourcePolyline.partIndex;
139 result.polylines.append(polyline);
140 }
141 }
142 for (const Parser::XpeditionSymbolText& sourceText : source.texts) {
143 IR::SymbolTextIR text; // 当前文本 IR。
144 text.text = sourceText.text;
145 text.position = toMillimeters(sourceText.position);
146 text.fontSizeMm = sourceText.size * kThousandthInchMm;
147 text.rotation = sourceText.rotation;
148 text.partIndex = sourceText.partIndex;
149 result.texts.append(text);
150 }
151
152 if (source.symbolType == 0 && diagnostics)
153 diagnostics->add(Parser::ParseSeverity::Warning,
154 Parser::ParseScope::Symbol,
155 QStringLiteral("Xpedition 符号缺少类型字段"),
156 result.name);
157 return result;
158}
