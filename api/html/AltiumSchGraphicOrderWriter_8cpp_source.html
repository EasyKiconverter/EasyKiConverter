24void AltiumSchGraphicOrderWriter::write(AltiumBinaryWriter& writer,
25 const AltiumSchComponent& component,
26 const AltiumSchGraphicOrder& order) {
27 const auto matchesPart = [&order](int sourcePartIndex) { return sourcePartIndex == order.partIndex; };
28 if (order.type == QStringLiteral("P")) {
29 int localIndex = 0; // 当前部件范围内的引脚局部索引。
30 for (const AltiumSchPin& pin : component.pins) {
31 const bool matchesSourcePart =
32 order.partIndex < 0 ? pin.ownerPartId == -1 : pin.sourcePartIndex == order.partIndex;
33 if (!matchesSourcePart)
34 continue;
35 if (localIndex == order.index) {
36 m_owner.writePinRecord(writer, pin);
37 return;
38 }
39 ++localIndex;
40 }
41 return;
42 }
43 if (order.type == QStringLiteral("R")) {
44 for (const AltiumSchRoundRectangle& rect : component.roundRectangles) {
45 if (rect.sourceGraphicIndex == order.index && matchesPart(rect.sourcePartIndex)) {
46 m_owner.writeRoundRectangleRecord(writer, rect);
47 return;
48 }
49 }
50 for (const AltiumSchRectangle& rect : component.rectangles) {
51 if (rect.sourceGraphicIndex == order.index && matchesPart(rect.sourcePartIndex)) {
52 m_owner.writeRectangleRecord(writer, rect);
53 return;
54 }
55 }
56 return;
57 }
58 if (order.type == QStringLiteral("C") || order.type == QStringLiteral("E")) {
59 for (const AltiumSchEllipse& ellipse : component.ellipses) {
60 if (ellipse.sourceGraphicType == order.type && ellipse.sourceGraphicIndex == order.index &&
61 matchesPart(ellipse.sourcePartIndex)) {
62 m_owner.writeEllipseRecord(writer, ellipse);
63 return;
64 }
65 }
66 return;
67 }
68 if (order.type == QStringLiteral("A")) {
69 for (const AltiumSchArc& arc : component.arcs) {
70 if (arc.sourceGraphicType == order.type && arc.sourceGraphicIndex == order.index &&
71 matchesPart(arc.sourcePartIndex)) {
72 m_owner.writeArcRecord(writer, arc);
73 return;
74 }
75 }
76 return;
77 }
78 if (order.type == QStringLiteral("PL")) {
79 for (const AltiumSchPolyline& polyline : component.polylines) {
80 if (polyline.sourceGraphicIndex == order.index && matchesPart(polyline.sourcePartIndex)) {
81 m_owner.writePolylineRecord(writer, polyline);
82 return;
83 }
84 }
85 return;
86 }
87 if (order.type == QStringLiteral("PG")) {
88 for (const AltiumSchPolygon& polygon : component.polygons) {
89 if (polygon.sourceGraphicIndex == order.index && matchesPart(polygon.sourcePartIndex)) {
90 m_owner.writePolygonRecord(writer, polygon);
91 return;
92 }
93 }
94 return;
95 }
96 if (order.type == QStringLiteral("T")) {
97 for (const AltiumSchText& text : component.texts) {
98 if (!text.isPinLabel && text.sourceGraphicIndex == order.index && matchesPart(text.sourcePartIndex)) {
99 m_owner.writeTextRecord(writer, text);
100 return;
101 }
102 }
103 return;
104 }
105 if (order.type == QStringLiteral("I")) {
106 for (const AltiumSchImage& image : component.images) {
107 if (image.sourceGraphicIndex == order.index && matchesPart(image.sourcePartIndex)) {
108 m_owner.writeImageRecord(writer, image);
109 return;
110 }
111 }
112 return;
113 }
114 if (order.type != QStringLiteral("PT"))
115 return;
116
117 for (const AltiumSchPath& path : component.paths) {
118 if (path.sourceGraphicType == order.type && path.sourceGraphicIndex == order.index &&
119 matchesPart(path.sourcePartIndex) && path.sourceSegmentIndex < 0) {
120 m_owner.writePathRecord(writer, path);
121 return;
122 }
123 }
124
125 QSet<int> segmentIndexSet; // 当前路径中已发现的片段索引集合。
126 const auto collectSegmentIndices = [&order, &matchesPart, &segmentIndexSet](const auto& graphics) {
127 for (const auto& graphic : graphics) {
128 if (graphic.sourceGraphicType == order.type && graphic.sourceGraphicIndex == order.index &&
129 matchesPart(graphic.sourcePartIndex) && graphic.sourceSegmentIndex >= 0) {
130 segmentIndexSet.insert(graphic.sourceSegmentIndex);
131 }
132 }
133 };
134 collectSegmentIndices(component.paths);
135 collectSegmentIndices(component.beziers);
136 collectSegmentIndices(component.arcs);
137 collectSegmentIndices(component.ellipticalArcs);
138
139 QList<int> segmentIndices = segmentIndexSet.values(); // 按序写入的路径片段索引列表。
140 std::sort(segmentIndices.begin(), segmentIndices.end());
141 for (const int segmentIndex : segmentIndices) {
142 for (const AltiumSchPath& path : component.paths) {
143 if (path.sourceGraphicType == order.type && path.sourceGraphicIndex == order.index &&
144 matchesPart(path.sourcePartIndex) && path.sourceSegmentIndex == segmentIndex) {
145 m_owner.writePathRecord(writer, path);
146 break;
147 }
148 }
149 for (const AltiumSchBezier& bezier : component.beziers) {
150 if (bezier.sourceGraphicType == order.type && bezier.sourceGraphicIndex == order.index &&
151 matchesPart(bezier.sourcePartIndex) && bezier.sourceSegmentIndex == segmentIndex) {
152 m_owner.writeBezierRecord(writer, bezier);
153 break;
154 }
155 }
156 for (const AltiumSchArc& arc : component.arcs) {
157 if (arc.sourceGraphicType == order.type && arc.sourceGraphicIndex == order.index &&
158 matchesPart(arc.sourcePartIndex) && arc.sourceSegmentIndex == segmentIndex) {
159 m_owner.writeArcRecord(writer, arc);
160 break;
161 }
162 }
163 for (const AltiumSchEllipticalArc& arc : component.ellipticalArcs) {
164 if (arc.sourceGraphicType == order.type && arc.sourceGraphicIndex == order.index &&
165 matchesPart(arc.sourcePartIndex) && arc.sourceSegmentIndex == segmentIndex) {
166 m_owner.writeEllipticalArcRecord(writer, arc);
167 break;
168 }
169 }
170 }
171}
