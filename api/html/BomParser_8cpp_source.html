21QStringList splitCsvLine(const QString& line) {
22 QStringList cells; // 拆分后的单元格列表。
23 QString cell; // 当前正在构造的单元格。
24 bool inQuotes = false; // 当前是否位于引号包围区域。
25
26 for (int i = 0; i < line.size(); ++i) {
27 const QChar character = line.at(i); // 当前 CSV 字符。
28 if (character == QChar('"')) {
29 if (inQuotes && i + 1 < line.size() && line.at(i + 1) == QChar('"')) {
30 cell += QChar('"');
31 ++i;
32 } else {
33 inQuotes = !inQuotes;
34 }
35 } else if (character == QChar(',') && !inQuotes) {
36 cells.append(cell);
37 cell.clear();
38 } else {
39 cell += character;
40 }
41 }
42 cells.append(cell);
43 return cells;
44}

87bool BomParser::validateId(const QString& componentId) {
88 // LCSC 元件ID格式：以 'C' 或 'c' 开头，后面跟至少4位数字
89 static const QRegularExpression re("^[Cc]\\d{4,}$"); // LCSC 编号格式。
90 bool matches = re.match(componentId).hasMatch(); // 格式匹配结果。
91
92 if (matches) {
93 return !getExcludedIds().contains(componentId.toUpper());
94 }
95 return false;
96}

103QStringList BomParser::extractIdsFromText(const QString& text) {
104 QStringList extractedIds; // 提取结果列表。
105 static const QRegularExpression re(QStringLiteral("[Cc]\\d{4,}")); // 编号匹配表达式。
106 QRegularExpressionMatchIterator it = re.globalMatch(text); // 全局匹配迭代器。
107
108 while (it.hasNext()) {
109 const QString id = it.next().captured().toUpper(); // 当前规范化编号。
110 if (validateId(id) && !extractedIds.contains(id)) {
111 extractedIds.append(id);
112 }
113 }
114 return extractedIds;
115}

122QStringList BomParser::parse(const QString& filePath) {
123 qDebug() << "BomParser: Parsing file:" << filePath;
124
125 QStringList componentIds; // 解析得到的元件编号。
126 QFileInfo fileInfo(filePath); // 输入文件信息。
127
128 if (!fileInfo.exists()) {
129 qWarning() << "BomParser: File does not exist:" << filePath;
130 return componentIds;
131 }
132
133 QString suffix = fileInfo.suffix().toLower(); // 小写文件扩展名。
134
135 if (suffix == "csv" || suffix == "txt") {
136 componentIds = parseCsv(filePath);
137 } else if (suffix == "xlsx" || suffix == "xls") {
138 componentIds = parseExcel(filePath);
139 } else {
140 qWarning() << "BomParser: Unsupported file format:" << suffix;
141 }
142
143 qDebug() << "BomParser: Extracted" << componentIds.size() << "IDs";
144 return componentIds;

152QStringList BomParser::parseCsv(const QString& filePath) {
153 QStringList componentIds; // 解析得到的元件编号。
154 QFile file(filePath); // 输入文本文件。
155
156 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
157 qWarning() << "BomParser: Failed to open CSV:" << file.errorString();
158 return componentIds;
159 }
160
161 QTextStream in(&file); // UTF-8 文本流。
162 in.setEncoding(QStringConverter::Utf8);
163
164 QString record; // 当前可能跨行的 CSV 记录。
165 while (!in.atEnd()) {
166 QString line = in.readLine(); // 当前输入行。
167 if (!record.isEmpty())
168 record.append('\n');
169 record.append(line);
170
171 if (csvRecordHasOpenQuote(record))
172 continue;
173
174 const QStringList cells = splitCsvLine(record); // 当前记录的单元格。
175 for (const QString& cell : cells) {
176 processCellText(cell, componentIds);
177 }
178 record.clear();
179 }
180
181 if (!record.isEmpty()) {
182 const QStringList cells = splitCsvLine(record); // 文件尾部记录的单元格。
183 for (const QString& cell : cells)
184 processCellText(cell, componentIds);
185 }
186
187 file.close();
188 return componentIds;
189}

196QStringList BomParser::parseExcel(const QString& filePath) {
197 QStringList componentIds; // 解析得到的元件编号。
198 QXlsx::Document xlsx(filePath); // Excel 文档对象。
199
200 if (!xlsx.load()) {
201 qWarning() << "BomParser: Failed to load Excel file";
202 return componentIds;
203 }
204
205 for (const QString& sheetName : xlsx.sheetNames()) { // 遍历工作表。
206 xlsx.selectSheet(sheetName);
207 QXlsx::CellRange range = xlsx.dimension(); // 当前工作表有效单元格范围。
208
209 for (int row = range.firstRow(); row <= range.lastRow(); ++row) {
210 for (int col = range.firstColumn(); col <= range.lastColumn(); ++col) {
211 if (auto cell = xlsx.cellAt(row, col)) {
212 processCellText(cell->readValue().toString(), componentIds);
213 }
214 }
215 }
216 }
217
218 return componentIds;
219}

226void BomParser::processCellText(const QString& text, QStringList& result) {
227 QString trimmedCell = text.trimmed(); // 去除空白后的单元格文本。
228
229 // 去除可能的引号
230 if (trimmedCell.startsWith('"') && trimmedCell.endsWith('"')) {
231 trimmedCell = trimmedCell.mid(1, trimmedCell.length() - 2);
232 }
233
234 if (validateId(trimmedCell)) {
235 QString id = trimmedCell.toUpper(); // 规范化后的元件编号。
236 if (!result.contains(id)) {
237 result.append(id);
238 }
239 }
240}
