19QString FootprintGraphicsGenerator::generatePad(const IR::FootprintPadIR& pad, double originX, double originY) const {
20 QString content; // 当前焊盘序列化结果。
21
22 // 坐标已为 mm，计算相对位置
23 double x = roundTo2(pad.position.x() - originX); // 焊盘相对原点的 X 坐标。
24 double y = roundTo2(pad.position.y() - originY); // 焊盘相对原点的 Y 坐标。
25 double width = roundTo2(pad.size.width()); // 焊盘宽度。
26 double height = roundTo2(pad.size.height()); // 焊盘高度。
27 double holeRadius = roundTo2(pad.holeSize / 2.0); // 通孔半径。
28
29 QString padNumber = pad.number; // 经过括号清理后的焊盘编号。
30 if (padNumber.contains("(") && padNumber.contains(")")) {
31 int start = padNumber.indexOf("("); // 编号中左括号的位置。
32 int end = padNumber.indexOf(")"); // 编号中右括号的位置。
33 padNumber = padNumber.mid(start + 1, end - start - 1);
34 }
35
36 double rotation = pad.rotation; // 焊盘旋转角度。
37 if (rotation > 180.0) {
38 rotation = rotation - 360.0;
39 }
40
41 // 使用 IR 映射获取 KiCad 焊盘形状
42 bool isCustomShape = (pad.shape == IR::PadShape::Polygon); // 是否使用自定义多边形焊盘。
43 QString polygonStr; // 自定义多边形的 KiCad 文本。
44
45 if (isCustomShape) {
46 if (pad.customShapePoints.isEmpty()) {
47 qWarning() << "Pad" << pad.number << "is a custom polygon, but has no points defined";
48 } else {
49 width = 0.005;
50 height = 0.005;
51 rotation = 0.0;
52
53 QString path; // 多边形顶点序列化文本。
54 // customShapePoints 已经是相对于焊盘中心的坐标（在 FootprintDataConverter 中计算）
55 for (const QPointF& pt : pad.customShapePoints) {
56 double relX = roundTo2(pt.x()); // 顶点相对焊盘中心的 X 坐标。
57 double relY = roundTo2(pt.y()); // 顶点相对焊盘中心的 Y 坐标。
58 path += QString("(xy %1 %2) ").arg(relX, 0, 'f', 2).arg(relY, 0, 'f', 2);
59 }
60
61 polygonStr = QString(
62 "\n (primitives\n (gr_poly\n (pts %1)\n (width 0.1)\n (fill "
63 "yes)\n )\n )\n ")
64 .arg(path);
65 }
66 } else {
67 width = qMax(width, 0.01);
68 height = qMax(height, 0.01);
69 }
70
71 QString kicadShape = KiCadTypeMap::toKicadPadShape(pad.shape); // KiCad 焊盘形状名称。
72
73 QString kicadType = padTypeToKicad(pad); // KiCad 焊盘电气类型。
74 QString layers = padLayersToKicad(pad.layer); // KiCad 焊盘层列表。
75
76 QString drillStr; // 钻孔描述文本。
77 if (pad.isThroughHole()) {
78 double holeRadiusMm = roundTo2(pad.holeSize / 2.0); // 毫米单位的钻孔半径。
79 if (holeRadiusMm > 0 && pad.holeLength > 0) {
80 double holeLengthMm = roundTo2(pad.holeLength); // 槽孔长度。
81 double maxDistanceHole = qMax(holeRadiusMm * 2, holeLengthMm); // 槽孔最大外形尺寸。
82 double pos0 = height - maxDistanceHole; // 槽孔沿水平轴放置时的剩余间距。
83 double pos90 = width - maxDistanceHole; // 槽孔沿垂直轴放置时的剩余间距。
84 double maxDistance = qMax(pos0, pos90); // 用于判断槽孔主轴方向的间距。
85
86 if (qAbs(maxDistance - pos0) < qAbs(maxDistance - pos90)) {
87 drillStr = QString(" (drill oval %1 %2)").arg(holeRadiusMm * 2, 0, 'f', 2).arg(holeLengthMm, 0, 'f', 2);
88 } else {
89 drillStr = QString(" (drill oval %1 %2)").arg(holeLengthMm, 0, 'f', 2).arg(holeRadiusMm * 2, 0, 'f', 2);
90 }
91 } else if (holeRadiusMm > 0) {
92 double drillDiameter = holeRadiusMm * 2; // 圆孔直径。
93 drillStr = QString(" (drill %1)").arg(drillDiameter, 0, 'f', 2);
94 }
95 }
96
97 content += QString(" (pad %1 %2 %3 (at %4 %5 %6) (size %7 %8) (layers %9)%10%11)\n")
98 .arg(padNumber)
99 .arg(kicadType)
100 .arg(kicadShape)
101 .arg(x, 0, 'f', 2)
102 .arg(y, 0, 'f', 2)
103 .arg(rotation, 0, 'f', 2)
104 .arg(width, 0, 'f', 2)
105 .arg(height, 0, 'f', 2)
106 .arg(layers)
107 .arg(drillStr)
108 .arg(polygonStr);
109
110 return content;
111}

249QString FootprintGraphicsGenerator::generateArc(const IR::FootprintArcIR& arc, double originX, double originY) const {
250 QString content; // 当前圆弧序列化结果。
251
252 // IR 弧线已解析为 center/radius/startAngle/endAngle
253 if (arc.radius > 0) {
254 double cx = roundTo2(arc.center.x() - originX); // 圆心相对原点的 X 坐标。
255 double cy = roundTo2(arc.center.y() - originY); // 圆心相对原点的 Y 坐标。
256
257 double startAngle = arc.startAngle; // 当前分段的起始角度。
258 double sweepAngle = arc.endAngle - arc.startAngle; // 尚未序列化的扫掠角度。
259 if (sweepAngle < 0) {
260 startAngle += sweepAngle;
261 sweepAngle = -sweepAngle;
262 }
263
264 const QString layer = layerTypeToKicad(arc.layer); // 圆弧目标图层名称。
265 const double strokeWidth = roundTo2(arc.width); // 圆弧线宽。
266 while (sweepAngle > 0.1) {
267 const double step = qMin(180.0, sweepAngle); // 当前三点弧线分段角度。
268 const double midAngle = startAngle + step / 2.0; // 当前分段中点角度。
269 const double endAngle = startAngle + step; // 当前分段终点角度。
270 const double startRad = startAngle * M_PI / 180.0; // 起始角度对应的弧度。
271 const double midRad = midAngle * M_PI / 180.0; // 中点角度对应的弧度。
272 const double endRad = endAngle * M_PI / 180.0; // 终点角度对应的弧度。
273 const double startX =
274 roundTo2(arc.center.x() + arc.radius * std::cos(startRad) - originX); // 起点 X 坐标。
275 const double startY =
276 roundTo2(arc.center.y() + arc.radius * std::sin(startRad) - originY); // 起点 Y 坐标。
277 const double midX = roundTo2(arc.center.x() + arc.radius * std::cos(midRad) - originX); // 中点 X 坐标。
278 const double midY = roundTo2(arc.center.y() + arc.radius * std::sin(midRad) - originY); // 中点 Y 坐标。
279 const double endX = roundTo2(arc.center.x() + arc.radius * std::cos(endRad) - originX); // 终点 X 坐标。
280 const double endY = roundTo2(arc.center.y() + arc.radius * std::sin(endRad) - originY); // 终点 Y 坐标。
281
282 // KiCad 8/9 的封装弧线使用三点语法，不再接受旧的 angle 字段。
283 content +=
284 QString(
285 " (fp_arc (start %1 %2) (mid %3 %4) (end %5 %6) (stroke (width %7) (type default)) (layer %8))\n")
286 .arg(startX, 0, 'f', 2)
287 .arg(startY, 0, 'f', 2)
288 .arg(midX, 0, 'f', 2)
289 .arg(midY, 0, 'f', 2)
290 .arg(endX, 0, 'f', 2)
291 .arg(endY, 0, 'f', 2)
292 .arg(strokeWidth, 0, 'f', 2)
293 .arg(layer);
294
295 sweepAngle -= step;
296 startAngle = endAngle;
297 }
298 } else {
299 qWarning() << "Warning: Arc has zero radius, skipping";
300 }
301
302 return content;
303}

312QString FootprintGraphicsGenerator::generateText(const IR::FootprintTextIR& text,
313 double originX,
314 double originY) const {
315 QString content; // 当前文本或轮廓序列化结果。
316 double x = roundTo2(text.position.x() - originX); // 文本相对原点的 X 坐标。
317 double y = roundTo2(text.position.y() - originY); // 文本相对原点的 Y 坐标。
318
319 QString layer = layerTypeToKicad(text.layer); // 文本目标图层名称。
320
321 // 装配层文本映射
322 if (text.isFabrication) {
323 layer = layer.replace(".SilkS", ".Fab");
324 }
325
326 QString displayStr = text.isDisplayed ? "" : " hide"; // 文本隐藏标记。
327
328 bool isNonASCII = false; // 是否包含需要转为轮廓的非 ASCII 字符。
329 for (int i = 0; i < text.text.length(); ++i) {
330 if (text.text[i].unicode() > 127) {
331 isNonASCII = true;
332 break;
333 }
334 }
335 if (isNonASCII && !text.textPathPoints.isEmpty()) {
336 qWarning() << "Warning: Converting non-ASCII text to polygon:" << text.text;
337
338 // IR 中已解析为 QList<QPointF>
339 const QList<QPointF>& points = text.textPathPoints; // 文本轮廓顶点。
340
341 if (points.size() >= 2) {
342 // 计算相对坐标
343 QList<QPointF> relPoints; // 相对封装原点的轮廓顶点。
344 for (const QPointF& pt : points) {
345 relPoints.append(QPointF(roundTo2(pt.x() - originX), roundTo2(pt.y() - originY)));
346 }
347
348 for (int i = 1; i < relPoints.size(); ++i) {
349 content += QString(" (fp_line (start %1 %2) (end %3 %4) (layer %5) (width %6))\n")
350 .arg(relPoints[i - 1].x(), 0, 'f', 2)
351 .arg(relPoints[i - 1].y(), 0, 'f', 2)
352 .arg(relPoints[i].x(), 0, 'f', 2)
353 .arg(relPoints[i].y(), 0, 'f', 2)
354 .arg(layer)
355 .arg(roundTo2(text.strokeWidth) * 0.8, 0, 'f', 2);
356 }
357 }
358 } else {
359 bool isBottomLayer = layer.startsWith("B"); // 是否位于底层。
360 QString mirrorStr = isBottomLayer ? " mirror" : ""; // 底层文本的镜像标记。
361
362 content += QString(" (fp_text user %1 (at %2 %3 %4) (layer %5)%6\n")
363 .arg(text.text)
364 .arg(x, 0, 'f', 2)
365 .arg(y, 0, 'f', 2)
366 .arg(text.rotation, 0, 'f', 2)
367 .arg(layer)
368 .arg(displayStr);
369
370 // IR 中 fontSize 和 strokeWidth 已为 mm
371 double fontSize = roundTo2(text.fontSize); // 文本字体高度。
372 double thickness = roundTo2(text.strokeWidth); // 文本笔画宽度。
373 fontSize = qMax(fontSize, 1.0);
374 thickness = qMax(thickness, 0.01);
375 content += QString(" (effects (font (size %1 %2) (thickness %3)) (justify left%4))\n")
376 .arg(fontSize, 0, 'f', 2)
377 .arg(fontSize, 0, 'f', 2)
378 .arg(thickness, 0, 'f', 2)
379 .arg(mirrorStr);
380 content += " )\n";
381 }
382
383 return content;
384}

393QString FootprintGraphicsGenerator::generateSolidRegion(const IR::FootprintRegionIR& region,

396 QString content; // 当前区域序列化结果。
397
398 QString layer = layerTypeToKicad(region.layer); // 区域目标图层名称。

422 content += " (fp_poly\n";
423 content += " (pts";

475QString FootprintGraphicsGenerator::generateModel3D(const IR::Model3DIR& model3D, const QString& model3DPath) const {
476 QString content; // 当前三维模型节点序列化结果。
477
478 QString finalPath = model3DPath.isEmpty() ? model3D.name() : model3DPath; // 最终写入的模型路径。
479
480 // IR 中 translation 和 rotation 已为 mm/度
481 double z = -model3D.translation().z; // KiCad 坐标系下的 Z 偏移。
482
483 double rotX = fmod(360.0 - model3D.rotation().x, 360.0); // 绕 X 轴的 KiCad 旋转角度。
484 if (rotX < 0.0)
485 rotX += 360.0;
486
487 double rotY = fmod(360.0 - model3D.rotation().y, 360.0); // 绕 Y 轴的 KiCad 旋转角度。
488 if (rotY < 0.0)
489 rotY += 360.0;
490
491 double rotZ = fmod(360.0 - model3D.rotation().z, 360.0); // 绕 Z 轴的 KiCad 旋转角度。
492 if (rotZ < 0.0)
493 rotZ += 360.0;
494
495 // STEP 文件需要额外的 stepOffsetMm 补偿
496 double finalOffsetX = 0.0; // STEP 模型的 X 轴补偿偏移。
497 double finalOffsetY = 0.0; // STEP 模型的 Y 轴补偿偏移。
498
499 if (finalPath.endsWith(QStringLiteral(".step"), Qt::CaseInsensitive) ||
500 finalPath.endsWith(QStringLiteral(".stp"), Qt::CaseInsensitive)) {
501 finalOffsetX = model3D.stepOffsetMm().x;
502 finalOffsetY = model3D.stepOffsetMm().y;
503 z += model3D.stepOffsetMm().z;
504 }
505
506 content += QString(" (model \"%1\"\n").arg(finalPath);
507
508 content += QString(" (offset (xyz %1 %2 %3))\n")
509 .arg(finalOffsetX, 0, 'f', 3)
510 .arg(finalOffsetY, 0, 'f', 3)
511 .arg(z, 0, 'f', 3);
512
513 content += " (scale (xyz 1 1 1))\n";
514
515 content += QString(" (rotate (xyz %1 %2 %3))\n").arg(rotX, 0, 'f', 0).arg(rotY, 0, 'f', 0).arg(rotZ, 0, 'f', 0);
516
517 content += " )\n";
518
519 return content;
520}

536QString FootprintGraphicsGenerator::layerTypeToKicad(IR::LayerType layerType) {
537 // 按 IR 层语义选择 KiCad 的目标图层名称，未知层统一降级到装配层并记录警告。
538 switch (layerType) {
539 case IR::LayerType::TopCopper:
540 return "F.Cu";
541 case IR::LayerType::BottomCopper:
542 return "B.Cu";
543 case IR::LayerType::InnerCopper1:
544 return "In1.Cu";
545 case IR::LayerType::InnerCopper2:
546 return "In2.Cu";
547 case IR::LayerType::InnerCopper3:
548 return "In3.Cu";
549 case IR::LayerType::InnerCopper4:
550 return "In4.Cu";
551 case IR::LayerType::TopSilk:
552 return "F.SilkS";
553 case IR::LayerType::BottomSilk:
554 return "B.SilkS";
555 case IR::LayerType::TopPaste:
556 return "F.Paste";
557 case IR::LayerType::BottomPaste:
558 return "B.Paste";
559 case IR::LayerType::TopMask:
560 return "F.Mask";
561 case IR::LayerType::BottomMask:
562 return "B.Mask";
563 case IR::LayerType::TopOverlay:
564 return "F.SilkS";
565 case IR::LayerType::BottomOverlay:
566 return "B.SilkS";
567 case IR::LayerType::TopAssembly:
568 return "F.Fab";
