34 void fetchPreviewImageAsync(const QString& lcscId,
35 const QString& imageUrl,
36 int imageIndex,
37 QAtomicInt* cancelled,
38 bool weakNetwork,
39 const MediaFetchCallback& onComplete) {
40 if (imageIndex < 0 || imageIndex >= 3) {
41 ComponentExportStatus::NetworkDiagnostics diag; // 参数校验失败诊断。
42 diag.url = imageUrl;
43 diag.errorString = QStringLiteral("Invalid preview image index");
44 onComplete(QByteArray(), diag);
45 return;
46 }
47
48 ComponentCacheService* cache = ComponentCacheService::instance(); // 缓存服务实例。
49 if (!cache) {
50 ComponentExportStatus::NetworkDiagnostics diag; // 缓存服务不可用诊断。
51 diag.url = imageUrl;
52 diag.errorString = QStringLiteral("Cache service unavailable");
53 onComplete(QByteArray(), diag);
54 return;
55 }
56
57 if (cancelled && cancelled->loadRelaxed()) {
58 ComponentExportStatus::NetworkDiagnostics diag; // 启动前取消诊断。
59 diag.url = imageUrl;
60 diag.errorString = QStringLiteral("Cancelled before start");
61 onComplete(QByteArray(), diag);
62 return;
63 }
64
65 const QByteArray cachedImage = cache->loadPreviewImage(lcscId, imageIndex); // 已缓存预览图数据。
66 if (!cachedImage.isEmpty()) {
67 ComponentExportStatus::NetworkDiagnostics diag; // 缓存命中诊断。
68 diag.url = imageUrl;
69 diag.statusCode = 200;
70 diag.retryCount = 0;
71 diag.wasRateLimited = false;
72 onComplete(cachedImage, diag);
73 return;
74 }
75
76 const RetryPolicy policy =
77 RetryPolicy::fromProfile(RequestProfiles::previewImage(), weakNetwork); // 下载重试策略。
78 const uint64_t gen = cache->currentGeneration(); // 当前缓存代次。
79 AsyncNetworkRequest* request = // 异步网络请求对象。
80 NetworkClient::instance().getAsync(QUrl(imageUrl), ResourceType::PreviewImage, policy);
81
82 QObject::connect(request, &AsyncNetworkRequest::finished, request, [=](const NetworkResult& result) {
83 ComponentExportStatus::NetworkDiagnostics diag; // 网络响应诊断。
84 diag.url = imageUrl;
85 diag.statusCode = result.statusCode;
86 diag.retryCount = result.retryCount;
87 diag.wasRateLimited = result.diagnostic.wasRateLimited;
88 diag.responseContentType = result.diagnostic.responseContentType;
89 diag.retryAfter = result.diagnostic.retryAfter;
90 diag.rateLimitRemaining = result.diagnostic.rateLimitRemaining;
91 diag.rateLimitReset = result.diagnostic.rateLimitReset;
92 diag.responseSummary = result.diagnostic.responseSummary;
93 diag.hasRateLimitHint = result.diagnostic.hasRateLimitHint;
94 diag.latencyMs = result.elapsedMs;
95
96 if ((cancelled && cancelled->loadRelaxed()) || result.wasCancelled) {
97 diag.errorString = QStringLiteral("Cancelled");
98 onComplete(QByteArray(), diag);
99 } else if (result.success) {
100 if (ComponentCacheService::isValidPreviewImageData(result.data)) {
101 cache->savePreviewImage(lcscId, result.data, imageIndex, gen);
102 onComplete(result.data, diag);
103 } else {
104 diag.errorString = QStringLiteral("Invalid preview image data");
105 onComplete(QByteArray(), diag);
106 }
107 } else {
108 diag.errorString = result.error;
109 onComplete(QByteArray(), diag);
110 }
111
112 request->deleteLater();
113 });
114 }

142 onComplete(QByteArray(), diag);

147 datasheetUrl.toLower().contains(QStringLiteral(".html")) ? QStringLiteral("html") : QStringLiteral("pdf");
148 const QByteArray cachedDatasheet = cache->loadDatasheet(lcscId); // 已缓存数据手册。

150 ComponentExportStatus::NetworkDiagnostics diag; // 缓存命中诊断。

161 const uint64_t gen = cache->currentGeneration(); // 当前缓存代次。
162 AsyncNetworkRequest* request =
163 NetworkClient::instance().getAsync(QUrl(datasheetUrl), ResourceType::Datasheet, policy);
164
165 QObject::connect(request, &AsyncNetworkRequest::finished, request, [=](const NetworkResult& result) mutable {
166 ComponentExportStatus::NetworkDiagnostics diag; // 网络响应诊断。
167 diag.url = datasheetUrl;
168 diag.statusCode = result.statusCode;
169 diag.retryCount = result.retryCount;
170 diag.wasRateLimited = result.diagnostic.wasRateLimited;
171 diag.responseContentType = result.diagnostic.responseContentType;
172 diag.retryAfter = result.diagnostic.retryAfter;
173 diag.rateLimitRemaining = result.diagnostic.rateLimitRemaining;
174 diag.rateLimitReset = result.diagnostic.rateLimitReset;
175 diag.responseSummary = result.diagnostic.responseSummary;
176 diag.hasRateLimitHint = result.diagnostic.hasRateLimitHint;
177 diag.latencyMs = result.elapsedMs;
178
179 if ((cancelled && cancelled->loadRelaxed()) || result.wasCancelled) {
180 diag.errorString = QStringLiteral("Cancelled");
181 onComplete(QByteArray(), diag);
182 } else if (result.success) {
183 if (ext == QStringLiteral("pdf") && result.data.size() >= 5 && !result.data.startsWith("%PDF-")) {
184 ext = QStringLiteral("html");
185 }
186 if (ComponentCacheService::isValidDatasheetData(result.data, ext)) {
187 cache->saveDatasheet(lcscId, result.data, ext, gen);
188 onComplete(result.data, diag);
189 } else {
190 diag.errorString = QStringLiteral("Invalid datasheet data");
191 onComplete(QByteArray(), diag);
192 }
193 } else {
194 diag.errorString = result.error;
195 onComplete(QByteArray(), diag);
196 }

198 request->deleteLater();
199 });
