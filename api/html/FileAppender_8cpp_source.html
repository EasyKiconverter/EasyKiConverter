76bool FileAppender::validateAndNormalizePath(const QString& filePath) {
77 if (filePath.isEmpty()) {
78 return false;
79 }
80
81 // 检查路径中的危险字符和模式
82 QString cleanPath = filePath;
83
84 // 检查路径遍历攻击模式
85 if (cleanPath.contains("..") || cleanPath.contains("~")) {
86 return false;
87 }
88
89 // 检查绝对路径是否指向系统敏感目录
90 QFileInfo fileInfo(cleanPath);
91 QString absolutePath = fileInfo.absoluteFilePath();
92
93 // 规范化路径（解析符号链接等）
94 QDir dir = fileInfo.absoluteDir();
95 m_baseDir = dir.absolutePath();
96
97 // 获取规范的绝对路径
98 QString canonicalDir = dir.canonicalPath();
99 if (!canonicalDir.isEmpty()) {
100 m_baseDir = canonicalDir;
101 }
102
103 // 检查路径是否在允许的目录范围内
104 if (!isPathInAllowedDir(absolutePath)) {
105 return false;
106 }
107
108 // 存储规范化的路径
109 m_filePath = absolutePath;
110 m_canonicalPath = absolutePath;
111
112 return true;
113}

120bool FileAppender::isPathInAllowedDir(const QString& path) const {
121 // 允许的根目录列表
122 QStringList allowedRoots;
123
124 // 应用程序目录
125 if (QCoreApplication::instance()) {
126 allowedRoots << QCoreApplication::applicationDirPath();
127 }
128
129 // 用户数据目录
130 allowedRoots << QStandardPaths::writableLocation(QStandardPaths::AppDataLocation);
131 allowedRoots << QStandardPaths::writableLocation(QStandardPaths::AppLocalDataLocation);
132 allowedRoots << QStandardPaths::writableLocation(QStandardPaths::TempLocation);
133
134 // 用户主目录
135 allowedRoots << QDir::homePath();
136
137 // 当前工作目录
138 allowedRoots << QDir::currentPath();
139
140 // 检查路径是否在允许的根目录下
141 for (const QString& root : allowedRoots) {
142 if (path.startsWith(root)) {
143 return true;
144 }
145 }

147 return false;

156 Q_UNUSED(record);
157

161

184void FileAppender::flush() {
185 if (!m_pathValid) {
186 return;
187 }
188
189 if (m_async) {
190 // 请求刷新
191 m_flushRequested.storeRelaxed(1);
192 m_queueCondition.wakeAll();
193
194 // 等待队列清空（带超时保护）
195 int timeout = 5000;
196 int waited = 0;
197 while (waited < timeout) {
198 QMutexLocker locker(&m_queueMutex);
199 if (m_writeQueue.isEmpty()) {
200 break;
201 }
202 locker.unlock();
203 QThread::msleep(10);
204 waited += 10;
205 }
206 }
207
208 // 刷新文件缓冲
209 QMutexLocker fileLocker(&m_fileMutex);
210 if (m_file.isOpen()) {
211 m_file.flush();
212 }
213}

218void FileAppender::close() {
219 // 停止异步线程
220 if (m_async && m_writerThread) {
221 m_running.storeRelaxed(0);
222 m_queueCondition.wakeAll();
223
224 // 等待线程结束（带超时保护）
225 if (m_writerThread) {
226 if (!m_writerThread->wait(3000)) {
227 m_writerThread->terminate();
228 m_writerThread->wait();
229 }
230 delete m_writerThread;
231 m_writerThread = nullptr;
232 }
233 }
234
235 // 关闭文件
236 QMutexLocker locker(&m_fileMutex);
237 if (m_file.isOpen()) {
238 m_file.flush();
239 m_file.close();
240 }

267void FileAppender::rollOver() {
268 // 使用递归锁，允许在 writeDirect 中调用
269 QMutexLocker locker(&m_fileMutex);
270
271 if (m_file.isOpen()) {
272 m_file.flush();
273 m_file.close();
274 }
275
276 // 清理旧文件，为新文件腾出空间
277 cleanupOldFiles();
278
279 // 重命名现有文件
280 QString rolledName = generateRolledFileName(1);
281
282 // 确保目标文件不存在
283 if (QFile::exists(rolledName)) {
284 QFile::remove(rolledName);
285 }
286
287 if (!QFile::rename(m_filePath, rolledName)) {
288 // 重命名失败，尝试直接删除并重新创建
289 QFile::remove(m_filePath);
290 }
291
292 // 重新打开新文件
293 m_file.setFileName(m_filePath);
294 if (!m_file.open(QIODevice::WriteOnly | QIODevice::Append | QIODevice::Text)) {
295 m_pathValid = false;
296 m_lastError = "Failed to reopen log file after rollover";
297 }

303void FileAppender::cleanupOldFiles() {
304 // 注意：此函数在 rollOver 中调用，已持有 m_fileMutex
305 QDir dir = QFileInfo(m_filePath).absoluteDir();
306 QString baseName = QFileInfo(m_filePath).baseName();
307 QString suffix = QFileInfo(m_filePath).completeSuffix();
308 QString pattern = baseName + "_*";
309 if (!suffix.isEmpty()) {
310 pattern += "." + suffix;
311 }
312
313 QStringList filters;
314 filters << pattern;
315 QStringList files = dir.entryList(filters, QDir::Files, QDir::Name);
316
317 // 删除超出数量限制的旧文件
318 while (files.size() >= m_maxFiles - 1) {
319 QString toRemove = dir.absoluteFilePath(files.takeFirst());
320 QFile::remove(toRemove);
321 }
322}

348void FileAppender::writerThreadFunc() {
349 while (m_running.loadRelaxed()) {
350 QString message;
351 bool overflowed = false;
352
353 {
354 QMutexLocker locker(&m_queueMutex);
355
356 // 等待新消息或刷新请求
357 while (m_writeQueue.isEmpty() && m_running.loadRelaxed() && !m_flushRequested.loadRelaxed()) {
358 m_queueCondition.wait(&m_queueMutex, 100);
359 }
360
361 if (!m_writeQueue.isEmpty()) {
362 message = m_writeQueue.dequeue();
363 }
364
365 // 检查是否有溢出发生过
366 if (m_queueOverflow.loadRelaxed()) {
367 overflowed = true;
368 m_queueOverflow.storeRelaxed(0);
369 }
370 }
371
372 // 如果发生过溢出，添加警告消息
373 if (overflowed) {
374 writeDirect("[WARNING] Log queue overflow occurred, some messages were dropped");
375 }
376
377 if (!message.isEmpty()) {
378 writeDirect(message);
379 }
380
381 // 处理刷新请求
382 if (m_flushRequested.loadRelaxed()) {
383 QMutexLocker fileLocker(&m_fileMutex);
384 if (m_file.isOpen()) {
385 m_file.flush();
386 }
387 m_flushRequested.storeRelaxed(0);
388 }
389 }
390
391 // 写入剩余消息
392 while (true) {
393 QString message;
394 {
395 QMutexLocker locker(&m_queueMutex);
396 if (m_writeQueue.isEmpty()) {
397 break;
398 }
399 message = m_writeQueue.dequeue();
400 }
401
402 if (!message.isEmpty()) {
403 writeDirect(message);
404 }
405 }
406}

412void FileAppender::writeDirect(const QString& formatted) {
413 QMutexLocker locker(&m_fileMutex);
414
415 if (!m_pathValid) {
416 return;
417 }
418
419 if (!m_file.isOpen()) {
420 openFile();
421 }
422
423 if (m_file.isOpen()) {
424 m_file.write(formatted.toUtf8());
425 m_file.write("\n");
426
427 // 检查是否需要滚动（避免在锁内调用 rollOver，因为它也使用同一个递归锁）
428 if (needsRollover()) {
429 locker.unlock(); // 释放锁
430 rollOver(); // rollOver 会重新获取锁
431 }
432 }
