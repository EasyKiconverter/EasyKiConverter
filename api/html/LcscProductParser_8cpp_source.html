18QStringList extractPreviewImageUrls(const QJsonObject& product) {
19 // 保存从产品字段解析出的预览图地址，最终限制为最多三项。
20 QStringList imageUrls;
21 // 读取 LCSC 使用 <$> 分隔的原始图片地址字段。
22 const QString imagesString = product.value(QStringLiteral("image")).toString();
23 if (!imagesString.isEmpty()) {
24 imageUrls = imagesString.split(QStringLiteral("<$>"), Qt::SkipEmptyParts);
25 }
26
27 while (imageUrls.size() > 3) {
28 imageUrls.removeLast();
29 }
30 return UrlUtils::deduplicateAndNormalizeUrls(imageUrls);
31}

38QString extractComponentCode(const QJsonObject& product) {
39 // 按兼容性顺序检查产品对象中可能承载元件编号的字段。
40 static const QStringList directKeys = {QStringLiteral("component_code"),
41 QStringLiteral("productCode"),
42 QStringLiteral("product_code"),
43 QStringLiteral("productNo"),
44 QStringLiteral("product_no"),
45 QStringLiteral("number"),
46 QStringLiteral("code"),
47 QStringLiteral("lcscPart"),
48 QStringLiteral("lcscPartNumber")};
49
50 // 在指定 JSON 对象中查找第一个非空的直接编号字段。
51 const auto findValue = [](const QJsonObject& object) {
52 for (const QString& key : directKeys) {
53 // 读取当前候选字段并去除空白，统一返回大写编号。
54 const QString value = object.value(key).toString().trimmed();
55 if (!value.isEmpty()) {
56 return value.toUpper();
57 }
58 }
59 return QString();
60 };
61
62 // 优先从产品顶层字段提取编号。
63 const QString directCode = findValue(product);
64 if (!directCode.isEmpty()) {
65 return directCode;
66 }
67
68 // 读取产品中的设备信息对象作为第二级编号来源。
69 const QJsonObject deviceInfo = product.value(QStringLiteral("device_info")).toObject();
70 // 尝试从设备信息对象的直接字段提取编号。
71 const QString deviceCode = findValue(deviceInfo);
72 if (!deviceCode.isEmpty()) {
73 return deviceCode;
74 }
75
76 // 读取设备属性对象作为最后一级编号来源。
77 const QJsonObject attributes = deviceInfo.value(QStringLiteral("attributes")).toObject();
78 // 属性接口可能使用不同的名称，按已知别名依次匹配。
79 static const QStringList attributeKeys = {QStringLiteral("LCSC Part"),
80 QStringLiteral("LCSC Part #"),
81 QStringLiteral("LCSC Part Number"),
82 QStringLiteral("Part Number")};
83 for (const QString& key : attributeKeys) {
84 // 读取当前属性别名对应的元件编号。
85 const QString value = attributes.value(key).toString().trimmed();
86 if (!value.isEmpty()) {
87 return value.toUpper();
88 }
89 }
90 return QString();
91}

99QJsonObject selectProduct(const QString& componentId, const QJsonArray& products) {
100 // 统一请求编号格式，避免大小写和首尾空白导致匹配失败。
101 const QString normalizedId = componentId.trimmed().toUpper();
102 for (const QJsonValue& value : products) {
103 // 将当前搜索结果转换为产品对象后进行编号比对。
104 const QJsonObject product = value.toObject();
105 if (extractComponentCode(product) == normalizedId) {
106 return product;
107 }
108 }
109 return QJsonObject();
110}

120std::optional<LcscProductInfo> LcscProductParser::parse(const QString& componentId, const QByteArray& responseJson) {
121 // 将网络响应解析为 JSON 文档，解析失败时立即返回空结果。
122 const QJsonDocument document = QJsonDocument::fromJson(responseJson);
123 if (document.isNull()) {
124 return std::nullopt;
125 }
126
127 // 读取 API 响应中的结果对象和产品数组。
128 const QJsonObject result = document.object().value(QStringLiteral("result")).toObject();
129 // 保存搜索结果中的产品条目，供编号精确匹配使用。
130 const QJsonArray products = result.value(QStringLiteral("productList")).toArray();
131 if (products.isEmpty()) {
132 return std::nullopt;
133 }
134
135 // 从搜索结果中选择编号精确匹配的产品。
136 const QJsonObject product = selectProduct(componentId, products);
137 if (product.isEmpty()) {
138 return std::nullopt;
139 }
140
141 // 组装供缓存和导出流程使用的产品媒体信息。
142 LcscProductInfo info;
143 info.imageUrls = extractPreviewImageUrls(product);
144 // 读取产品设备属性中的制造商料号和数据手册地址。
145 const QJsonObject attributes =
146 product.value(QStringLiteral("device_info")).toObject().value(QStringLiteral("attributes")).toObject();
147 info.manufacturerPart = attributes.value(QStringLiteral("Manufacturer Part")).toString();
148 info.datasheetUrl = attributes.value(QStringLiteral("Datasheet")).toString();
149 return info;
150}
