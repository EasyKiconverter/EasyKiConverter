70bool readWideStringIndices(const QByteArray& data, QSet<quint32>* indices) {
71 if (indices == nullptr)
72 return false;
73 indices->clear();
74
75 AltiumBinaryReader reader(data);
76 QMap<QString, QString> parameters; // 参数块键值表。
77 if (!reader.readCStringParameterBlock(&parameters) || reader.hasError() || reader.remaining() != 0)
78 return false;
79
80 for (auto it = parameters.cbegin(); it != parameters.cend(); ++it) {
81 if (!it.key().startsWith(QStringLiteral("ENCODEDTEXT")))
82 continue;
83 const QString suffix = it.key().mid(QStringLiteral("ENCODEDTEXT").size()); // 文本索引后缀。
84 bool ok = false; // 文本索引转换是否成功。
85 const quint32 index = suffix.toUInt(&ok); // 解析得到的文本索引。
86 if (!ok || suffix.isEmpty() || indices->contains(index))
87 return false;
88 indices->insert(index);
89 }
90 for (quint32 expected = 0; expected < static_cast<quint32>(indices->size()); ++expected) {
91 if (!indices->contains(expected))
92 return false;
93 }
94 return true;
95}

346bool validatePrimitiveFields(const AltiumPcbLibReader::PrimitiveRecord& object, QString* error) {
347 auto reject = [error](const QString& message) { // 记录校验错误并返回失败。
348 if (error != nullptr)
349 *error = message;
350 return false;
351 };
352
353 if (object.hasTrackFields || object.hasArcFields || object.hasPadFields || object.hasFillFields ||
354 object.hasRegionFields || object.hasComponentBodyFields || object.hasTextFields) {
355 if (object.layer < 1 || object.layer > 74)
356 return reject(QStringLiteral("PcbLib 图元层号必须在 1..74 范围内"));
357 }
358
359 if (object.hasArcFields) {
360 if (object.arc.radius <= 0)
361 return reject(QStringLiteral("PcbLib 弧线半径必须为正"));
362 if (object.arc.width <= 0)
363 return reject(QStringLiteral("PcbLib 弧线宽度必须为正"));
364 if (!std::isfinite(object.arc.startAngle) || !std::isfinite(object.arc.endAngle))
365 return reject(QStringLiteral("PcbLib 弧线角度必须为有限值"));
366 }
367 if (object.hasTrackFields && object.track.width <= 0)
368 return reject(QStringLiteral("PcbLib 走线宽度必须为正"));
369 if (object.hasPadFields) {
370 const auto& pad = object.pad; // 当前焊盘字段。
371 const auto isValidPadShape = [](quint8 shape) { // 检查焊盘形状编码。
372 return shape == AltiumConstants::PCB_PAD_SHAPE_ROUND ||
373 shape == AltiumConstants::PCB_PAD_SHAPE_RECTANGULAR ||
374 shape == AltiumConstants::PCB_PAD_SHAPE_OCTAGONAL ||
375 shape == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT;
376 };
377 if (pad.sizeTopX <= 0 || pad.sizeTopY <= 0 || pad.sizeMidX <= 0 || pad.sizeMidY <= 0 || pad.sizeBotX <= 0 ||
378 pad.sizeBotY <= 0)
379 return reject(QStringLiteral("PcbLib 焊盘尺寸必须为正"));
380 if (!isValidPadShape(pad.shapeTop) || !isValidPadShape(pad.shapeMid) || !isValidPadShape(pad.shapeBot))
381 return reject(QStringLiteral("PcbLib 焊盘形状无效"));
382 if (pad.holeSize < 0 || pad.holeSlotLengthRaw < 0)
383 return reject(QStringLiteral("PcbLib 焊盘孔尺寸不能为负"));
384 if (object.layer == AltiumConstants::PCB_LAYER_MULTI && pad.holeSize <= 0)
385 return reject(QStringLiteral("PcbLib 通孔焊盘孔径必须为正"));
386 if (pad.cornerRadiusPercentage > 100 || pad.mode > 3 || pad.powerPlaneConnectStyle > 2 ||
387 (pad.reliefEntries != 2 && pad.reliefEntries != 4) || pad.drillType > 2 || pad.holeType > 2)
388 return reject(QStringLiteral("PcbLib 焊盘扩展属性无效"));
389 if (pad.holeType == 2 && pad.holeSlotLengthRaw <= 0)
390 return reject(QStringLiteral("PcbLib 焊盘槽孔长度必须为正"));
391 if (!std::isfinite(pad.rotation) || !std::isfinite(pad.holeRotation))
392 return reject(QStringLiteral("PcbLib 焊盘旋转角度必须为有限值"));
393 }
394 if (object.hasFillFields && !std::isfinite(object.fill.rotation))
395 return reject(QStringLiteral("PcbLib 填充旋转角度必须为有限值"));
396 if (object.hasTextFields) {
397 if (object.textFields.height <= 0 || object.textFields.strokeWidth < 0)
398 return reject(QStringLiteral("PcbLib 文本尺寸必须有效"));
399 if (!std::isfinite(object.textFields.rotation))
400 return reject(QStringLiteral("PcbLib 文本旋转角度必须为有限值"));
401 }
402 if (object.hasRegionFields) {
403 if (object.region.vertices.size() < 3)
404 return reject(QStringLiteral("PcbLib 区域至少需要三个顶点"));
405 for (const QPointF& vertex : object.region.vertices) {
406 if (!std::isfinite(vertex.x()) || !std::isfinite(vertex.y()))
407 return reject(QStringLiteral("PcbLib 区域顶点必须为有限值"));
408 }
409 }
410 if (object.hasComponentBodyFields) {
411 for (const QPointF& vertex : object.componentBody.outline) {
412 if (!std::isfinite(vertex.x()) || !std::isfinite(vertex.y()))
413 return reject(QStringLiteral("PcbLib 三维元件体轮廓顶点必须为有限值"));
414 }
415 const auto& parameters = object.componentBody.parameters; // 元件体参数表。
416 if (parameters.contains(QStringLiteral("BODYOPACITY3D"))) {
417 bool opacityOk = false; // 透明度文本转换是否成功。
418 const double opacity =
419 parameters.value(QStringLiteral("BODYOPACITY3D")).toDouble(&opacityOk); // 透明度值。
420 if (!opacityOk || !std::isfinite(opacity) || opacity < 0.0 || opacity > 1.0)
421 return reject(QStringLiteral("PcbLib 三维元件体透明度必须在 0..1 范围内"));
422 }
423 }
424 return true;
425}

447bool AltiumPcbLibReader::open(const QString& filePath) {
448 m_components.clear();
449 m_fileVersion.clear();
450 m_libraryMetadata.clear();
451 m_errorMessage.clear();
452
453 if (!m_oleReader.open(filePath))
454 return fail(m_oleReader.errorString());
455
456 QByteArray fileHeaderData; // FileHeader 流数据。
457 if (!m_oleReader.readStream(QStringLiteral("FileHeader"), &fileHeaderData))
458 return fail(QStringLiteral("PcbLib 缺少 FileHeader 流"));
459 AltiumBinaryReader fileHeaderReader(fileHeaderData);
460 int32_t versionLength = 0; // 版本字符串长度。
461 if (!fileHeaderReader.readInt32(&versionLength) || versionLength < 0 || versionLength > 255 ||
462 !fileHeaderReader.readPascalShortString(&m_fileVersion) || m_fileVersion.toLatin1().size() != versionLength ||
463 fileHeaderReader.remaining() != 0)
464 return fail(QStringLiteral("PcbLib FileHeader 无效"));
465
466 QByteArray libraryData; // Library/Data 流数据。
467 if (!m_oleReader.readStream(QStringLiteral("Library/Data"), &libraryData))
468 return fail(QStringLiteral("PcbLib 缺少 Library/Data 流"));
469 AltiumBinaryReader libraryReader(libraryData);
470 if (!libraryReader.readBlock(&m_libraryMetadata))
471 return fail(QStringLiteral("PcbLib Library/Data 元数据块无效: %1").arg(libraryReader.errorString()));
472 if (!m_libraryMetadata.isEmpty() && m_libraryMetadata.endsWith('\0'))
473 m_libraryMetadata.chop(1);
474
475 uint32_t componentCount = 0; // 库中封装数量。
476 if (!libraryReader.readUInt32(&componentCount) ||
477 componentCount > static_cast<uint32_t>(libraryReader.remaining() / 5))
478 return fail(QStringLiteral("PcbLib Library/Data 的封装数量无效"));
479
480 QVector<QString> names; // 封装名称列表。
481 names.reserve(static_cast<int>(componentCount));
482 QSet<QString> componentNames; // 大小写折叠后的封装名称集合。
483 for (uint32_t i = 0; i < componentCount; ++i) {
484 QString name; // 当前封装名称。
485 if (!libraryReader.readStringBlock(&name) || name.isEmpty())
486 return fail(QStringLiteral("PcbLib Library/Data 的封装名称无效"));
487 const QString foldedName = name.toCaseFolded(); // 用于重复检查的折叠名称。
488 if (componentNames.contains(foldedName))
489 return fail(QStringLiteral("PcbLib Library/Data 的封装名称重复: %1").arg(name));
490 componentNames.insert(foldedName);
491 names.append(name);
492 }
493 if (libraryReader.hasError() || libraryReader.remaining() != 0)
494 return fail(QStringLiteral("PcbLib Library/Data 末尾包含无效数据"));
495
496 QMap<QString, QString> sectionKeys; // 封装名称到 OLE 节名称的映射。
497 if (m_oleReader.containsStream(QStringLiteral("SectionKeys"))) {
498 QByteArray sectionKeyData; // SectionKeys 流数据。
499 if (!m_oleReader.readStream(QStringLiteral("SectionKeys"), &sectionKeyData))
500 return fail(QStringLiteral("无法读取 PcbLib SectionKeys 流"));
501 AltiumBinaryReader sectionKeyReader(sectionKeyData);
502 uint32_t keyCount = 0; // 节映射数量。
503 if (!sectionKeyReader.readUInt32(&keyCount) ||
504 keyCount > static_cast<uint32_t>(sectionKeyReader.remaining() / 9))
505 return fail(QStringLiteral("PcbLib SectionKeys 的数量无效"));
506 QSet<QString> mappedComponents; // 已映射的封装名称集合。
507 QSet<QString> mappedSectionKeys; // 已使用的节名称集合。
508 for (uint32_t i = 0; i < keyCount; ++i) {
509 QString libRef; // SectionKeys 中的封装名称。
510 QString sectionKey; // 对应的 OLE 节名称。
511 if (!sectionKeyReader.readPascalString(&libRef) || !sectionKeyReader.readStringBlock(&sectionKey) ||
512 libRef.isEmpty() || sectionKey.isEmpty())
513 return fail(QStringLiteral("PcbLib SectionKeys 映射不完整"));
514 if (!names.contains(libRef))
515 return fail(QStringLiteral("PcbLib SectionKeys 包含未知封装: %1").arg(libRef));
516 if (mappedComponents.contains(libRef.toCaseFolded()) ||
517 mappedSectionKeys.contains(sectionKey.toCaseFolded()))
518 return fail(QStringLiteral("PcbLib SectionKeys 映射重复: %1").arg(libRef));
519 mappedComponents.insert(libRef.toCaseFolded());
520 mappedSectionKeys.insert(sectionKey.toCaseFolded());
521 sectionKeys.insert(libRef, sectionKey);
522 }
523 if (sectionKeyReader.hasError() || sectionKeyReader.remaining() != 0)
524 return fail(QStringLiteral("PcbLib SectionKeys 末尾包含无效数据"));
525 }
526
527 QSet<QString> resolvedSectionKeys; // 已解析的 OLE 节名称集合。
528 for (const QString& name : names) {
529 const QString sectionKey = sectionKeys.value(name, name); // 当前封装对应的节名称。
530 if (resolvedSectionKeys.contains(sectionKey.toCaseFolded()))
531 return fail(QStringLiteral("PcbLib 封装存储键重复: %1").arg(sectionKey));
532 resolvedSectionKeys.insert(sectionKey.toCaseFolded());
533 const QString dataPath = sectionKey + QStringLiteral("/Data"); // 封装 Data 流路径。
534 if (!m_oleReader.containsStream(dataPath))
535 return fail(QStringLiteral("PcbLib 缺少封装 Data 流: %1").arg(dataPath));
536 m_components.append({name, sectionKey});
537 }
538 return true;
539}

613bool AltiumPcbLibReader::readFootprintObjects(int componentIndexValue, QVector<PrimitiveRecord>* objects) const {
614 if (objects == nullptr) {
615 m_errorMessage = QStringLiteral("读取 PcbLib 图元时输出容器为空");
616 return false;
617 }
618 objects->clear();
619 m_errorMessage.clear();
620
621 const auto failRead = [this, objects](const QString& message) {
622 objects->clear();
623 m_errorMessage = message;
624 return false;
625 };
626
627 QByteArray data; // 封装 Data 流数据。
628 if (!readFootprintStream(componentIndexValue, QStringLiteral("Data"), &data))
629 return failRead(QStringLiteral("无法读取 PcbLib 封装 Data 流"));
630
631 AltiumBinaryReader reader(data);
632 QString footprintName; // Data 流头部声明的封装名称。
633 if (!reader.readStringBlock(&footprintName) || footprintName.isEmpty())
634 return failRead(QStringLiteral("PcbLib 封装 Data 首部无效: %1").arg(reader.errorString()));
635
636 QSet<quint32> wideStringIndices; // WideStrings 中已声明的文本索引。
637 bool wideStringsLoaded = false; // 是否已经加载 WideStrings 流。
638
639 while (reader.remaining() > 0) {
640 const int objectStart = reader.position(); // 当前图元起始偏移。
641 uint8_t objectId = 0; // 当前图元对象类型编号。
642 if (!reader.readUInt8(&objectId)) {
643 return failRead(QStringLiteral("读取 PcbLib 图元 ID 失败: %1").arg(reader.errorString()));
644 }
645 const int blockCount = primitiveBlockCount(objectId); // 当前图元的子块数量。
646 if (blockCount < 0)
647 return failRead(QStringLiteral("PcbLib 图元 ID %1 不受支持").arg(objectId));
648
649 PrimitiveRecord object; // 当前图元解析结果。
650 object.objectId = objectId;
651 object.blocks.reserve(blockCount);
652 for (int blockIndex = 0; blockIndex < blockCount; ++blockIndex) {
653 const int blockStart = reader.position(); // 当前子块起始偏移。
654 QByteArray payload; // 子块载荷。
655 uint8_t flags = 0; // 子块标志位。
656 if (!reader.readBlock(&payload, &flags))
657 return failRead(QStringLiteral("读取 PcbLib 图元 %1 的子块 %2 失败: %3")
658 .arg(objectId)
659 .arg(blockIndex)
660 .arg(reader.errorString()));
661 if (payload.isEmpty())
662 return failRead(QStringLiteral("PcbLib 图元 %1 的子块 %2 为空").arg(objectId).arg(blockIndex));
663 const bool isPadStringBlock =
664 objectId == AltiumConstants::PCB_OBJECT_PAD && blockIndex < 3; // 是否为焊盘字符串块。
665 const bool isTextStringBlock =
666 objectId == AltiumConstants::PCB_OBJECT_TEXT && blockIndex == 1; // 是否为文本字符串块。
667 if ((isPadStringBlock || isTextStringBlock) && !isStringBlockPayload(payload)) {
668 return failRead(QStringLiteral("PcbLib 图元 %1 的字符串子块 %2 无效").arg(objectId).arg(blockIndex));
669 }
670 const bool isMainBlock =
671 (objectId == AltiumConstants::PCB_OBJECT_PAD && blockIndex == 4) ||
672 (objectId != AltiumConstants::PCB_OBJECT_PAD && blockIndex == 0); // 是否为主字段块。
673 if (isMainBlock && !readCommonPrimitiveHeader(payload, &object.layer, &object.primitiveFlags))
674 return failRead(QStringLiteral("PcbLib 图元 %1 的公共头部不完整").arg(objectId));
675 if (isMainBlock && objectId == AltiumConstants::PCB_OBJECT_TRACK) {
676 if (!parseTrackFields(payload, &object.track))
677 return failRead(QStringLiteral("PcbLib 走线图元字段不完整"));
678 object.hasTrackFields = true;
679 } else if (isMainBlock && objectId == AltiumConstants::PCB_OBJECT_ARC) {
680 if (!parseArcFields(payload, &object.arc))
681 return failRead(QStringLiteral("PcbLib 弧线图元字段不完整"));
682 object.hasArcFields = true;
683 } else if (isMainBlock && objectId == AltiumConstants::PCB_OBJECT_PAD) {
684 if (!parsePadMainFields(payload, &object.pad))
685 return failRead(QStringLiteral("PcbLib 焊盘图元主块字段不完整"));
686 object.hasPadFields = true;
687 } else if (isMainBlock && objectId == AltiumConstants::PCB_OBJECT_FILL) {
688 if (!parseFillFields(payload, &object.fill))
689 return failRead(QStringLiteral("PcbLib 填充图元字段不完整"));
690 object.hasFillFields = true;
691 } else if (isMainBlock && objectId == AltiumConstants::PCB_OBJECT_REGION) {
692 if (!parseRegionFields(payload, &object.region))
693 return failRead(QStringLiteral("PcbLib 区域图元字段不完整"));
694 object.hasRegionFields = true;
695 } else if (isMainBlock && objectId == AltiumConstants::PCB_OBJECT_COMPONENT_BODY) {
696 if (!parseComponentBodyFields(payload, &object.componentBody))
697 return failRead(QStringLiteral("PcbLib 三维元件体图元字段不完整"));
698 object.hasComponentBodyFields = true;
699 } else if (isMainBlock && objectId == AltiumConstants::PCB_OBJECT_TEXT) {
700 if (!parseTextFields(payload, &object.textFields))
701 return failRead(QStringLiteral("PcbLib 文本图元字段不完整"));
702 object.hasTextFields = true;
703 }
704 PrimitiveBlock block; // 当前子块的原始记录。
705 block.flags = flags;
706 block.payload = payload;
707 block.encoded = data.mid(blockStart, reader.position() - blockStart);
708 object.blocks.append(block);
709 }
710 if (objectId == AltiumConstants::PCB_OBJECT_PAD) {
711 if (!readStringBlockPayload(object.blocks.at(0).payload, &object.designator) ||
712 !parsePadExtendedFields(object.blocks.at(5).payload, &object.pad))
713 return failRead(QStringLiteral("PcbLib 焊盘图元扩展字段不完整"));
714 }
715 if (objectId == AltiumConstants::PCB_OBJECT_TEXT &&
716 !readStringBlockPayload(object.blocks.at(1).payload, &object.text))
717 return failRead(QStringLiteral("PcbLib 文本图元字符串内容无效"));
718 QString semanticError; // 图元语义校验错误说明。
719 if (!validatePrimitiveFields(object, &semanticError))
720 return failRead(semanticError);
721 if (object.hasTextFields) {
722 if (!wideStringsLoaded) {
723 QByteArray wideStrings; // WideStrings 流数据。
724 if (!readFootprintStream(componentIndexValue, QStringLiteral("WideStrings"), &wideStrings))
725 return failRead(QStringLiteral("无法读取 PcbLib 封装 WideStrings 流"));
726 if (!readWideStringIndices(wideStrings, &wideStringIndices))
727 return failRead(QStringLiteral("PcbLib 封装 WideStrings 流无效"));
728 wideStringsLoaded = true;
729 }
730 if (!wideStringIndices.contains(object.textFields.wideStringIndex))
731 return failRead(QStringLiteral("PcbLib 文本图元引用了不存在的 WideStrings 索引: %1")
732 .arg(object.textFields.wideStringIndex));
733 }
734 object.encoded = data.mid(objectStart, reader.position() - objectStart);
735 objects->append(object);
736 }
737 return true;
738}
