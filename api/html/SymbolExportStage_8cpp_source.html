39void SymbolExportStage::start(const QStringList& componentIds,
40 const QMap<QString, QSharedPointer<ComponentData>>& cachedData) {
41 if (m_isExporting.load()) {
42 qWarning() << "SymbolExportStage: Export already in progress";
43 return;
44 }
45
46 if (componentIds.isEmpty()) {
47 qWarning() << "SymbolExportStage: No components to export";
48 emit completed(0, 0, 0);
49 return;
50 }
51
52 m_componentIds = componentIds;
53 m_cachedData = cachedData;
54 m_cancelled.store(false);
55 m_isRunning.store(true);
56
57 {
58 QMutexLocker locker(&m_progressMutex);
59 m_progress = ExportTypeProgress();
60 m_progress.typeName = QStringLiteral("Symbol");
61 m_progress.totalCount = componentIds.size();
62 m_progress.completedCount = 0;
63 m_progress.successCount = 0;
64 m_progress.failedCount = 0;
65 m_progress.skippedCount = 0;
66 m_progress.inProgressCount = componentIds.size();
67 for (const QString& componentId : componentIds) { // 当前元器件编号。
68 ExportItemStatus itemStatus;
69 itemStatus.status = ExportItemStatus::Status::InProgress;
70 itemStatus.startTime = QDateTime::currentDateTime();
71 m_progress.itemStatus[componentId] = itemStatus;
72 }
73 }
74
75 m_tempManager.setOutputPath(m_options.outputPath);
76
77 m_isExporting.store(true);
78 m_workerThread = QThread::create([this, componentIds, cachedData]() { doLibraryExport(componentIds, cachedData); });
79 connect(m_workerThread, &QThread::finished, this, [this]() { m_workerThread = nullptr; });
80 m_workerThread->start();
81}

119void SymbolExportStage::doLibraryExport(const QStringList& componentIds,
120 const QMap<QString, QSharedPointer<ComponentData>>& cachedData) {
121 qDebug() << "SymbolExportStage: Starting library export in worker thread for" << componentIds.size()
122 << "components";
123
124 QList<SymbolData> symbolList;
125 QStringList collectedIds;
126 QStringList failedIds;
127 int successCount = 0;
128 int skippedCount = 0;
129
130 // 库级导出只有一次 writer 调用，必须在提交前把每个组件的最终状态固化到阶段快照。
131 const auto finalizeProgress = [this, &componentIds, &collectedIds, &failedIds](bool librarySucceeded,
132 const QString& errorMessage) {
133 ExportTypeProgress progressSnapshot;
134 {
135 QMutexLocker locker(&m_progressMutex);
136 for (const QString& componentId : componentIds) { // 当前元器件编号。
137 auto statusIt = m_progress.itemStatus.find(componentId);
138 if (statusIt == m_progress.itemStatus.end() || statusIt->isComplete()) {
139 continue;
140 }
141
142 const bool failed = // 当前元器件是否应标记为失败。
143 failedIds.contains(componentId) || (!librarySucceeded && collectedIds.contains(componentId));
144 statusIt->status = failed ? ExportItemStatus::Status::Failed : ExportItemStatus::Status::Success;
145 if (failed && !errorMessage.isEmpty()) {
146 statusIt->errorMessage = errorMessage;
147 }
148 statusIt->endTime = QDateTime::currentDateTime();
149 ++m_progress.completedCount;
150 if (failed) {
151 ++m_progress.failedCount;
152 } else {
153 ++m_progress.successCount;
154 }
155 m_progress.inProgressCount = qMax(0, m_progress.inProgressCount - 1);
156 }
157 progressSnapshot = m_progress;
158 }
159 emit progressChanged(progressSnapshot);
160 };
161
162 for (const QString& componentId : componentIds) { // 当前元器件编号。
163 if (m_cancelled.load()) {
164 qDebug() << "SymbolExportStage: Export cancelled during data collection";
165 break;
166 }
167
168 auto it = cachedData.find(componentId);
169 if (it == cachedData.end() || !it.value()) {
170 qWarning() << "SymbolExportStage: No data for component:" << componentId;
171 failedIds.append(componentId);
172 ExportItemStatus status;
173 status.status = ExportItemStatus::Status::Failed;
174 status.errorMessage = "No component data";
175 emit itemStatusChanged(componentId, status);
176 continue;
177 }
178
179 QSharedPointer<ComponentData> data = it.value();
180
181 if (!data->symbolData()) {
182 qWarning() << "SymbolExportStage: No symbol data for component:" << componentId;
183 failedIds.append(componentId);
184 ExportItemStatus status;
185 status.status = ExportItemStatus::Status::Failed;
186 status.errorMessage = "No symbol data";
187 emit itemStatusChanged(componentId, status);
188 continue;
189 }
190
191 symbolList.append(*data->symbolData());
192 collectedIds.append(componentId);
193 successCount++;
194
195 ExportItemStatus status;
196 status.status = ExportItemStatus::Status::Success;
197 status.diagnostics = data->symbolData()->validationErrors();
198 if (!status.diagnostics.isEmpty()) {
199 qWarning() << "SymbolExportStage: Input diagnostics for" << componentId << status.diagnostics;
200 }
201 emit itemStatusChanged(componentId, status);
202
203 if (m_options.debugMode) {
204 QThreadPool::globalInstance()->start([componentId, data, outputPath = m_options.outputPath]() {
205 DebugExportHelper::exportDebugData(componentId, data, outputPath);
206 });
207 }
208
209 qDebug() << "SymbolExportStage: Collected symbol for" << componentId;
210 }
211
212 if (m_cancelled.load()) {
213 m_isExporting.store(false);
214 m_isRunning.store(false);
215 emit completed(0, componentIds.size(), 0);
216 return;
217 }
218
219 if (symbolList.isEmpty()) {
220 qWarning() << "SymbolExportStage: No valid symbols to export";
221 m_isExporting.store(false);
222 m_isRunning.store(false);
223 emit completed(0, componentIds.size(), 0);
224 return;
225 }
226
227 const auto failCollectedSymbols = [this, &collectedIds, &failedIds, &successCount](const QString& errorMessage) {
228 for (const QString& componentId : collectedIds) { // 当前已收集元器件编号。
229 failedIds.append(componentId);
230 ExportItemStatus status;
231 status.status = ExportItemStatus::Status::Failed;
232 status.errorMessage = errorMessage;
233 status.endTime = QDateTime::currentDateTime();
234 emit itemStatusChanged(componentId, status);
235 }
236 successCount = 0;
237 };
238
239 const auto abortExport = [&](const QString& errorMessage) {
240 qCritical() << "SymbolExportStage:" << errorMessage;
241 failCollectedSymbols(errorMessage);
242 finalizeProgress(false, errorMessage);
243 {
244 QMutexLocker locker(&m_progressMutex);
245 if (!m_progress.diagnostics.contains(errorMessage))
246 m_progress.diagnostics.append(errorMessage);
247 const ExportTypeProgress progressSnapshot = m_progress;
248 locker.unlock();
249 emit progressChanged(progressSnapshot);
250 }
251 m_tempManager.rollbackAll();
252 m_isExporting.store(false);
253 m_isRunning.store(false);
254 emit completed(0, failedIds.size(), 0);
255 };
256
257 QString libName =
258 m_options.libName.isEmpty() ? QStringLiteral("EasyKiConverter") : m_options.libName;
259
260 // 创建导出器（提前创建以获取格式自描述信息）
261 auto exporter = ExporterFactory::createSymbolExporter(m_options.targetFormat);
262 if (!exporter) {
263 abortExport(QStringLiteral("Failed to create symbol exporter for target format"));
264 return;
265 }
266
267 // 从导出器获取文件扩展名
268 const QString fileExt = exporter->libraryFileExtension();
269 QString fileName = libName + fileExt;
270 QString outputDir = m_options.outputPath;
271 if (outputDir.isEmpty()) {
272 outputDir = QDir::currentPath() + QStringLiteral("/export");
273 }
274 QString finalPath = outputDir + QDir::separator() + fileName;
275
276 QDir dir;
277 if (!dir.mkpath(outputDir)) {
278 abortExport(QStringLiteral("Failed to create output directory: %1").arg(outputDir));
279 return;
280 }
281
282 const bool isDirOutput = exporter->isDirectoryOutput();
283 const bool finalOutputExists = QFileInfo::exists(finalPath);
284
285 if (m_options.targetFormat == TargetEdaFormat::Xpedition && (m_options.updateMode || m_options.retryMode)) {
286 abortExport(QStringLiteral("Xpedition 符号 ZIP 暂不支持更新或重试模式，请选择覆盖导出"));
287 return;
288 }
289 if (m_options.targetFormat == TargetEdaFormat::Xpedition && finalOutputExists &&
290 !m_options.overwriteExistingFiles) {
291 abortExport(QStringLiteral("Xpedition 符号 ZIP 已存在且当前禁止覆盖: %1").arg(finalPath));
292 return;
293 }
294
295 if (m_options.targetFormat == TargetEdaFormat::Horizon && (m_options.updateMode || m_options.retryMode)) {
296 abortExport(QStringLiteral("Horizon 符号 Pool 暂不支持更新或重试模式，请选择覆盖导出"));
297 return;
298 }
299 if (m_options.targetFormat == TargetEdaFormat::Horizon && finalOutputExists && !m_options.overwriteExistingFiles) {
300 abortExport(QStringLiteral("Horizon 符号 Pool 已存在且当前禁止覆盖: %1").arg(finalPath));
301 return;
302 }
303
304 // 路径。
305 QString tempPath = isDirOutput ? m_tempManager.createTempDirectoryPath(fileName)
306 : m_tempManager.createSymbolTempPath(libName, fileExt);
307 if (tempPath.isEmpty()) {
308 abortExport(isDirOutput ? QStringLiteral("Failed to create temp directory path")
309 : QStringLiteral("Failed to create temp file path"));
310 return;
311 }
312
313 // 追加/更新到现有符号库时，先把最终文件复制到临时文件，再在临时文件上执行 merge。
314 if (!isDirOutput && finalOutputExists &&
315 (!m_options.overwriteExistingFiles || m_options.updateMode || m_options.retryMode)) {
316 const QString tempDirPath = QFileInfo(tempPath).absolutePath();
317 if (!QDir().mkpath(tempDirPath)) {
318 abortExport(QStringLiteral("Failed to create temp symbol library directory: %1").arg(tempDirPath));
319 return;
320 }
321
322 QFile tempFile(tempPath);
323 if (tempFile.exists() && !tempFile.remove()) {
324 abortExport(QStringLiteral("Failed to remove stale temp symbol library: %1").arg(tempPath));
325 return;
326 }
327
328 QFile sourceFile(finalPath);
329 if (!sourceFile.open(QIODevice::ReadOnly)) {
330 if (!QFileInfo::exists(finalPath)) {
331 qWarning()
332 << "SymbolExportStage: Existing symbol library disappeared before copy, creating new library:"
333 << finalPath;
334 } else {
335 abortExport(QStringLiteral("Failed to open existing symbol library for copy: %1").arg(finalPath));
336 return;
337 }
338 } else if (!tempFile.open(QIODevice::WriteOnly | QIODevice::Truncate)) {
339 abortExport(QStringLiteral("Failed to create temp symbol library for copy: %1").arg(tempPath));
340 return;
341 } else {
342 constexpr qint64 kChunkSize = 65536;
343 char buf[kChunkSize];
344 bool copyError = false;
345 while (!sourceFile.atEnd()) {
346 qint64 bytesRead = sourceFile.read(buf, kChunkSize);
347 if (bytesRead < 0) {
348 abortExport(QStringLiteral("Failed to read existing symbol library: %1").arg(finalPath));
349 copyError = true;
350 break;
351 }
352 if (tempFile.write(buf, bytesRead) != bytesRead) {
353 abortExport(QStringLiteral("Failed to write temp symbol library copy: %1").arg(tempPath));
354 copyError = true;
355 break;
356 }
357 }
358 if (copyError) {
359 return;
360 }
361 tempFile.close();
362 }
363 }
364
365 qDebug() << "SymbolExportStage: Exporting" << symbolList.size() << "symbols to temp:" << tempPath;
366 qDebug() << "SymbolExportStage: targetFormat:" << static_cast<int>(m_options.targetFormat);
367
368 bool exportSuccess = false;
369 ISymbolExporter::CompanionFiles companionFiles;
370 QString libraryDescription = m_options.symbolLibraryDescription;
371 {
372 // 只有目标库已经存在且禁止覆盖时才表示追加；首次创建空库不能被误判为追加模式。
373 const bool appendMode = finalOutputExists && !m_options.overwriteExistingFiles;
374 // 转换旧类型列表到 IR 类型
375 QList<IR::SymbolComponentIR> irSymbolList;
376 irSymbolList.reserve(symbolList.size());
377 for (const SymbolData& sd : symbolList) { // 当前符号数据。
378 irSymbolList.append(IR::toSymbolIR(sd));
379 }
380 exportSuccess = exporter->exportSymbolLibrary(
381 irSymbolList, libName, tempPath, appendMode, m_options.updateMode, libraryDescription);
382 companionFiles = exporter->companionFiles();
383 const QStringList exporterDiagnostics = exporter->diagnostics();
384 if (!exporterDiagnostics.isEmpty()) {
385 QMutexLocker locker(&m_progressMutex);
386 for (const QString& diagnostic : exporterDiagnostics) { // 当前导出诊断。
387 if (!m_progress.diagnostics.contains(diagnostic))
388 m_progress.diagnostics.append(diagnostic);
389 }
390 const ExportTypeProgress progressSnapshot = m_progress;
391 locker.unlock();
392 emit progressChanged(progressSnapshot);
393 }
394 }
395
396 if (m_cancelled.load()) {
397 m_tempManager.rollbackAll();
398 m_isExporting.store(false);
399 m_isRunning.store(false);
400 emit completed(0, symbolList.size(), 0);
401 return;
402 }
403
404 if (!exportSuccess) {
405 abortExport(QStringLiteral("Failed to export symbol library"));
406 return;
407 }
408
409 QStringList horizonDiagnostics;
410 if (m_options.targetFormat == TargetEdaFormat::Horizon &&
411 !HorizonPoolIntegration::updatePool(tempPath, horizonDiagnostics)) {
412 abortExport(QStringLiteral("Horizon Pool 官方更新失败，原有库未修改：%1")
413 .arg(horizonDiagnostics.join(QStringLiteral("; "))));
414 return;
415 }
416
417 QVector<TempFileManager::CommitItem> commitItems;
418 commitItems.append({tempPath, finalPath, isDirOutput});
419 for (auto it = companionFiles.cbegin(); it != companionFiles.cend(); ++it) { // 当前伴随文件记录。
420 if (isDirOutput) {
421 abortExport(QStringLiteral("目录型符号库不支持伴随文件: %1").arg(it.key()));
422 return;
423 }
424 const QString companionName = it.key();
425 if (companionName.isEmpty() || QFileInfo(companionName).fileName() != companionName ||
426 companionName == QFileInfo(finalPath).fileName()) {
427 abortExport(QStringLiteral("Invalid symbol companion file name: %1").arg(companionName));
428 return;
429 }
430 const QString companionFinalPath = QDir(outputDir).filePath(companionName);
431 // 主库不存在时没有可合并的旧内容，禁止覆盖残留的伴随文件，避免部分库被静默替换。
432 if (!finalOutputExists && !m_options.overwriteExistingFiles && QFile::exists(companionFinalPath)) {
433 abortExport(QStringLiteral("Symbol companion file already exists and overwrite is disabled: %1")
434 .arg(companionFinalPath));
435 return;
436 }
437 const QString companionTempPath =
438 QDir(m_tempManager.tempDirectory()).filePath(companionName);
439 QFile companionFile(companionTempPath);
440 if (!companionFile.open(QIODevice::WriteOnly | QIODevice::Truncate) ||
441 companionFile.write(it.value()) != it.value().size()) {
442 abortExport(QStringLiteral("Failed to write symbol companion file: %1").arg(companionName));
443 return;
444 }
445 companionFile.close();
446 m_tempManager.registerTempFile(companionTempPath);
447 commitItems.append({companionTempPath, companionFinalPath, false});
448 }
449 // 提交临时文件是否成功。
450 const bool commitSuccess = isDirOutput ? m_tempManager.commitDirectoryWithBackup(tempPath, finalPath)
451 : m_tempManager.commitBatch(commitItems);
452 if (!commitSuccess) {
453 abortExport(QStringLiteral("Failed to commit symbol library files"));
454 return;
455 }
456 qDebug() << "SymbolExportStage: Successfully exported to:" << finalPath;
457
458 if (m_options.targetFormat == TargetEdaFormat::Horizon) {
459 if (!HorizonPoolIntegration::registerPool(finalPath, horizonDiagnostics)) {
460 abortExport(QStringLiteral("Horizon Pool 已生成，但官方注册失败：%1")
461 .arg(horizonDiagnostics.join(QStringLiteral("; "))));
462 return;
463 }
464 }
465
466 // KiCad 特有：生成 sym-lib-table 并注册库
467 if (m_options.targetFormat == TargetEdaFormat::KiCad && !libraryDescription.isEmpty()) {
468 KiCadLibraryTableManager::registerSymbolLibrary(outputDir, libName, finalPath, libraryDescription);
469 }
470
471 // 注意：不在这里清理临时目录，由 ParallelExportService 统一管理
472 // 避免与其他 Stage（如 FootprintExportStage）的临时文件冲突
473
474 qDebug() << "SymbolExportStage: Completed. Success:" << successCount << "Failed:" << failedIds.size();
475
476 finalizeProgress(true, QString());
477
478 m_isExporting.store(false);
479 m_isRunning.store(false);
480 emit completed(successCount, failedIds.size(), skippedCount);
481}
