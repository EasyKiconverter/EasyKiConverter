20void ComponentListBatchCoordinator::addComponents(ComponentListViewModel& viewModel, const QStringList& componentIds) {
21 viewModel.clearAttentionHints();
22 // 保存规范化后待加入的元器件编号。
23 QStringList newIds;
24 for (const QString& rawId : componentIds) {
25 // 规范化当前输入的元器件编号。
26 QString id = rawId.trimmed().toUpper();
27 if (id.isEmpty()) {
28 continue;
29 }
30
31 if (!viewModel.validateComponentId(id)) {
32 // 从非标准输入中提取可能的元器件编号。
33 const QStringList extracted = viewModel.extractComponentIdFromText(id);
34 for (const QString& extractedId : extracted) {
35 if (!viewModel.componentExists(extractedId) && !newIds.contains(extractedId)) {
36 newIds.append(extractedId);
37 }
38 }
39 continue;
40 }
41
42 if (!viewModel.componentExists(id) && !newIds.contains(id)) {
43 newIds.append(id);
44 }
45 }
46
47 if (newIds.isEmpty()) {
48 return;
49 }
50
51 viewModel.m_bomImportComplete = false;
52 // 大量元器件导入时启用 BOM 导入模式，降低 UI 更新频率。
53 if (newIds.count() >= 20 && !viewModel.m_bomImportMode) {
54 viewModel.m_bomImportMode = true;
55 }
56
57 viewModel.m_pendingComponentIds.append(newIds);
58 viewModel.m_pendingBatchValidationCount += newIds.count();
59 if (!viewModel.m_batchAddTimer->isActive()) {
60 viewModel.m_batchAddTimer->start();
61 }
62}

68void ComponentListBatchCoordinator::processNext(ComponentListViewModel& viewModel) {
69 if (viewModel.m_pendingComponentIds.isEmpty()) {
70 viewModel.m_batchAddTimer->stop();
71 return;
72 }
73
74 // 保存本次从待处理队列取出的编号批次。
75 QStringList batch;
76 // 计算本次批处理的编号数量。
77 const int count = qMin(ComponentListViewModel::BATCH_ADD_SIZE, viewModel.m_pendingComponentIds.size());
78 for (int i = 0; i < count; ++i) {
79 batch.append(viewModel.m_pendingComponentIds.takeFirst());
80 }
81
82 // 保存本批次待插入的列表项数量。
83 const int pendingCount = batch.count();
84 // 保存列表插入起始位置。
85 int startIndex;
86 // 标记当前批次是否为最后一批。
87 bool isLastBatch;
88 {
89 QMutexLocker locker(&viewModel.m_listMutex);
90 startIndex = viewModel.m_componentList.count();
91 isLastBatch = viewModel.m_pendingComponentIds.isEmpty();
92 }
93
94 // 在锁外创建列表项，避免构造对象时持有共享列表锁。
95 QList<ComponentListItemData*> newItems;
96 for (const QString& id : batch) {
97 // 创建并初始化当前编号对应的列表项。
98 auto item = new ComponentListItemData(id, &viewModel);
99 item->setFetching(true);
100 item->setValid(false);
101 item->setValidationPhase("validating");
102 newItems.append(item);
103 }
104
105 viewModel.beginInsertRows(QModelIndex(), startIndex, startIndex + pendingCount - 1);
106 {
107 QMutexLocker locker(&viewModel.m_listMutex);
108 for (auto item : newItems) {
109 viewModel.m_componentList.append(item);
110 viewModel.m_componentIdIndex.insert(item->componentId(), viewModel.m_componentList.count() - 1);
111 }
112 }
113 viewModel.endInsertRows();
114
115 if (isLastBatch) {
116 // 保存本轮批处理产生的验证任务数量。
117 const int validationCount = viewModel.m_pendingBatchValidationCount;
118 viewModel.m_pendingBatchValidationCount = 0;
119
120 // BOM 导入结束后恢复普通 UI 更新，并合并已累积的状态变更。
121 if (viewModel.m_bomImportMode) {
122 // 标记 BOM 导入期间是否积累了列表更新。
123 const bool hadPendingUpdates = viewModel.m_bomImportPendingUpdates > 0;
124 viewModel.m_bomImportMode = false;
125 if (hadPendingUpdates) {
126 viewModel.m_bomImportPendingUpdates = 0;
127 viewModel.scheduleListUpdate();
128 }
129 }
130
131 if (validationCount > 0) {
132 if (viewModel.m_validationStateManager->pendingCount() > 0 || !viewModel.m_validationQueue.isEmpty() ||
133 viewModel.m_validationQueue.hasInFlight()) {
134 viewModel.m_validationStateManager->addValidation(validationCount);
135 } else {
136 viewModel.m_validationStateManager->startValidation(validationCount);
137 }
138 }
139
140 viewModel.m_bomImportComplete = true;
141 viewModel.startValidationQueue();
142 }
143
144 viewModel.scheduleListUpdate();
145}

152void ComponentListBatchCoordinator::selectBomFile(ComponentListViewModel& viewModel, const QString& filePath) {
153 qDebug() << "BOM file selected:" << filePath;
154
155 if (viewModel.m_bomFilePath != filePath) {
156 viewModel.m_bomFilePath = filePath;
157 emit viewModel.bomFilePathChanged();
158 }
159
160 // 保存转换为本地文件系统格式后的 BOM 路径。
161 QString localPath = filePath;
162 if (filePath.startsWith("file:///")) {
163 localPath = QUrl(filePath).toLocalFile();
164 qDebug() << "Converted URL to local path:" << localPath;
165 }
166
167 viewModel.m_bomResult = "Parsing BOM file...";
168 emit viewModel.bomResultChanged();
169
170 // 保存后台 BOM 解析任务。
171 QFuture<QStringList> future =
172 QtConcurrent::run([&viewModel, localPath]() { return viewModel.m_service->parseBomFile(localPath); });
173
174 // 监视后台 BOM 解析任务的完成状态。
175 auto* watcher = new QFutureWatcher<QStringList>(&viewModel);
176 QPointer<ComponentListViewModel> self(&viewModel);
177 QObject::connect(watcher, &QFutureWatcher<QStringList>::finished, &viewModel, [self, watcher, localPath]() {
178 if (!self) {
179 return;
180 }
181 // 读取后台解析得到的元器件编号。
182 const QStringList componentIds = watcher->result();
183 watcher->deleteLater();
184
185 if (componentIds.isEmpty()) {
186 self->m_bomResult = "No valid component IDs found in BOM file";
187 qWarning() << "No component IDs found in BOM file:" << localPath;
188 emit self->bomResultChanged();
189 return;
190 }
191
192 // 保存去重后待加入列表的编号。
193 QStringList newIds;
194 // 统计已存在而跳过的编号数量。
195 int skipped = 0;
196 for (const QString& id : componentIds) {
197 if (!self->componentExists(id)) {
198 newIds.append(id);
199 } else {
200 ++skipped;
201 }
202 }
203
204 if (!newIds.isEmpty()) {
205 ComponentListBatchCoordinator::addComponents(*self, newIds);
206 }
207
208 const QString resultMessage =
209 QString("BOM file imported: %1 components added, %2 skipped").arg(newIds.count()).arg(skipped);
210 self->m_bomResult = resultMessage;
211 qDebug() << resultMessage;
212 emit self->bomResultChanged();
213 });
214 watcher->setFuture(future);
215}
