210bool readBinaryRecordMetadata(const QByteArray& payload,
211 int* recordType,
212 int* ownerPartId,
213 int* ownerPartDisplayMode = nullptr) {
214 if (recordType == nullptr || ownerPartId == nullptr || payload.size() < 7)
215 return false;
216 AltiumBinaryReader reader(payload);
217 qint32 type = -1; // 记录类型。
218 quint8 unknown = 0; // 记录头部保留字节。
219 qint16 owner = -1; // 所属部件编号。
220 if (!reader.readInt32(&type) || !reader.readUInt8(&unknown) || !reader.readInt16(&owner))
221 return false;
222 *recordType = type;
223 *ownerPartId = owner;
224 if (ownerPartDisplayMode != nullptr)
225 *ownerPartDisplayMode = -1;
226 if (ownerPartDisplayMode != nullptr && type == 2 && payload.size() >= 8) {
227 AltiumBinaryReader displayReader(payload);
228 qint32 ignoredType = -1; // 显示模式读取时忽略的记录类型。
229 quint8 ignoredUnknown = 0; // 显示模式读取时忽略的保留字节。
230 qint16 ignoredOwner = -1; // 显示模式读取时忽略的部件编号。
231 quint8 displayMode = 0; // 部件显示模式。
232 if (!displayReader.readInt32(&ignoredType) || !displayReader.readUInt8(&ignoredUnknown) ||
233 !displayReader.readInt16(&ignoredOwner) || !displayReader.readUInt8(&displayMode))
234 return false;

236 }

240constexpr qint64 kMaxImageDecompressedSize = 512LL * 1024LL * 1024LL;

247bool isValidZlibPayload(const QByteArray& compressedData) {
248 if (compressedData.isEmpty())
249 return false;
250
251 z_stream stream{}; // zlib 解压状态。
252 if (inflateInit(&stream) != Z_OK)
253 return false;
254
255 stream.next_in = reinterpret_cast<Bytef*>(const_cast<char*>(compressedData.constData()));
256 stream.avail_in = static_cast<uInt>(compressedData.size());
257 QByteArray buffer(8192, Qt::Uninitialized); // 解压输出缓冲区。
258 qint64 decompressedSize = 0; // 已解压字节数。
259 int result = Z_OK; // 最近一次 zlib 操作结果。
260 while (result == Z_OK) {
261 stream.next_out = reinterpret_cast<Bytef*>(buffer.data());
262 stream.avail_out = static_cast<uInt>(buffer.size());
263 result = inflate(&stream, Z_NO_FLUSH);
264 decompressedSize += buffer.size() - stream.avail_out;
265 if (decompressedSize > kMaxImageDecompressedSize) {
266 inflateEnd(&stream);
267 return false;
268 }
269 }
270
271 const bool valid =
272 result == Z_STREAM_END && stream.avail_in == 0 && decompressedSize > 0; // 解压结果是否完整有效。
273 inflateEnd(&stream);
274 return valid;
275}

297bool AltiumSchLibReader::open(const QString& filePath) {
298 m_components.clear();
299 m_headerParameters.clear();
300 m_fonts.clear();
301 m_errorMessage.clear();
302
303 if (!m_oleReader.open(filePath))
304 return fail(m_oleReader.errorString());
305
306 QByteArray headerData; // FileHeader 流数据。
307 if (!m_oleReader.readStream(QStringLiteral("FileHeader"), &headerData))
308 return fail(QStringLiteral("SchLib 缺少 FileHeader 流"));
309
310 AltiumBinaryReader headerReader(headerData);
311 if (!headerReader.readCStringParameterBlock(&m_headerParameters))
312 return fail(QStringLiteral("SchLib FileHeader 参数块无效: %1").arg(headerReader.errorString()));
313
314 bool fontCountOk = true; // 字体数量转换是否成功。
315 const int fontCount = m_headerParameters.contains(QStringLiteral("FontIdCount")) // 字体表条目数量。
316 ? m_headerParameters.value(QStringLiteral("FontIdCount")).toInt(&fontCountOk)
317 : 0;
318 if (!fontCountOk || fontCount < 0)
319 return fail(QStringLiteral("SchLib FileHeader 的 FontIdCount 无效"));
320 m_fonts.reserve(fontCount);
321 for (int i = 1; i <= fontCount; ++i) {
322 FontInfo font; // 当前字体表条目。
323 font.name = m_headerParameters.value(QStringLiteral("FontName%1").arg(i));
324 font.size = parameterInt(m_headerParameters, QStringLiteral("Size%1").arg(i), 0);
325 font.bold = m_headerParameters.value(QStringLiteral("Bold%1").arg(i)) == QStringLiteral("T");
326 font.italic = m_headerParameters.value(QStringLiteral("Italic%1").arg(i)) == QStringLiteral("T");
327 font.underline = m_headerParameters.value(QStringLiteral("Underline%1").arg(i)) == QStringLiteral("T");
328 if (font.name.isEmpty() || font.size <= 0)
329 return fail(QStringLiteral("SchLib FileHeader 的字体 %1 无效").arg(i));
330 m_fonts.append(font);
331 }
332
333 bool componentCountOk = false; // 组件数量转换是否成功。
334 const int componentCount =
335 m_headerParameters.value(QStringLiteral("COMPCOUNT")).toInt(&componentCountOk); // 组件数量。
336 if (!componentCountOk || componentCount < 0)
337 return fail(QStringLiteral("SchLib FileHeader 的 COMPCOUNT 无效"));
338
339 int32_t serializedCount = 0; // FileHeader 序列化的组件数量。
340 if (!headerReader.readInt32(&serializedCount) || serializedCount != componentCount)
341 return fail(QStringLiteral("SchLib FileHeader 的组件数量不一致"));
342
343 QVector<QString> names; // 组件名称列表。
344 names.reserve(componentCount);
345 QSet<QString> componentNames; // 大小写折叠后的组件名称集合。
346 for (int i = 0; i < componentCount; ++i) {
347 QString name; // 当前组件名称。
348 if (!headerReader.readStringBlock(&name) || name.isEmpty())
349 return fail(QStringLiteral("SchLib FileHeader 的组件名称无效"));
350 const QString foldedName = name.toCaseFolded(); // 用于重复检查的折叠名称。
351 if (componentNames.contains(foldedName))
352 return fail(QStringLiteral("SchLib FileHeader 的组件名称重复: %1").arg(name));
353 componentNames.insert(foldedName);
354 names.append(name);
355 }
356 if (headerReader.hasError() || headerReader.remaining() != 0)
357 return fail(QStringLiteral("SchLib FileHeader 末尾包含无效数据"));
358
359 QMap<QString, QString> sectionKeys; // 组件名称到 OLE 节名称的映射。
360 if (m_oleReader.containsStream(QStringLiteral("SectionKeys"))) {
361 QByteArray sectionKeyData; // SectionKeys 流数据。
362 if (!m_oleReader.readStream(QStringLiteral("SectionKeys"), &sectionKeyData))
363 return fail(QStringLiteral("无法读取 SchLib SectionKeys 流"));
364 AltiumBinaryReader sectionKeyReader(sectionKeyData);
365 QMap<QString, QString> parameters; // SectionKeys 参数表。
366 if (!sectionKeyReader.readCStringParameterBlock(&parameters))
367 return fail(QStringLiteral("SchLib SectionKeys 参数块无效: %1").arg(sectionKeyReader.errorString()));
368 bool keyCountOk = false; // 映射数量转换是否成功。
369 const int keyCount = parameters.value(QStringLiteral("KeyCount")).toInt(&keyCountOk); // 映射数量。
370 if (!keyCountOk || keyCount < 0)
371 return fail(QStringLiteral("SchLib SectionKeys 的 KeyCount 无效"));
372 QSet<QString> mappedComponents; // 已映射的组件名称集合。
373 QSet<QString> mappedSectionKeys; // 已使用的节名称集合。
374 for (int i = 0; i < keyCount; ++i) {
375 const QString libRef = parameters.value(QStringLiteral("LibRef%1").arg(i)); // 组件名称。
376 const QString sectionKey = parameters.value(QStringLiteral("SectionKey%1").arg(i)); // OLE 节名称。
377 if (libRef.isEmpty() || sectionKey.isEmpty())
378 return fail(QStringLiteral("SchLib SectionKeys 映射不完整"));
379 if (!names.contains(libRef))
380 return fail(QStringLiteral("SchLib SectionKeys 包含未知组件: %1").arg(libRef));
381 if (mappedComponents.contains(libRef.toCaseFolded()) ||
382 mappedSectionKeys.contains(sectionKey.toCaseFolded()))
383 return fail(QStringLiteral("SchLib SectionKeys 映射重复: %1").arg(libRef));
384 mappedComponents.insert(libRef.toCaseFolded());
385 mappedSectionKeys.insert(sectionKey.toCaseFolded());
386 sectionKeys.insert(libRef, sectionKey);
387 }
388 if (sectionKeyReader.hasError() || sectionKeyReader.remaining() != 0)
389 return fail(QStringLiteral("SchLib SectionKeys 末尾包含无效数据"));
390 }
391
392 QSet<QString> resolvedSectionKeys; // 已解析的组件存储键集合。
393 for (int i = 0; i < names.size(); ++i) {
394 const QString& name = names.at(i); // 当前组件名称。
395 const QString sectionKey = sectionKeys.value(name, name); // 当前组件存储键。
396 const QString foldedSectionKey = sectionKey.toCaseFolded(); // 用于重复检查的折叠存储键。
397 if (resolvedSectionKeys.contains(foldedSectionKey))
398 return fail(QStringLiteral("SchLib 组件存储键重复: %1").arg(sectionKey));
399 resolvedSectionKeys.insert(foldedSectionKey);
400 const QString dataPath = sectionKey + QStringLiteral("/Data"); // 组件 Data 流路径。
401 if (!m_oleReader.containsStream(dataPath))
402 return fail(QStringLiteral("SchLib 缺少组件 Data 流: %1").arg(dataPath));
403 const int serializedPartCount =
404 parameterInt(m_headerParameters, QStringLiteral("PARTCOUNT%1").arg(i), 2); // 序列化部件数量。
405 if (serializedPartCount < 1)
406 return fail(QStringLiteral("SchLib 组件 %1 的 PARTCOUNT 无效").arg(name));
407 m_components.append({name, sectionKey, qMax(1, serializedPartCount - 1)});
408 }
409 return true;
410}

477bool AltiumSchLibReader::readComponentRecords(int componentIndexValue, QVector<Record>* records) const {
478 if (records == nullptr) {
479 m_errorMessage = QStringLiteral("读取 SchLib 记录时输出容器为空");
480 return false;
481 }
482 records->clear();
483 m_errorMessage.clear();
484
485 const auto failRead = [this, records](const QString& message) {
486 records->clear();
487 m_errorMessage = message;
488 return false;
489 };
490
491 QByteArray data; // 组件 Data 流数据。
492 if (!readComponentData(componentIndexValue, &data))
493 return failRead(QStringLiteral("无法读取 SchLib 组件 Data 流"));
494 AltiumBinaryReader reader(data);
495 int lastIndexInSheet = -1; // 最近一次记录的图纸索引。
496 int nextContentIndex = -1; // 期望的下一个内容索引。
497 while (reader.remaining() > 0) {
498 const int startPosition = reader.position(); // 当前记录起始偏移。
499 QByteArray payload; // 当前记录载荷。
500 uint8_t flags = 0; // 当前记录标志位。
501 if (!reader.readBlock(&payload, &flags))
502 return failRead(QStringLiteral("读取 SchLib 组件记录失败: %1").arg(reader.errorString()));
503 if (payload.isEmpty())
504 return failRead(QStringLiteral("SchLib 组件记录为空，偏移量 %1").arg(startPosition));
505 Record record; // 当前组件记录。
506 record.flags = flags;
507 record.payload = payload;
508 record.encoded = data.mid(startPosition, reader.position() - startPosition);
509 if (payload.startsWith('|')) {
510 AltiumBinaryReader parameterReader(payload);
511 if (!parameterReader.parseCStringParameterData(payload, &record.parameters)) {
512 return failRead(QStringLiteral("SchLib 组件参数记录无效，偏移量 %1: %2")
513 .arg(startPosition)
514 .arg(parameterReader.errorString()));
515 }
516 record.hasParameters = true;
517 const auto readOptionalParameterAlias =
518 [&record](const QString& primary, const QString& alias, int fallback, int* value) {
519 const bool hasPrimary = record.parameters.contains(primary); // 是否存在主字段。
520 const bool hasAlias = record.parameters.contains(alias); // 是否存在别名字段。
521 if (hasPrimary && hasAlias) {
522 int primaryValue = fallback; // 主字段解析值。
523 int aliasValue = fallback; // 别名字段解析值。
524 if (!readOptionalParameterInt(record.parameters, primary, fallback, &primaryValue) ||
525 !readOptionalParameterInt(record.parameters, alias, fallback, &aliasValue) ||
526 primaryValue != aliasValue)
527 return false;
528 *value = primaryValue;
529 return true;
530 }
531 const QString selectedName = hasPrimary ? primary : alias; // 实际使用的字段名称。
532 return readOptionalParameterInt(record.parameters, selectedName, fallback, value);
533 };
534 if (!readOptionalParameterInt(record.parameters, QStringLiteral("RECORD"), -1, &record.recordType) ||
535 !readOptionalParameterAlias(
536 QStringLiteral("OWNERPARTID"), QStringLiteral("OwnerPartId"), -1, &record.ownerPartId) ||
537 !readOptionalParameterAlias(QStringLiteral("OWNERPARTDISPLAYMODE"),
538 QStringLiteral("OwnerPartDisplayMode"),
539 -1,
540 &record.ownerPartDisplayMode) ||
541 !readOptionalParameterInt(
542 record.parameters, QStringLiteral("IndexInSheet"), -1, &record.indexInSheet)) {
543 return failRead(QStringLiteral("SchLib 组件参数记录的数值字段无效，偏移量 %1").arg(startPosition));
544 }
545 if (record.ownerPartDisplayMode < -1)
546 return failRead(
547 QStringLiteral("SchLib 组件参数记录的 OWNERPARTDISPLAYMODE 无效，偏移量 %1").arg(startPosition));
548 const bool hasFontId = record.parameters.contains(QStringLiteral("FontID")) || // 是否存在字体编号字段。
549 record.parameters.contains(QStringLiteral("FONTID"));
550 if (hasFontId) {
551 int mixedCaseFontId = -1; // 混合大小写字段中的字体编号。
552 int upperCaseFontId = -1; // 全大写字段中的字体编号。
553 if (!readOptionalParameterInt(record.parameters, QStringLiteral("FontID"), -1, &mixedCaseFontId) ||
554 !readOptionalParameterInt(record.parameters, QStringLiteral("FONTID"), -1, &upperCaseFontId)) {
555 return failRead(QStringLiteral("SchLib 组件参数记录的 FONTID 无效，偏移量 %1").arg(startPosition));
556 }
557 if (mixedCaseFontId >= 0 && upperCaseFontId >= 0 && mixedCaseFontId != upperCaseFontId)
558 return failRead(
559 QStringLiteral("SchLib 组件参数记录的 FONTID 字段不一致，偏移量 %1").arg(startPosition));
560 record.fontId = mixedCaseFontId >= 0 ? mixedCaseFontId : upperCaseFontId;
561 if (record.fontId < 1 || record.fontId > m_fonts.size())
562 return failRead(
563 QStringLiteral("SchLib 组件参数记录的 FONTID 超出字体表范围，偏移量 %1").arg(startPosition));
564 }
565 if (record.ownerPartId < -1)
566 return failRead(
567 QStringLiteral("SchLib 组件参数记录的 OWNERPARTID 小于 -1，偏移量 %1").arg(startPosition));
568 if (record.ownerPartId >= 0 && record.ownerPartId > m_components.at(componentIndexValue).partCount) {
569 return failRead(
570 QStringLiteral("SchLib 组件参数记录的 OWNERPARTID 超出部件范围，偏移量 %1").arg(startPosition));
571 }
572 QString geometryError; // 几何参数校验错误说明。
573 if (!validateGraphicParameters(record.parameters, record.recordType, &geometryError))
574 return failRead(
575 QStringLiteral("SchLib 组件几何记录无效（%1），偏移量 %2").arg(geometryError).arg(startPosition));
576 if (record.indexInSheet < -1 || (record.indexInSheet >= 0 && record.indexInSheet <= lastIndexInSheet)) {
577 return failRead(
578 QStringLiteral("SchLib 组件参数记录的 IndexInSheet 非递增，偏移量 %1").arg(startPosition));
579 }
580 if (record.indexInSheet >= 0)
581 lastIndexInSheet = record.indexInSheet;
582 } else {
583 if (!readBinaryRecordMetadata(
584 payload, &record.recordType, &record.ownerPartId, &record.ownerPartDisplayMode)) {
585 return failRead(QStringLiteral("SchLib 二进制记录元数据无效，偏移量 %1").arg(startPosition));
586 }
587 if (record.ownerPartId < -1)
588 return failRead(
589 QStringLiteral("SchLib 二进制记录的 OWNERPARTID 小于 -1，偏移量 %1").arg(startPosition));
590 if (record.ownerPartId >= 0 && record.ownerPartId > m_components.at(componentIndexValue).partCount) {
591 return failRead(
592 QStringLiteral("SchLib 二进制记录的 OWNERPARTID 超出部件范围，偏移量 %1").arg(startPosition));
593 }
594 }
595
596 // 图元参数记录和二进制引脚共享同一个内容序号。首条内容记录的
597 // IndexInSheet=0 可以省略，但后续记录必须紧接前一条内容记录；
598 // 否则写入端可能漏掉图元，Altium 的绘制顺序也会发生偏移。
599 const bool hasOwnerPartField =
600 record.parameters.contains(QStringLiteral("OWNERPARTID")) || // 是否存在部件字段。
601 record.parameters.contains(QStringLiteral("OwnerPartId"));
602 // 判断当前参数记录是否参与内容序号连续性校验。
603 const bool isIndexedParameter = record.hasParameters && hasOwnerPartField && record.recordType != 1 &&
604 record.recordType != 34 && (record.recordType != 41 || record.ownerPartId >= 1);
605 const bool isContentRecord =
606 (!record.hasParameters && record.recordType == 2) || isIndexedParameter; // 是否为有序内容记录。
607 if (isContentRecord) {
608 if (nextContentIndex < 0) {
609 nextContentIndex = record.indexInSheet >= 0 ? record.indexInSheet + 1 : 1;
610 } else if (!record.hasParameters) {
611 ++nextContentIndex;
612 } else if (record.indexInSheet < 0 || record.indexInSheet != nextContentIndex) {
613 return failRead(
614 QStringLiteral("SchLib 组件的内容记录 IndexInSheet 不连续，偏移量 %1").arg(startPosition));
615 } else {
616 ++nextContentIndex;
617 }
618 }
619 records->append(record);
620 }
621 if (records->isEmpty())
622 return failRead(QStringLiteral("SchLib 组件 Data 流不包含记录"));
623 return true;
624}

641bool AltiumSchLibReader::readImageStorage(QVector<ImageStorageEntry>* entries) const {
642 if (entries == nullptr) {
643 m_errorMessage = QStringLiteral("读取 SchLib 图片 Storage 时输出容器为空");
644 return false;
645 }
646 entries->clear();
647 m_errorMessage.clear();
648
649 if (!m_oleReader.containsStream(QStringLiteral("Storage")))
650 return true;
651
652 QByteArray storage; // 图片 Storage 流数据。
653 if (!m_oleReader.readStream(QStringLiteral("Storage"), &storage)) {
654 m_errorMessage = QStringLiteral("无法读取 SchLib 图片 Storage 流");
655 return false;
656 }
657
658 AltiumBinaryReader reader(storage);
659 QMap<QString, QString> storageParameters; // Storage 头部参数表。
660 if (!reader.readCStringParameterBlock(&storageParameters)) {
661 m_errorMessage = QStringLiteral("SchLib 图片 Storage 头部无效: %1").arg(reader.errorString());
662 return false;
663 }
664 bool expectedCountOk = true; // 声明数量转换是否成功。
665 const int expectedCount = storageParameters.contains(QStringLiteral("Weight")) // 声明的图片条目数量。
666 ? storageParameters.value(QStringLiteral("Weight")).toInt(&expectedCountOk)
667 : -1;
668 if (!expectedCountOk || expectedCount < 0) {
669 m_errorMessage = QStringLiteral("SchLib 图片 Storage 的 Weight 无效");
670 return false;
671 }
672
673 QSet<QString> names; // 已读取图片名称集合。
674 while (reader.remaining() > 0) {
675 QByteArray entryData; // 图片条目载荷。
676 quint8 flags = 0; // 图片条目标志位。
677 if (!reader.readBlock(&entryData, &flags)) {
678 m_errorMessage = QStringLiteral("读取 SchLib 图片 Storage 条目失败: %1").arg(reader.errorString());
679 entries->clear();
680 return false;
681 }
682 if (flags != 1) {
683 m_errorMessage = QStringLiteral("SchLib 图片 Storage 条目标志无效: %1").arg(flags);
684 entries->clear();
685 return false;
686 }
687
688 AltiumBinaryReader entryReader(entryData);
689 quint8 marker = 0; // 图片条目标记字节。
690 quint8 nameSize = 0; // 文件名字节数。
691 QByteArray nameData; // 文件名原始字节。
692 quint32 compressedSize = 0; // 压缩数据字节数。
693 QByteArray compressedData; // 压缩图片数据。
694 if (!entryReader.readUInt8(&marker) || marker != 0xD0 || !entryReader.readUInt8(&nameSize) || nameSize == 0 ||
695 !entryReader.readBytes(nameSize, &nameData) || nameData.contains('\0') ||
696 !entryReader.readUInt32(&compressedSize) || compressedSize == 0 ||
697 compressedSize > static_cast<quint32>(entryReader.remaining()) ||
698 !entryReader.readBytes(static_cast<int>(compressedSize), &compressedData) || entryReader.remaining() != 0) {
699 m_errorMessage = QStringLiteral("SchLib 图片 Storage 条目内容无效");
700 entries->clear();
701 return false;
702 }
703
704 if (!isValidZlibPayload(compressedData)) {
705 m_errorMessage = QStringLiteral("SchLib 图片 Storage 压缩数据无效");
706 entries->clear();
707 return false;
708 }
709
710 const QString name = QString::fromLocal8Bit(nameData); // 图片文件名。
711 if (name.toLocal8Bit() != nameData) {
712 m_errorMessage = QStringLiteral("SchLib 图片 Storage 文件名编码无效");
713 entries->clear();
714 return false;
715 }
716 const QString foldedName = name.toCaseFolded(); // 用于重复检查的折叠文件名。
717 if (!AltiumWriterUtils::isValidImageStorageName(name) || names.contains(foldedName)) {
718 m_errorMessage = QStringLiteral("SchLib 图片 Storage 包含重复或非法文件名: %1").arg(name);
719 entries->clear();
720 return false;
721 }
722 names.insert(foldedName);
723 entries->append({flags, name, compressedData});
724 }
725
726 if (expectedCount >= 0 && expectedCount != entries->size()) {
727 m_errorMessage =
728 QStringLiteral("SchLib 图片 Storage 数量不一致，声明 %1，实际 %2").arg(expectedCount).arg(entries->size());
729 entries->clear();
730 return false;
731 }
732 return true;
733}
