24AltiumSchText AltiumSymbolAnnotationConverter::convertText(const IR::SymbolTextIR& text) {
25 AltiumSchText altiumText; // 转换后的 Altium 文本记录。
26 altiumText.locationX = AltiumCoord::mmToRaw(text.position.x());
27 altiumText.locationY = AltiumCoord::mmToRaw(text.position.y());
28 altiumText.text = text.text;
29 altiumText.fontId = 0;
30 altiumText.fontName = text.fontFamily;
31 altiumText.fontSizeMm = text.fontSizeMm;
32 altiumText.bold = text.bold;
33 altiumText.italic = text.italic;
34 altiumText.anchor = text.anchor.trimmed().isEmpty() ? QStringLiteral("middle") : text.anchor.trimmed();
35 altiumText.color = toAltiumColor(text.color);
36 altiumText.isHidden = !text.visible;
37 altiumText.orientation = toAltiumOrientation(text.rotation);
38 altiumText.ownerPartId = toAltiumOwnerPartId(text.partIndex);
39 return altiumText;
40}

47AltiumSchTextFrame AltiumSymbolAnnotationConverter::convertTextFrame(const IR::SymbolTextFrameIR& frame) {
48 AltiumSchTextFrame altiumFrame; // 转换后的 Altium 文本框记录。
49 altiumFrame.locationX = AltiumCoord::mmToRaw(frame.x0);
50 altiumFrame.locationY = AltiumCoord::mmToRaw(frame.y0);
51 altiumFrame.cornerX = AltiumCoord::mmToRaw(frame.x1);
52 altiumFrame.cornerY = AltiumCoord::mmToRaw(frame.y1);
53 altiumFrame.lineWidth = AltiumCoord::lineWidthMmToIndex(frame.strokeWidth);
54 altiumFrame.lineStyle = toAltiumLineStyle(frame.strokeStyle);
55 altiumFrame.color = toAltiumColor(frame.strokeColor);
56 altiumFrame.areaColor = frame.isFilled ? toAltiumColor(frame.fillColor) : 0;
57 altiumFrame.textColor = toAltiumColor(frame.textColor);
58 altiumFrame.fontName = frame.fontFamily;
59 altiumFrame.fontSizeMm = frame.fontSizeMm;
60 altiumFrame.bold = frame.bold;
61 altiumFrame.italic = frame.italic;
62 altiumFrame.fontId = qMax(0, frame.fontId);
63 altiumFrame.orientation = ((frame.orientation % 4) + 4) % 4;
64 altiumFrame.alignment = qMax(0, frame.alignment);
65 altiumFrame.textMargin = AltiumCoord::mmToRaw(qMax(0.0, frame.textMargin));
66 altiumFrame.text = frame.text;
67 altiumFrame.isSolid = frame.isFilled;
68 altiumFrame.showBorder = frame.showBorder;
69 altiumFrame.wordWrap = frame.wordWrap;
70 altiumFrame.clipToRect = frame.clipToRect;
71 altiumFrame.transparent = frame.transparent;
72 altiumFrame.ownerPartId = toAltiumOwnerPartId(frame.partIndex);
73 return altiumFrame;
74}
