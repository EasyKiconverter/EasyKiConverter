195void ComponentListItemData::updatePreviewImagesCache() const {
196 m_previewImagesCache.clear();
197 for (const QImage& image : m_previewImages) {
198 if (image.isNull()) {
199 m_previewImagesCache.append(QVariant());
200 continue;
201 }
202 QByteArray byteArray; // 当前预览图的 PNG 编码字节。
203 QBuffer buffer(&byteArray);
204 buffer.open(QIODevice::WriteOnly);
205 image.save(&buffer, "PNG");
206 m_previewImagesCache.append(QString::fromLatin1(byteArray.toBase64().data()));
207 }
208}

225int ComponentListItemData::previewImageCount() const {
226 int cacheCount = 0; // 编码缓存中的有效预览图数量。
227 for (const QVariant& cached : m_previewImagesCache) {
228 if (cached.isValid() && !cached.toString().isEmpty()) {
229 ++cacheCount;
230 }
231 }
232
233 int rawCount = 0; // 原始图像列表中的有效预览图数量。
234 for (const QImage& image : m_previewImages) {
235 if (!image.isNull()) {
236 ++rawCount;
237 }
238 }
239
240 return qMax(cacheCount, rawCount);
241}
