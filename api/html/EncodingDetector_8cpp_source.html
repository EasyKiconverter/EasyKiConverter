11bool isValidUtf8(const QByteArray& data) {
12 int remaining = 0;
13 for (unsigned char byte : data) {
14 if (remaining == 0) {
15 if (byte <= 0x7F)
16 continue;
17 if (byte >= 0xC2 && byte <= 0xDF)
18 remaining = 1;
19 else if (byte >= 0xE0 && byte <= 0xEF)
20 remaining = 2;
21 else if (byte >= 0xF0 && byte <= 0xF4)
22 remaining = 3;
23 else
24 return false;
25 } else {
26 if (byte < 0x80 || byte > 0xBF)
27 return false;
28 --remaining;
29 }
30 }
31 return remaining == 0;
32}

40QString decodeUtf16(const QByteArray& data, bool littleEndian) {
41 const int byteCount = data.size() - 2;
42 if (byteCount < 0 || byteCount % 2 != 0)
43 return {};
44 QString result;
45 result.reserve(byteCount / 2);
46 for (int index = 2; index < data.size(); index += 2) {
47 const quint16 first = static_cast<unsigned char>(data.at(index));
48 const quint16 second = static_cast<unsigned char>(data.at(index + 1));
49 const quint16 value = littleEndian ? first | (second << 8) : (first << 8) | second;
50 result.append(QChar(value));
51 }
52 return result;
53}

62EncodingDetectionResult EncodingDetector::detect(const QByteArray& data) {
63 if (data.startsWith(QByteArray::fromHex("EFBBBF")))
64 return {TextEncoding::Utf8, true, 100};
65 if (data.startsWith(QByteArray::fromHex("FFFE")))
66 return {TextEncoding::Utf16LittleEndian, data.size() >= 2 && (data.size() - 2) % 2 == 0, 100};
67 if (data.startsWith(QByteArray::fromHex("FEFF")))
68 return {TextEncoding::Utf16BigEndian, data.size() >= 2 && (data.size() - 2) % 2 == 0, 100};
69 if (isValidUtf8(data))
70 return {TextEncoding::Utf8, true, 90};
71 if (!data.isEmpty())
72 return {TextEncoding::Latin1, true, 30};
73 return {TextEncoding::Utf8, true, 90};
74}

83QString EncodingDetector::decode(const QByteArray& data, ParseDiagnostics* diagnostics, const QString& filePath) {
84 if (diagnostics)
85 diagnostics->setFilePath(filePath);
86 const EncodingDetectionResult result = detect(data);
87 if (!result.valid) {
88 if (diagnostics)
89 diagnostics->add(ParseSeverity::Error, ParseScope::File, QStringLiteral("文本编码标记不完整"));
90 return {};
91 }
92 // 根据已确认的编码分支解码，未知编码不会静默转换。
93 switch (result.encoding) {
94 case TextEncoding::Utf8: {
95 const int offset = data.startsWith(QByteArray::fromHex("EFBBBF")) ? 3 : 0;
96 return QString::fromUtf8(data.constData() + offset, data.size() - offset);
97 }
98 case TextEncoding::Utf16LittleEndian:
99 return decodeUtf16(data, true);
100 case TextEncoding::Utf16BigEndian:
101 return decodeUtf16(data, false);
102 case TextEncoding::Latin1:
103 if (diagnostics)
104 diagnostics->add(ParseSeverity::Warning,
105 ParseScope::File,
106 QStringLiteral("无法确认文本编码，已按 Latin-1 回退解码"));
107 return QString::fromLatin1(data);
108 case TextEncoding::Unknown:
109 break;
110 }
111 if (diagnostics)
112 diagnostics->add(ParseSeverity::Error, ParseScope::File, QStringLiteral("不支持的文本编码"));
113 return {};
114}
