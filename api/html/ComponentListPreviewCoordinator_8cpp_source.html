13void ComponentListPreviewCoordinator::processCacheImages(ComponentListViewModel& owner) {
14 // 取出待处理的预览图更新快照。
15 const ComponentListPreviewUpdateBuffer::Snapshot snapshot = owner.m_previewUpdateBuffer.take();
16 // 获取完整预览图更新集合。
17 const auto& pending = snapshot.completeImages;
18 // 获取增量预览图更新集合。
19 const auto& incrementalPending = snapshot.incrementalImages;
20
21 if (pending.isEmpty() && incrementalPending.isEmpty()) {
22 return;
23 }
24
25 for (auto it = incrementalPending.cbegin(); it != incrementalPending.cend(); ++it) {
26 // 查找需要应用增量预览图的列表项。
27 auto item = owner.findItemData(it.key());
28 if (!item) {
29 continue;
30 }
31
32 // 标记当前列表项是否收到预览图更新。
33 bool changed = false;
34 for (auto imageIt = it.value().cbegin(); imageIt != it.value().cend(); ++imageIt) {
35 item->setEncodedPreviewImageAt(imageIt.value(), imageIt.key(), false);
36 changed = true;
37 }
38
39 if (changed) {
40 item->notifyPreviewImagesChanged();
41 }
42 }
43
44 for (auto it = pending.cbegin(); it != pending.cend(); ++it) {
45 // 查找需要应用完整预览图的列表项。
46 auto item = owner.findItemData(it.key());
47 if (item) {
48 item->setEncodedPreviewImages(it.value());
49 if (item->validationPhase() == "fetching_preview") {
50 item->setValidationPhase("completed");
51 }
52 }
53 }
54
55 owner.scheduleListUpdate();
56}

63void ComponentListPreviewCoordinator::fetchImages(ComponentListViewModel& owner, const QStringList& componentIds) {
64 // 保存去重后的有效元器件编号。
65 QStringList validIds;
66 // 记录已经处理过的元器件编号。
67 QSet<QString> seenIds;
68 for (const QString& componentId : componentIds) {
69 if (componentId.isEmpty() || seenIds.contains(componentId)) {
70 continue;
71 }
72 seenIds.insert(componentId);
73
74 // 查找当前有效的列表项。
75 auto item = owner.findItemData(componentId);
76 if (item && item->isValid()) {
77 validIds.append(componentId);
78 }
79 }
80
81 if (validIds.isEmpty()) {
82 qDebug() << "No valid components to fetch preview images for";
83 owner.m_previewReadyHint = false;
84 owner.m_validationReadyHint = false;
85 emit owner.attentionStateChanged();
86 return;
87 }
88
89 qDebug() << "Fetching preview images for" << validIds.count() << "valid components";
90 owner.m_pendingPreviewFetchIds = QSet<QString>(validIds.begin(), validIds.end());
91 owner.m_previewReadyHint = false;
92 emit owner.attentionStateChanged();
93
94 for (const QString& componentId : std::as_const(validIds)) {
95 // 查找需要进入预览图获取状态的列表项。
96 auto item = owner.findItemData(componentId);
97 if (item && item->isValid() && item->previewImageCount() == 0) {
98 item->setValidationPhase("fetching_preview");
99 }
100 }
101 owner.scheduleListUpdate();
102
103 // 使用批量获取预览图接口，避免为每个组件创建单独的定时器。
104 // LcscImageService 会自动处理缓存加载和队列管理。
105 owner.m_service->fetchBatchPreviewImages(validIds);
106}

113void ComponentListPreviewCoordinator::markFetchCompleted(ComponentListViewModel& owner, const QString& componentId) {
114 if (componentId.isEmpty() || !owner.m_pendingPreviewFetchIds.contains(componentId)) {
115 return;
116 }
117
118 owner.m_pendingPreviewFetchIds.remove(componentId);
119 if (owner.m_pendingPreviewFetchIds.isEmpty()) {
120 owner.m_validationReadyHint = false;
121 owner.m_previewReadyHint = true;
122 emit owner.attentionStateChanged();
123 }
