104bool OLECompoundReader::open(const QString& filePath) {
105 clear();
106
107 QFile file(filePath);
108 if (!file.open(QIODevice::ReadOnly))
109 return fail(QStringLiteral("无法打开 OLE 文件: %1").arg(file.errorString()));
110 const QByteArray fileData = file.readAll();
111 if (fileData.size() < static_cast<qsizetype>(kSectorSize * 2))
112 return fail(QStringLiteral("OLE 文件大小不足"));
113
114 static constexpr unsigned char signature[] = {0xD0, 0xCF, 0x11, 0xE0, 0xA1, 0xB1, 0x1A, 0xE1};
115 if (memcmp(fileData.constData(), signature, sizeof(signature)) != 0)
116 return fail(QStringLiteral("不是有效的 OLE 复合文档"));
117 if (readU16(fileData, 26) != 3 || readU16(fileData, 28) != 0xFFFE || readU16(fileData, 30) != 9 ||
118 readU16(fileData, 32) != 6)
119 return fail(QStringLiteral("仅支持 OLE V3、512 字节扇区和 64 字节迷你扇区"));
120
121 const quint32 sectorCount = static_cast<quint32>((fileData.size() - kSectorSize) / kSectorSize);
122 if (sectorCount == 0 || static_cast<qsizetype>(sectorCount + 1) * kSectorSize != fileData.size())
123 return fail(QStringLiteral("OLE 文件包含不完整的扇区"));
124
125 auto sector = [&](quint32 index, QByteArray* result) {
126 if (index >= sectorCount)
127 return false;
128 *result = fileData.mid(static_cast<qsizetype>(index + 1) * kSectorSize, kSectorSize);
129 return result->size() == static_cast<int>(kSectorSize);
130 };
131
132 const quint32 firstDirectorySector = readU32(fileData, 48);
133 const quint32 firstMiniFatSector = readU32(fileData, 60);
134 const quint32 miniFatSectorCount = readU32(fileData, 64);
135 const quint32 firstDifatSector = readU32(fileData, 68);
136 const quint32 difatSectorCount = readU32(fileData, 72);
137 const quint32 fatSectorCount = readU32(fileData, 44);
138
139 QVector<quint32> fatSectors;
140 for (int i = 0; i < 109; ++i) {
141 const quint32 value = readU32(fileData, 76 + i * 4);
142 if (value != kFreeSector)
143 fatSectors.append(value);
144 }
145 quint32 difatSector = firstDifatSector;
146 QSet<quint32> visitedDifat;
147 for (quint32 i = 0; i < difatSectorCount && difatSector != kEndOfChain; ++i) {
148 if (difatSector >= sectorCount || visitedDifat.contains(difatSector))
149 return fail(QStringLiteral("DIFAT 链无效"));
150 visitedDifat.insert(difatSector);
151 QByteArray data;
152 if (!sector(difatSector, &data))
153 return fail(QStringLiteral("无法读取 DIFAT 扇区"));
154 for (int j = 0; j < 127; ++j) {
155 const quint32 value = readU32(data, j * 4);
156 if (value != kFreeSector)
157 fatSectors.append(value);
158 }
159 difatSector = readU32(data, 508);
160 }
161 if (fatSectors.size() < static_cast<int>(fatSectorCount))
162 return fail(QStringLiteral("FAT 扇区数量不足"));
163
164 QVector<quint32> fat(static_cast<int>(sectorCount), kFreeSector);
165 for (quint32 i = 0; i < fatSectorCount; ++i) {
166 if (fatSectors.at(static_cast<int>(i)) >= sectorCount)
167 return fail(QStringLiteral("FAT 扇区索引越界"));
168 QByteArray data;
169 sector(fatSectors.at(static_cast<int>(i)), &data);
170 for (int j = 0; j < 128; ++j) {
171 const quint32 index = i * 128U + static_cast<quint32>(j);
172 if (index < sectorCount)
173 fat[static_cast<int>(index)] = readU32(data, j * 4);
174 }
175 }

177 auto readChain =
178 [&](quint32 start, quint64 byteCount, const QVector<quint32>& table, quint32 unitSize, QByteArray* result) {
179 result->clear();
180 if (byteCount == 0)
181 return true;
182 if (byteCount > static_cast<quint64>(std::numeric_limits<int>::max()))
183 return false;
184 quint32 current = start;
185 QSet<quint32> visited;
186 while (result->size() < static_cast<qsizetype>(byteCount)) {
187 if (current >= static_cast<quint32>(table.size()) || current == kEndOfChain || current == kFreeSector ||
188 visited.contains(current))
189 return false;
190 visited.insert(current);
191 if (unitSize == kSectorSize) {
192 QByteArray data;
193 if (!sector(current, &data))
194 return false;
195 result->append(data);
196 } else {
197 const qsizetype offset = static_cast<qsizetype>(current) * unitSize;
198 if (offset < 0 || offset + unitSize > result->capacity())
199 return false;
200 }
201 current = table.at(static_cast<int>(current));
202 if (result->size() > static_cast<qsizetype>(byteCount) + unitSize)
203 break;
204 }
205 if (result->size() < static_cast<qsizetype>(byteCount))
206 return false;
207 *result = result->left(static_cast<int>(byteCount));
208 return true;
209 };
210
211 QVector<quint32> miniFat;
212 if (miniFatSectorCount > 0) {
213 QByteArray miniFatData;
214 if (!readChain(firstMiniFatSector,
215 static_cast<quint64>(miniFatSectorCount) * kSectorSize,
216 fat,
217 kSectorSize,
218 &miniFatData))
219 return fail(QStringLiteral("无法读取 Mini-FAT"));
220 miniFat.reserve(miniFatData.size() / 4);
221 for (qsizetype offset = 0; offset + 4 <= miniFatData.size(); offset += 4)
222 miniFat.append(readU32(miniFatData, offset));
223 }
224
225 QByteArray directoryData;
226 QSet<quint32> directorySectors;
227 quint32 currentDirectorySector = firstDirectorySector;
228 while (currentDirectorySector != kEndOfChain) {
229 if (currentDirectorySector >= sectorCount || directorySectors.contains(currentDirectorySector))
230 return fail(QStringLiteral("目录扇区链无效"));
231 directorySectors.insert(currentDirectorySector);
232 QByteArray data;
233 sector(currentDirectorySector, &data);
234 directoryData.append(data);
235 currentDirectorySector = fat.at(static_cast<int>(currentDirectorySector));
236 }
237 if (directoryData.isEmpty() || directoryData.size() % kDirectoryEntrySize != 0)
238 return fail(QStringLiteral("目录数据无效"));
239
240 QVector<DirectoryEntry> entries;
241 entries.reserve(directoryData.size() / kDirectoryEntrySize);
242 for (qsizetype offset = 0; offset < directoryData.size(); offset += kDirectoryEntrySize) {
243 DirectoryEntry entry;
244 const quint16 nameSize = readU16(directoryData, offset + 64);
245 if (nameSize > 64 || nameSize % 2 != 0 || (nameSize != 0 && nameSize < 2))
246 return fail(QStringLiteral("目录名称长度无效"));
247 const int nameChars = nameSize == 0 ? 0 : nameSize / 2 - 1;
248 for (int i = 0; i < nameChars; ++i)
249 entry.name.append(QChar(readU16(directoryData, offset + i * 2)));
250 entry.type = static_cast<quint8>(directoryData.at(offset + 66));
251 if (entry.type != 0 && entry.type != 1 && entry.type != 2 && entry.type != 5)
252 return fail(QStringLiteral("目录对象类型无效"));
253 entry.left = readU32(directoryData, offset + 68);
254 entry.right = readU32(directoryData, offset + 72);
255 entry.child = readU32(directoryData, offset + 76);
256 entry.startSector = readU32(directoryData, offset + 116);
257 entry.streamSize = readU64(directoryData, offset + 120);
258 entries.append(entry);
259 }
260 if (entries.isEmpty() || entries.at(0).type != 5)
261 return fail(QStringLiteral("OLE 根目录条目无效"));
262
263 QByteArray miniStreamData;
264 if (entries.at(0).streamSize > 0 &&
265 !readChain(entries.at(0).startSector, entries.at(0).streamSize, fat, kSectorSize, &miniStreamData))
266 return fail(QStringLiteral("无法读取 Mini-Stream"));
267
268 auto readMiniStream = [&](const DirectoryEntry& entry, QByteArray* result) {
269 result->clear();
270 if (entry.streamSize == 0)
271 return true;
272 if (entry.streamSize > static_cast<quint64>(std::numeric_limits<int>::max()) || miniFat.isEmpty())
273 return false;
274 quint32 current = entry.startSector;
275 QSet<quint32> visited;
276 while (result->size() < static_cast<qsizetype>(entry.streamSize)) {
277 if (current >= static_cast<quint32>(miniFat.size()) || visited.contains(current))
278 return false;
279 visited.insert(current);
280 const qsizetype offset = static_cast<qsizetype>(current) * kMiniSectorSize;
281 if (offset + kMiniSectorSize > miniStreamData.size())
282 return false;
283 result->append(miniStreamData.constData() + offset, kMiniSectorSize);
284 current = miniFat.at(static_cast<int>(current));
285 if (result->size() > static_cast<qsizetype>(entry.streamSize) + kMiniSectorSize)
286 break;
287 }
288 *result = result->left(static_cast<int>(entry.streamSize));
289 return result->size() == static_cast<int>(entry.streamSize);
290 };
291
292 QSet<quint32> visitedEntries;
293 std::function<bool(quint32, const QString&, int)> visitSiblings;
294 visitSiblings = [&](quint32 index, const QString& parentPath, int depth) {
295 if (index == kNoStream)
296 return true;
297 if (depth > kMaxDirectoryDepth || index >= static_cast<quint32>(entries.size()) ||
298 visitedEntries.contains(index))
299 return false;
300 const DirectoryEntry& entry = entries.at(static_cast<int>(index));
301 if (!visitSiblings(entry.left, parentPath, depth + 1))
302 return false;
303 if (entry.type != 1 && entry.type != 2)
304 return false;
305 const QString path = parentPath.isEmpty() ? entry.name : parentPath + "/" + entry.name;
306 if (entry.type == 1) {
307 if (!visitSiblings(entry.child, path, depth + 1))
308 return false;
309 } else {
310 QByteArray streamData;
311 const bool regular = entry.streamSize >= kMiniStreamCutoff;
312 // 校验结果。
313 const bool valid = regular ? readChain(entry.startSector, entry.streamSize, fat, kSectorSize, &streamData)
314 : readMiniStream(entry, &streamData);
315 if (!valid || m_streams.contains(path))
316 return false;
317 m_streams.insert(path, streamData);
318 }
319 visitedEntries.insert(index);
320 return visitSiblings(entry.right, parentPath, depth + 1);
321 };
322 if (!visitSiblings(entries.at(0).child, QString(), 0))
323 return fail(QStringLiteral("OLE 目录树无效"));
324
325 m_opened = true;
326 return true;
327}
