92void ComponentQueueManager::checkAndProcessNext() {
93 if (!m_isRunning) {
94 m_checkTimer->stop();
95 return;
96 }
97
98 int activeCount = m_pendingComponents.size(); // 当前正在处理的请求数量。
99
100 // 如果所有槽位都满了，启用定时器作为安全网（防止某些请求完成但信号未触发的情况）
101 // 否则禁用定时器（正常情况下 requestCompleted 会直接触发处理）
102 if (activeCount >= m_maxConcurrentRequests && !m_requestQueue.isEmpty()) {
103 if (!m_checkTimer->isActive()) {
104 m_checkTimer->start();
105 }
106 } else {
107 m_checkTimer->stop();
108 }
109
110 while (activeCount < m_maxConcurrentRequests && !m_requestQueue.isEmpty()) {
111 QString componentId = m_requestQueue.takeFirst(); // 从队首取出的元器件编号。
112
113 if (m_pendingComponents.contains(componentId)) {
114 qDebug() << "Component" << componentId << "already pending, skipping";
115 continue;
116 }
117
118 m_pendingComponents.insert(componentId);
119 activeCount++;
120
121 qDebug() << "ComponentQueueManager: Request ready for:" << componentId << "(Active:" << activeCount << "/"
122 << m_maxConcurrentRequests << "Remaining:" << m_requestQueue.size() << ")";
123
124 emit requestReady(componentId);
125 }
126
127 if (m_requestQueue.isEmpty() && activeCount == 0) {
128 qDebug() << "ComponentQueueManager: All components processed. Queue empty.";
129 m_checkTimer->stop();
130 emit queueEmpty();
131 stop();
132 }
133}
