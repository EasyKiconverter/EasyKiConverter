85bool AltiumPcbLibWriter::write(const QList<AltiumPcbComponent>& components,
86 const QString& filePath,
87 const QString& libraryName) {
88 m_diagnostics.clear();
89 if (!validateComponents(components, filePath)) {
90 return false;
91 }
92 m_wideStrings.clear();
93
94 qDebug() << "AltiumPcbLibWriter::write: components:" << components.size() << "filePath:" << filePath;
95
96 OLECompoundWriter ole; // 当前输出文件的 OLE 写入器。
97 if (!ole.create()) {
98 qWarning() << "AltiumPcbLibWriter::write: Failed to create OLE compound document";
99 return false;
100 }
101
102 QStringList names; // 原始封装名称列表。
103 for (const AltiumPcbComponent& component : components) {
104 names.append(component.name);
105 }
106 const QStringList sectionKeys = AltiumWriterUtils::makeUniqueSectionKeys(names); // 唯一 OLE 节名称列表。
107
108 // 只输出 PcbLib 核心流。未知或不完整的可选 storage 比缺省更危险，
109 // 因为 Altium 会尝试按其声明的版本反序列化它们。
110 writeFileHeader(ole);
111 writeSectionKeys(ole, components, sectionKeys);
112 writeLibraryStorage(ole, components, filePath);
113
114 // 写入每个封装
115 for (int i = 0; i < components.size(); ++i) {
116 writeFootprintStorage(ole, components[i], sectionKeys[i]);
117 }
118
119 if (ole.hasError()) {
120 qWarning() << "AltiumPcbLibWriter: Failed to construct OLE document:" << ole.errorString();
121 return false;
122 }
123
124 bool result = ole.saveToFile(filePath); // OLE 文件保存结果。
125 qDebug() << "AltiumPcbLibWriter::write: saveToFile result:" << result << "filePath:" << filePath;
126 return result;
127}

133void AltiumPcbLibWriter::writeFileHeader(OLECompoundWriter& ole) {
134 QByteArray header; // FileHeader 流数据。
135 AltiumBinaryWriter writer(header);
136
137 // 此流的第一个 i32 是版本文本本身的长度，随后是 Pascal short string。
138 // 它不是通用 string block，后面也没有版本 double 或随机 ID。
139 QByteArray versionStr = "PCB 6.0 Binary Library File"; // PcbLib 版本文本。
140 writer.writeInt32(versionStr.size());
141 writer.writePascalShortString(QString::fromLatin1(versionStr));
142
143 ole.writeStream("FileHeader", header);
144}

152void AltiumPcbLibWriter::writeSectionKeys(OLECompoundWriter& ole,
153 const QList<AltiumPcbComponent>& components,
154 const QStringList& sectionKeys) {
155 QByteArray data; // SectionKeys 流数据。
156 AltiumBinaryWriter writer(data);
157
158 // 计算需要映射的条目数
159 QVector<QPair<QString, QString>> mappings; // 需要显式写入的名称映射。
160 for (int i = 0; i < components.size(); ++i) {
161 const AltiumPcbComponent& comp = components[i]; // 当前封装。
162 const QString& key = sectionKeys[i]; // 当前封装的唯一节名称。
163 if (key != comp.name) {
164 mappings.append({comp.name, key});
165 }
166 }
167
168 writer.writeUInt32(static_cast<uint32_t>(mappings.size()));
169 for (const auto& mapping : mappings) {
170 writer.writePascalString(mapping.first);
171 writer.writeStringBlock(mapping.second);
172 }
173
174 if (!mappings.isEmpty()) {
175 ole.writeStream("SectionKeys", data);
176 }

256QString AltiumPcbLibWriter::buildLibraryMetadata(const QString& filePath) const {
257 auto safeValue = [](QString value) { // 清理元数据中的分隔符。
258 value.replace('|', ' ');
259 value.replace('\0', ' ');
260 return value;
261 };
262 auto append = [&safeValue](QStringList& fields, const QString& key, const QString& value) { // 追加一个键值字段。
263 fields.append(key + "=" + safeValue(value));
264 };
265
266 QString absolutePath = QFileInfo(filePath).absoluteFilePath(); // 规范化后的绝对文件路径。
267 absolutePath.replace('/', '\\');
268 const QDateTime now = QDateTime::currentDateTime(); // 生成元数据时刻。
269
270 // Altium serializes library settings as a sequence of Board records. We build
271 // that profile from typed defaults so it remains reviewable and evolvable
272 // instead of embedding an opaque captured blob.
273 QList<QStringList> records; // Board 记录列表。
274 records.append(QStringList{});
275 QStringList& identity = records.last(); // 库身份记录。
276 append(identity, "FILENAME", absolutePath);
277 append(identity, "KIND", "Protel_Advanced_PCB_Library");
278 append(identity, "VERSION", "3.00");
279 append(identity, "DATE", QLocale::c().toString(now.date(), "M/d/yyyy"));
280 append(identity, "TIME", QLocale::c().toString(now.time(), "h:mm:ss AP"));
281 append(identity, "TOPTYPE", "3");
282 append(identity, "TOPCONST", "3.500");
283 append(identity, "TOPHEIGHT", "0.4mil");
284 append(identity, "TOPMATERIAL", "Solder Resist");
285 append(identity, "BOTTOMTYPE", "3");
286 append(identity, "BOTTOMCONST", "3.500");
287 append(identity, "BOTTOMHEIGHT", "0.4mil");
288 append(identity, "BOTTOMMATERIAL", "Solder Resist");
289 append(identity, "LAYERSTACKSTYLE", "0");
290
291 auto layerName = [](int layer) -> QString { // 将层编号转换为 Altium 层名称。
292 if (layer == 1)
293 return "Top Layer";
294 if (layer >= 2 && layer <= 31)
295 return QString("Mid-Layer %1").arg(layer - 1);
296 if (layer == 32)
297 return "Bottom Layer";
298 static const QStringList standardNames = {// 标准覆盖层名称。
299 "Top Overlay",
300 "Bottom Overlay",
301 "Top Paste",
302 "Bottom Paste",
303 "Top Solder",
304 "Bottom Solder"};
305 if (layer >= 33 && layer <= 38)
306 return standardNames[layer - 33];
307 if (layer >= 39 && layer <= 54)
308 return QString("Internal Plane %1").arg(layer - 38);
309 if (layer == 55)
310 return "Drill Guide";
311 if (layer == 56)
312 return "Keep-Out Layer";
313 if (layer >= 57 && layer <= 72)
314 return QString("Mechanical %1").arg(layer - 56);
315 static const QStringList auxiliaryNames = {// 辅助层名称。
316 "Drill Drawing",
317 "Multi-Layer",
318 "Connections",
319 "Background",
320 "DRC Error Markers",
321 "Selections",
322 "Visible Grid 1",
323 "Visible Grid 2",
324 "Pad Holes",
325 "Via Holes"};
326 return auxiliaryNames.value(layer - 73, QString("Layer %1").arg(layer));
327 };
328
329 for (int layer = 1; layer <= 82; ++layer) {
330 if ((layer - 1) % 5 == 0 && layer != 1) {
331 records.append(QStringList{});
332 }
333 QStringList& fields = records.last(); // 当前层的字段列表。
334 const QString prefix = QString("LAYER%1").arg(layer); // 当前层字段前缀。
335 append(fields, prefix + "NAME", layerName(layer));
336 append(fields, prefix + "PREV", layer == 1 ? "0" : (layer == 32 ? "1" : "0"));
337 append(fields, prefix + "NEXT", layer == 1 ? "32" : "0");
338 append(fields, prefix + "MECHENABLED", layer == 57 ? "TRUE" : "FALSE");
339 append(fields, prefix + "COPTHICK", "1.4mil");
340 append(fields, prefix + "DIELTYPE", "0");
341 append(fields, prefix + "DIELCONST", "4.800");
342 append(fields, prefix + "DIELHEIGHT", "12.6mil");
343 append(fields, prefix + "DIELMATERIAL", "FR-4");
344 }
345
346 records.append(QStringList{});
347 QStringList& grid = records.last(); // 网格设置记录。
348 append(grid, "BIGVISIBLEGRIDSIZE", "0.000000");
349 append(grid, "VISIBLEGRIDSIZE", "0.000000");
350 append(grid, "SNAPGRIDSIZE", "50000.000000");
351 append(grid, "SNAPGRIDSIZEX", "50000.000000");
352 append(grid, "SNAPGRIDSIZEY", "50000.000000");
353 append(grid, "ELECTRICALGRIDRANGE", "8mil");
354 append(grid, "ELECTRICALGRIDENABLED", "TRUE");
355 append(grid, "DISPLAYUNIT", "1");
356 append(grid, "CURRENT2D3DVIEWSTATE", "2D");
357
358 QStringList serializedRecords; // 已序列化的 Board 记录。
359 for (const QStringList& record : records) {
360 if (!record.isEmpty()) {
361 serializedRecords.append(record.join('|'));
362 }
363 }
364 return serializedRecords.join("\r|RECORD=Board|");
365}

372void AltiumPcbLibWriter::writeModelsStorage(OLECompoundWriter& ole, const QList<AltiumPcbComponent>& components) {
373 ole.addStorage("Library", "Models");
374
375 // 收集所有 3D 模型
376 QList<AltiumPcbComponent::Model3D> allModels; // 所有封装去重前的模型列表。
377 for (const AltiumPcbComponent& comp : components) {
378 for (const AltiumPcbComponent::Model3D& model : comp.models) {
379 allModels.append(model);
380 }
381 }
382
383 // Header 流
384 QByteArray hdrData; // Models/Header 流数据。
385 AltiumBinaryWriter hdr(hdrData); // Models/Header 写入器。
386 hdr.writeUInt32(static_cast<uint32_t>(allModels.size()));
387 ole.writeStream("Library/Models", "Header", hdrData);
388
389 // Data 流（模型元数据）
390 QByteArray modelData; // Models/Data 流数据。
391 AltiumBinaryWriter modelWriter(modelData); // 模型元数据写入器。
392 const auto safeMetadataValue = [](QString value) { // 清理模型元数据分隔符。
393 value.replace('|', ' ');
394 value.replace('\0', ' ');
395 value.replace('\r', ' ');
396 value.replace('\n', ' ');
397 return value;
398 };
399 for (const AltiumPcbComponent::Model3D& model : allModels) {
400 const QString id = safeMetadataValue(model.id.isEmpty() ? model.name : model.id); // 模型标识。
401 const QString metadata =
402 QString("EMBED=TRUE|MODELSOURCE=Undefined|ID=%1|ROTX=%2|ROTY=%3|ROTZ=%4|DZ=%5|CHECKSUM=0|NAME=%6")
403 .arg(id,
404 QString::number(normalizeFiniteValue(model.rotX, 0.0, QStringLiteral("3D 模型 X 旋转")), 'f', 6),
405 QString::number(normalizeFiniteValue(model.rotY, 0.0, QStringLiteral("3D 模型 Y 旋转")), 'f', 6),
406 QString::number(normalizeFiniteValue(model.rotZ, 0.0, QStringLiteral("3D 模型 Z 旋转")), 'f', 6),
407 QString::number(
408 AltiumCoord::mmToRaw(normalizeFiniteValue(model.dz, 0.0, QStringLiteral("3D 模型 Z 偏移")))),
409 safeMetadataValue(model.name));
410 QByteArray encoded = metadata.toLatin1(); // 模型元数据编码结果。
411 encoded.append('\0');
412 modelWriter.writeInt32(encoded.size());
413 modelWriter.writeBytes(encoded);
414 }
415 ole.writeStream("Library/Models", "Data", modelData);
416
417 // 写入压缩的 STEP 数据（去掉 qCompress 的 4 字节头，Altium 期望原始 zlib 流）
418 for (int i = 0; i < allModels.size(); ++i) {
419 if (!allModels[i].stepData.isEmpty()) {
420 QByteArray compressed = qCompress(allModels[i].stepData, 9); // STEP 压缩数据。
421 ole.writeStream("Library/Models", QString::number(i), compressed.mid(4));
422 }
423 }
424}

432void AltiumPcbLibWriter::writeFootprintStorage(OLECompoundWriter& ole,
433 const AltiumPcbComponent& component,
434 const QString& sectionKey) {
435 ole.addStorage(sectionKey);
436 QString basePath = sectionKey; // 封装存储根路径。
437
438 // Header 流（图元数量）
439 QByteArray hdrData; // 封装 Header 流数据。
440 AltiumBinaryWriter hdr(hdrData); // 封装 Header 写入器。
441 hdr.writeUInt32(static_cast<uint32_t>(countPrimitives(component)));
442 ole.writeStream(basePath, "Header", hdrData);
443
444 // Parameters 流
445 QByteArray paramsData; // Parameters 流数据。
446 writeFootprintParameters(paramsData, component);
447 ole.writeStream(basePath, "Parameters", paramsData);
448
449 // WideStrings 流
450 QByteArray wsData; // WideStrings 流数据。
451 writeWideStrings(wsData, component);
452 ole.writeStream(basePath, "WideStrings", wsData);
453
454 // Data 流
455 QByteArray dataData; // Data 流数据。
456 writeFootprintData(dataData, component);
457 ole.writeStream(basePath, "Data", dataData);
458
459 // UniqueIdPrimitiveInformation 流
460 ole.addStorage(basePath, "UniqueIdPrimitiveInformation");
461 QByteArray uidHdrData; // 唯一标识 Header 流数据。
462 AltiumBinaryWriter uidHdr(uidHdrData); // 唯一标识 Header 写入器。
463 uidHdr.writeUInt32(static_cast<uint32_t>(countPrimitives(component)));
464 ole.writeStream(basePath + "/UniqueIdPrimitiveInformation", "Header", uidHdrData);
465
466 QByteArray uidData; // 唯一标识 Data 流数据。
467 writeUniqueIdPrimitiveInformation(uidData, component);
468 ole.writeStream(basePath + "/UniqueIdPrimitiveInformation", "Data", uidData);
469
470 // ExtendedPrimitiveInformation 流（仅当有扩展信息时写入）
471 if (!component.extendedPrimitives.isEmpty()) {
472 ole.addStorage(basePath, "ExtendedPrimitiveInformation");
473 QByteArray extHdrData; // 扩展信息 Header 流数据。
474 AltiumBinaryWriter extHdr(extHdrData); // 扩展信息 Header 写入器。
475 extHdr.writeUInt32(static_cast<uint32_t>(component.extendedPrimitives.size()));
476 ole.writeStream(basePath + "/ExtendedPrimitiveInformation", "Header", extHdrData);
477
478 QByteArray extData; // 扩展信息 Data 流数据。
479 writeExtendedPrimitiveInformation(extData, component);
480 ole.writeStream(basePath + "/ExtendedPrimitiveInformation", "Data", extData);
481 }

616void AltiumPcbLibWriter::writeUniqueIdPrimitiveInformation(QByteArray& buffer, const AltiumPcbComponent& component) {
617 AltiumBinaryWriter writer(buffer);
618
619 // 按图元顺序写入唯一标识信息，确保后续记录可以通过索引稳定关联。
620 auto writeEntry = [&](const char* objectName, int index) {
621 QMap<QString, QString> params; // 当前图元唯一标识参数。
622 if (index > 0) {
623 params["PRIMITIVEINDEX"] = QString::number(index);
624 }
625 params["PRIMITIVEOBJECTID"] = QString::fromLatin1(objectName);
626 writer.writeCStringParameterBlock(params);
627 };
628
629 int idx = 0; // 当前图元序号。
630 for (int i = 0; i < component.pads.size(); ++i, ++idx)
631 writeEntry("Pad", idx);
632 for (int i = 0; i < component.tracks.size(); ++i, ++idx)
633 writeEntry("Track", idx);
634 for (int i = 0; i < component.arcs.size(); ++i, ++idx)
635 writeEntry("Arc", idx);
636 for (int i = 0; i < component.texts.size(); ++i, ++idx)
637 writeEntry("Text", idx);
638 for (int i = 0; i < component.fills.size(); ++i, ++idx)
639 writeEntry("Fill", idx);
640 for (int i = 0; i < component.regions.size(); ++i, ++idx)
641 writeEntry("Region", idx);
642 for (int i = 0; i < component.bodies.size(); ++i, ++idx)
643 writeEntry("ComponentBody", idx);
