54ExportOptions CliContext::createExportOptions() const {
55 ExportOptions options; // 按命令行参数构造的导出选项。
56
57 // 设置输出路径
58 options.outputPath = m_parser.outputDir();
59 // 清洗 libName，防止路径穿越（如 ../foo）
60 options.libName = PathSecurity::sanitizeFilename(m_parser.libName());
61
62 // 设置导出类型（CLI 默认：符号库、封装库；3D 模型通过参数启用）
63 options.exportSymbol = m_parser.exportSymbol();
64 options.exportFootprint = m_parser.exportFootprint();
65 options.exportModel3D = m_parser.export3DModel();
66 options.exportModel3DFormat = model3DFormatFromString(m_parser.model3DFormat());
67 options.exportPreviewImages = m_parser.exportPreview();
68 options.exportDatasheet = m_parser.exportDatasheet();
69
70 // 设置默认值
71 options.debugMode = m_parser.isDebugMode();
72
73 // 弱网模式
74 options.weakNetworkSupport = m_parser.weakNetworkSupport();
75
76 // 更新模式
77 options.updateMode = m_parser.updateMode();
78
79 // 3D 模型路径模式
80 const QString pathMode = m_parser.model3DPathMode(); // 3D 模型路径模式文本。
81 options.exportModel3DPathMode = (pathMode == QStringLiteral("absolute")) ? ExportOptions::MODEL_3D_PATH_ABSOLUTE
82 : ExportOptions::MODEL_3D_PATH_RELATIVE;
83
84 // 覆盖模式（默认 true，--no-overwrite 禁用）
85 options.overwriteExistingFiles = m_parser.overwriteExistingFiles();
86
87 // 库描述
88 options.symbolLibraryDescription = m_parser.symbolDescription();
89 options.footprintLibraryDescription = m_parser.footprintDescription();
90
91 // 目标 EDA 格式
92 const QString format = m_parser.targetFormat(); // 目标格式文本。
93 if (format == QStringLiteral("altium")) {
94 options.targetFormat = TargetEdaFormat::Altium;
95 } else if (format == QStringLiteral("xpedition")) {
96 options.targetFormat = TargetEdaFormat::Xpedition;
97 } else if (format == QStringLiteral("allegro")) {
98 options.targetFormat = TargetEdaFormat::Allegro;
99 } else if (format == QStringLiteral("pads")) {
100 options.targetFormat = TargetEdaFormat::Pads;
101 } else if (format == QStringLiteral("eagle")) {
102 options.targetFormat = TargetEdaFormat::Eagle;
103 } else if (format == QStringLiteral("pcad")) {
104 options.targetFormat = TargetEdaFormat::Pcad;
105 } else if (format == QStringLiteral("cadstar")) {
106 options.targetFormat = TargetEdaFormat::Cadstar;
107 } else if (format == QStringLiteral("orcad")) {
108 options.targetFormat = TargetEdaFormat::Orcad;
109 } else if (format == QStringLiteral("librepcb")) {
110 options.targetFormat = TargetEdaFormat::LibrePcb;
111 } else if (format == QStringLiteral("horizon")) {
112 options.targetFormat = TargetEdaFormat::Horizon;
113 } else {
114 options.targetFormat = TargetEdaFormat::KiCad;
115 }
116 if (options.targetFormat == TargetEdaFormat::Altium &&
117 (options.exportModel3DFormat & ExportOptions::MODEL_3D_FORMAT_WRL)) {
118 // Altium PcbLib 仅可靠嵌入 STEP；CLI 即使收到 wrl/both 也自动收敛到 STEP。
119 options.exportModel3DFormat = ExportOptions::MODEL_3D_FORMAT_STEP;
120 }
121 if (options.targetFormat == TargetEdaFormat::Allegro) {
122 if (options.exportModel3D)
123 options.exportModel3DFormat = ExportOptions::MODEL_3D_FORMAT_STEP;
124 }
125 if (options.targetFormat == TargetEdaFormat::Orcad) {
126 options.exportFootprint = false;
127 }
128
129 return options;
130}
