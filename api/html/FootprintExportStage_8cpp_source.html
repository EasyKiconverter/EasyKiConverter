30QString librePcbProjectRoot(const QString& outputPath) {

34 candidate = QDir::cleanPath(candidate);

92bool copyDirectoryTree(const QString& sourcePath, const QString& targetPath) {
93 const QDir source(sourcePath);
94 if (!source.exists())
95 return true;
96 if (!QDir().mkpath(targetPath))
97 return false;
98
99 const QFileInfoList entries = // 源目录中的所有可见及隐藏目录项。
100 source.entryInfoList(QDir::NoDotAndDotDot | QDir::AllEntries | QDir::Hidden | QDir::System);
101 for (const QFileInfo& entry : entries) { // 当前目录项。
102 if (entry.isSymLink())
103 return false;
104
105 const QString target = QDir(targetPath).filePath(entry.fileName());
106 if (entry.isDir()) {
107 if (!copyDirectoryTree(entry.absoluteFilePath(), target))
108 return false;
109 } else {
110 if (QFile::exists(target) && !QFile::remove(target))
111 return false;
112 if (!QFile::copy(entry.absoluteFilePath(), target))
113 return false;
114 }
115 }
116 return true;
117}

160void FootprintExportStage::start(const QStringList& componentIds,
161 const QMap<QString, QSharedPointer<ComponentData>>& cachedData) {
162 if (m_isExporting.load()) {
163 qWarning() << "FootprintExportStage: Export already in progress";
164 return;
165 }
166
167 if (componentIds.isEmpty()) {
168 qWarning() << "FootprintExportStage: No components to export";
169 emit completed(0, 0, 0);
170 return;
171 }
172
173 m_componentIds = componentIds;
174 m_cachedData = cachedData;
175 m_cancelled.store(false);
176 m_isRunning.store(true);
177
178 {
179 QMutexLocker locker(&m_progressMutex);
180 m_progress = ExportTypeProgress();
181 m_progress.typeName = QStringLiteral("Footprint");
182 m_progress.totalCount = componentIds.size();
183 m_progress.completedCount = 0;
184 m_progress.successCount = 0;
185 m_progress.failedCount = 0;
186 m_progress.skippedCount = 0;
187 m_progress.inProgressCount = componentIds.size();
188 for (const QString& componentId : componentIds) { // 当前元器件编号。
189 ExportItemStatus itemStatus;
190 itemStatus.status = ExportItemStatus::Status::InProgress;
191 itemStatus.startTime = QDateTime::currentDateTime();
192 m_progress.itemStatus[componentId] = itemStatus;
193 }
194 }
195
196 m_tempManager.setOutputPath(m_options.outputPath);
197
198 m_isExporting.store(true);
199 m_workerThread = QThread::create([this, componentIds, cachedData]() { doLibraryExport(componentIds, cachedData); });
200 connect(m_workerThread, &QThread::finished, this, [this]() { m_workerThread = nullptr; });
201 m_workerThread->start();
202}

241 const QMap<QString, QSharedPointer<ComponentData>>& cachedData) {
242 qDebug() << "FootprintExportStage: Starting library export in worker thread for" << componentIds.size()
243 << "components";
244 const uint64_t gen = ComponentCacheService::instance()->currentGeneration();

246 QList<FootprintData> footprintList;
247 QList<IR::ComponentIR> componentIrList;

254 // 统一更新阶段进度，确保库级导出与普通 Worker 阶段提供一致的快照。
255 const auto publishItemStatus = [this](const QString& componentId, const ExportItemStatus& status) {
256 ExportTypeProgress progressSnapshot;
257 {
258 QMutexLocker locker(&m_progressMutex);
259 auto item = m_progress.itemStatus.find(componentId);
260 if (item == m_progress.itemStatus.end())
261 return;
262 item.value() = status;
263 m_progress.completedCount = 0;
264 m_progress.successCount = 0;
265 m_progress.failedCount = 0;
266 m_progress.skippedCount = 0;
267 m_progress.inProgressCount = 0;
268 for (const ExportItemStatus& itemStatus : std::as_const(m_progress.itemStatus)) { // 当前状态记录。
269 if (itemStatus.status == ExportItemStatus::Status::Success) {
270 ++m_progress.successCount;
271 } else if (itemStatus.status == ExportItemStatus::Status::Failed) {
272 ++m_progress.failedCount;
273 } else if (itemStatus.status == ExportItemStatus::Status::Skipped) {
274 ++m_progress.skippedCount;
275 } else if (itemStatus.status == ExportItemStatus::Status::InProgress) {
276 ++m_progress.inProgressCount;
277 }
278 }
279 m_progress.completedCount = m_progress.successCount + m_progress.failedCount + m_progress.skippedCount;
280 progressSnapshot = m_progress;
281 }
282 emit itemStatusChanged(componentId, status);
283 emit progressChanged(progressSnapshot);
284 };

286 // Altium、Allegro、LibrePCB 和 Horizon 将 3D 模型写入封装输出，阶段提前失败时也必须结束对应状态。
287 const auto publishEmbeddedModelFailure = [this](const QString& componentId, const QString& errorMessage) {
288 if ((m_options.targetFormat != TargetEdaFormat::Altium && m_options.targetFormat != TargetEdaFormat::Allegro &&
289 m_options.targetFormat != TargetEdaFormat::LibrePcb &&
290 m_options.targetFormat != TargetEdaFormat::Horizon) ||
291 !m_options.exportModel3D) {
292 return;
293 }
294 ExportItemStatus modelStatus;
295 modelStatus.status = ExportItemStatus::Status::Failed;
296 modelStatus.errorMessage = errorMessage;
297 modelStatus.endTime = QDateTime::currentDateTime();
298 emit embeddedModel3DStatusChanged(componentId, modelStatus);
299 };
300
301 for (const QString& componentId : componentIds) { // 当前元器件编号。
302 if (m_cancelled.load()) {
303 qDebug() << "FootprintExportStage: Export cancelled during data collection";
304 break;
305 }
306
307 auto it = cachedData.find(componentId);

309 qWarning() << "FootprintExportStage: No data for component:" << componentId;
310 failedIds.append(componentId);
311 ExportItemStatus status;
312 status.status = ExportItemStatus::Status::Failed;
313 status.errorMessage = "No component data";
314 publishItemStatus(componentId, status);
315 publishEmbeddedModelFailure(componentId, QStringLiteral("No component data"));
316 continue;
317 }
318
319 QSharedPointer<ComponentData> data = it.value();
320
321 const bool combinedTarget = // 是否为需要组合符号和封装输出的目标格式。
322 m_options.targetFormat == TargetEdaFormat::Eagle || m_options.targetFormat == TargetEdaFormat::Cadstar ||
323 m_options.targetFormat == TargetEdaFormat::Allegro || m_options.targetFormat == TargetEdaFormat::LibrePcb ||
324 m_options.targetFormat == TargetEdaFormat::Horizon;
325 const bool needsSymbol = combinedTarget && m_options.exportSymbol;
326 const bool needsFootprint =
327 m_options.exportFootprint || m_options.exportModel3D;
328 if (needsSymbol && !data->symbolData()) {
329 qWarning() << "FootprintExportStage: No symbol data for component:" << componentId;
330 failedIds.append(componentId);
331 ExportItemStatus status;
332 status.status = ExportItemStatus::Status::Failed;
333 status.errorMessage = "No symbol data";
334 publishItemStatus(componentId, status);
335 publishEmbeddedModelFailure(componentId, QStringLiteral("No symbol data"));
336 continue;
337 }
338 if (needsFootprint && !data->footprintData()) {
339 qWarning() << "FootprintExportStage: No footprint data for component:" << componentId;
340 failedIds.append(componentId);
341 ExportItemStatus status;
342 status.status = ExportItemStatus::Status::Failed;
343 status.errorMessage = "No footprint data";
344 publishItemStatus(componentId, status);
345 publishEmbeddedModelFailure(componentId, QStringLiteral("No footprint data"));
346 continue;
347 }
348
349 FootprintData footprint;
350 if (data->footprintData()) {
351 footprint = *data->footprintData();
352 }
353 if (needsFootprint && m_options.needsEmbeddedModel3DStep()) {
354 FootprintModel3DPreparation::prepare(footprint, data, componentId, gen);
355 }
356
357 if (data->footprintData())
358 footprintList.append(footprint);
359 if (combinedTarget && data->symbolData()) {
360 // 组合库必须使用完成 STEP 准备后的封装副本，否则模型准备结果会在重新转换组件时丢失。

365 symbolList.append(IR::toSymbolIR(*data->symbolData()));
366 }
367 collectedIds.append(componentId);
368 successCount++;
369
370 ExportItemStatus status;
371 status.status = ExportItemStatus::Status::Success;
372 if (data->symbolData())
373 status.diagnostics.append(data->symbolData()->validationErrors());
374 if (data->footprintData())
375 status.diagnostics.append(footprint.validationErrors());
376 if (!status.diagnostics.isEmpty())
377 qWarning() << "FootprintExportStage: Input diagnostics for" << componentId << status.diagnostics;
378 publishItemStatus(componentId, status);
379
380 if ((m_options.targetFormat == TargetEdaFormat::Altium || m_options.targetFormat == TargetEdaFormat::Allegro ||
381 m_options.targetFormat == TargetEdaFormat::LibrePcb ||
382 m_options.targetFormat == TargetEdaFormat::Horizon) &&
383 m_options.exportModel3D && data->footprintData()) {
384 ExportItemStatus modelStatus;
385 if (!footprint.model3D().step().isEmpty()) {
386 modelStatus.status = ExportItemStatus::Status::Success;
387 } else {
388 modelStatus.status = ExportItemStatus::Status::Failed;
389 modelStatus.errorMessage = m_options.targetFormat == TargetEdaFormat::Allegro
390 ? QStringLiteral("Allegro Import Package 缺少 STEP 3D 模型")
391 : m_options.targetFormat == TargetEdaFormat::LibrePcb
392 ? QStringLiteral("LibrePCB 库缺少 STEP 3D 模型")
393 : m_options.targetFormat == TargetEdaFormat::Horizon
394 ? QStringLiteral("Horizon Pool 缺少 STEP 3D 模型")
395 : QStringLiteral("STEP 3D model was not embedded in PcbLib");
396 }
397 emit embeddedModel3DStatusChanged(componentId, modelStatus);
398 }
399
400 qDebug() << "FootprintExportStage: Collected footprint for" << componentId;
401 }
402
403 if (m_options.targetFormat == TargetEdaFormat::Altium) {
404 // Altium PcbLib 要求封装名称（不区分大小写）唯一。实际 BOM 中多个
405 // 元件经常共用同一个封装，例如多个 C0603，但它们仍可能关联不同
406 // 的元件数据或三维模型，因此不能简单丢弃重复项。
407 QSet<QString> usedNames;
408 for (int index = 0; index < footprintList.size(); ++index) { // 当前封装索引。
409 FootprintData& footprint = footprintList[index];
410 FootprintInfo info = footprint.info();
411 const QString baseName = info.name.trimmed();
412 QString uniqueName = baseName;
413 int suffix = 1;
414 while (usedNames.contains(uniqueName.toCaseFolded())) {
415 const QString componentId = collectedIds.value(index);
416 uniqueName = QStringLiteral("%1_%2").arg(baseName, componentId);
417 if (suffix > 1)
418 uniqueName += QStringLiteral("_%1").arg(suffix);
419 ++suffix;
420 }
421
422 if (uniqueName != info.name) {
423 info.name = uniqueName;
424 footprint.setInfo(info);
425 qWarning() << "FootprintExportStage: Renamed duplicate Altium footprint" << baseName << "to"
426 << uniqueName;
427 }
428 usedNames.insert(uniqueName.toCaseFolded());
429 }
430 }
431
432 if (m_cancelled.load()) {
433 m_isExporting.store(false);
434 m_isRunning.store(false);
435 emit completed(0, componentIds.size(), 0);
436 return;
437 }
438
439 if (footprintList.isEmpty() && componentIrList.isEmpty() && symbolList.isEmpty()) {
440 qWarning() << "FootprintExportStage: No valid library data to export";
441 m_isExporting.store(false);
442 m_isRunning.store(false);
443 emit completed(0, componentIds.size(), 0);
444 return;
445 }

622 return;

639 abortExport(

641 return;
642 }
643 }

645 QDir(finalPath).exists() &&
646 (!m_options.overwriteExistingFiles || m_options.updateMode || m_options.retryMode);
647 if (preserveExistingFootprints) {
648 if (!QDir().mkpath(tempPath)) {
649 abortExport(QStringLiteral("Failed to create temp dir for merge: %1").arg(tempPath));
650 return;
651 }
652 if (m_options.targetFormat != TargetEdaFormat::Horizon) {
653 const QStringList existingFiles =
654 QDir(finalPath).entryList({"*.kicad_mod"}, QDir::Files);
655 for (const QString& file : existingFiles) { // 当前既有封装文件名。
656 if (!QFile::copy(finalPath + QDir::separator() + file, tempPath + QDir::separator() + file)) {
657 qWarning() << "FootprintExportStage: Failed to copy existing footprint:" << file;
658 }
659 }

661 }
662 }
663 }

665 qDebug() << "FootprintExportStage: Exporting" << footprintList.size() << "footprints to temp:" << tempPath;
666 qDebug() << "FootprintExportStage: fileExt:" << fileExt << "isDirOutput:" << isDirOutput;
667
668 bool exportSuccess = false;
669 QStringList exporterDiagnostics;
670 QString libraryDescription = m_options.footprintLibraryDescription;
671 {
672 // 当前目标格式是否优先使用 WRL 模型。
673 const bool preferWrl = m_options.needsModel3DWrl() &&
674 m_options.targetFormat != TargetEdaFormat::Altium;
675 const bool exportStep = m_options.needsEmbeddedModel3DStep();
676 QString libraryKeywords = m_options.footprintLibraryKeywords;

678 QList<IR::FootprintComponentIR> irFootprintList;
679 irFootprintList.reserve(footprintList.size());

681 irFootprintList.append(IR::toFootprintIR(fd));
682 }

684 m_options.targetFormat == TargetEdaFormat::Eagle || m_options.targetFormat == TargetEdaFormat::Cadstar ||
685 m_options.targetFormat == TargetEdaFormat::Allegro || m_options.targetFormat == TargetEdaFormat::LibrePcb ||
686 m_options.targetFormat == TargetEdaFormat::Horizon;

688 exportSuccess = exporter->exportComponentLibrary(
689 componentIrList, libName, tempPath, m_options.exportModel3D, outputDir);

693 exporterDiagnostics.append(QStringLiteral("目标格式未提供组合库符号导出器"));
694 } else {
695 exportSuccess = symbolExporter->exportSymbolLibrary(
696 symbolList, libName, tempPath, false, m_options.updateMode, libraryDescription);
697 }

699 exportSuccess = exporter->exportFootprintLibrary(
700 irFootprintList,
701 libName,
702 tempPath,
703 preferWrl,
704 exportStep,
705 libraryDescription,
706 libraryKeywords,

708 outputDir);
709 }
710 const QStringList exporterReportedDiagnostics = exporter->diagnostics();

712 if (!exporterDiagnostics.contains(diagnostic))
713 exporterDiagnostics.append(diagnostic);
714 }
715 if (!exporterDiagnostics.isEmpty()) {
716 QMutexLocker locker(&m_progressMutex);
717 for (const QString& diagnostic : exporterDiagnostics) { // 当前待发布诊断。
718 if (!m_progress.diagnostics.contains(diagnostic))
719 m_progress.diagnostics.append(diagnostic);
720 }
721 const ExportTypeProgress progressSnapshot = m_progress;
722 locker.unlock();
723 emit progressChanged(progressSnapshot);
724 }
725 qDebug() << "FootprintExportStage: exportFootprintLibrary result:" << exportSuccess;
726 }
727
728 if (m_cancelled.load()) {
729 abortExport(QStringLiteral("Export cancelled"));
730 return;
731 }
732

735 if (!exporterDiagnostics.isEmpty())
736 errorMessage += QStringLiteral(": ") + exporterDiagnostics.join(QStringLiteral("; "));
737 abortExport(errorMessage);
738 return;
739 }

742 QStringList horizonDiagnostics;

744 abortExport(QStringLiteral("Horizon Pool 官方更新失败，原有库未修改：%1")
745 .arg(horizonDiagnostics.join(QStringLiteral("; "))));
746 return;

752 workspaceTempPath = m_tempManager.createTempDirectoryPath(libName + QStringLiteral("-workspace-library"));
753 if (workspaceTempPath.isEmpty() || !copyDirectoryTree(tempPath, workspaceTempPath)) {
754 abortExport(
755 QStringLiteral("LibrePCB 工作区库准备失败，未修改项目或工作区库: %1").arg(librePcbWorkspaceLibrary));
756 return;
757 }

760 }

764 m_tempManager.createTempDirectoryPath(libName + QStringLiteral("-project-library"));

766 abortExport(QStringLiteral("LibrePCB 项目库合并失败，已保留原项目库: %1").arg(finalPath));
767 return;
768 }
769 tempPath = mergedTempPath;
770 }
771
772 // 提交临时文件/目录到最终路径
773 bool commitSuccess = false;
774 if (exportIntoLibrePcbProject) {
775 QVector<TempFileManager::CommitItem> commitItems;
776 commitItems.append({tempPath, finalPath, true});
777 if (!workspaceTempPath.isEmpty())
778 commitItems.append({workspaceTempPath, librePcbWorkspaceLibrary, true});
779 commitSuccess = m_tempManager.commitBatch(commitItems);
780 } else if (!isDirOutput) {
781 commitSuccess = m_tempManager.commitWithBackup(tempPath, finalPath);
782 } else {
783 commitSuccess = m_tempManager.commitDirectoryWithBackup(tempPath, finalPath);
784 }
785
786 if (commitSuccess) {
787 qDebug() << "FootprintExportStage: Successfully exported to:" << finalPath;
788 if (!librePcbWorkspaceLibrary.isEmpty())
789 qDebug() << "FootprintExportStage: Installed LibrePCB workspace library to:" << librePcbWorkspaceLibrary;
790 } else {
791 abortExport(QStringLiteral("Failed to commit temp path"));
792 return;
793 }
794
795 if (m_options.targetFormat == TargetEdaFormat::Horizon) {
796 QStringList horizonDiagnostics;
797 if (!HorizonPoolIntegration::registerPool(finalPath, horizonDiagnostics)) {
798 abortExport(QStringLiteral("Horizon Pool 已生成，但官方注册失败：%1")
799 .arg(horizonDiagnostics.join(QStringLiteral("; "))));
800 return;
801 }
802 if (!horizonDiagnostics.isEmpty()) {
803 QMutexLocker locker(&m_progressMutex);
804 for (const QString& diagnostic : std::as_const(horizonDiagnostics)) {
805 if (!m_progress.diagnostics.contains(diagnostic))
806 m_progress.diagnostics.append(diagnostic);
807 }
808 const ExportTypeProgress progressSnapshot = m_progress;
809 locker.unlock();
810 emit progressChanged(progressSnapshot);
811 for (const QString& diagnostic : std::as_const(horizonDiagnostics))
812 qWarning() << "FootprintExportStage:" << diagnostic;
813 }
814 }
815
816 // 目录输出模式下注册库（如 KiCad 库表）
817 if (isDirOutput && m_options.targetFormat == TargetEdaFormat::KiCad && !libraryDescription.isEmpty()) {
818 KiCadLibraryTableManager::registerFootprintLibrary(outputDir, libName, finalPath, libraryDescription);
819 }
820
821 // 注意：不在这里清理临时目录，由 ParallelExportService 统一管理
822
823 qDebug() << "FootprintExportStage: Completed. Success:" << successCount << "Failed:" << failedIds.size();
824
825 m_isExporting.store(false);
826 m_isRunning.store(false);
827 emit completed(successCount, failedIds.size(), skippedCount);
828}
