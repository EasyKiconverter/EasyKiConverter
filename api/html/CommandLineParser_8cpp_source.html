120bool CommandLineParser::parse() {
121 // 重要：使用存储的 argv 而非 QCoreApplication::arguments()
122 // 因为此函数可能在 QCoreApplication 创建前被调用
123 QStringList args;
124 if (m_argv) {
125 args.reserve(m_argc);
126 for (int i = 0; i < m_argc; ++i) {
127 args.append(QString::fromLocal8Bit(m_argv[i]));
128 }
129 }
130
131 // 保存命令行解析结果。
132 bool result = m_parser.parse(args);
133
134 if (result) {
135 // 检测 CLI 子命令
136 // 使用 args 而非 QCoreApplication::arguments() 保证可靠性
137
138 // 检查是否包含 "convert" 命令
139 for (int i = 1; i < args.size(); ++i) {
140 // 获取当前命令行参数的小写形式。
141 QString arg = args[i].toLower();
142 if (arg == "convert") {
143 m_hasConvertCommand = true;
144 // 检查下一个参数
145 if (i + 1 < args.size()) {
146 // 获取 convert 命令后的子命令。
147 QString subcommand = args[i + 1].toLower();
148 if (subcommand == "bom") {
149 m_hasBomSubcommand = true;
150 m_cliMode = CliMode::ConvertBom;
151 } else if (subcommand == "component") {
152 m_hasComponentSubcommand = true;
153 m_cliMode = CliMode::ConvertComponent;
154 } else if (subcommand == "batch") {
155 m_hasBatchSubcommand = true;
156 m_cliMode = CliMode::ConvertBatch;
157 }
158 }
159 break;
160 }
161 }
162 }
163
164 return result;
165}

293bool CommandLineParser::validate() const {
294 // 验证日志级别
295 if (m_parser.isSet(m_logLevelOption)) {
296 // 获取用户指定的日志级别。
297 QString level = logLevel();
298 // 定义允许的日志级别集合。
299 QStringList validLevels = {"trace", "debug", "info", "warn", "error", "fatal"};
300 if (!validLevels.contains(level)) {
301 return false;
302 }
303 }
304
305 // 验证语言设置
306 if (m_parser.isSet(m_languageOption)) {
307 // 获取用户指定的界面语言。
308 QString lang = language();
309 // 定义允许的界面语言集合。
310 QStringList validLangs = {"zh_CN", "en"};
311 if (!validLangs.contains(lang)) {
312 return false;
313 }
314 }
315
316 // 验证主题设置
317 if (m_parser.isSet(m_themeOption)) {
318 // 获取用户指定的界面主题。
319 QString theme = this->theme();
320 // 定义允许的界面主题集合。
321 QStringList validThemes = {"dark", "light"};
322 if (!validThemes.contains(theme)) {
323 return false;
324 }
325 }
326
327 // CLI 模式验证
328 if (m_parser.isSet(m_cacheDirOption) && cacheDir().trimmed().isEmpty()) {
329 return false;
330 }
331
332 if (m_parser.isSet(m_diskCacheLimitOption)) {
333 // 保存磁盘缓存大小的解析状态。
334 bool ok = false;
335 // 读取用户指定的磁盘缓存大小。
336 const int sizeMB = m_parser.value(m_diskCacheLimitOption).toInt(&ok);
337 if (!ok || sizeMB <= 0) {
338 return false;
339 }
340 }
341
342 // CLI 模式验证
343 if (isCliMode()) {
344 if (m_parser.isSet(m_3dModelFormatOption)) {
345 if (!VALID_3D_MODEL_FORMATS.contains(model3DFormat())) {
346 return false;
347 }
348 }
349
350 if (m_parser.isSet(m_3dPathModeOption)) {
351 if (!VALID_3D_PATH_MODES.contains(model3DPathMode())) {
352 return false;
353 }
354 }
355
356 // 验证输出目录
357 if (!m_parser.isSet(m_outputOption)) {
358 return false;
359 }
360
361 // BOM 转换需要输入文件
362 if (m_cliMode == CliMode::ConvertBom && !m_parser.isSet(m_inputOption)) {
363 return false;
364 }
365
366 // 单个元器件转换需要组件编号
367 if (m_cliMode == CliMode::ConvertComponent && !m_parser.isSet(m_componentOption)) {
368 return false;
369 }
370
371 // 批量转换需要输入文件
372 if (m_cliMode == CliMode::ConvertBatch && !m_parser.isSet(m_inputOption)) {
373 return false;
374 }
375 }
376
377 return true;
378}

383QString CommandLineParser::validationError() const {
384 // 收集所有命令行参数校验错误。
385 QStringList errors;
386
387 // 检查日志级别
388 if (m_parser.isSet(m_logLevelOption)) {
389 // 获取用户指定的日志级别。
390 QString level = logLevel();
391 // 定义允许的日志级别集合。
392 QStringList validLevels = {"trace", "debug", "info", "warn", "error", "fatal"};
393 if (!validLevels.contains(level)) {
394 errors.append(QCoreApplication::translate("CommandLineParser", "无效的日志级别: %1（有效值: %2）")
395 .arg(level)
396 .arg(validLevels.join(", ")));
397 }
398 }
399
400 // 检查语言设置
401 if (m_parser.isSet(m_languageOption)) {
402 // 获取用户指定的界面语言。
403 QString lang = language();
404 // 定义允许的界面语言集合。
405 QStringList validLangs = {"zh_CN", "en"};
406 if (!validLangs.contains(lang)) {
407 errors.append(QCoreApplication::translate("CommandLineParser", "无效的语言设置: %1（有效值: %2）")
408 .arg(lang)
409 .arg(validLangs.join(", ")));
410 }
411 }
412
413 // 检查主题设置
414 if (m_parser.isSet(m_themeOption)) {
415 // 获取用户指定的界面主题。
416 QString theme = this->theme();
417 // 定义允许的界面主题集合。
418 QStringList validThemes = {"dark", "light"};
419 if (!validThemes.contains(theme)) {
420 errors.append(QCoreApplication::translate("CommandLineParser", "无效的主题设置: %1（有效值: %2）")
421 .arg(theme)
422 .arg(validThemes.join(", ")));
423 }
424 }
425
426 // CLI 模式验证错误
427 if (m_parser.isSet(m_cacheDirOption) && cacheDir().trimmed().isEmpty()) {
428 errors.append(QCoreApplication::translate("CommandLineParser", "缓存目录不能为空"));
429 }
430
431 if (m_parser.isSet(m_diskCacheLimitOption)) {
432 // 保存磁盘缓存大小的解析状态。
433 bool ok = false;
434 // 读取用户指定的磁盘缓存大小。
435 const int sizeMB = m_parser.value(m_diskCacheLimitOption).toInt(&ok);
436 if (!ok || sizeMB <= 0) {
437 errors.append(
438 QCoreApplication::translate("CommandLineParser", "磁盘缓存大小必须是大于 0 的整数（单位: MB）"));
439 }
440 }
441
442 // CLI 模式验证错误
443 if (isCliMode()) {
444 if (m_parser.isSet(m_3dModelFormatOption)) {
445 // 获取用户指定的三维模型格式。
446 const QString format = model3DFormat();
447 if (!VALID_3D_MODEL_FORMATS.contains(format)) {
448 errors.append(QCoreApplication::translate("CommandLineParser", "无效的 3D 模型格式: %1（有效值: %2）")
449 .arg(format)
450 .arg(VALID_3D_MODEL_FORMATS.join(", ")));
451 }
452 }
453
454 if (m_parser.isSet(m_3dPathModeOption)) {
455 // 获取用户指定的三维模型路径模式。
456 const QString pathMode = model3DPathMode();
457 if (!VALID_3D_PATH_MODES.contains(pathMode)) {
458 errors.append(
459 QCoreApplication::translate("CommandLineParser", "无效的 3D 模型路径模式: %1（有效值: %2）")
460 .arg(pathMode, VALID_3D_PATH_MODES.join(", ")));
461 }
462 }
463
464 if (!m_parser.isSet(m_outputOption)) {
465 errors.append(QCoreApplication::translate("CommandLineParser", "CLI 模式必须指定输出目录 (-o/--output)"));
466 }
467
468 if (m_cliMode == CliMode::ConvertBom && !m_parser.isSet(m_inputOption)) {
469 errors.append(QCoreApplication::translate("CommandLineParser", "BOM 表转换必须指定输入文件 (-i/--input)"));
470 }
471
472 if (m_cliMode == CliMode::ConvertComponent && !m_parser.isSet(m_componentOption)) {
473 errors.append(
474 QCoreApplication::translate("CommandLineParser", "单个元器件转换必须指定 LCSC 编号 (-c/--component)"));
475 }
476
477 if (m_cliMode == CliMode::ConvertBatch && !m_parser.isSet(m_inputOption)) {
478 errors.append(QCoreApplication::translate("CommandLineParser", "批量转换必须指定输入文件 (-i/--input)"));
479 }
480 }
481
482 return errors.join("\n");
483}
