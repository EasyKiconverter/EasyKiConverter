42QString safeName(const QString& value) {
43 QString result; // 清洗后的名称。
44 for (const QChar character : value) { // 逐字符处理原始名称。
45 const ushort code = character.unicode(); // 当前字符编码。
46 if ((code >= 'A' && code <= 'Z') || (code >= 'a' && code <= 'z') || (code >= '0' && code <= '9') ||
47 character == QChar('_') || character == QChar('-'))
48 result.append(character);
49 else
50 result.append(QChar('_'));
51 }
52 return result.left(64);
53}

111 switch (pad.shape) {

121 case IR::PadShape::Trapezoid:

168bool validateFootprint(const IR::FootprintComponentIR& footprint, QStringList& diagnostics) {
169 if (safeName(footprint.name).isEmpty()) {
170 diagnostics.append(QStringLiteral("P-CAD: 封装名称清洗后为空"));
171 return false;
172 }
173 if (!isAscii(footprint.name)) {
174 diagnostics.append(QStringLiteral("P-CAD: 封装名称必须是 ASCII，无法安全编码：%1").arg(footprint.name));
175 return false;
176 }
177 if (!footprint.holes.isEmpty()) {
178 diagnostics.append(
179 QStringLiteral("P-CAD: 封装 %1 含独立安装孔，当前 Pattern 输出无法表达无编号机械孔").arg(footprint.name));
180 return false;
181 }
182 if (!footprint.models3d.isEmpty())
183 diagnostics.append(QStringLiteral("P-CAD: Pattern ASCII 不包含三维模型关联，将由独立三维阶段输出 %1 个模型")
184 .arg(footprint.models3d.size()));
185 for (const IR::FootprintPadIR& pad : footprint.pads) {
186 if (pad.number.isEmpty() || !isAscii(pad.number)) {
187 diagnostics.append(QStringLiteral("P-CAD: 焊盘编号必须为非空 ASCII 文本：%1").arg(pad.number));
188 return false;
189 }
190 if (!std::isfinite(pad.position.x()) || !std::isfinite(pad.position.y()) || !std::isfinite(pad.size.width()) ||
191 !std::isfinite(pad.size.height()) || !std::isfinite(pad.holeSize) || !std::isfinite(pad.holeLength)) {
192 diagnostics.append(QStringLiteral("P-CAD: 焊盘 %1 包含非法坐标、尺寸或孔参数").arg(pad.number));
193 return false;
194 }
195 if (pad.holeLength > 0.0) {
196 diagnostics.append(
197 QStringLiteral("P-CAD: 焊盘 %1 的槽孔当前没有可靠的 ASCII Pad Style 映射").arg(pad.number));
198 return false;
199 }
200 if (pad.isSmd() && pad.layer != IR::LayerType::TopCopper && pad.layer != IR::LayerType::BottomCopper) {
201 diagnostics.append(QStringLiteral("P-CAD: SMD 焊盘 %1 使用无法映射的铜层").arg(pad.number));
202 return false;
203 }
204 if (padShape(pad, diagnostics).isEmpty())
205 return false;
206 }
207 for (const IR::FootprintTextIR& text : footprint.texts) {
208 if (text.isDisplayed && !isAscii(text.text)) {
209 diagnostics.append(QStringLiteral("P-CAD: 文本必须是 ASCII，无法安全编码：%1").arg(text.text));
210 return false;
211 }
212 }
213 return true;
214}

233bool writeGraphics(QTextStream& stream, const IR::FootprintComponentIR& footprint, QStringList& diagnostics) {
234 const auto layerOf = [&diagnostics](IR::LayerType layer) -> std::optional<int> {
235 const std::optional<int> numberValue = layerNumber(layer); // P-CAD 图层编号。
236 if (!numberValue.has_value())
237 diagnostics.append(QStringLiteral("P-CAD: 图元使用无法映射的 IR 图层"));
238 return numberValue;
239 };
240
241 for (const IR::FootprintTrackIR& track : footprint.tracks) {
242 if (track.points.size() < 2)
243 continue;
244 const std::optional<int> layer = layerOf(track.layer); // 走线目标图层。
245 if (!layer.has_value())
246 return false;
247 stream << " (layerContents (layerNumRef " << *layer << ')';
248 for (int index = 0; index + 1 < track.points.size(); ++index) {
249 stream << "\n (line";
250 writePoints(stream, {track.points.at(index), track.points.at(index + 1)});
251 stream << " (width " << number(track.width) << "))";
252 }
253 stream << "\n )\n";
254 }
255 for (const IR::FootprintCircleIR& circle : footprint.circles) {
256 const std::optional<int> layer = layerOf(circle.layer); // 圆形图元目标图层。
257 if (!layer.has_value())
258 return false;
259 stream << " (layerContents (layerNumRef " << *layer << ")\n (circle";
260 writePoints(stream, {circle.center});
261 stream << " (radius " << number(circle.radius) << ") (width " << number(circle.strokeWidth) << "))\n"
262 << " )\n";
263 }
264 for (const IR::FootprintRectangleIR& rectangle : footprint.rectangles) {
265 const std::optional<int> layer = layerOf(rectangle.layer); // 矩形图元目标图层。
266 if (!layer.has_value())
267 return false;
268 const QRectF bounds = rectangle.bounds; // 矩形边界。
269 const QList<QPointF> corners = {// 矩形闭合顶点。
270 bounds.topLeft(),
271 bounds.topRight(),
272 bounds.bottomRight(),
273 bounds.bottomLeft(),
274 bounds.topLeft()};
275 stream << " (layerContents (layerNumRef " << *layer << ')';
276 for (int index = 0; index + 1 < corners.size(); ++index) {
277 stream << "\n (line";
278 writePoints(stream, {corners.at(index), corners.at(index + 1)});
279 stream << " (width " << number(rectangle.strokeWidth) << "))";
280 }
281 stream << "\n )\n";
282 }
283 for (const IR::FootprintRegionIR& region : footprint.regions) {
284 if (region.vertices.size() < 3)
285 continue;
286 const std::optional<int> layer = layerOf(region.layer); // 区域目标图层。
287 if (!layer.has_value())
288 return false;
289 stream << " (layerContents (layerNumRef " << *layer << ")\n (pcbPoly";
290 writePoints(stream, region.vertices);
291 stream << ")\n )\n";
292 }
293 for (const IR::FootprintOutlineIR& outline : footprint.outlines) {
294 if (outline.points.size() < 2)
295 continue;
296 const std::optional<int> layer = layerOf(outline.layer); // 轮廓目标图层。
297 if (!layer.has_value())
298 return false;
299 stream << " (layerContents (layerNumRef " << *layer << ')';
300 for (int index = 0; index + 1 < outline.points.size(); ++index) {
301 stream << "\n (line";
302 writePoints(stream, {outline.points.at(index), outline.points.at(index + 1)});
303 stream << " (width " << number(outline.strokeWidth) << "))";
304 }
305 stream << "\n )\n";
306 }
307 for (const IR::FootprintTextIR& text : footprint.texts) {
308 if (!text.isDisplayed || text.text.isEmpty())
309 continue;
310 const std::optional<int> layer = layerOf(text.layer); // 文本目标图层。
311 if (!layer.has_value())
312 return false;
313 stream << " (layerContents (layerNumRef " << *layer << ")\n"
314 << " (text " << quote(text.text) << "\n"
315 << " (pt " << number(text.position.x()) << ' ' << number(-text.position.y()) << ")\n"
316 << " (rotation " << QString::number(text.rotation * 10.0, 'f', 3) << ")\n"
317 << " (isFlipped " << (text.mirror ? "True" : "False") << ")\n"
318 << " )\n )\n";
319 }
320 return true;
321}

331bool writeLibrary(const QList<IR::FootprintComponentIR>& footprints,
332 const QString& libName,
333 const QString& path,
334 QStringList& diagnostics) {
335 if (footprints.isEmpty()) {
336 diagnostics.append(QStringLiteral("P-CAD: 没有可导出的封装"));
337 return false;
338 }
339 QSet<QString> names; // 已使用的封装名称。
340 QHash<QString, QString> padStyles; // Pad Style 去重键到名称的映射。
341 QList<const IR::FootprintPadIR*> stylePads; // 每种 Pad Style 的代表焊盘。
342 for (const IR::FootprintComponentIR& footprint : footprints) {
343 const QString name = safeName(footprint.name); // 当前封装名称。
344 if (!validateFootprint(footprint, diagnostics))
345 return false;
346 if (names.contains(name)) {
347 diagnostics.append(QStringLiteral("P-CAD: 封装名称清洗后冲突：%1").arg(name));
348 return false;
349 }
350 names.insert(name);
351 for (const IR::FootprintPadIR& pad : footprint.pads) {
352 const QString shape = padShape(pad, diagnostics); // 当前焊盘形状文本。
353 if (shape.isEmpty())
354 return false;
355 const QString key = padKey(pad, shape); // 当前焊盘去重键。
356 if (!padStyles.contains(key)) {
357 padStyles.insert(key, QStringLiteral("EK_%1").arg(padStyles.size() + 1));
358 stylePads.append(&pad);
359 }
360 }
361 }
362
363 QFile file(path); // P-CAD 库输出文件。
364 if (!file.open(QIODevice::WriteOnly | QIODevice::Truncate | QIODevice::Text)) {
365 diagnostics.append(QStringLiteral("P-CAD: 无法写入文件：%1").arg(path));
366 return false;
367 }
368 QTextStream stream(&file); // UTF-8 P-CAD 输出流。
369 stream.setEncoding(QStringConverter::Utf8);
370 stream << "(ACCEL_ASCII " << quote(libName.isEmpty() ? QStringLiteral("EasyKiConverter") : libName) << "\n"
371 << " (asciiHeader\n (asciiVersion 4 0)\n (fileUnits MM)\n )\n"
372 << " (library\n";
373 for (const IR::FootprintPadIR* pad : stylePads) {
374 const QString shape = padShape(*pad, diagnostics); // Pad Style 形状文本。
375 stream << " (padStyleDef " << quote(padStyles.value(padKey(*pad, shape))) << "\n"
376 << " (holeDiam " << number(pad->isThroughHole() ? pad->holeSize : 0.0) << ")\n"
377 << " (isHolePlated " << (pad->isPlated ? "True" : "False") << ")\n";
378 if (pad->isThroughHole()) {
379 writePadShape(stream, *pad, shape, 1);
380 writePadShape(stream, *pad, shape, 2);
381 } else {
382 writePadShape(stream, *pad, shape, pad->layer == IR::LayerType::BottomCopper ? 2 : 1);
383 }
384 stream << " )\n";
385 }
386 for (const IR::FootprintComponentIR& footprint : footprints) {
387 const QString name = safeName(footprint.name); // 当前封装名称。
388 stream << " (patternDef " << quote(name) << "\n"
389 << " (originalName " << quote(name) << ")\n"
390 << " (multiLayer\n";
391 for (const IR::FootprintPadIR& pad : footprint.pads) {
392 const QString shape = padShape(pad, diagnostics); // 当前焊盘形状文本。
393 stream << " (pad (padNum " << quote(pad.number) << ") (padStyleRef "
394 << quote(padStyles.value(padKey(pad, shape))) << ") (pt " << number(pad.position.x()) << ' '
395 << number(-pad.position.y()) << ") (rotation " << QString::number(pad.rotation * 10.0, 'f', 3)
396 << "))\n";
397 }
398 stream << " )\n";
399 if (!writeGraphics(stream, footprint, diagnostics))
400 return false;
401 stream << " )\n";
402 }
403 stream << " )\n)\n";
404 stream.flush();
405 if (stream.status() != QTextStream::Ok) {
406 diagnostics.append(QStringLiteral("P-CAD: 写入文件时发生 I/O 错误：%1").arg(path));
407 return false;
408 }
409 return true;
410}
