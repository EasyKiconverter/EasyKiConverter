90bool parsePair(const QString& text,
91 LengthUnit unit,
92 ParseDiagnostics* diagnostics,
93 const QString& field,
94 int line,
95 QPointF& result) {
96 const QString value = text.trimmed(); // 去除坐标文本外围空白。
97 const bool hasOpeningParenthesis = value.startsWith(QChar('(')); // 是否有左括号。
98 const bool hasClosingParenthesis = value.endsWith(QChar(')')); // 是否有右括号。
99 if (hasOpeningParenthesis != hasClosingParenthesis) {
100 if (diagnostics)
101 diagnostics->add(ParseSeverity::Error,
102 ParseScope::Field,
103 QStringLiteral("坐标字段括号不匹配：%1").arg(text),
104 field,
105 line);
106 return false;
107 }
108 // 去除成对括号后保留两个坐标分量。
109 const QString pairText = hasOpeningParenthesis ? value.mid(1, value.size() - 2).trimmed() : value;
110 static const QRegularExpression pattern(QStringLiteral(R"(^\s*([^,\s()]+)\s*(?:,|\s+)\s*([^,\s()]+)\s*$)"));
111 const QRegularExpressionMatch match = pattern.match(pairText); // 坐标两分量匹配结果。
112 if (!match.hasMatch()) {
113 if (diagnostics)
114 diagnostics->add(
115 ParseSeverity::Error, ParseScope::Field, QStringLiteral("坐标字段格式错误：%1").arg(text), field, line);
116 return false;
117 }
118 const int diagnosticCount = diagnostics ? diagnostics->items().size() : 0; // 解析前诊断数量。
119 const double x = StrictNumberParser::parseDouble(match.captured(1), diagnostics, field, line); // X 坐标。
120 const double y = StrictNumberParser::parseDouble(match.captured(2), diagnostics, field, line); // Y 坐标。
121 if (diagnostics && diagnostics->items().size() != diagnosticCount)
122 return false;
123 result = QPointF(scale(x, unit), scale(y, unit));
124 return true;
125}

178QList<QPointF> parsePoints(const QString& text,
179 LengthUnit unit,
180 ParseDiagnostics* diagnostics,
181 const QString& field,
182 int line) {
183 QList<QPointF> points; // 保存解析出的坐标点。
184 const QString value = text.trimmed(); // 清理坐标文本外围空白。
185 if (!value.contains(QChar('(')) && !value.contains(QChar(')'))) {
186 QPointF point; // 单坐标文本解析结果。
187 if (parsePair(text, unit, diagnostics, field, line, point))
188 points.append(point);
189 return points;
190 }
191
192 int position = 0; // 当前扫描位置。
193 while (position < value.size()) {
194 while (position < value.size() && value.at(position).isSpace())
195 ++position;
196 if (position >= value.size())
197 break;
198 if (value.at(position) != QChar('(')) {
199 if (diagnostics)
200 diagnostics->add(ParseSeverity::Error,
201 ParseScope::Field,
202 QStringLiteral("坐标点之间存在无法解析的文本：%1").arg(value.mid(position)),
203 field,
204 line);
205 break;
206 }
207 const int closing = value.indexOf(QChar(')'), position + 1); // 当前坐标点的右括号位置。
208 if (closing < 0) {
209 if (diagnostics)
210 diagnostics->add(ParseSeverity::Error,
211 ParseScope::Field,
212 QStringLiteral("坐标字段缺少右括号：%1").arg(value.mid(position)),
213 field,
214 line);
215 break;
216 }
217 QPointF point; // 当前括号坐标解析结果。
218 if (parsePair(value.mid(position, closing - position + 1), unit, diagnostics, field, line, point))
219 points.append(point);
220 position = closing + 1;
221 }
222 return points;
223}

247QString uniqueName(const QString& name,
248 const QStringList& names,
249 ParseDiagnostics* diagnostics,
250 ParseScope scope,
251 int line) {
252 if (!names.contains(name))
253 return name;
254 duplicateWarning(diagnostics, scope, name, line);
255 int suffix = 2; // 稳定名称后缀起始值。
256 QString candidate; // 当前候选名称。
257 do {
258 candidate = QStringLiteral("%1_%2").arg(name).arg(suffix++);
259 } while (names.contains(candidate));

271 if (upper(node.keyword) == QStringLiteral("ROUND") || upper(node.keyword) == QStringLiteral("SQUARE")) {
272 if (const SectionNode* diameter = child(node, QStringLiteral("DIAMETER"))) {

287 unit));

300 const QString rawName = valueOf(node); // 来源 Pad 名称。
301 XpeditionPadDefinition pad; // 当前 Pad 定义。
302 pad.name = uniqueName(
303 rawName,
304 [&]() {
305 QStringList names; // 已使用的 Pad 名称。
306 for (const auto& item : model.pads)
307 names.append(item.name);
308 return names;
309 }(),
310 diagnostics,
311 ParseScope::Footprint,
312 node.line);
313 pad.line = node.line;
314 const SectionNode* geometry = nullptr; // 首个可识别的 Pad 几何节点。
315 for (const SectionNode& childNode : node.children) {
316 const XpeditionPadShape candidate = shapeOf(childNode.keyword); // 子节点对应的形状。
317 if (candidate != XpeditionPadShape::Unknown && geometry == nullptr) {
318 geometry = &childNode;
319 pad.shape = candidate;
320 }
321 }
322 if (geometry != nullptr) {
323 parseSize(*geometry, unit, diagnostics, pad.size);
324 if (const SectionNode* offset = child(*geometry, QStringLiteral("OFFSET")))
325 parsePair(valueOf(*offset), unit, diagnostics, QStringLiteral("OFFSET"), offset->line, pad.offset);
326 for (const SectionNode* xy : childrenOf(*geometry, QStringLiteral("XY")))
327 pad.polygon.append(parsePoints(valueOf(*xy), unit, diagnostics, QStringLiteral("XY"), xy->line));
328 } else {
329 if (diagnostics)
330 diagnostics->add(ParseSeverity::Skipped,
331 ParseScope::Footprint,
332 QStringLiteral("Pad 没有可识别的几何定义"),
333 pad.name,
334 node.line);
335 }
336 model.pads.append(pad);
337 model.padNameVariants[rawName].append(pad.name);

443void parseCell(const SectionNode& node, LengthUnit unit, XpeditionHkpModel& model, ParseDiagnostics* diagnostics) {
444 XpeditionCellDefinition cell; // 当前 Cell 定义。
445 cell.name = uniqueName(
446 valueOf(node),
447 [&]() {
448 QStringList names; // 已使用的 Cell 名称。
449 for (const auto& item : model.cells)
450 names.append(item.name);
451 return names;
452 }(),
453 diagnostics,
454 ParseScope::Footprint,
455 node.line);
456 cell.line = node.line;
457 for (const SectionNode& item : node.children) {
458 // 规范化当前 Cell 子节点关键字。
459 const QString keyword = upper(item.keyword);
460 if (keyword == QStringLiteral("PACKAGE_GROUP"))
461 cell.packageGroup = valueOf(item);
462 else if (keyword == QStringLiteral("MOUNT_TYPE"))
463 cell.mountType = valueOf(item);
464 else if (keyword == QStringLiteral("NUMBER_LAYERS"))
465 cell.numberOfLayers = static_cast<int>(StrictNumberParser::parseInteger(
466 valueOf(item), diagnostics, QStringLiteral("NUMBER_LAYERS"), item.line));
467 // 解析引脚位置、Padstack、旋转和镜像信息。
468 else if (keyword == QStringLiteral("PIN")) {
469 XpeditionCellPin pin; // 当前封装引脚。
470 pin.number = valueOf(item);
471 pin.line = item.line;
472 if (const SectionNode* xy = child(item, QStringLiteral("XY")))
473 parsePair(valueOf(*xy), unit, diagnostics, QStringLiteral("XY"), xy->line, pin.position);
474 if (const SectionNode* padstack = child(item, QStringLiteral("PADSTACK")))
475 pin.padstack = valueOf(*padstack);
476 if (const SectionNode* rotation = child(item, QStringLiteral("ROTATION")))
477 pin.rotation = StrictNumberParser::parseDouble(
478 valueOf(*rotation), diagnostics, QStringLiteral("ROTATION"), rotation->line);
479 if (const SectionNode* mirror = child(item, QStringLiteral("MIRROR")))
480 pin.mirror = parseBool(valueOf(*mirror), diagnostics, QStringLiteral("MIRROR"), mirror->line);
481 cell.pins.append(pin);
482 } else if (keyword.endsWith(QStringLiteral("OUTLINE"))) {
483 XpeditionCellOutline outline; // 当前封装轮廓。
484 outline.layer = item.keyword;
485 outline.line = item.line;
486 for (const SectionNode& shape : item.children) {
487 for (const SectionNode* xy : childrenOf(shape, QStringLiteral("XY")))
488 outline.points.append(parsePoints(valueOf(*xy), unit, diagnostics, QStringLiteral("XY"), xy->line));
489 }
490 if (!outline.points.isEmpty())
491 cell.outlines.append(outline);
492 }
493 }
494 if (cell.pins.isEmpty() && diagnostics)
495 diagnostics->add(
496 ParseSeverity::Skipped, ParseScope::Footprint, QStringLiteral("空 Cell 被跳过"), cell.name, cell.line);
497 model.cells.append(cell);
498 model.cellNameVariants[valueOf(node)].append(cell.name);
499}

543void parsePart(const SectionNode& node, XpeditionHkpModel& model, ParseDiagnostics* diagnostics) {
544 XpeditionPartDefinition part; // 当前器件定义。
545 part.number = valueOf(node);
546 part.name = part.number;
547 part.line = node.line;
548 for (const SectionNode& item : node.children) {
549 const QString keyword = upper(item.keyword); // 当前属性关键字。
550 const QString value = valueOf(item); // 当前属性值。
551 if (keyword == QStringLiteral("NAME"))
552 part.name = value;
553 else if (keyword == QStringLiteral("LABEL"))
554 part.label = value;
555 else if (keyword == QStringLiteral("DESC"))
556 part.description = value;
557 else if (keyword == QStringLiteral("REFPREFIX"))
558 part.referencePrefix = value;
559 else if (keyword == QStringLiteral("TOPCELL"))
560 part.topCell = value;
561 else if (keyword == QStringLiteral("BOTTOMCELL"))
562 part.bottomCell = value;
563 else if (keyword == QStringLiteral("SYMBOL"))
564 part.symbol = value;
565 else if (keyword == QStringLiteral("SLOTS"))
566 collectSlotPinNumbers(item, part.slotPinNumbers);
567 // 解析 PDB 的键值属性，并拒绝缺少名称或值的记录。
568 else if (keyword == QStringLiteral("PROP")) {
569 // 按逗号拆分 PDB 属性名称和值。
570 const QStringList values = item.value.split(QRegularExpression(QStringLiteral("\\s*,\\s*")));
571 if (values.size() >= 2)
572 part.properties.insert(unquote(values.at(0)), unquote(values.at(1)));
573 else if (diagnostics)
574 diagnostics->add(ParseSeverity::Warning,
575 ParseScope::Component,
576 QStringLiteral("器件属性缺少名称或值"),
577 part.number,
578 item.line);
579 }
580 if (keyword == QStringLiteral("SYMBOL"))
581 collectSectionValues(item, QStringLiteral("PINNAME"), part.symbolPinNames);
582 }
583 if (part.topCell.isEmpty() && part.bottomCell.isEmpty() && diagnostics)
584 diagnostics->add(ParseSeverity::Warning,
585 ParseScope::Component,
586 QStringLiteral("器件没有封装 Cell 关联"),
587 part.number,
588 part.line);
589 model.parts.append(part);
590}

657XpeditionHkpModel XpeditionHkpModelParser::parse(const QList<SectionNode>& sections,
658 LengthUnit unit,
659 ParseDiagnostics* diagnostics) {
660 XpeditionHkpModel model; // 保存解析得到的 Xpedition HKP 模型。
661 for (const SectionNode& node : sections) {
662 const QString keyword = upper(node.keyword); // 当前顶层节点关键字。
663 if (keyword == QStringLiteral("PAD"))
664 parsePad(node, unit, model, diagnostics);
665 else if (keyword == QStringLiteral("HOLE"))
666 parseHole(node, unit, model, diagnostics);
667 else if (keyword == QStringLiteral("PADSTACK"))
668 parsePadstack(node, model, diagnostics);
669 else if (keyword == QStringLiteral("PACKAGE_CELL"))
670 parseCell(node, unit, model, diagnostics);
671 else if (keyword == QStringLiteral("NUMBER"))
672 parsePart(node, model, diagnostics);
673 }
674
675 for (const auto& padstack : model.padstacks) {
676 // 收集 Padstack 引用的全部焊盘名称，供关联校验使用。
677 const QStringList padNames = {padstack.topPad,
678 padstack.bottomPad,
679 padstack.topSolderMaskPad,
680 padstack.bottomSolderMaskPad,
681 padstack.topSolderPastePad,
682 padstack.bottomSolderPastePad};
683 for (const QString& name : padNames) {
684 if (name.isEmpty() || !diagnostics)
685 continue;
686 if (model.isPadAmbiguous(name))
687 diagnostics->add(ParseSeverity::Error,
688 ParseScope::Footprint,
689 QStringLiteral("Padstack 引用的 Pad 名称存在歧义：%1").arg(name),
690 padstack.name,
691 padstack.line);
692 else if (std::none_of(
693 model.pads.cbegin(), model.pads.cend(), [&](const auto& pad) { return pad.name == name; }))
694 diagnostics->add(ParseSeverity::Warning,
695 ParseScope::Footprint,
696 QStringLiteral("Padstack 引用了不存在的 Pad：%1").arg(name),
697 padstack.name,
698 padstack.line);
699 }
700 if (!padstack.holeName.isEmpty() && diagnostics) {
701 if (model.isHoleAmbiguous(padstack.holeName))
702 diagnostics->add(ParseSeverity::Error,
703 ParseScope::Footprint,
704 QStringLiteral("Padstack 引用的孔名称存在歧义：%1").arg(padstack.holeName),
705 padstack.name,
706 padstack.line);
707 // 唯一候选也不存在时保留可恢复的缺失关联诊断。
708 else if (std::none_of(model.holes.cbegin(), model.holes.cend(), [&](const auto& hole) {
709 return hole.name == padstack.holeName;
710 }))
711 diagnostics->add(ParseSeverity::Warning,
712 ParseScope::Footprint,
713 QStringLiteral("Padstack 引用了不存在的孔：%1").arg(padstack.holeName),
714 padstack.name,
715 padstack.line);
716 }
717 }
718 for (const auto& cell : model.cells) {
719 for (const auto& pin : cell.pins) {
720 if (pin.padstack.isEmpty() || !diagnostics)
721 continue;
722 if (model.isPadstackAmbiguous(pin.padstack))
723 diagnostics->add(ParseSeverity::Error,
724 ParseScope::Footprint,
725 QStringLiteral("引脚引用的 Padstack 名称存在歧义：%1").arg(pin.padstack),
726 cell.name,
727 pin.line);
728 else if (!model.findPadstack(pin.padstack))
729 diagnostics->add(ParseSeverity::Warning,
730 ParseScope::Footprint,
731 QStringLiteral("引脚引用了不存在的 Padstack：%1").arg(pin.padstack),
732 cell.name,
733 pin.line);
734 }
735 }
736 for (const auto& part : model.parts) {
737 for (const QString& cellName : {part.topCell, part.bottomCell}) {
738 if (cellName.isEmpty() || !diagnostics)
739 continue;
740 if (model.isCellAmbiguous(cellName))
741 diagnostics->add(ParseSeverity::Error,
742 ParseScope::Component,
743 QStringLiteral("器件引用的 Cell 名称存在歧义：%1").arg(cellName),
744 part.number,
745 part.line);
746 else if (!model.findCell(cellName))
747 diagnostics->add(ParseSeverity::Warning,
748 ParseScope::Component,
749 QStringLiteral("器件引用了不存在的 Cell：%1").arg(cellName),
750 part.number,
