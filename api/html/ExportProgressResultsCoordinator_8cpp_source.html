11void ExportProgressResultsCoordinator::updateResultsList(ExportProgressViewModel& owner) {
12 // 统计已成功完成的元器件结果数量。
13 int success = 0;
14 // 统计导出失败的元器件结果数量。
15 int failure = 0;
16 for (const auto& item : owner.m_resultsList) {
17 // 将结果项转换为状态字段映射。

22 ++failure;
23 }
24 }

27 emit owner.successCountChanged();
28 }
29 if (owner.m_failureCount != failure) {
30 owner.m_failureCount = failure;
31 emit owner.failureCountChanged();
32 }

47QVariantList ExportProgressResultsCoordinator::filteredResultsList(const ExportProgressViewModel& owner) {

52 // 保存符合当前过滤模式的结果项。
53 QVariantList filtered;

55 // 将结果项转换为状态字段映射。
56 const QVariantMap map = item.toMap();
57 if (owner.m_filterMode == "success" && map.value("status") == "success") {
58 filtered.append(item);
59 } else if (owner.m_filterMode == "failed" && map.value("status") == "failed") {
60 filtered.append(item);

62 (map.value("status") == "pending" || map.value("status") == "in_progress")) {

64 }
65 }

145void ExportProgressResultsCoordinator::updateOverallItemStatus(const ExportProgressViewModel& owner,
146 QVariantMap& result) {
147 struct TypeRule {
148 // 标记导出类型是否参与总体状态计算。
149 bool enabled;
150 // 保存该类型在结果映射中的状态字段名。
151 QString statusKey;
152 };
153
154 const QList<TypeRule> rules = {
155 {owner.m_exportSymbolEnabled, "symbolStatus"},
156 {owner.m_exportFootprintEnabled, "footprintStatus"},
157 {owner.m_exportModel3DEnabled, "model3DStatus"},
158 {owner.m_exportPreviewEnabled, "previewStatus"},
159 {owner.m_exportDatasheetEnabled, "datasheetStatus"},
160 };
161
162 // 标记是否至少启用了一种导出类型。
163 bool hasEnabledType = false;
164 // 标记是否存在失败的导出类型。
165 bool anyFailed = false;
166 // 标记是否存在仍在进行的导出类型。
167 bool anyInProgress = false;
168 // 标记所有启用类型是否都已结束。
169 bool allDone = true;
170 for (const TypeRule& rule : rules) {
171 if (!rule.enabled) {
172 continue;
173 }
174 hasEnabledType = true;
175 // 读取当前导出类型的状态。
176 const QString status = result.value(rule.statusKey, "pending").toString();
177 anyFailed = anyFailed || status == "failed";
178 anyInProgress = anyInProgress || status == "in_progress";
179 allDone = allDone && (status == "success" || status == "failed" || status == "skipped");
180 }
181
182 if (!hasEnabledType) {
183 result["status"] = "success";
184 } else if (allDone) {
185 result["status"] = anyFailed ? "failed" : "success";
186 } else if (anyInProgress) {
187 result["status"] = "in_progress";
188 } else {
189 result["status"] = "pending";
190 }
191}
