76AltiumPcbComponent ExporterAltiumFootprint::convertFootprint(const IR::FootprintComponentIR& data,
77 const QString& model3DPath) {
78 AltiumPcbComponent component; // 转换后的 Altium 封装组件。
79 component.name = data.name;
80 component.description = data.description;
81 component.height = data.height; // IR height 已经是 mm
82
83 // 转换焊盘
84 for (const IR::FootprintPadIR& pad : data.pads) {
85 component.pads.append(convertPad(pad));
86 }
87
88 // IR 折线必须逐段展开。PcbLib Track 原语只表示一个起点和终点，
89 // 直接取首尾点会把折线错误地压成一条对角线。
90 for (const IR::FootprintTrackIR& track : data.tracks) {
91 for (int i = 1; i < track.points.size(); ++i) {
92 IR::FootprintTrackIR segment = track; // 当前折线段的临时 IR 表示。
93 segment.points = {track.points[i - 1], track.points[i]};
94 component.tracks.append(convertTrack(segment));
95 }
96 }
97
98 // 独立安装孔用非电镀 MultiLayer pad 表示，使孔径、钻孔属性和图元归属
99 // 都进入同一套已验证的 pad 序列化路径。
100 for (const IR::FootprintHoleIR& hole : data.holes) {
101 AltiumPcbPad pad; // 由独立机械孔生成的非电镀焊盘。
102 pad.locationX = AltiumCoord::mmToRaw(hole.center.x());
103 pad.locationY = AltiumCoord::mmToRaw(hole.center.y());
104 pad.holeSize = AltiumCoord::mmToRaw(hole.radius * 2.0);
105 pad.sizeTopX = pad.sizeMidX = pad.sizeBotX = pad.holeSize;
106 pad.sizeTopY = pad.sizeMidY = pad.sizeBotY = pad.holeSize;
107 pad.shapeTop = pad.shapeMid = pad.shapeBot = AltiumConstants::PCB_PAD_SHAPE_ROUND;
108 pad.layer = AltiumConstants::PCB_LAYER_MULTI;
109 pad.isSMD = false;
110 pad.isPlated = false;
111 pad.isLocked = hole.isLocked;
112 component.pads.append(pad);
113 }
114
115 // 转换圆
116 for (const IR::FootprintCircleIR& circle : data.circles) {
117 component.arcs.append(convertCircle(circle));
118 }
119
120 // 转换弧线
121 for (const IR::FootprintArcIR& arc : data.arcs) {
122 component.arcs.append(convertArc(arc));
123 }
124
125 // 转换矩形
126 for (const IR::FootprintRectangleIR& rect : data.rectangles) {
127 component.fills.append(convertRectangle(rect));
128 }
129
130 // 转换文本
131 for (const IR::FootprintTextIR& text : data.texts) {
132 component.texts.append(convertText(text));
133 }
134
135 // 转换区域
136 for (const IR::FootprintRegionIR& region : data.regions) {
137 component.regions.append(convertSolidRegion(region));
138 }
139
140 // 板框同样是逐段线原语；保留 IR 的层和线宽，不引入导入器特有假设。
141 for (const IR::FootprintOutlineIR& outline : data.outlines) {
142 for (int i = 1; i < outline.points.size(); ++i) {
143 IR::FootprintTrackIR segment; // 当前板框线段的临时 IR 表示。
144 segment.points = {outline.points[i - 1], outline.points[i]};
145 segment.width = outline.strokeWidth;
146 segment.layer = outline.layer;
147 segment.isLocked = outline.isLocked;
148 component.tracks.append(convertTrack(segment));
149 }
150 }
151
152 // 加载 3D 模型
153 QByteArray stepData; // 待嵌入的 STEP 文件数据。
154 QString modelName; // 3D 模型显示名称。
155 if (!model3DPath.isEmpty()) {
156 stepData = loadStepData(model3DPath);
157 modelName = QFileInfo(model3DPath).fileName();
158 } else if (!data.models3d.isEmpty()) {
159 // 从 IR 的 3D 模型中加载 STEP 数据
160 for (const IR::Model3DIR& model3d : data.models3d) {
161 if (model3d.hasStepData()) {
162 AltiumPcbComponent::Model3D model; // 从 IR 转换的三维模型记录。
163 model.name = model3d.name();
164 model.stepData = model3d.stepData();
165 const auto& rot = model3d.rotation(); // 模型旋转变换。
166 model.rotX = rot.x;
167 model.rotY = rot.y;
168 model.rotZ = rot.z;
169 const auto& translation = model3d.translation(); // 模型平移变换。
170 const auto& stepOffset = model3d.stepOffsetMm(); // STEP 原点偏移。
171 model.x = AltiumCoord::mmToRaw(translation.x + stepOffset.x);
172 model.y = AltiumCoord::mmToRaw(translation.y + stepOffset.y);
173 model.dz = translation.z + stepOffset.z;
174 component.models.append(model);
175 }
176 }
177 if (!component.models.isEmpty()) {
178 stepData = component.models.first().stepData;
179 modelName = component.models.first().name;
180 }
181 }
182
183 // 如果有 3D 模型数据但 models 列表为空，补全
184 if (!stepData.isEmpty() && component.models.isEmpty()) {
185 AltiumPcbComponent::Model3D model; // 从单独 STEP 文件补建的三维模型记录。
186 model.name = modelName;
187 model.stepData = stepData;
188 component.models.append(model);
189 }
190
191 // 封装图元和 3D 模型必须经过同一原点平移，避免焊盘与模型错位。
192 centerComponent(component);
193
194 // 为 3D 模型生成 ComponentBody（基于封装包围盒）
195 if (!component.models.isEmpty()) {
196 generateComponentBody(component);
197 }
198
199 return component;
200}

207AltiumPcbPad ExporterAltiumFootprint::convertPad(const IR::FootprintPadIR& pad) {
208 AltiumPcbPad altiumPad; // 转换后的 Altium 焊盘记录。
209 altiumPad.designator = pad.number;
210 altiumPad.locationX = AltiumCoord::mmToRaw(pad.position.x());
211 altiumPad.locationY = AltiumCoord::mmToRaw(pad.position.y());
212
213 // IR size 已经是 mm
214 int width = AltiumCoord::mmToRaw(pad.size.width()); // 焊盘宽度的 Altium 原始单位。
215 int height = AltiumCoord::mmToRaw(pad.size.height()); // 焊盘高度的 Altium 原始单位。
216
217 // 确定焊盘类型
218 altiumPad.isSMD = pad.isSmd();
219 altiumPad.isPlated = pad.isPlated;
220 altiumPad.isLocked = pad.isLocked;
221
222 if (pad.isThroughHole()) {
223 // 通孔焊盘
224 altiumPad.layer = AltiumConstants::PCB_LAYER_MULTI;
225 altiumPad.holeSize = AltiumCoord::mmToRaw(pad.holeSize);
226
227 // 槽孔检测：holeLength > holeSize 时为槽孔
228 if (pad.holeLength > pad.holeSize + 0.001) {
229 altiumPad.holeType = 2; // Slot
230 altiumPad.holeSlotLengthRaw = AltiumCoord::mmToRaw(pad.holeLength);
231 // 槽孔旋转：长轴方向
232 altiumPad.holeRotation = (pad.size.height() > pad.size.width()) ? 90.0 : 0.0;
233 }
234 } else {
235 // 表贴焊盘：从 IR LayerType 映射到 Altium 层字节
236 altiumPad.layer = static_cast<uint8_t>(AltiumLayerMap::fromLayerTypeToAltium(pad.layer));
237 }
238
239 // 形状：从 IR PadShape 映射到 Altium 焊盘形状字节
240 uint8_t shape = AltiumLayerMap::toAltiumPadShape(pad.shape); // Altium 焊盘形状编号。
241 altiumPad.shapeTop = shape;
242 altiumPad.shapeMid = shape;
243 altiumPad.shapeBot = shape;
244
245 // 圆角矩形的默认圆角百分比
246 if (shape == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT) {
247 altiumPad.cornerRadiusPercentage = 50;
248 }
249
250 // 尺寸
251 altiumPad.sizeTopX = width;
252 altiumPad.sizeTopY = height;
253 altiumPad.sizeMidX = width;
254 altiumPad.sizeMidY = height;
255 altiumPad.sizeBotX = width;
256 altiumPad.sizeBotY = height;
257
258 altiumPad.rotation = pad.rotation;
259
260 return altiumPad;
261}

268AltiumPcbTrack ExporterAltiumFootprint::convertTrack(const IR::FootprintTrackIR& track) {
269 AltiumPcbTrack altiumTrack; // 转换后的 Altium 走线记录。
270 altiumTrack.layer = static_cast<uint8_t>(AltiumLayerMap::fromLayerTypeToAltium(track.layer));
271 altiumTrack.width = AltiumCoord::mmToRaw(track.width);
272
273 // IR points 已经是解析后的 QList<QPointF>，单位 mm
274 if (track.points.size() >= 2) {
275 altiumTrack.startX = AltiumCoord::mmToRaw(track.points.first().x());
276 altiumTrack.startY = AltiumCoord::mmToRaw(track.points.first().y());
277 altiumTrack.endX = AltiumCoord::mmToRaw(track.points.last().x());
278 altiumTrack.endY = AltiumCoord::mmToRaw(track.points.last().y());
279 }
280
281 return altiumTrack;

407void ExporterAltiumFootprint::generateComponentBody(AltiumPcbComponent& component) {
408 if (component.models.isEmpty())
409 return;
410
411 const AltiumFootprintBounds bounds =
412 AltiumFootprintGeometryNormalizer::computeBounds(component, true); // 用于生成轮廓的包围盒。
413 if (!bounds.valid)
414 return; // 无图元，无法生成轮廓
415
416 const auto makeModelId = [&component](int modelIndex) {
417 // 使用组件名称和模型序号构造稳定的模型标识种子。
418 const QString seed = modelIndex == 0 ? component.name + QStringLiteral("|body")
419 : component.name + QStringLiteral("|body|") + QString::number(modelIndex);
420 const QByteArray seedBytes = seed.toUtf8(); // 稳定 UUID 计算用的 UTF-8 种子。
421 uint64_t hash = 0xCBF29CE484222325ULL; // FNV-1a 哈希初始值。
422 for (char byte : seedBytes) {
423 hash ^= static_cast<uint8_t>(byte);
424 hash *= 0x100000001B3ULL;
425 }
426 return QString("{%1-%2-%3-%4-%5}")
427 .arg((hash >> 32) & 0xFFFFFFFF, 8, 16, QChar('0'))
428 .arg((hash >> 16) & 0xFFFF, 4, 16, QChar('0'))
429 .arg(hash & 0xFFFF, 4, 16, QChar('0'))
430 .arg(((~hash) >> 48) & 0xFFFF, 4, 16, QChar('0'))
431 .arg((~hash) & 0xFFFFFFFFFFFFLL, 12, 16, QChar('0'))
432 .toUpper();
433 };
434
435 // 每个模型都需要独立的 ComponentBody，否则虽然模型数据写入了 Library/Models，
436 // Altium 仍然只会通过第一个元件体显示第一个模型。
437 for (int modelIndex = 0; modelIndex < component.models.size(); ++modelIndex) {
438 auto& model = component.models[modelIndex]; // 当前待关联的三维模型。
439 const QString modelId = makeModelId(modelIndex); // 当前模型的稳定标识符。
440 model.id = modelId;
441
442 AltiumPcbComponentBody body; // 当前模型对应的 ComponentBody 记录。
443 body.modelId = modelId;
444 body.modelName = model.name;
445 body.model2dRotX = model.x;
446 body.model2dRotY = model.y;
447 body.model3dRotX = model.rotX;
448 body.model3dRotY = model.rotY;
449 body.model3dRotZ = model.rotZ;
450 body.model3dDzRaw = AltiumCoord::mmToRaw(model.dz);
451 body.overallHeightRaw = AltiumCoord::mmToRaw(qMax(component.height, 0.2));
452
453 // 矩形轮廓
454 body.outline = {QPointF(bounds.minX, bounds.minY),
455 QPointF(bounds.maxX, bounds.minY),
456 QPointF(bounds.maxX, bounds.maxY),
457 QPointF(bounds.minX, bounds.maxY)};
458
459 component.bodies.append(body);
460 }
