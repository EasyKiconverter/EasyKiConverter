20 const QString& componentId,

52void ComponentApiCallbackCoordinator::handleFetchErrorWithId(ComponentService& owner,
53 const QString& idOrUuid,
54 const QString& error) {
55 QString componentId = idOrUuid; // 当前错误对应的元器件编号。
56
57 // 并行模式下，CAD 请求错误可能只携带模型 UUID，需要从活动状态反查组件编号。
58 if (owner.m_parallelContext != nullptr) {
59 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
60 for (auto it = owner.m_fetchingComponents.begin(); it != owner.m_fetchingComponents.end();
61 ++it) { // 反查活动请求。
62 if (it.value().data.model3DData() && it.value().data.model3DData()->uuid() == idOrUuid) {
63 componentId = it.key();
64 qDebug() << "Resolved UUID" << idOrUuid << "to component ID" << componentId;
65 break;
66 }
67 }
68 }
69
70 emitFetchErrorAndClearState(owner, componentId, error);
71}

80void ComponentApiCallbackCoordinator::emitFetchErrorAndClearState(ComponentService& owner,
81 const QString& componentId,
82 const QString& error,
83 uint64_t expectedGeneration) {
84 qWarning() << "Fetch error for component" << componentId << ":" << error;
85
86 {
87 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
88 const auto it = owner.m_fetchingComponents.find(componentId.toUpper()); // 当前请求状态。
89 if (it == owner.m_fetchingComponents.end() ||
90 (expectedGeneration != 0 && it->cacheGeneration != expectedGeneration)) {
91 if (expectedGeneration != 0) {
92 qDebug() << "ComponentService: Discarding stale error for" << componentId;
93 }
94 return;
95 }
96 owner.m_fetchingComponents.erase(it);
97 }
98
99 owner.m_componentCache.removeIfInvalid(componentId.toUpper());
100 ComponentParallelFetchCoordinator::handleFetchError(owner, componentId, error);
101
102 // fetchError 必须最后发送，因为连接的槽函数可能会删除服务对象。
103 emit owner.fetchError(componentId, error);
104}
