22void AltiumPcbPrimitiveWriter::writePad(AltiumBinaryWriter& writer, const AltiumPcbPad& pad) {
23 writer.writeUInt8(AltiumConstants::PCB_OBJECT_PAD);
24
25 // 子记录 1: Designator
26 writer.writeStringBlock(pad.designator);
27
28 // 子记录 2: PadSubrecord2（空）
29 writer.writeStringBlock("");
30
31 // 子记录 3: Net 字符串
32 writer.writeStringBlock("|&|0");
33
34 // 子记录 4: 标记字节
35 writer.beginBlock();
36 writer.writeUInt8(0);
37 writer.endBlock();
38
39 // 子记录 5: 主焊盘数据
40 writer.beginBlock();
41 {
42 uint8_t layer = pad.isSMD ? pad.layer : AltiumConstants::PCB_LAYER_MULTI; // 焊盘写入层。
43 uint16_t flags = // 焊盘图元标志位。
44 m_owner.encodePrimitiveFlags(pad.isLocked, pad.isTentingTop, pad.isTentingBottom, pad.isKeepout);
45 m_owner.writeCommonPrimitiveHeader(writer, layer, flags);
46
47 // 位置
48 writer.writeInt32(pad.locationX);
49 writer.writeInt32(pad.locationY);
50
51 // 三层尺寸
52 writer.writeInt32(pad.sizeTopX);
53 writer.writeInt32(pad.sizeTopY);
54 writer.writeInt32(pad.sizeMidX);
55 writer.writeInt32(pad.sizeMidY);
56 writer.writeInt32(pad.sizeBotX);
57 writer.writeInt32(pad.sizeBotY);
58
59 // 孔径
60 writer.writeInt32(pad.holeSize);
61
62 // 三层形状
63 writer.writeUInt8(pad.shapeTop);
64 writer.writeUInt8(pad.shapeMid);
65 writer.writeUInt8(pad.shapeBot);
66
67 // 旋转 + 电镀
68 writer.writeDouble(m_owner.normalizeFiniteValue(pad.rotation, 0.0, QStringLiteral("焊盘旋转角度")));
69 writer.writeUInt8(pad.isPlated ? 1 : 0);
70
71 // 补齐主记录固定布局（61 字节之后的字段）
72 writer.writeUInt8(0); // stack mode = Simple
73 writer.writeUInt8(pad.mode);
74 writer.writeUInt8(pad.powerPlaneConnectStyle);
75 writer.writeInt32(pad.reliefAirGapRaw);
76 writer.writeInt32(pad.reliefConductorWidthRaw);
77 writer.writeInt16(pad.reliefEntries);
78 writer.writeInt32(pad.powerPlaneClearanceRaw);
79 writer.writeInt32(pad.powerPlaneReliefExpansionRaw);
80 writer.writeInt32(0); // reserved
81 writer.writeInt32(pad.pasteMaskExpansionRaw);
82 writer.writeInt32(pad.solderMaskExpansionRaw);
83 writer.writeBytes(QByteArray(7, 0)); // reserved
84 writer.writeUInt8(pad.pasteMaskExpansionRaw != 0 ? 2 : 0); // paste mask expansion mode
85 writer.writeUInt8(pad.solderMaskExpansionRaw != 0 ? 2 : 1); // solder mask expansion mode
86 writer.writeUInt8(pad.drillType);
87 writer.writeInt16(0); // reserved
88 writer.writeInt32(0); // reserved
89 writer.writeInt16(0); // jumper ID
90 writer.writeInt16(0); // reserved
91 }
92 writer.endBlock();
93
94 // 子记录 6: 扩展块（各层尺寸/形状覆盖 + 孔元数据）
95 writer.beginBlock();
96 writePadExtendedBlock(writer, pad);
97 writer.endBlock();
98}

109void AltiumPcbPrimitiveWriter::writePadExtendedBlock(AltiumBinaryWriter& writer, const AltiumPcbPad& pad) {
110 // 尺寸 X 覆盖：29 层 × 4 字节 = 116 字节
111 for (int i = 0; i < AltiumConstants::PCB_PAD_LAYER_COUNT; ++i) {
112 writer.writeInt32(pad.sizeMidX);
113 }
114 // 尺寸 Y 覆盖：29 层 × 4 字节 = 116 字节
115 for (int i = 0; i < AltiumConstants::PCB_PAD_LAYER_COUNT; ++i) {
116 writer.writeInt32(pad.sizeMidY);
117 }
118 // 形状覆盖：29 个铜层，随后一个保留字节
119 for (int i = 0; i < AltiumConstants::PCB_PAD_LAYER_COUNT; ++i) {
120 writer.writeUInt8(pad.shapeMid);
121 }
122 writer.writeUInt8(0);
123
124 // 孔元数据
125 writer.writeUInt8(pad.holeType);
126 writer.writeInt32(pad.holeSlotLengthRaw);
127 writer.writeDouble(m_owner.normalizeFiniteValue(pad.holeRotation, 0.0, QStringLiteral("焊盘孔旋转角度")));
128
129 // 保留区域
130 writer.writeBytes(QByteArray(32 * 4, 0));
131 writer.writeBytes(QByteArray(32 * 4, 0));
132
133 // 圆角矩形标记
134 bool hasRoundedRect = (pad.shapeTop == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT) || // 是否存在圆角矩形。
135 (pad.shapeMid == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT) ||
136 (pad.shapeBot == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT);
137 writer.writeUInt8(hasRoundedRect ? 1 : 0);
138
139 // 各层形状列表（32 字节：top + 30*mid + bot）
140 writer.writeUInt8(pad.shapeTop);
141 for (int i = 0; i < 30; ++i) {
142 writer.writeUInt8(pad.shapeMid);
143 }
144 writer.writeUInt8(pad.shapeBot);
145
146 // 各层圆角半径百分比（32 字节）
147 for (int i = 0; i < 32; ++i) {
148 writer.writeUInt8(pad.cornerRadiusPercentage);
149 }
150}

206void AltiumPcbPrimitiveWriter::writeText(AltiumBinaryWriter& writer, const AltiumPcbText& text) {
207 writer.writeUInt8(5); // Object ID
208
209 writer.beginBlock();
210 {
211 uint16_t flags = 0x08; // 文本图元标志位。
212 m_owner.writeCommonPrimitiveHeader(writer, text.layer, flags);
213
214 writer.writeInt32(text.locationX);
215 writer.writeInt32(text.locationY);
216 writer.writeInt32(text.height);
217 writer.writeInt16(0); // font ID
218 writer.writeDouble(m_owner.normalizeFiniteValue(text.rotation, 0.0, QStringLiteral("PCB 文本旋转角度")));
219 writer.writeUInt8(text.isMirrored ? 1 : 0);
220 writer.writeInt32(text.strokeWidth);
221 writer.writeUInt8(0); // is comment
222 writer.writeUInt8(0); // is designator
223 writer.writeUInt8(0); // char set
224 writer.writeUInt8(0); // base font type (Stroke)
225
226 // 剩余填充到文本记录总大小（已写字段精确占 44 字节）。
227 constexpr int WRITTEN_TEXT_HEADER_SIZE = 44;
228 QByteArray padding(AltiumConstants::PCB_TEXT_RECORD_SIZE - WRITTEN_TEXT_HEADER_SIZE, 0); // 文本尾部填充。
229 // wide string index
230 int wsIdx = m_owner.addWideString(text.text); // 宽字符串索引。
231 int wsBase = AltiumConstants::PCB_TEXT_WS_INDEX_FIELD_OFFSET - WRITTEN_TEXT_HEADER_SIZE; // 索引字段偏移。
232 padding[wsBase] = static_cast<char>(wsIdx & 0xFF);
233 padding[wsBase + 1] = static_cast<char>((wsIdx >> 8) & 0xFF);
234 padding[wsBase + 2] = static_cast<char>((wsIdx >> 16) & 0xFF);
235 padding[wsBase + 3] = static_cast<char>((wsIdx >> 24) & 0xFF);
236 // text kind = Stroke
237 padding[AltiumConstants::PCB_TEXT_KIND_FIELD_OFFSET - WRITTEN_TEXT_HEADER_SIZE] = 0;
238 // V7 layer ID
239 uint32_t v7id = m_owner.toV7LayerId(text.layer); // 文本对应的 V7 层标识。
240 int v7Base = AltiumConstants::PCB_TEXT_V7LAYER_FIELD_OFFSET - WRITTEN_TEXT_HEADER_SIZE; // 层字段偏移。
241 padding[v7Base] = static_cast<char>(v7id & 0xFF);
242 padding[v7Base + 1] = static_cast<char>((v7id >> 8) & 0xFF);
243 padding[v7Base + 2] = static_cast<char>((v7id >> 16) & 0xFF);
244 padding[v7Base + 3] = static_cast<char>((v7id >> 24) & 0xFF);
245
246 writer.writeBytes(padding);
247 }
248 writer.endBlock();
249
250 // 文本字符串块
251 writer.writeStringBlock(text.text);
252}

286void AltiumPcbPrimitiveWriter::writeRegion(AltiumBinaryWriter& writer, const AltiumPcbRegion& region) {
287 writer.writeUInt8(AltiumConstants::PCB_OBJECT_REGION);
288
289 writer.beginBlock();
290 {
291 uint16_t flags = m_owner.encodePrimitiveFlags(false, false, false, false); // 区域图元标志位。
292 m_owner.writeCommonPrimitiveHeader(writer, region.layer, flags);
293
294 writer.writeUInt32(0);
295 writer.writeUInt8(0);
296
297 // 嵌套参数块
298 QMap<QString, QString> params; // 区域参数键值表。
299 // 根据显式层名称或 Altium 层映射确定 V7 层名称。
300 QString v7Name = region.v7LayerName.isEmpty() ? AltiumLayerMap::toLayerName(region.layer)
301 : region.v7LayerName;
302 params["V7_LAYER"] = v7Name;
303 params["KIND"] = QString::number(region.kind);
304 params["SUBPOLYINDEX"] = "0";
305 params["UNIONINDEX"] = "0";
306 params["ARCRESOLUTION"] = "0mil";
307 if (region.isBoardCutout)
308 params["ISBOARDCUTOUT"] = "TRUE";
309 if (!region.net.isEmpty())
310 params["NET"] = region.net;
311 if (!region.uniqueId.isEmpty())
312 params["UNIQUEID"] = region.uniqueId;
313 if (!region.name.isEmpty())
314 params["NAME"] = region.name;
315 writer.writeCStringParameterBlock(params);
316
317 // 轮廓顶点
318 writer.writeUInt32(static_cast<uint32_t>(region.vertices.size()));
319 // 按输入顺序写入区域轮廓的每个顶点。
320 for (const QPointF& v : region.vertices) { // 当前待写入的区域轮廓顶点。
321 writer.writeDouble(m_owner.normalizeFiniteValue(v.x(), 0.0, QStringLiteral("PCB 区域 X 坐标")));
322 writer.writeDouble(m_owner.normalizeFiniteValue(v.y(), 0.0, QStringLiteral("PCB 区域 Y 坐标")));
323 }
324 }
325 writer.endBlock();

333void AltiumPcbPrimitiveWriter::writeComponentBody(AltiumBinaryWriter& writer, const AltiumPcbComponentBody& body) {
334 writer.writeUInt8(AltiumConstants::PCB_OBJECT_COMPONENT_BODY);
335
336 writer.beginBlock();
337 {
338 uint16_t flags = m_owner.encodePrimitiveFlags(false, false, false, false); // 元件体图元标志位。
339 uint8_t layer = 57; // 默认机械层编号。
340 const QString normalizedLayer = body.layerName.trimmed().toUpper(); // 规范化后的机械层名称。
341 if (normalizedLayer.startsWith("MECHANICAL")) {
342 bool ok = false; // 机械层编号转换是否成功。
343 const int number = normalizedLayer.mid(10).toInt(&ok); // 解析得到的机械层编号。
344 if (ok && number >= 1 && number <= 16) {
345 layer = static_cast<uint8_t>(56 + number);

348 m_owner.writeCommonPrimitiveHeader(writer, layer, flags);
349
350 writer.writeUInt32(0); // reserved
351 writer.writeUInt8(0); // reserved

353 QMap<QString, QString> params; // 元件体参数键值表。

355 params["NAME"] = body.name;

357 params["SUBPOLYINDEX"] = QString::number(body.subpolyIndex);
358 params["UNIONINDEX"] = QString::number(body.unionIndex);

369 params["TEXTURE"] = "";

371 params["TEXTURECENTERY"] = "0mil";
372 params["TEXTURESIZEX"] = "0mil";
373 params["TEXTURESIZEY"] = "0mil";
374 params["TEXTUREROTATION"] = "0.000000";
