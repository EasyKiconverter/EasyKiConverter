23void ComponentMediaCallbackCoordinator::handleImageReady(ComponentService& owner,
24 const QString& componentId,
25 const QByteArray& imageData,
26 int imageIndex) {
27 // 将原始字节解码为 Qt 图像，供界面预览和缓存事件同时使用。
28 QImage image = QImage::fromData(imageData);
29 if (image.isNull()) {
30 qWarning() << "Failed to load image from data for component:" << componentId << "index:" << imageIndex
31 << "data size:" << imageData.size();
32 return;
33 }
34
35 // 使用大写编号统一访问正在获取的元器件状态。
36 const QString normalizedId = componentId.toUpper();
37 // 保存合并后的组件快照，供缓存替换使用。
38 ComponentData updatedData;
39 // 标记本次回调是否命中了有效的活动请求。
40 bool hasValidUpdate = false;
41 // 记录请求创建时的缓存代次，避免旧回调覆盖新缓存。
42 uint64_t gen = 0;
43 {
44 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
45 // 在锁保护下查找当前元器件的活动请求。
46 const auto it = owner.m_fetchingComponents.find(normalizedId);
47 if (it == owner.m_fetchingComponents.end() ||
48 it->cacheGeneration != ComponentCacheService::instance()->currentGeneration()) {
49 qDebug() << "ComponentService: Discarding stale image callback for" << componentId;
50 return;
51 }
52 it->data.addPreviewImageData(imageData, imageIndex);
53 updatedData = it->data;
54 gen = it->cacheGeneration;
55 hasValidUpdate = true;
56 }
57
58 if (hasValidUpdate) {
59 emit owner.previewImageReady(normalizedId, image, imageIndex);
60 emit owner.previewImageDataReady(normalizedId, imageData, imageIndex);
61 if (ComponentCacheService::instance()->currentGeneration() != gen)
62 return;
63 owner.m_componentCache.replaceIfPresent(normalizedId, updatedData);
64 }
65}

75void ComponentMediaCallbackCoordinator::handleLcscDataReady(ComponentService& owner,
76 const QString& componentId,
77 const QString& manufacturerPart,
78 const QString& datasheetUrl,
79 const QStringList& imageUrls) {
80 qDebug() << "LCSC data ready for component:" << componentId
81 << "Manufacturer Part:" << (manufacturerPart.isEmpty() ? "none" : manufacturerPart)
82 << "Datasheet:" << (datasheetUrl.isEmpty() ? "none" : datasheetUrl) << "Images:" << imageUrls.size();
83
84 // 保存合并后的组件快照，供异步元数据持久化使用。
85 ComponentData updatedData;
86 // 标记是否找到可更新的活动请求。
87 bool hasValidUpdate = false;
88 // 记录请求创建时的缓存代次。
89 uint64_t gen = 0;
90 {
91 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
92 // 使用大写编号查找活动请求，兼容不同大小写来源。
93 const QString normalizedId = componentId.toUpper();
94 if (owner.m_fetchingComponents.contains(normalizedId)) {
95 // 获取待合并的活动组件状态。
96 ComponentService::FetchingComponent& fetchingComponent = owner.m_fetchingComponents[normalizedId];
97 if (fetchingComponent.cacheGeneration != ComponentCacheService::instance()->currentGeneration()) {
98 qDebug() << "ComponentService: Discarding stale LCSC data callback for" << componentId;
99 return;
100 }
101
102 if (!manufacturerPart.isEmpty()) {
103 fetchingComponent.data.setManufacturerPart(manufacturerPart);
104 qDebug() << "Manufacturer part saved to ComponentData:" << manufacturerPart;
105 }
106
107 if (!datasheetUrl.isEmpty()) {
108 fetchingComponent.data.setDatasheet(datasheetUrl);
109 // 根据数据手册地址推断其缓存格式。
110 QString format =
111 datasheetUrl.toLower().contains(".html") ? QStringLiteral("html") : QStringLiteral("pdf");
112 fetchingComponent.data.setDatasheetFormat(format);
113 qDebug() << "Datasheet saved to ComponentData:" << datasheetUrl << "format:" << format;
114 }
115
116 if (!imageUrls.isEmpty()) {
117 fetchingComponent.data.setPreviewImages(imageUrls);
118 qDebug() << "Preview images saved to ComponentData:" << imageUrls.size() << "images";
119 // 为新预览图地址预分配与地址数量一致的图像数据槽位。
120 QList<QByteArray> emptyImageDataList;
121 emptyImageDataList.resize(imageUrls.size());
122 fetchingComponent.data.setPreviewImageData(emptyImageDataList);
123 qDebug() << "Pre-allocated" << imageUrls.size() << "empty image data slots";
124 }
125
126 updatedData = fetchingComponent.data;
127 gen = fetchingComponent.cacheGeneration;
128 hasValidUpdate = true;
129 } else {
130 qWarning() << "Component" << componentId << "not found in m_fetchingComponents, cannot update LCSC data";
131 }
132 }
133
134 if (hasValidUpdate) {
135 if (ComponentCacheService::instance()->currentGeneration() != gen) {
136 qDebug() << "ComponentService: Discarding stale LCSC data update for" << componentId;
137 return;
138 }
139 // 使用大写编号转发事件并更新缓存。
140 const QString normalizedId = componentId.toUpper();
141 emit owner.lcscDataUpdated(normalizedId, manufacturerPart, datasheetUrl, imageUrls);
142 ComponentCacheService::instance()->saveComponentMetadataAsync(normalizedId, updatedData, gen);
143 owner.updateComponentCache(normalizedId, updatedData);
144 }
145}

153void ComponentMediaCallbackCoordinator::handleDatasheetReady(ComponentService& owner,
154 const QString& componentId,
155 const QByteArray& datasheetData) {
156 qDebug() << "Datasheet downloaded for component:" << componentId << "size:" << datasheetData.size() << "bytes";
157
158 // 记录数据手册格式，并在 PDF 内容无效时切换到 HTML。
159 QString format;
160 // 标记当前回调是否命中了活动请求。
161 bool hasValidUpdate = false;
162 // 标记所有并行下载是否已完成。
163 bool shouldMarkCompleted = false;
164 // 保存完成时的组件快照，供并行获取上下文消费。
165 ComponentData completedData;
166 // 记录活动请求的缓存代次。
167 uint64_t gen = 0;
168 // 使用大写编号访问活动请求和并行获取上下文。
169 const QString normalizedId = componentId.toUpper();
170 {
171 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
172 if (owner.m_fetchingComponents.contains(normalizedId)) {
173 // 获取待更新的数据手册所属组件状态。
174 ComponentService::FetchingComponent& fetchingComponent = owner.m_fetchingComponents[normalizedId];
175 gen = fetchingComponent.cacheGeneration;
176 if (gen != ComponentCacheService::instance()->currentGeneration()) {
177 qDebug() << "ComponentService: Discarding stale datasheet callback for" << componentId;
178 return;
179 }
180 fetchingComponent.data.setDatasheetData(datasheetData);
181 format = fetchingComponent.data.datasheetFormat();
182 if (format == "pdf" && !owner.isPDF(datasheetData)) {
183 format = "html";
184 fetchingComponent.data.setDatasheetFormat(format);
185 }
186 qDebug() << "Datasheet data saved to ComponentData, size:" << datasheetData.size() << "bytes"
187 << "format:" << format;
188 if (fetchingComponent.pendingAsyncDownloads > 0) {
189 fetchingComponent.pendingAsyncDownloads--;
190 qDebug() << "Datasheet download complete, pending async downloads:"
191 << fetchingComponent.pendingAsyncDownloads;
192 if (fetchingComponent.pendingAsyncDownloads == 0) {
193 shouldMarkCompleted = true;
194 completedData = fetchingComponent.data;
195 }
196 }
197 hasValidUpdate = true;
198 } else {
199 qWarning() << "Component" << componentId
200 << "not found in m_fetchingComponents, datasheet data will be saved directly to cache";
201 }
202 }
203
204 if (shouldMarkCompleted) {
205 if (ComponentCacheService::instance()->currentGeneration() != gen) {
206 qDebug() << "ComponentService: Discarding stale datasheet completion for" << componentId;
207 return;
208 }
209 // 读取当前并行获取上下文，避免在无锁状态下访问共享指针。
210 ParallelFetchContext* parallelContext = nullptr;
211 {
212 QMutexLocker locker(&owner.m_parallelContextMutex);
213 parallelContext = owner.m_parallelContext;
214 }
215 if (parallelContext != nullptr)
216 parallelContext->markCompleted(normalizedId, completedData);
217 if (owner.m_queueManager != nullptr)
218 owner.m_queueManager->requestCompleted(normalizedId);
219 }
220
221 if (hasValidUpdate) {
222 // 复制带有最新数据手册的组件状态，避免锁外访问共享对象。
223 ComponentData dataCopy;
224 // 标记是否成功复制了仍属于当前代次的组件状态。
225 bool hasDataCopy = false;
226 {
227 QMutexLocker fetchLocker(&owner.m_fetchingComponentsMutex);
228 if (owner.m_fetchingComponents.contains(normalizedId) &&
229 owner.m_fetchingComponents[normalizedId].cacheGeneration == gen) {
230 dataCopy = owner.m_fetchingComponents[normalizedId].data;
231 hasDataCopy = true;
232 }
233 }
234 if (hasDataCopy && owner.m_componentCache.replaceIfPresent(normalizedId, dataCopy))
235 qDebug() << "ComponentService: Updated cache with datasheet data for" << normalizedId;
236 } else {
237 qDebug() << "ComponentService: Discarding datasheet callback without active request for" << componentId;
238 }
239
240 if (hasValidUpdate)
241 emit owner.datasheetReady(componentId, datasheetData);
242}

281void ComponentMediaCallbackCoordinator::handleAllImagesReady(ComponentService& owner,
282 const QString& componentId,
283 const QStringList& imagePaths) {
284 qDebug() << "All images ready for component:" << componentId << "paths:" << imagePaths.size();
285 // 记录活动请求的缓存代次，防止异步编码结果写回旧请求。
286 uint64_t gen = 0;
287 {
288 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
289 // 在锁保护下确认元器件请求仍处于活动状态。
290 const auto it = owner.m_fetchingComponents.find(componentId.toUpper());
291 if (it == owner.m_fetchingComponents.end()) {
292 qDebug() << "ComponentService: Discarding all-images callback without active request for" << componentId;
293 return;
294 }
295 gen = it->cacheGeneration;
296 }
297
298 // 在线程池中编码所有预览图文件，避免阻塞组件服务线程。
299 QFuture<PreviewImageDataResult> future =
300 QtConcurrent::run([imagePaths]() { return PreviewImageDataEncoder::encodeFiles(imagePaths); });
301 // 监听后台编码完成事件，并在组件服务线程中合并结果。
302 auto* watcher = new QFutureWatcher<PreviewImageDataResult>(&owner);
303 QObject::connect(watcher,
304 &QFutureWatcher<PreviewImageDataResult>::finished,
305 &owner,
306 [&owner, watcher, componentId, imagePaths, gen]() {
307 // 读取后台编码结果并释放一次性监听器。
308 const PreviewImageDataResult imageResult = watcher->result();
309 watcher->deleteLater();
310 if (ComponentCacheService::instance()->currentGeneration() != gen) {
311 qDebug() << "ComponentService: Discarding stale all-images result for" << componentId;
312 return;
313 }
314 // 使用大写编号查找仍然有效的活动请求。
315 const QString normalizedId = componentId.toUpper();
316 {
317 QMutexLocker locker(&owner.m_fetchingComponentsMutex);
318 // 在锁保护下确认请求代次没有被替换。
319 const auto it = owner.m_fetchingComponents.find(normalizedId);
320 if (it == owner.m_fetchingComponents.end() || it->cacheGeneration != gen) {
321 qDebug() << "ComponentService: Discarding all-images result for replaced request"
322 << componentId;
323 return;
324 }
325 it->data.setPreviewImageData(imageResult.imageData);
326 qDebug() << "All image data updated in ComponentData for component:" << componentId
327 << "count:" << imageResult.imageData.size();
328 }
329 emit owner.previewImagesReady(componentId, imageResult.encodedImages);
330 emit owner.allImagesReady(componentId, imagePaths);
331 });
332 watcher->setFuture(future);
333}
