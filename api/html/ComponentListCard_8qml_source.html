236 // 将弹窗坐标限制在父窗口边界内。
237 function clampPopupCoordinate(value, size, boundary) {
238 var margin = AppStyle.spacing.xs;
239 var maxValue = Math.max(margin, boundary - size - margin);
240 return Math.max(margin, Math.min(value, maxValue));
241 }

243 // 按设备像素比对弹窗坐标取整，避免边缘模糊。
244 function snapPopupCoordinate(value) {
245 var win = componentListCard.Window.window;
246 var dpr = win && win.devicePixelRatio > 0 ? win.devicePixelRatio : 1;
247 return Math.round(value * dpr) / dpr;
248 }

250 // 判断鼠标是否位于预览弹窗及其安全间距内。
251 function cursorOverPopup(x, y, width, height, cursorX, cursorY, safetyGap) {
252 if (cursorX === undefined || cursorY === undefined || isNaN(cursorX) || isNaN(cursorY))
253 return false;
254 return cursorX >= x - safetyGap && cursorX <= x + width + safetyGap && cursorY >= y - safetyGap && cursorY <= y + height + safetyGap;
255 }

257 // 在鼠标重叠时计算一个不遮挡鼠标的弹窗位置。
258 function avoidCursorOverlap(x, y, width, height, boundaryWidth, boundaryHeight, cursorX, cursorY) {
259 var safetyGap = AppStyle.interactions.safeCursorGap;
260 if (!cursorOverPopup(x, y, width, height, cursorX, cursorY, safetyGap))
261 return Qt.point(x, y);
262 var candidates = [Qt.point(cursorX + safetyGap, y), Qt.point(cursorX - width - safetyGap, y), Qt.point(x, cursorY + safetyGap), Qt.point(x, cursorY - height - safetyGap)];
263 var bestPoint = Qt.point(x, y);
264 var bestDistance = Number.MAX_VALUE;
265 for (var i = 0; i < candidates.length; ++i) {
266 var candidateX = clampPopupCoordinate(candidates[i].x, width, boundaryWidth);
267 var candidateY = clampPopupCoordinate(candidates[i].y, height, boundaryHeight);
268 if (cursorOverPopup(candidateX, candidateY, width, height, cursorX, cursorY, safetyGap))
269 continue;
270 var distance = Math.abs(candidateX - x) + Math.abs(candidateY - y);
271 if (distance < bestDistance) {
272 bestDistance = distance;
273 bestPoint = Qt.point(candidateX, candidateY);
274 }
275 }
276 return bestPoint;
277 }

296 // 根据缩略图位置和窗口边界定位预览弹窗。
297 function positionPreviewPopup() {
298 var popupParent = _previewPopup.parent || componentListCard;
299 var winW = popupParent.width;
300 var winH = popupParent.height;
301 var gap = AppStyle.spacing.lg + 8;
302 var popupW = _previewPopup.width;
303 var popupH = _previewPopup.height;
304 if (winW <= 0 || winH <= 0 || popupW <= 0 || popupH <= 0)
305 return;
306 var thumbX = _previewPopup.anchorX;
307 var thumbY = _previewPopup.anchorY;
308 var thumbW = _previewPopup.anchorWidth;
309 var thumbH = _previewPopup.anchorHeight;
310 var spaceRight = winW - (thumbX + thumbW);
311 var spaceLeft = thumbX;
312 var targetX;
313 if (spaceRight >= popupW + gap) {
314 targetX = thumbX + thumbW + gap;
315 } else if (spaceLeft >= popupW + gap) {
316 targetX = thumbX - popupW - gap;
317 } else {
318 targetX = (winW - popupW) / 2;
319 }
320 targetX = clampPopupCoordinate(targetX, popupW, winW);
321 var targetY = thumbY + (thumbH - popupH) / 2;
322 targetY = clampPopupCoordinate(targetY, popupH, winH);
323 var adjustedPos = avoidCursorOverlap(targetX, targetY, popupW, popupH, winW, winH, _previewPopup.cursorX, _previewPopup.cursorY);
324 _previewPopup.x = snapPopupCoordinate(adjustedPos.x);
325 _previewPopup.y = snapPopupCoordinate(adjustedPos.y);
326 var thumbCenterX = thumbX + thumbW / 2;
327 var thumbCenterY = thumbY + thumbH / 2;
328 _previewPopup.scaleOriginX = Math.max(0, Math.min(popupW, thumbCenterX - _previewPopup.x));
329 _previewPopup.scaleOriginY = Math.max(0, Math.min(popupH, thumbCenterY - _previewPopup.y));
330 }
