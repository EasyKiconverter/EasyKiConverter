27bool copyDirectoryRecursively(const QString& sourcePath, const QString& targetPath) {
28 QDir sourceDir(sourcePath);
29 if (!sourceDir.exists()) {
30 return false;
31 }
32
33 if (!QDir().mkpath(targetPath)) {
34 return false;
35 }
36
37 const QFileInfoList entries =
38 sourceDir.entryInfoList(QDir::NoDotAndDotDot | QDir::AllEntries | QDir::Hidden | QDir::System);
39 for (const QFileInfo& entry : entries) {
40 const QString sourceEntryPath = entry.absoluteFilePath();
41 const QString targetEntryPath = QDir(targetPath).filePath(entry.fileName());
42 if (entry.isDir()) {
43 if (!copyDirectoryRecursively(sourceEntryPath, targetEntryPath)) {
44 return false;
45 }
46 } else {
47 if (QFile::exists(targetEntryPath) && !QFile::remove(targetEntryPath)) {
48 return false;
49 }
50 if (!QFile::copy(sourceEntryPath, targetEntryPath)) {
51 return false;
52 }
53 }
54 }
55
56 return true;
57}

64bool moveFileWithFallback(const QString& sourcePath, const QString& targetPath) {
65 if (!ensureParentDirectory(targetPath)) {
66 return false;
67 }
68
69 constexpr int kMaxRetries = 3;
70 for (int attempt = 0; attempt < kMaxRetries; ++attempt) { // 当前重试序号。
71 if (attempt > 0) {
72 std::this_thread::sleep_for(std::chrono::milliseconds(50 * attempt));
73 }
74 if (QFile::rename(sourcePath, targetPath)) {
75 return true;
76 }
77 }
78
79 if (!QFile::copy(sourcePath, targetPath)) {
80 return false;
81 }
82
83 return QFile::remove(sourcePath);
84}

182bool writeManifest(const QString& manifestPath,

184 const QString& outputRoot,
185 const QVector<BackupEntry>& entries,
186 bool committed) {
187 QJsonObject root;
188 root.insert(QStringLiteral("transactionId"), transactionId);
189 root.insert(QStringLiteral("outputRoot"), outputRoot);
190 root.insert(QStringLiteral("createdAt"), QDateTime::currentDateTimeUtc().toString(Qt::ISODateWithMs));
191 root.insert(QStringLiteral("appVersion"), QCoreApplication::applicationVersion());
192 root.insert(QStringLiteral("committed"), committed);
193
194 QJsonArray entryArray;
195 for (const BackupEntry& entry : entries) { // 当前备份条目。
196 QJsonObject entryObject;
197 entryObject.insert(QStringLiteral("finalPath"), entry.finalPath);
198 entryObject.insert(QStringLiteral("backupPath"), entry.backupPath);
199 entryObject.insert(QStringLiteral("tempPath"), entry.tempPath);
200 entryObject.insert(QStringLiteral("isDirectory"), entry.isDirectory);
201 entryObject.insert(QStringLiteral("existedBefore"), entry.existedBefore);
202 entryArray.append(entryObject);
203 }
204 root.insert(QStringLiteral("entries"), entryArray);
205
206 QFile manifestFile(manifestPath);
207 if (!manifestFile.open(QIODevice::WriteOnly | QIODevice::Truncate)) {
208 qWarning() << "TempFileManager: Failed to write manifest:" << manifestPath << manifestFile.errorString();
209 return false;
210 }
211 return manifestFile.write(QJsonDocument(root).toJson(QJsonDocument::Indented)) > 0;
212}

219QVector<BackupEntry> readManifestEntries(const QString& manifestPath, bool* committed) {
220 QVector<BackupEntry> entries;
221 QFile manifestFile(manifestPath);
222 if (!manifestFile.open(QIODevice::ReadOnly)) {
223 qWarning() << "TempFileManager: Failed to read manifest:" << manifestPath << manifestFile.errorString();
224 return entries;

226
227 const QJsonDocument document = QJsonDocument::fromJson(manifestFile.readAll());
228 const QJsonObject root = document.object();
229 if (committed != nullptr) {
230 *committed = root.value(QStringLiteral("committed")).toBool(false);
231 }
232
233 const QJsonArray entryArray = root.value(QStringLiteral("entries")).toArray();
234 for (const QJsonValue& value : entryArray) { // 当前事务条目值。
235 const QJsonObject object = value.toObject();
236 BackupEntry entry;
237 entry.finalPath = object.value(QStringLiteral("finalPath")).toString();
238 entry.backupPath = object.value(QStringLiteral("backupPath")).toString();
239 entry.tempPath = object.value(QStringLiteral("tempPath")).toString();
240 entry.isDirectory = object.value(QStringLiteral("isDirectory")).toBool(false);
241 entry.existedBefore = object.value(QStringLiteral("existedBefore")).toBool(false);
242 entries.append(entry);
243 }
