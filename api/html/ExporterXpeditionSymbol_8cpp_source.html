148QRectF symbolBounds(const IR::SymbolComponentIR& symbol, int partIndex) {
149 QRectF bounds;
150 bool initialized = false;
151 const auto addPoint = [&](const QPointF& point) {
152 if (!initialized) {
153 bounds = QRectF(point, QSizeF(0.0, 0.0));
154 initialized = true;
155 return;
156 }
157 bounds.setLeft(qMin(bounds.left(), point.x()));
158 bounds.setTop(qMin(bounds.top(), point.y()));
159 bounds.setRight(qMax(bounds.right(), point.x()));
160 bounds.setBottom(qMax(bounds.bottom(), point.y()));
161 };
162 for (const IR::SymbolPinIR& pin : symbol.pins) {
163 if (pin.partIndex != partIndex && !pin.commonToAllParts)
164 continue;
165 addPoint(pin.position);
166 addPoint(pin.position - directionVector(pin.direction) * pin.length);
167 }
168 for (const IR::SymbolRectangleIR& rect : symbol.rectangles) {
169 if (rect.partIndex != partIndex)
170 continue;
171 addPoint(QPointF(rect.x0, rect.y0));
172 addPoint(QPointF(rect.x1, rect.y1));
173 }
174 const auto addPointList = [&](const QList<QPointF>& points, int elementPartIndex) {
175 if (elementPartIndex != partIndex)
176 return;
177 for (const QPointF& point : points)
178 addPoint(point);
179 };
180 for (const IR::SymbolPolylineIR& polyline : symbol.polylines)
181 addPointList(polyline.points, polyline.partIndex);
182 for (const IR::SymbolPolygonIR& polygon : symbol.polygons)
183 addPointList(polygon.points, polygon.partIndex);
184 for (const IR::SymbolCircleIR& circle : symbol.circles) {
185 if (circle.partIndex != partIndex)
186 continue;
187 addPoint(circle.center - QPointF(circle.radius, circle.radius));
188 addPoint(circle.center + QPointF(circle.radius, circle.radius));
189 }
190 for (const IR::SymbolArcIR& arc : symbol.arcs) {
191 if (arc.partIndex != partIndex)
192 continue;
193 addPoint(arc.startPoint);
194 addPoint(arc.midPoint);
195 addPoint(arc.endPoint);
196 }
197 return initialized ? bounds.normalized() : QRectF(-1.0, -1.0, 2.0, 2.0);
198}

206void appendPin(QString& output, const IR::SymbolPinIR& pin, int index) {
207 const QPointF direction = directionVector(pin.direction);
208 const QPointF outer = pin.position;
209 const QPointF inner = outer - direction * pin.length;
210 // Xpedition 侧标识对应引脚方向的枚举值。
211 const int side = pin.direction == IR::PinDirection::Left ? 3
212 : pin.direction == IR::PinDirection::Up ? 0
213 : pin.direction == IR::PinDirection::Down ? 1
214 : 2;
215 output += QStringLiteral("P %1 %2 %3 %4 %5 0 %6 0\n")
216 .arg(index)
217 .arg(fmt(toTh(outer.x())))
218 .arg(fmt(toTh(outer.y())))
219 .arg(fmt(toTh(inner.x())))
220 .arg(fmt(toTh(inner.y())))
221 .arg(side);
222 const QPointF namePosition =
223 pinDisplayPosition(pin.namePosition, pin.hasNamePosition, outer);
224 const QPointF numberPosition =
225 pinDisplayPosition(pin.numberPosition, pin.hasNumberPosition, outer);
226 if (pin.display.showName && !pin.name.isEmpty())
227 output += QStringLiteral("L %1 %2 8 0 2 0 1 0 %3\n")
228 .arg(fmt(toTh(namePosition.x())))
229 .arg(fmt(toTh(namePosition.y())))
230 .arg(escapedText(pin.name));
231 if (pin.display.showDesignator && !pin.designator.isEmpty())
232 output += QStringLiteral("A %1 %2 8 0 3 3 #=%4\n")
233 .arg(fmt(toTh(numberPosition.x())))
234 .arg(fmt(toTh(numberPosition.y())))
235 .arg(escapedText(pin.designator));
236 const QString electricalType = pinElectricalTypeName(pin.electricalType);
237 if (!electricalType.isEmpty())
238 output += QStringLiteral("A %1 %2 10 0 2 0 PINTYPE=%3\n")
239 .arg(fmt(toTh(outer.x())))
240 .arg(fmt(toTh(outer.y())))
241 .arg(electricalType);
242}

282QByteArray ExporterXpeditionSymbol::symbolFile(const IR::SymbolComponentIR& symbol,
283 int partIndex,
284 const QString& outputName) const {
285 // 每个部件独立成文件，同时保留 commonToAllParts 引脚，避免多单元符号丢失公共引脚。
286 QString body;
287 body += QStringLiteral("V 50\nK 1000000000 %1\nY 1\nZ 0\ni 0\n").arg(outputName);
288 body += QStringLiteral("U 0 0 10 0 5 0 %1\n").arg(outputName);
289 body += QStringLiteral("U 0 0 5 0 5 0 REFDES=%1\n").arg(escapedText(symbol.designatorPrefix));
290 body += QStringLiteral("U 0 0 5 0 5 0 VALUE=%1\n").arg(escapedText(symbol.name));
291 const QRectF bounds = symbolBounds(symbol, partIndex);
292 body += QStringLiteral("b %1 %2 %3 %4\n")
293 .arg(fmt(toTh(bounds.left())))
294 .arg(fmt(toTh(bounds.top())))
295 .arg(fmt(toTh(bounds.right())))
296 .arg(fmt(toTh(bounds.bottom())));
297
298 int pinIndex = 1;
299 for (const IR::SymbolPinIR& pin : symbol.pins) {
300 if (pin.partIndex == partIndex || pin.commonToAllParts)
301 appendPin(body, pin, pinIndex++);
302 }
303 for (const IR::SymbolRectangleIR& rect : symbol.rectangles) {
304 if (rect.partIndex != partIndex)
305 continue;
306 appendLine(body, rect.x0, rect.y0, rect.x1, rect.y0);
307 appendLine(body, rect.x1, rect.y0, rect.x1, rect.y1);
308 appendLine(body, rect.x1, rect.y1, rect.x0, rect.y1);
309 appendLine(body, rect.x0, rect.y1, rect.x0, rect.y0);
310 }
311 for (const IR::SymbolPolylineIR& polyline : symbol.polylines) {
312 if (polyline.partIndex == partIndex)
313 appendPolyline(body, polyline.points);
314 }
315 for (const IR::SymbolPolygonIR& polygon : symbol.polygons) {
316 if (polygon.partIndex == partIndex)
317 appendPolyline(body, polygon.points);
318 }
319 for (const IR::SymbolCircleIR& circle : symbol.circles) {
320 if (circle.partIndex != partIndex)
321 continue;
322 body += QStringLiteral("c %1 %2 %3\n|GRPHSTL_EXT01 255 -1 0 1 1\n")
323 .arg(fmt(toTh(circle.center.x())))
324 .arg(fmt(toTh(circle.center.y())))
325 .arg(fmt(toTh(circle.radius)));
326 }
327 for (const IR::SymbolArcIR& arc : symbol.arcs) {
328 if (arc.partIndex != partIndex)
329 continue;
330 body += QStringLiteral("a %1 %2 %3 %4 %5 %6\n|GRPHSTL_EXT01 255 -1 0 1 1\n")
331 .arg(fmt(toTh(arc.startPoint.x())))
332 .arg(fmt(toTh(arc.startPoint.y())))
333 .arg(fmt(toTh(arc.midPoint.x())))
334 .arg(fmt(toTh(arc.midPoint.y())))
335 .arg(fmt(toTh(arc.endPoint.x())))
336 .arg(fmt(toTh(arc.endPoint.y())));
337 }
338 body += QStringLiteral("E\n");
339 return body.toUtf8();
340}

363bool ExporterXpeditionSymbol::exportSymbolLibrary(const QList<IR::SymbolComponentIR>& symbols,
364 const QString& libName,
365 const QString& filePath,
366 bool appendMode,
367 bool updateMode,
368 const QString& libraryDescription) {
369 Q_UNUSED(libName);
370 Q_UNUSED(libraryDescription);
371 // 清理任务级诊断后再写入各个部件，保证调用者读取到的是本次结果。
372 m_diagnostics.clear();
373 if (appendMode || updateMode) {
374 m_diagnostics.append(QStringLiteral("Xpedition 符号 ZIP 暂不支持追加或更新模式"));
375 return false;
376 }
377 XpeditionZipWriter archive;
378 QSet<QString> usedNames;
379 for (const IR::SymbolComponentIR& symbol : symbols) {
380 if (symbol.name.trimmed().isEmpty()) {
381 m_diagnostics.append(QStringLiteral("跳过名称为空的 Xpedition 符号"));
382 continue;
383 }
384 const QString baseName = safeName(symbol.name);
385 QString outputName = baseName;
386 int suffix = 2;
387 while (usedNames.contains(outputName))
388 outputName = QStringLiteral("%1_%2").arg(baseName).arg(suffix++);
389 if (outputName != baseName)
390 m_diagnostics.append(
391 QStringLiteral("Xpedition 符号名称重复，已重命名：%1 -> %2").arg(symbol.name, outputName));
392 usedNames.insert(outputName);
393 const int partCount = qMax(1, symbol.partCount);
394 if (!symbol.ellipses.isEmpty() || !symbol.pies.isEmpty() || !symbol.ellipticalArcs.isEmpty() ||
395 !symbol.paths.isEmpty() || !symbol.beziers.isEmpty() || !symbol.ieeeSymbols.isEmpty() ||
396 !symbol.texts.isEmpty() || !symbol.textFrames.isEmpty() || !symbol.images.isEmpty()) {
397 m_diagnostics.append(QStringLiteral("Xpedition 符号 %1 包含当前未写入的 IR 图元").arg(symbol.name));
398 }
399 if (std::any_of(symbol.pins.cbegin(), symbol.pins.cend(), [](const IR::SymbolPinIR& pin) {
400 return pin.style.hasAny();
401 })) {
402 m_diagnostics.append(QStringLiteral("Xpedition 符号 %1 包含当前未写入的引脚装饰").arg(symbol.name));
403 }
404 for (int partIndex = 0; partIndex < partCount; ++partIndex) {
405 if (!archive.addFile(symbolFileName(outputName, partIndex), symbolFile(symbol, partIndex, outputName))) {
406 m_diagnostics.append(QStringLiteral("Xpedition 符号文件名重复：%1").arg(symbol.name));
407 return false;
408 }
409 }
410 }
411 if (archive.write(filePath))
412 return true;
413 m_diagnostics.append(QStringLiteral("无法写入 Xpedition 符号 ZIP：%1").arg(QFileInfo(filePath).fileName()));
414 return false;
415}
