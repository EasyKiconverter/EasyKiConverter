74bool AltiumSchInputValidator::validate(AltiumSchLibWriter& owner, const QList<AltiumSchComponent>& components) {
75 QStringList& diagnostics = owner.m_diagnostics; // 主写入器共享的诊断列表。
76 QSet<QString> componentNames; // 已出现组件名称的大小写折叠集合。
77 for (const AltiumSchComponent& component : components) {
78 if (component.name.trimmed().isEmpty()) {
79 return reject(diagnostics, QStringLiteral("Altium SchLib 组件名称为空，已拒绝写入"));
80 }
81 if (component.name.toLatin1().size() > 255) {
82 return reject(diagnostics,
83 QStringLiteral("Altium SchLib 组件 %1 名称超过 255 字节，已拒绝写入").arg(component.name));
84 }
85 if (QString::fromLatin1(component.name.toLatin1()) != component.name) {
86 return reject(
87 diagnostics,
88 QStringLiteral("Altium SchLib 组件名称包含无法编码的字符: %1，已拒绝写入").arg(component.name));
89 }
90
91 if (!validateParameterValue(diagnostics, component, component.name, QStringLiteral("组件名称")) ||
92 !validateParameterValue(diagnostics, component, component.description, QStringLiteral("组件描述")) ||
93 !validateParameterValue(diagnostics, component, component.designatorPrefix, QStringLiteral("位号前缀"))) {
94 return false;
95 }
96 for (const QString& alias : component.aliases) {
97 if (!validateParameterValue(diagnostics, component, alias, QStringLiteral("组件别名"))) {
98 return false;
99 }
100 }
101 for (const AltiumSchPin& pin : component.pins) {
102 if (!validateParameterValue(diagnostics, component, pin.name, QStringLiteral("引脚名称")) ||
103 !validateParameterValue(diagnostics, component, pin.designator, QStringLiteral("引脚编号"))) {
104 return false;
105 }
106 }
107 for (const AltiumSchText& text : component.texts) {
108 if (!validateParameterValue(diagnostics, component, text.text, QStringLiteral("文本内容")) ||
109 !validateParameterValue(diagnostics, component, text.fontName, QStringLiteral("文本字体名称")) ||
110 !validateParameterValue(diagnostics, component, text.anchor, QStringLiteral("文本对齐锚点"))) {
111 return false;
112 }
113 }
114 for (const AltiumSchTextFrame& frame : component.textFrames) {
115 if (!validateParameterValue(diagnostics, component, frame.text, QStringLiteral("文本框内容")) ||
116 !validateParameterValue(diagnostics, component, frame.fontName, QStringLiteral("文本框字体名称"))) {
117 return false;
118 }
119 }
120 for (const AltiumSchImage& image : component.images) {
121 // 有效的嵌入图片名称由图片 Storage 协调器负责诊断并跳过；只有
122 // 外部引用或嵌入数据为空时，文件名才会直接进入参数块。
123 if ((!image.embedImage || image.data.isEmpty()) &&
124 !validateParameterValue(diagnostics, component, image.fileName, QStringLiteral("图片文件名"))) {
125 return false;
126 }
127 }
128 for (const AltiumSchParameter& parameter : component.parameters) {
129 if (!validateParameterValue(diagnostics, component, parameter.value, QStringLiteral("参数值"))) {
130 return false;
131 }
132 }
133 for (const auto& implementation : component.implementations) {
134 if (!validateParameterValue(
135 diagnostics, component, implementation.modelName, QStringLiteral("实现模型名称")) ||
136 !validateParameterValue(
137 diagnostics, component, implementation.modelType, QStringLiteral("实现模型类型")) ||
138 !validateParameterValue(
139 diagnostics, component, implementation.dataFileKind, QStringLiteral("实现数据文件类型")) ||
140 !validateParameterValue(
141 diagnostics, component, implementation.dataFileEntity, QStringLiteral("实现数据文件实体"))) {
142 return false;
143 }
144 for (auto it = implementation.parameters.cbegin(); it != implementation.parameters.cend(); ++it) {
145 if (!validateParameterValue(diagnostics, component, it.value(), QStringLiteral("实现参数值"))) {
146 return false;
147 }
148 }
149 for (auto it = implementation.pinMappings.cbegin(); it != implementation.pinMappings.cend(); ++it) {
150 if (!validateParameterValue(diagnostics, component, it.value(), QStringLiteral("引脚映射值"))) {
151 return false;
152 }
153 }
154 }
155
156 const QString foldedName = component.name.trimmed().toCaseFolded(); // 用于检测重名的规范化名称。
157 if (componentNames.contains(foldedName)) {
158 return reject(
159 diagnostics,
160 QStringLiteral("Altium SchLib 组件名称重复（不区分大小写）: %1，已拒绝写入").arg(component.name));
161 }
162 componentNames.insert(foldedName);
163
164 if (component.partCount > 32767) {
165 return reject(diagnostics,
166 QStringLiteral("Altium SchLib 组件 %1 的 partCount 超出支持范围: %2，已拒绝写入")
167 .arg(component.name)
168 .arg(component.partCount));
169 }
170 for (const AltiumSchComponent::Implementation& implementation : component.implementations) {
171 for (auto it = implementation.parameters.cbegin(); it != implementation.parameters.cend(); ++it) {
172 if (it.key().trimmed().isEmpty() || it.key().contains(QChar('|')) || it.key().contains(QChar::Null)) {
173 return reject(diagnostics,
174 QStringLiteral("Altium SchLib 组件 %1 的实现参数键无效: %2，已拒绝写入")
175 .arg(component.name, it.key()));
176 }
177 }
178 }
179 for (auto it = component.sourceMetadata.cbegin(); it != component.sourceMetadata.cend(); ++it) {
180 if (!validateParameterName(diagnostics, component, it.key(), QStringLiteral("源元数据")) ||
181 !validateParameterValue(diagnostics, component, it.value(), QStringLiteral("源元数据值"))) {
182 return false;
183 }
184 }
185 for (const AltiumSchParameter& parameter : component.parameters) {
186 if (!validateParameterName(diagnostics, component, parameter.name, QStringLiteral("参数"))) {
187 return false;
188 }
189 }
190 if (!owner.validateGeometry(component) || !owner.validatePartOwnership(component)) {
191 return false;
192 }
193 if (component.partCount <= 0) {
194 diagnostics.append(
195 QStringLiteral("Altium SchLib 组件 %1 的 partCount 无效，已规范化为 1").arg(component.name));
196 }
197 }
198 return true;
199}
