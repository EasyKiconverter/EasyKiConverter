63 const QPointF& origin,
64 double rotationDegrees,
65 bool mirrorX,
66 bool mirrorY) {
67 QPointF translated = point - origin; // 相对变换中心的坐标。
68 if (mirrorX)
69 translated.setX(-translated.x());
70 if (mirrorY)
71 translated.setY(-translated.y());
72 constexpr double pi = 3.14159265358979323846; // 圆周率常量。
73 const double radians = rotationDegrees * pi / 180.0; // 旋转角度的弧度值。
74 const double cosine = std::cos(radians); // 旋转矩阵余弦项。
75 const double sine = std::sin(radians); // 旋转矩阵正弦项。
76 return QPointF(translated.x() * cosine - translated.y() * sine + origin.x(),
77 translated.x() * sine + translated.y() * cosine + origin.y());
