100bool isHostAllowed(const QString& host, const QSet<QString>& allowedHosts) {
101 if (host.isEmpty()) {
102 return false;
103 }
104
105 // 先执行完整 host 的精确匹配。
106 if (allowedHosts.contains(host)) {
107 return true;
108 }
109
110 // 再检查 host 是否为允许 host 的子域名。
111 for (const QString& allowed : allowedHosts) {
112 if (host.endsWith(QLatin1Char('.') + allowed)) {
113 return true;
114 }
115 }
116
117 return false;
118}

127QString normalizePreviewImageUrl(const QString& imageUrl) {
128 QString normalizedUrl = imageUrl.trimmed(); // 待规范化的 URL 副本。
129 if (normalizedUrl.isEmpty()) {
130 return QString();
131 }
132
133 if (normalizedUrl.startsWith(QStringLiteral("//"))) {
134 normalizedUrl.prepend(QStringLiteral("https:"));
135 } else if (normalizedUrl.startsWith(QStringLiteral("/image.lceda.cn/")) ||
136 normalizedUrl.startsWith(QStringLiteral("/file.elecfans.com/")) ||
137 normalizedUrl.startsWith(QStringLiteral("/www.lcsc.com/"))) {
138 normalizedUrl.remove(0, 1);
139 normalizedUrl.prepend(QStringLiteral("https://"));
140 } else if (normalizedUrl.startsWith(QStringLiteral("/web1/")) ||
141 normalizedUrl.startsWith(QStringLiteral("/M00/"))) {
142 normalizedUrl.remove(0, 1);
143 normalizedUrl.prepend(QStringLiteral("https://file.elecfans.com/"));
144 } else if (normalizedUrl.startsWith('/')) {
145 normalizedUrl.remove(0, 1);
146 normalizedUrl.prepend(QStringLiteral("https://image.lceda.cn/"));
147 } else if (normalizedUrl.contains("alimg.szlcsc.com")) {
148 if (!normalizedUrl.startsWith("http")) {
149 normalizedUrl = "https://" + normalizedUrl;
150 }
151 } else if (!normalizedUrl.contains("://")) {
152 normalizedUrl = "https://image.lceda.cn/" + normalizedUrl;
153 }
154
155 normalizedUrl.replace(QStringLiteral("https://image.lceda.cn//image.lceda.cn/"),
156 QStringLiteral("https://image.lceda.cn/"));
157 normalizedUrl.replace(QStringLiteral("http://image.lceda.cn//image.lceda.cn/"),
158 QStringLiteral("https://image.lceda.cn/"));
159 normalizedUrl.replace(QStringLiteral("https://image.lceda.cn/image.lceda.cn/"),
160 QStringLiteral("https://image.lceda.cn/"));
161
162 return normalizedUrl;
163}

188bool isAllowedUrl(const QUrl& url, ResourceType resourceType, QString* errorMessage) {
189 if (!url.isValid()) {
190 if (errorMessage) {
191 *errorMessage = QStringLiteral("Invalid URL: %1").arg(url.toString());
192 }
193 return false;
194 }
195
196 // 仅允许 https scheme
197 const QString scheme = url.scheme().toLower(); // 规范化后的协议名。
198 if (scheme != QStringLiteral("https")) {
199 if (errorMessage) {
200 *errorMessage = QStringLiteral("URL scheme must be https, got: %1").arg(scheme);
201 }
202 return false;
203 }
204
205 const QString host = url.host().toLower(); // 规范化后的 host。
206
207 // Unknown 类型：仅允许已知域名（拒绝未知 host）
208 if (resourceType == ResourceType::Unknown) {
209 static const QSet<QString> allKnown = // 所有资源类型合并后的 host 白名单。
210 allowedHostsForType(ResourceType::ComponentInfo) + allowedHostsForType(ResourceType::ProductSearch) +
211 allowedHostsForType(ResourceType::PreviewImage) + allowedHostsForType(ResourceType::Datasheet) +
212 allowedHostsForType(ResourceType::Model3DObj) + allowedHostsForType(ResourceType::UpdateCheck) +
213 allowedHostsForType(ResourceType::WorkerRequest);
214 if (!isHostAllowed(host, allKnown)) {
215 if (errorMessage) {
216 *errorMessage = QStringLiteral("URL host '%1' is not in any known allowlist").arg(host);
217 }
218 return false;
219 }
220 return true;
221 }
222 const QSet<QString> allowed = allowedHostsForType(resourceType); // 当前资源类型的 host 白名单。
223
224 if (allowed.isEmpty()) {
225 // 空白名单 = 允许所有（向后兼容）
226 return true;
227 }
228
229 if (!isHostAllowed(host, allowed)) {
230 if (errorMessage) {
231 *errorMessage = QStringLiteral("URL host '%1' is not allowed for resource type %2")
232 .arg(host)
233 .arg(static_cast<int>(resourceType));
234 }
235 return false;
236 }
237
238 return true;
239}
