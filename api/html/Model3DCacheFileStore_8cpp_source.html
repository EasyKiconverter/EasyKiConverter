19QByteArray Model3DCacheFileStore::read(const QString& filePath, const QString& extension) {
20 if (!QFileInfo::exists(filePath)) {
21 return QByteArray();
22 }
23
24 QDir modelDir = QFileInfo(filePath).dir(); // 模型文件所在的组件缓存目录。
25 if (!modelDir.cdUp() || !CacheSafety::isOwnedModel3DFile(modelDir.absolutePath(), filePath)) {
26 return QByteArray();
27 }
28
29 QFile file(filePath);
30 if (!file.open(QIODevice::ReadOnly)) {
31 return QByteArray();
32 }
33
34 const QByteArray data = file.readAll(); // 模型文件原始字节。
35 // Windows 不允许删除仍被 QFile 打开的文件，先显式关闭再清理损坏缓存。
36 file.close();
37 if (CacheDataValidator::isUsableModel3D(data, extension)) {
38 return data;
39 }
40
41 QString trashError; // 回收站操作失败时的诊断信息。
42 CacheSafety::moveToTrash(filePath, &trashError);
43 return QByteArray();
44}

67bool Model3DCacheFileStore::copy(const QString& sourcePath, const QString& destinationPath, const QString& extension) {
68 const QByteArray sourceData = read(sourcePath, extension); // 经校验的源模型数据。
69 if (sourceData.isEmpty()) {
70 return false;
71 }
72
73 const QFileInfo destinationInfo(destinationPath); // 目标文件信息。
74 QDir destinationDir = destinationInfo.dir(); // 目标文件所在目录。
75 if (!destinationDir.exists() && !destinationDir.mkpath(destinationDir.path())) {
76 return false;
77 }
78
79 if (QFile::exists(destinationPath)) {

84 }
