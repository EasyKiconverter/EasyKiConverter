225bool SymbolPinSerializer::fromJson(SymbolPin& pin, const QJsonObject& json) {
226 if (!json.contains("settings") || !json.contains("name")) {
227 return false;
228 }
229 if (!fromJson(pin.settings, json["settings"].toObject())) {
230 return false;
231 }
232 fromJson(pin.pinDot, json["pin_dot"].toObject());
233 fromJson(pin.pinPath, json["pin_path"].toObject());
234 fromJson(pin.name, json["name"].toObject());
235 if (json.contains("number")) {
236 fromJson(pin.number, json["number"].toObject());
237 }
238 fromJson(pin.dot, json["dot"].toObject());
239 fromJson(pin.clock, json["clock"].toObject());
