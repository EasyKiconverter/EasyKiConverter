37IR::PinDirection directionFor(const QPointF& start, const QPointF& end) {
38 const QPointF delta = end - start; // 引脚端点之间的方向向量。
39 if (qAbs(delta.x()) >= qAbs(delta.y()))
40 return delta.x() < 0.0 ? IR::PinDirection::Right : IR::PinDirection::Left;
41 return delta.y() < 0.0 ? IR::PinDirection::Up : IR::PinDirection::Down;
42}

49IR::PinElectricalType electricalTypeFor(const QString& value) {
50 const QString type = value.trimmed().toUpper(); // 统一大小写后的电气类型。
51 if (type == QStringLiteral("IN") || type == QStringLiteral("INPUT"))
52 return IR::PinElectricalType::Input;
53 if (type == QStringLiteral("OUT") || type == QStringLiteral("OUTPUT"))
54 return IR::PinElectricalType::Output;
55 if (type == QStringLiteral("BI") || type == QStringLiteral("BIDIRECTIONAL"))
56 return IR::PinElectricalType::Bidirectional;
57 if (type == QStringLiteral("PASSIVE"))
58 return IR::PinElectricalType::Passive;
59 if (type == QStringLiteral("POWER") || type == QStringLiteral("PWR"))
60 return IR::PinElectricalType::Power;
61 return IR::PinElectricalType::Unspecified;

71const Parser::CadstarPad* findPad(const Parser::CadstarLibrary& library,
72 const QString& name,
73 Parser::ParseDiagnostics* diagnostics) {
74 if (library.isPadAmbiguous(name)) {
75 if (diagnostics)
76 diagnostics->add(Parser::ParseSeverity::Error,
77 Parser::ParseScope::Footprint,
78 QStringLiteral("Cadstar 焊盘引用存在歧义：%1").arg(name),
79 name);
80 return nullptr;
81 }
82 const Parser::CadstarPad* result = nullptr; // 当前唯一匹配的焊盘。
83 for (const Parser::CadstarPad& pad : library.pads) {
84 if (pad.name != name)
85 continue;
86 if (result != nullptr) {
87 if (diagnostics)
88 diagnostics->add(Parser::ParseSeverity::Error,
89 Parser::ParseScope::Footprint,
90 QStringLiteral("Cadstar 焊盘引用存在歧义：%1").arg(name),
91 name);
92 return nullptr;
93 }
94 result = &pad;
95 }
96 if (result == nullptr && diagnostics)
97 diagnostics->add(Parser::ParseSeverity::Error,
98 Parser::ParseScope::Footprint,
99 QStringLiteral("Cadstar 封装引用了不存在的焊盘：%1").arg(name),
100 name);
101 return result;
102}

112IR::FootprintPadIR makePad(const Parser::CadstarPad& source,
113 const Parser::CadstarPackagePin& pin,
114 Parser::LengthUnit unit,
115 Parser::ParseDiagnostics* diagnostics) {
116 IR::FootprintPadIR result; // 转换后的统一焊盘。
117 result.number = pin.number;
118 result.position = millimeters(pin.position, unit);
119 result.position += millimeters(QPointF(source.offsetX, source.offsetY), unit);
120 result.rotation = pin.rotation;
121 if (source.holeDiameter > 0.0 || source.holeWidth > 0.0 || source.holeHeight > 0.0) {
122 result.padType = IR::PadType::ThroughHole;
123 result.layer = IR::LayerType::MultiLayer;
124 result.holeSize = millimeters(source.holeDiameter, unit);
125 result.holeLength = millimeters(qMax(source.holeWidth, source.holeHeight), unit);
126 }
127 // 按来源形状设置统一焊盘形状和尺寸。
128 switch (source.shape) {
129 case Parser::CadstarPadShape::Round:
130 result.shape = IR::PadShape::Ellipse;
131 result.size = QSizeF(millimeters(source.diameter, unit), millimeters(source.diameter, unit));
132 break;
133 case Parser::CadstarPadShape::Square:
134 result.shape = IR::PadShape::Rect;
135 result.size = QSizeF(millimeters(source.width > 0.0 ? source.width : source.diameter, unit),
136 millimeters(source.height > 0.0 ? source.height : source.diameter, unit));
137 break;
138 case Parser::CadstarPadShape::Rectangle:
139 result.shape = IR::PadShape::Rect;
140 result.size = QSizeF(millimeters(source.width, unit), millimeters(source.height, unit));
141 break;
142 case Parser::CadstarPadShape::Oblong:
143 result.shape = IR::PadShape::Oval;
144 result.size = QSizeF(millimeters(source.width, unit), millimeters(source.height, unit));
145 break;
146 case Parser::CadstarPadShape::Octagon:
147 result.shape = IR::PadShape::Polygon;
148 result.size = QSizeF(millimeters(source.diameter, unit), millimeters(source.diameter, unit));
149 break;
150 case Parser::CadstarPadShape::Custom:
151 result.shape = IR::PadShape::Polygon;
152 result.customShapePoints.reserve(source.polygon.size());
153 for (const QPointF& point : source.polygon)
154 result.customShapePoints.append(millimeters(point, unit));
155 break;
156 case Parser::CadstarPadShape::Unknown:
157 result.shape = IR::PadShape::Ellipse;
158 if (diagnostics)
159 diagnostics->add(Parser::ParseSeverity::Warning,
160 Parser::ParseScope::Footprint,
161 QStringLiteral("Cadstar 未知焊盘形状已降级为圆形：%1").arg(source.name),
162 source.name);
163 break;
164 }
165 return result;
166}

174void appendFootprintGraphic(const Parser::CadstarGraphic& source,
175 Parser::LengthUnit unit,
176 IR::FootprintComponentIR& result) {
177 QList<QPointF> converted; // 已换算为毫米的图形顶点。
178 for (const QPointF& point : source.points)
179 converted.append(millimeters(point, unit));
180 // 按图形类型写入封装 IR，所有坐标已经统一为毫米。
181 switch (source.type) {
182 case Parser::CadstarGraphicType::Line:
183 case Parser::CadstarGraphicType::Polyline: {
184 IR::FootprintTrackIR track; // 转换后的封装走线。
185 track.points = converted;
186 track.width = millimeters(source.width, unit);
187 track.layer = IR::LayerType::TopSilk;
188 result.tracks.append(track);
189 break;
190 }
191 case Parser::CadstarGraphicType::Rectangle:
192 if (converted.size() >= 2) {
193 IR::FootprintRectangleIR rectangle; // 转换后的封装矩形。
194 rectangle.bounds = QRectF(converted.at(0), converted.at(1));
195 rectangle.strokeWidth = millimeters(source.width, unit);
196 rectangle.layer = IR::LayerType::TopSilk;
197 result.rectangles.append(rectangle);
198 }
199 break;
200 case Parser::CadstarGraphicType::Circle:
201 if (!converted.isEmpty()) {
202 IR::FootprintCircleIR circle; // 转换后的封装圆形。
203 circle.center = converted.first();
204 circle.radius = millimeters(source.radius, unit);
205 circle.strokeWidth = millimeters(source.width, unit);
206 circle.layer = IR::LayerType::TopSilk;
207 result.circles.append(circle);
208 }
209 break;
210 case Parser::CadstarGraphicType::Arc:
211 if (converted.size() >= 3) {
212 const QPointF& start = converted.at(0); // 圆弧起点。
213 const QPointF& center = converted.at(1); // 圆弧中心。
214 const QPointF& end = converted.at(2); // 圆弧终点。
215 IR::FootprintArcIR arc; // 转换后的封装圆弧。
216 arc.center = center;
217 arc.radius = QLineF(center, start).length();
218 arc.startAngle = qRadiansToDegrees(qAtan2(-(start.y() - center.y()), start.x() - center.x()));
219 arc.endAngle = qRadiansToDegrees(qAtan2(-(end.y() - center.y()), end.x() - center.x()));
220 arc.width = millimeters(source.width, unit);
221 arc.layer = IR::LayerType::TopSilk;
222 result.arcs.append(arc);
223 }
224 break;
225 case Parser::CadstarGraphicType::Polygon: {
226 IR::FootprintRegionIR region; // 转换后的封装区域。
227 region.vertices = converted;
228 region.layer = IR::LayerType::TopSilk;
229 result.regions.append(region);
230 break;
231 }
232 }

241void appendSymbolGraphic(const Parser::CadstarGraphic& source, Parser::LengthUnit unit, IR::SymbolComponentIR& result) {
242 QList<QPointF> converted; // 已换算为毫米的图形顶点。
243 for (const QPointF& point : source.points)
244 converted.append(millimeters(point, unit));
245 // 符号图形使用现有 IR 图元，不把 Cadstar 字段带入 IR。
246 switch (source.type) {
247 case Parser::CadstarGraphicType::Line:
248 case Parser::CadstarGraphicType::Polyline: {
249 IR::SymbolPolylineIR polyline; // 转换后的符号折线。
250 polyline.points = converted;
251 result.polylines.append(polyline);
252 break;
253 }
254 case Parser::CadstarGraphicType::Polygon: {
255 IR::SymbolPolygonIR polygon; // 转换后的符号多边形。
256 polygon.points = converted;
257 polygon.isFilled = true;
258 result.polygons.append(polygon);
259 break;
260 }
261 case Parser::CadstarGraphicType::Rectangle:
262 if (converted.size() >= 2) {
263 const QPointF& first = converted.at(0); // 矩形第一个顶点。
264 const QPointF& second = converted.at(1); // 矩形第二个顶点。
265 IR::SymbolRectangleIR rectangle; // 转换后的符号矩形。
266 rectangle.x0 = first.x();
267 rectangle.y0 = first.y();
268 rectangle.x1 = second.x();
269 rectangle.y1 = second.y();
270 result.rectangles.append(rectangle);
271 }
272 break;
273 case Parser::CadstarGraphicType::Circle:
274 if (!converted.isEmpty())
275 result.circles.append({converted.first(), millimeters(source.radius, unit)});
276 break;
277 case Parser::CadstarGraphicType::Arc:
278 if (converted.size() >= 3)
279 result.arcs.append({converted.at(0), converted.at(1), converted.at(2)});
280 break;
281 }

293IR::FootprintComponentIR CadstarAdapter::toFootprint(const Parser::CadstarLibrary& library,
294 const Parser::CadstarPackage& packageModel,
295 Parser::ParseDiagnostics* diagnostics) {
296 IR::FootprintComponentIR result; // 转换后的封装 IR。
297 result.name = packageModel.name;
298 result.description = packageModel.description;
299 for (const Parser::CadstarPackagePin& sourcePin : packageModel.pins) {
300 const Parser::CadstarPad* pad = findPad(library, sourcePin.padName, diagnostics); // 关联焊盘定义。
301 if (pad == nullptr)
302 continue;
303 result.pads.append(makePad(*pad, sourcePin, library.unit, diagnostics));
304 }
305 for (const Parser::CadstarGraphic& graphic : packageModel.graphics)
306 appendFootprintGraphic(graphic, library.unit, result);
307 return result;
308}

317IR::SymbolComponentIR CadstarAdapter::toSymbol(const Parser::CadstarLibrary& library,
318 const Parser::CadstarComponent& component,
319 Parser::ParseDiagnostics* diagnostics) {
320 Q_UNUSED(diagnostics);
321 IR::SymbolComponentIR result; // 转换后的符号 IR。
322 result.name = component.name;
323 result.partCount = 1;
324 result.designatorPrefix = component.properties.value(QStringLiteral("REFDES"));
325 for (auto iterator = component.properties.cbegin(); iterator != component.properties.cend(); ++iterator)
326 result.sourceMetadata.insert(iterator.key(), iterator.value());
327 for (const Parser::CadstarComponentPin& sourcePin : component.pins) {
328 IR::SymbolPinIR pin; // 转换后的符号引脚。
329 pin.designator = sourcePin.numbers.join(QStringLiteral(","));
330 pin.name = sourcePin.label;
331 pin.position = millimeters(sourcePin.start, library.unit);
332 pin.length =
333 QLineF(sourcePin.start, sourcePin.end).length() * Parser::UnitConverter::toMillimeters(1.0, library.unit);
334 pin.direction = directionFor(sourcePin.start, sourcePin.end);
335 pin.electricalType = electricalTypeFor(sourcePin.pinType);
336 pin.style.inverted = sourcePin.inverted;
337 pin.display.showName = sourcePin.labelVisible;
338 pin.display.showDesignator = sourcePin.numberVisible;
339 pin.hasNamePosition = !sourcePin.label.isEmpty();
340 pin.namePosition = millimeters(sourcePin.labelPosition, library.unit);
341 result.pins.append(pin);
342 }
343 for (const Parser::CadstarGraphic& graphic : component.graphics)
344 appendSymbolGraphic(graphic, library.unit, result);
345 return result;
346}

354CadstarConversionResult CadstarAdapter::toIR(const Parser::CadstarLibrary& library,
355 Parser::ParseDiagnostics* diagnostics) {
356 CadstarConversionResult result; // 汇总后的转换结果。
357 for (const Parser::CadstarPackage& packageModel : library.packages)
358 result.footprints.append(toFootprint(library, packageModel, diagnostics));
359 for (const Parser::CadstarComponent& component : library.components)
360 result.symbols.append(toSymbol(library, component, diagnostics));
361
362 for (const Parser::CadstarPart& part : library.parts) {
363 const bool componentAmbiguous = library.isComponentAmbiguous(part.componentName); // 符号关联是否歧义。
364 const bool packageAmbiguous = library.isPackageAmbiguous(part.packageName); // 封装关联是否歧义。
365 bool componentFound = false; // 是否找到关联符号。
366 bool packageFound = false; // 是否找到关联封装。
367 for (const Parser::CadstarComponent& component : library.components)
368 componentFound = componentFound || component.name == part.componentName;
369 for (const Parser::CadstarPackage& packageModel : library.packages)
370 packageFound = packageFound || packageModel.name == part.packageName;
371 if (diagnostics && componentAmbiguous)
372 diagnostics->add(Parser::ParseSeverity::Error,
373 Parser::ParseScope::Component,
374 QStringLiteral("Cadstar Part 符号关联存在歧义：%1").arg(part.componentName),
375 part.name);
376 if (diagnostics && packageAmbiguous)
377 diagnostics->add(Parser::ParseSeverity::Error,
378 Parser::ParseScope::Footprint,
379 QStringLiteral("Cadstar Part 封装关联存在歧义：%1").arg(part.packageName),
380 part.name);
381 if (diagnostics && !componentFound)
382 diagnostics->add(Parser::ParseSeverity::Error,
383 Parser::ParseScope::Component,
384 QStringLiteral("Cadstar Part 缺少符号关联：%1").arg(part.componentName),
385 part.name);
386 if (diagnostics && !packageFound)
387 diagnostics->add(Parser::ParseSeverity::Error,
388 Parser::ParseScope::Footprint,
389 QStringLiteral("Cadstar Part 缺少封装关联：%1").arg(part.packageName),
390 part.name);
391
392 if (componentAmbiguous || packageAmbiguous || !componentFound || !packageFound)
393 continue;
394
395 const Parser::CadstarComponent* sourceComponent = nullptr; // 关联的源符号。
396 const Parser::CadstarPackage* sourcePackage = nullptr; // 关联的源封装。
397 for (const Parser::CadstarComponent& component : library.components) {
398 if (component.name == part.componentName) {
399 sourceComponent = &component;
400 break;
401 }
402 }
403 for (const Parser::CadstarPackage& packageModel : library.packages) {
404 if (packageModel.name == part.packageName) {
405 sourcePackage = &packageModel;
406 break;
407 }
408 }
409 if (sourceComponent == nullptr || sourcePackage == nullptr)
410 continue;
411
412 IR::ComponentIR component; // 转换后的完整组件。
413 component.name = part.name;
414 component.description = part.description;
415 component.package = part.packageName;
416 component.symbol = toSymbol(library, *sourceComponent, diagnostics);
417 component.footprint = toFootprint(library, *sourcePackage, diagnostics);
418 for (auto iterator = part.properties.cbegin(); iterator != part.properties.cend(); ++iterator)
419 component.sourceMetadata.insert(iterator.key(), iterator.value());
420 result.components.append(component);
421 }
422 return result;
423}
