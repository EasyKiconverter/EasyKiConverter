197bool AltiumBinaryReader::readBlock(QByteArray* payload, uint8_t* flags) {
198 uint32_t header = 0; // 块长度和标志组合值。
199 if (!readUInt32(&header))
200 return false;
201 const int size = static_cast<int>(header & 0x00FFFFFFU); // 块内容长度。
202 if (flags != nullptr)
203 *flags = static_cast<uint8_t>(header >> 24);
204 return readBytes(size, payload);

230bool AltiumBinaryReader::readStringBlock(QString* value) {
231 uint32_t blockSize = 0; // 字符串块总长度。
232 if (!readUInt32(&blockSize) || blockSize < 1 || blockSize > static_cast<uint32_t>(remaining()))
233 return fail(QStringLiteral("字符串块长度无效"));
234 const int payloadStart = m_position; // 字符串块载荷起始偏移。
235 uint8_t stringSize = 0; // 字符串内容长度。
236 if (!readUInt8(&stringSize) || stringSize > blockSize - 1)
237 return fail(QStringLiteral("字符串块内容长度无效"));
238 QByteArray data; // 字符串原始字节。
239 if (!readBytes(stringSize, &data))
240 return false;
241 m_position = payloadStart + static_cast<int>(blockSize);
242 if (value == nullptr)
243 return fail(QStringLiteral("读取字符串块时输出指针为空"));
244 *value = QString::fromLatin1(data);
245 return true;
246}

253bool AltiumBinaryReader::readPascalString(QString* value) {
254 QByteArray block; // Pascal 字符串块数据。
255 if (!readBlock(&block))
256 return false;
257 AltiumBinaryReader blockReader(block); // 字符串块读取器。
258 if (!blockReader.readPascalShortString(value))
259 return fail(blockReader.errorString());
260 uint8_t terminator = 0; // Pascal 字符串终止字节。
261 if (!blockReader.readUInt8(&terminator) || terminator != 0 || blockReader.remaining() != 0)
262 return fail(QStringLiteral("Pascal 字符串块终止符无效"));
263 return true;

286bool AltiumBinaryReader::parseCStringParameterData(const QByteArray& data, QMap<QString, QString>* params) {
287 if (params == nullptr)
288 return fail(QStringLiteral("解析参数数据时输出指针为空"));
289 params->clear();
290 QByteArray block = data; // 去除末尾 NUL 后的参数块数据。
291 if (!block.isEmpty() && block.endsWith('\0'))
292 block.chop(1);
293 const QList<QByteArray> fields = block.split('|'); // 参数字段列表。
294 QSet<QString> rawKeys; // ANSI 参数键集合。
295 QSet<QString> utf8Keys; // UTF-8 覆盖键集合。
296 for (const QByteArray& field : fields) {
297 if (field.isEmpty())
298 continue;
299 const int separator = field.indexOf('='); // 键值分隔符位置。
300 if (separator <= 0)
301 return fail(QStringLiteral("参数块字段缺少键值分隔符"));
302 const QByteArray key = field.left(separator); // 参数键原始字节。
303 const QByteArray rawValue = field.mid(separator + 1); // 参数值原始字节。
304 if (key.startsWith("%UTF8%")) {
305 const QString baseKey = QString::fromUtf8(key.mid(6)); // UTF-8 覆盖对应的基础键。
306 if (baseKey.isEmpty() || utf8Keys.contains(baseKey))
307 return fail(QStringLiteral("参数块包含重复或空的 UTF-8 键"));
308 utf8Keys.insert(baseKey);
309 (*params)[baseKey] = QString::fromUtf8(rawValue);
310 } else {
311 const QString baseKey = QString::fromLatin1(key); // ANSI 参数基础键。
312 if (rawKeys.contains(baseKey) || utf8Keys.contains(baseKey))
313 return fail(QStringLiteral("参数块包含重复键: %1").arg(baseKey));
314 rawKeys.insert(baseKey);
315 (*params)[baseKey] = QString::fromLatin1(rawValue);
316 }
317 }
318 return true;
319}
