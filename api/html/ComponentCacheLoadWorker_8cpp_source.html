46ComponentCacheLoadResult ComponentCacheLoadWorker::load(const QString& componentId,
47 bool fetch3DModel,
48 ComponentCacheService* cache) {
49 ComponentCacheLoadResult result; // 当前元器件的缓存加载结果。
50 result.componentId = componentId;
51
52 QSharedPointer<ComponentData> cachedData = cache->loadComponentData(componentId); // 一级元器件缓存。
53 if (!cachedData) {
54 return result;
55 }
56
57 const QByteArray cadJsonData = cache->loadCadDataJson(componentId); // CAD JSON 原始数据。
58 if (!cadJsonData.isEmpty()) {
59 QJsonParseError parseError; // CAD JSON 解析错误。
60 const QJsonDocument cadDoc = QJsonDocument::fromJson(cadJsonData, &parseError); // 解析后的 JSON 文档。
61 if (parseError.error == QJsonParseError::NoError && cadDoc.isObject()) {
62 const QJsonObject cadDataObj = cadDoc.object(); // CAD JSON 根对象。
63 EasyedaSymbolImporter symbolImporter; // 符号导入器。
64 result.symbolData = symbolImporter.importSymbolData(cadDataObj);
65 EasyedaFootprintImporter footprintImporter; // 封装导入器。
66 result.footprintData = footprintImporter.importFootprintData(cadDataObj);
67 }
68 }
69
70 const QJsonObject metadata = cache->loadMetadata(componentId); // 缓存元数据。
71 if (metadata.contains(QStringLiteral("model3duuid"))) {
72 const QString uuid = metadata.value(QStringLiteral("model3duuid")).toString(); // 三维模型标识。
73 if (!uuid.isEmpty()) {
74 if (!cachedData->model3DData()) {
75 auto model3DData = QSharedPointer<Model3DData>::create(); // 从元数据恢复的三维模型。
76 model3DData->setUuid(uuid);
77 cachedData->setModel3DData(model3DData);
78 } else if (cachedData->model3DData()->uuid().isEmpty()) {
79 cachedData->model3DData()->setUuid(uuid);
80 }
81 }
82 }
83
84 restoreModel3DFromFootprint(*cachedData, result.footprintData);
85 if (!result.symbolData || !result.footprintData) {
86 return result;
87 }
88
89 result.cachedData = cachedData;
90 result.success = true;
91 result.encodedPreviewImages = QStringList(3); // 供界面使用的 Base64 预览图列表。
92 for (int i = 0; i < 3; ++i) {
93 const QByteArray imageData = cache->loadPreviewImage(componentId, i); // 当前序号的预览图数据。
94 if (!imageData.isEmpty()) {
95 result.previewImageData.append({i, imageData});
96 result.encodedPreviewImages[i] = QString::fromLatin1(imageData.toBase64());
97 }
98 }
99
100 result.datasheetData = cache->loadDatasheet(componentId);
101 if (fetch3DModel && cachedData->model3DData() && !cachedData->model3DData()->uuid().isEmpty()) {
102 const QByteArray objData = // 缓存中的 OBJ 模型数据。
103 cache->loadModel3D(cachedData->model3DData()->uuid(), QStringLiteral("obj"));
104 if (!objData.isEmpty()) {
105 cachedData->setModel3DObjRaw(objData);
106 }
107 }
108
109 return result;
110}
