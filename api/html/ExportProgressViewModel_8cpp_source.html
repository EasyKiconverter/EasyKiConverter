156void ExportProgressViewModel::startExport(const QStringList& componentIds,
157 const QString& outputPath,
158 const QString& libName,
159 bool exportSymbol,
160 bool exportFootprint,
161 bool exportModel3D,
162 int exportModel3DFormat,
163 int exportModel3DPathMode,
164 bool exportPreviewImages,
165 bool exportDatasheet,
166 bool overwriteExistingFiles,
167 bool updateMode,
168 bool debugMode,
169 const QString& symbolLibraryDescription,
170 const QString& footprintLibraryDescription,
171 const QString& footprintLibraryKeywords,
172 int targetFormat) {
173 qDebug() << "ExportProgressViewModel: Starting export for" << componentIds.size() << "components"
174 << "targetFormat:" << targetFormat;
175
176 if (m_isExporting) {
177 qWarning() << "Export already in progress";
178 return;
179 }
180
181 if (!m_exportService) {
182 qWarning() << "ParallelExportService is not available";
183 setStatus("Export service not available");
184 return;
185 }
186
187 if (componentIds.isEmpty()) {
188 qWarning() << "No components to export";
189 setStatus("No components to export");
190 return;
191 }
192
193 // Store component IDs
194 const TargetEdaFormat selectedFormat = static_cast<TargetEdaFormat>(targetFormat);
195 // 保存切换目标格式前的重试能力，供变化通知判断使用。
196 const bool oldRetrySupported = m_retrySupported;
197 m_retrySupported = selectedFormat != TargetEdaFormat::Xpedition && selectedFormat != TargetEdaFormat::Allegro &&
198 selectedFormat != TargetEdaFormat::Pads && selectedFormat != TargetEdaFormat::Eagle &&
199 selectedFormat != TargetEdaFormat::Pcad && selectedFormat != TargetEdaFormat::Cadstar &&
200 selectedFormat != TargetEdaFormat::LibrePcb;
201 if (m_retrySupported != oldRetrySupported)
202 emit retrySupportedChanged();
203 m_eagleCombinedLibrary =
204 (selectedFormat == TargetEdaFormat::Eagle || selectedFormat == TargetEdaFormat::Cadstar ||
205 selectedFormat == TargetEdaFormat::Allegro || selectedFormat == TargetEdaFormat::LibrePcb ||
206 selectedFormat == TargetEdaFormat::Horizon) &&
207 // 仅在同时导出符号和封装时生成合并封装库。
208 exportSymbol && exportFootprint;
209 m_exportSymbolEnabled = exportSymbol;
210 m_exportFootprintEnabled = exportFootprint;
211 m_exportModel3DEnabled = exportModel3D;
212 m_exportPreviewEnabled = exportPreviewImages;
213 m_exportDatasheetEnabled = exportDatasheet;
214
215 m_componentIds = componentIds;
216 m_successCount = 0;
217 m_failureCount = 0;
218 m_pendingUpdate = false;
219 if (m_throttleTimer) {
220 m_throttleTimer->stop();
221 }
222 setHasCompletedExport(false);
223 m_resultsList.clear();
224 m_idToIndexMap.clear();
225
226 for (int i = 0; i < componentIds.size(); ++i) {
227 m_idToIndexMap[componentIds[i]] = i;
228 // 创建一个元器件的初始导出结果项。
229 QVariantMap item;
230 item["componentId"] = componentIds[i];
231 item["status"] = "pending";
232 item["symbolSuccess"] = !m_exportSymbolEnabled;
233 item["footprintSuccess"] = !m_exportFootprintEnabled;
234 item["model3DSuccess"] = !m_exportModel3DEnabled;
235 item["previewSuccess"] = !m_exportPreviewEnabled;
236 item["datasheetSuccess"] = !m_exportDatasheetEnabled;
237 item["symbolStatus"] = m_exportSymbolEnabled ? "pending" : "disabled";
238 item["footprintStatus"] = m_exportFootprintEnabled ? "pending" : "disabled";
239 item["model3DStatus"] = m_exportModel3DEnabled ? "pending" : "disabled";
240 item["previewStatus"] = m_exportPreviewEnabled ? "pending" : "disabled";
241 item["datasheetStatus"] = m_exportDatasheetEnabled ? "pending" : "disabled";
242 m_resultsList.append(item);
243 }
244
245 // Build export options
246 ExportOptions options;
247 options.outputPath = outputPath;
248 options.libName = libName;
249 options.exportSymbol = exportSymbol;
250 options.exportFootprint = exportFootprint;
251 options.exportModel3D = exportModel3D;
252 options.exportModel3DFormat = exportModel3DFormat;
253 options.exportModel3DPathMode = ExportOptions::normalizePathMode(exportModel3DPathMode);
254 options.exportPreviewImages = exportPreviewImages;
255 options.exportDatasheet = exportDatasheet;
256 // 这些目标格式只能完整重建库；界面的“覆盖”选项不能作为更新模式传入。
257 const bool requiresFullReplacement =
258 selectedFormat == TargetEdaFormat::Xpedition || selectedFormat == TargetEdaFormat::Allegro ||
259 selectedFormat == TargetEdaFormat::Pads || selectedFormat == TargetEdaFormat::Eagle ||
260 selectedFormat == TargetEdaFormat::Pcad || selectedFormat == TargetEdaFormat::Cadstar ||
261 selectedFormat == TargetEdaFormat::LibrePcb;
262 options.overwriteExistingFiles = overwriteExistingFiles || requiresFullReplacement;
263 options.updateMode = updateMode && !requiresFullReplacement;
264 options.debugMode = debugMode;
265 options.exportSymbolDescription = true;
266 options.exportFootprintDescription = true;
267 options.symbolLibraryDescription = symbolLibraryDescription;
268 options.footprintLibraryDescription = footprintLibraryDescription;
269 options.footprintLibraryKeywords = footprintLibraryKeywords;
270 options.targetFormat = static_cast<TargetEdaFormat>(targetFormat);
271
272 m_exportService->setOptions(options);
273 m_exportService->setOutputPath(outputPath);
274
275 beginExportRun(componentIds, QStringLiteral("Preloading component data..."));
276 emit successCountChanged();
277 emit failureCountChanged();
278
279 // Start preload - actual export will begin after preload completes
280 m_exportService->startPreload(componentIds);
281}

288
289 if (!m_exportService) {

293 // 记录取消前服务是否仍在运行，以决定是否需要本地收敛状态。

295 if (!m_isStopping) {

396void ExportProgressViewModel::handleItemStatusChanged(const QString& componentId,
397 const QString& typeName,
398 const ExportItemStatus& status) {
399 if (m_idToIndexMap.contains(componentId)) {
400 // 查找状态变化对应的结果项索引。
401 int index = m_idToIndexMap[componentId];
402 // 复制待更新的结果项映射。
403 QVariantMap result = m_resultsList[index].toMap();
404
405 // Update individual type success flags
406 if (typeName == "Symbol") {
407 result["symbolSuccess"] = (status.status == ExportItemStatus::Status::Success);
408 } else if (typeName == "Footprint") {
409 result["footprintSuccess"] = (status.status == ExportItemStatus::Status::Success);
410 } else if (typeName == "Model3D") {
411 result["model3DSuccess"] = (status.status == ExportItemStatus::Status::Success);
412 } else if (typeName == "PreviewImages") {
413 result["previewSuccess"] = (status.status == ExportItemStatus::Status::Success);
414 } else if (typeName == "Datasheet") {
415 result["datasheetSuccess"] = (status.status == ExportItemStatus::Status::Success);
416 }
417
418 if (status.status == ExportItemStatus::Status::Failed) {
419 result["error"] = status.errorMessage;
420 }
421
422 // 将内部状态转换为 QML 使用的字符串。
423 QString statusText = "pending";
424 // 将内部枚举状态转换为供 QML 使用的稳定字符串。
425 switch (status.status) {
426 case ExportItemStatus::Status::Pending:
427 statusText = "pending";
428 break;
429 case ExportItemStatus::Status::InProgress:
430 statusText = "in_progress";
431 break;
432 case ExportItemStatus::Status::Success:
433 statusText = "success";
434 break;
435 case ExportItemStatus::Status::Failed:
436 statusText = "failed";
437 break;
438 case ExportItemStatus::Status::Skipped:
439 statusText = "skipped";
440 break;
441 }
442 result[typeStatusKey(typeName)] = statusText;
443 if (m_eagleCombinedLibrary && typeName == QStringLiteral("Footprint")) {
444 // 组合目标在封装阶段同时写入符号；让符号状态跟随同一原子导出结果。
445 result["symbolSuccess"] = result["footprintSuccess"];
446 result["symbolStatus"] = statusText;
447 }
448
449 updateOverallItemStatus(result);
450
451 m_resultsList[index] = result;
452
453 markResultsDirty();
454 }
455}

500void ExportProgressViewModel::handleCancelled() {
501 qInfo() << "Export cancelled";
502
503 // 将所有未完成的项标记为失败，并同步各类型状态，避免小球状态残留
504 for (int i = 0; i < m_resultsList.size(); ++i) {
505 // 复制当前结果项，写入取消诊断和失败状态。
506 QVariantMap result = m_resultsList[i].toMap();
507 // 读取结果项当前的总体状态。
508 QString currentStatus = result["status"].toString();
509 if (currentStatus == "pending" || currentStatus == "in_progress") {
510 result["error"] = "Export cancelled";
511 const QStringList statusKeys = {
512 QStringLiteral("symbolStatus"),
513 QStringLiteral("footprintStatus"),
514 QStringLiteral("model3DStatus"),
515 QStringLiteral("previewStatus"),
516 QStringLiteral("datasheetStatus"),
517 };
518 for (const QString& key : statusKeys) {
519 // 读取该导出类型当前的状态。
520 const QString currentTypeStatus = result.value(key).toString();
521 if (currentTypeStatus == "pending" || currentTypeStatus == "in_progress") {
522 result[key] = "failed";
523 }
524 }
525 updateOverallItemStatus(result);
526 m_resultsList[i] = result;
527 }
528 }
529
530 if (m_isStopping) {
531 m_isStopping = false;
532 emit isStoppingChanged();
533 }
534 setIsExporting(false);
535 setHasCompletedExport(false);
536 setProgress(0);
537 setStatus("Export cancelled");
538 m_fetchProgress = 0;
539 m_processProgress = 0;
540 m_writeProgress = 0;
541 updateResultsList(); // 更新失败数量统计
542 emit stageProgressChanged();
543 flushPendingUpdates();
544}

772void ExportProgressViewModel::removeResult(const QString& componentId) {
773 if (m_idToIndexMap.contains(componentId)) {
774 // 取得待移除结果项在列表中的索引。
775 int index = m_idToIndexMap[componentId];
776 m_resultsList.removeAt(index);
777 m_idToIndexMap.remove(componentId);
778 // Reindex remaining items
779 for (int i = index; i < m_resultsList.size(); ++i) {
780 // 读取结果项中的元器件编号并重建索引。
781 QString id = m_resultsList[i].toMap().value("componentId").toString();
782 m_idToIndexMap[id] = i;
783 }
784 // Update counts first, then notify UI of list change
785 markResultsDirty();
