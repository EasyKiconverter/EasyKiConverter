44QJsonObject CacheMetadataStore::read(const QString& metadataPath) {
45 if (!QFileInfo::exists(metadataPath)) {
46 return QJsonObject();
47 }
48
49 QFile file(metadataPath);
50 if (!file.open(QIODevice::ReadOnly)) {
51 return QJsonObject();
52 }
53
54 const QByteArray data = file.readAll(); // 元数据文件原始内容。
55 QJsonParseError error; // JSON 解析错误信息。
56 const QJsonDocument document = QJsonDocument::fromJson(data, &error); // 解析后的 JSON 文档。
57 if (error.error != QJsonParseError::NoError) {
58 LOG_WARN(LogModule::Core, "JSON parse error: {}", error.errorString());
59 return QJsonObject();
60 }
61
62 return document.object();
63}

70bool CacheMetadataStore::hasValidModel3D(const QJsonObject& metadata) {
71 if (!metadata.contains(QStringLiteral("model3duuid"))) {
72 return true;
73 }
74
75 const QJsonValue uuid = metadata.value(QStringLiteral("model3duuid")); // 三维模型 UUID 字段。
76 if (!uuid.isString() || uuid.toString().isEmpty()) {
77 return false;
78 }
79
80 if (metadata.contains(QStringLiteral("model3dName")) && !metadata.value(QStringLiteral("model3dName")).isString()) {
81 return false;
82 }
83
84 const auto validateVector = [&metadata](const QString& name) {
85 if (!metadata.contains(name)) {
86 return true;
87 }
88 const QJsonValue value = metadata.value(name); // 当前变换字段值。
89 if (!value.isObject()) {
90 return false;
91 }
92 Model3DBase vector; // 待解析的三维变换向量。
93 return vector.fromJson(value.toObject());
94 };
95
96 return validateVector(QStringLiteral("model3dTranslation")) && validateVector(QStringLiteral("model3dRotation"));
97}

105QJsonObject CacheMetadataStore::build(const QString& componentId, const ComponentData& data) {
106 QJsonObject metadata; // 待写入磁盘的缓存元数据。
107 metadata[QStringLiteral("cacheOwner")] = QStringLiteral("EasyKiConverter");
108 metadata[QStringLiteral("cacheEntryVersion")] = 1;
109 metadata["lcscId"] = componentId;
110 metadata["name"] = data.name();
111 metadata["prefix"] = data.prefix();
112 metadata["package"] = data.package();
113 metadata["manufacturer"] = data.manufacturer();
114 metadata["manufacturerPart"] = data.manufacturerPart();

117 metadata["cachedAt"] = QDateTime::currentDateTime().toString(Qt::ISODate);

119 QJsonArray previewUrls; // 通过规范化校验的预览图地址数组。
120 for (const QString& url : data.previewImages()) {
121 const QString normalizedUrl = UrlUtils::normalizePreviewImageUrl(url); // 当前预览图地址的规范化结果。

154QJsonObject CacheMetadataStore::merge(const QJsonObject& existing, const QJsonObject& incoming) {
155 QJsonObject merged = existing; // 合并结果，初始保留已有字段。
156 for (auto it = incoming.begin(); it != incoming.end(); ++it) {
157 const QJsonValue& value = it.value(); // 当前增量字段值。
158 bool shouldWrite = true; // 是否将当前字段写入合并结果。
159
160 if (value.isString() && value.toString().isEmpty()) {
161 shouldWrite = false;
162 } else if (value.isArray() && value.toArray().isEmpty()) {
163 shouldWrite = false;
164 } else if (value.isObject() && value.toObject().isEmpty()) {
165 shouldWrite = false;
166 }
167
168 if (shouldWrite) {
169 merged[it.key()] = value;
170 }
171 }
172
173 merged["cachedAt"] = incoming.value("cachedAt").toString(QDateTime::currentDateTime().toString(Qt::ISODate));
174 return merged;
175}

183bool CacheMetadataStore::writeAtomically(const QString& path, const QByteArray& data) {
184 QFileInfo info(path); // 目标文件及其父目录信息。
185 if (!info.absoluteDir().exists() && !QDir().mkpath(info.absolutePath())) {
186 return false;
187 }
188
189 QSaveFile file(path); // 提供原子提交语义的临时写入文件。
190 if (!file.open(QIODevice::WriteOnly)) {
191 return false;
192 }
193
194 if (file.write(data) != data.size()) {
195 file.cancelWriting();
196 return false;
197 }
198
199 return file.commit();
200}
