44QString padShape(const IR::FootprintPadIR& pad) {
45 // 目标格式可以直接表达圆、长圆、矩形和自定义多边形；其余形状必须先有可用顶点。
46 switch (pad.shape) {
47 case IR::PadShape::Ellipse:
48 return qFuzzyCompare(pad.size.width(), pad.size.height()) ? QStringLiteral("ROUND")
49 : QStringLiteral("OBLONG");
50 case IR::PadShape::Oval:
51 return QStringLiteral("OBLONG");
52 case IR::PadShape::Polygon:
53 case IR::PadShape::RoundRect:
54 case IR::PadShape::Trapezoid:
55 return pad.customShapePoints.size() >= 3 ? QStringLiteral("CUSTOM") : QString();
56 case IR::PadShape::Rect:
57 return QStringLiteral("RECTANGLE");
58 default:
59 return QString();
60 }
61}

79QString padName(const IR::FootprintPadIR& pad) {
80 // 几何对象。
81 QString geometry = QStringLiteral("%1_%2x%3")
82 .arg(padShape(pad))
83 .arg(fmt(toTh(pad.size.width())))
84 .arg(fmt(toTh(pad.size.height())));
85 if (pad.shape == IR::PadShape::Polygon && !pad.customShapePoints.isEmpty()) {
86 // 自定义多边形顶点参与名称计算，避免相同外接尺寸的异形焊盘错误复用。
87 QByteArray points;
88 for (const QPointF& point : pad.customShapePoints)
89 points += QByteArray::number(point.x(), 'f', 6) + ',' + QByteArray::number(point.y(), 'f', 6) + ';';
90 geometry += QStringLiteral("_P%1").arg(
91 QString::fromLatin1(QCryptographicHash::hash(points, QCryptographicHash::Sha1).toHex().left(10)));
92 }
93 if (pad.isThroughHole()) {
94 // 通孔几何参数参与名称计算，避免不同孔径或槽长共享同一个钻孔定义。
95 geometry += QStringLiteral("_H%1_L%2").arg(fmt(toTh(pad.holeSize))).arg(fmt(toTh(pad.holeLength)));
96 // 镀层属性决定孔定义的工艺选项，必须参与名称计算以避免错误复用。
97 if (!pad.isPlated)
98 geometry += QStringLiteral("_NONPLATED");
99 }
100 return QStringLiteral("PAD_") + geometry;
101}

122QList<QPointF> rectanglePoints(const IR::FootprintRectangleIR& rectangle) {
123 const QPointF center = rectangle.bounds.center();
124 const double angle = qDegreesToRadians(rectangle.rotation);
125 const double cosine = std::cos(angle);
126 const double sine = std::sin(angle);
127 // 矩形顶点列表。
128 const QList<QPointF> corners = {rectangle.bounds.topLeft(),
129 rectangle.bounds.topRight(),
130 rectangle.bounds.bottomRight(),
131 rectangle.bounds.bottomLeft()};
132 QList<QPointF> points;
133 points.reserve(corners.size() + 1);
134 for (const QPointF& corner : corners) {
135 const QPointF relative = corner - center;
136 points.append(
137 center + QPointF(relative.x() * cosine - relative.y() * sine, relative.x() * sine + relative.y() * cosine));
138 }
139 points.append(points.first());
140 return points;
141}

148LayerSection layerSection(IR::LayerType layer) {
149 // 每组图层同时决定目标段名和安装面，未知图层返回空值由调用方跳过。
150 switch (layer) {
151 case IR::LayerType::TopSilk:
152 case IR::LayerType::BottomSilk:
153 case IR::LayerType::TopOverlay:
154 case IR::LayerType::BottomOverlay:
155 return {QStringLiteral("SILKSCREEN_OUTLINE"),
156 layer == IR::LayerType::TopSilk || layer == IR::LayerType::TopOverlay ? QStringLiteral("MNT_SIDE")
157 : QStringLiteral("OPP_SIDE")};
158 case IR::LayerType::TopAssembly:
159 case IR::LayerType::BottomAssembly:
160 return {QStringLiteral("ASSEMBLY_OUTLINE"),
161 layer == IR::LayerType::TopAssembly ? QStringLiteral("MNT_SIDE") : QStringLiteral("OPP_SIDE")};
162 case IR::LayerType::TopPaste:
163 case IR::LayerType::BottomPaste:
164 return {QStringLiteral("SOLDER_PASTE"),
165 layer == IR::LayerType::TopPaste ? QStringLiteral("MNT_SIDE") : QStringLiteral("OPP_SIDE")};
166 case IR::LayerType::TopMask:
167 case IR::LayerType::BottomMask:
168 return {QStringLiteral("SOLDER_MASK"),
169 layer == IR::LayerType::TopMask ? QStringLiteral("MNT_SIDE") : QStringLiteral("OPP_SIDE")};
170 case IR::LayerType::TopCopper:
171 case IR::LayerType::BottomCopper:
172 return {QStringLiteral("GRAPHIC"),
173 layer == IR::LayerType::TopCopper ? QStringLiteral("MNT_SIDE") : QStringLiteral("OPP_SIDE")};
174 case IR::LayerType::EdgeCuts:
175 return {QStringLiteral("ASSEMBLY_OUTLINE"), QStringLiteral("MNT_SIDE")};
176 default:
177 return {};
178 }

246QRectF footprintBounds(const IR::FootprintComponentIR& footprint) {
247 double minX = 0.0;
248 double minY = 0.0;
249 double maxX = 0.0;
250 double maxY = 0.0;
251 bool initialized = false;
252 const auto addPoint = [&](const QPointF& point) {
253 if (!initialized) {
254 minX = maxX = point.x();
255 minY = maxY = point.y();
256 initialized = true;
257 return;
258 }
259 minX = qMin(minX, point.x());
260 minY = qMin(minY, point.y());
261 maxX = qMax(maxX, point.x());
262 maxY = qMax(maxY, point.y());
263 };
264 for (const auto& pad : footprint.pads) {
265 addPoint(pad.position - QPointF(pad.size.width() / 2.0, pad.size.height() / 2.0));
266 addPoint(pad.position + QPointF(pad.size.width() / 2.0, pad.size.height() / 2.0));
267 }
268 for (const auto& circle : footprint.circles) {
269 addPoint(circle.center - QPointF(circle.radius, circle.radius));
270 addPoint(circle.center + QPointF(circle.radius, circle.radius));
271 }
272 for (const auto& rect : footprint.rectangles) {
273 for (const QPointF& point : rectanglePoints(rect))
274 addPoint(point);
275 }
276 for (const auto& arc : footprint.arcs) {
277 // 用完整圆弧外接框参与原点计算，保证任意圆弧不会被遗漏；不会改变实际写出的弧段。
278 addPoint(arc.center - QPointF(arc.radius, arc.radius));
279 addPoint(arc.center + QPointF(arc.radius, arc.radius));
280 }
281 for (const auto& region : footprint.regions) {
282 for (const QPointF& point : region.vertices)
283 addPoint(point);
284 }
285 for (const auto& text : footprint.texts)
286 if (text.isDisplayed && !text.text.isEmpty())
287 addPoint(text.position);
288 for (const auto& outline : footprint.outlines) {
289 for (const auto& point : outline.points)
290 addPoint(point);
291 }
292 for (const auto& track : footprint.tracks) {
293 for (const auto& point : track.points)
294 addPoint(point);
295 }
296 for (const auto& hole : footprint.holes) {
297 addPoint(hole.center - QPointF(hole.radius, hole.radius));
298 addPoint(hole.center + QPointF(hole.radius, hole.radius));
299 }
300 return initialized ? QRectF(QPointF(minX, minY), QPointF(maxX, maxY)) : QRectF(-1, -1, 2, 2);
301}

311QString shapeBlock(const QString& kind, const QList<QPointF>& points, double width = 0.0, bool filled = false) {
312 if (points.isEmpty())
313 return {};
314 // 输出对象。
315 QString output = QStringLiteral(" ...%1\n ....WIDTH %2\n ....XY (%3, %4)")
316 .arg(kind)
317 .arg(fmt(toTh(width)))
318 .arg(fmt(toTh(points.first().x())))
319 .arg(fmt(toTh(points.first().y())));
320 for (int index = 1; index < points.size(); ++index)
321 output +=
322 QStringLiteral("\n ...XY (%1, %2)").arg(fmt(toTh(points[index].x()))).arg(fmt(toTh(points[index].y())));
323 if (filled)
324 output += QStringLiteral("\n ....SHAPE_OPTIONS FILLED");
325 output += QLatin1Char('\n');
326 return output;
327}

373QByteArray ExporterXpeditionFootprint::padstackFile(const IR::FootprintComponentIR& footprint) const {
374 // Pad 与 Padstack 分开去重，保证相同尺寸的焊盘只生成一次定义。
375 QString output = QStringLiteral(".FILETYPE PADSTACK_LIBRARY\n.VERSION \"VB99.0\"\n.SCHEMA_VERSION 13\n") +
376 QStringLiteral(".CREATOR \"EasyKiConverter\"\n\n.UNITS TH\n\n");
377 QSet<QString> writtenPads;
378 QSet<QString> writtenStacks;
379
380 // IR 没有独立的阻焊扩展字段，使用明确的 8 TH 默认扩展生成工艺层焊盘。
381 const auto appendPad = [&](const QString& name, const IR::FootprintPadIR& pad, double expansionTh) {
382 if (writtenPads.contains(name))
383 return;
384 const double width = toTh(pad.size.width()) + expansionTh;
385 const double height = toTh(pad.size.height()) + expansionTh;
386 output += QStringLiteral(".PAD \"%1\"\n..PAD_OPTIONS USER_GENERATED_NAME\n..OFFSET (0, 0)\n..%2\n")
387 .arg(name, padShape(pad));
388 if (pad.shape == IR::PadShape::Ellipse && qFuzzyCompare(pad.size.width(), pad.size.height())) {
389 output += QStringLiteral("...DIAMETER %1\n").arg(fmt(width));
390 } else if (hasCustomPadGeometry(pad)) {
391 // 自定义焊盘按 X/Y 尺寸比例扩展，保持多边形的相对形状。
392 const double scaleX = pad.size.width() > 0.0 ? width / toTh(pad.size.width()) : 1.0;
393 const double scaleY =
394 pad.size.height() > 0.0 ? height / toTh(pad.size.height()) : 1.0;
395 output += QStringLiteral("...POLYLINE_SHAPE\n....XY");
396 for (const QPointF& point : pad.customShapePoints)
397 output +=
398 QStringLiteral(" (%1, %2)").arg(fmt(toTh(point.x()) * scaleX)).arg(fmt(toTh(point.y()) * scaleY));
399 output += QStringLiteral("\n....SHAPE_OPTIONS FILLED\n");
400 } else {
401 output += QStringLiteral("...WIDTH %1\n...HEIGHT %2\n").arg(fmt(width)).arg(fmt(height));
402 }
403 writtenPads.insert(name);
404 };
405
406 for (const auto& pad : footprint.pads) {
407 const QString baseName = padName(pad);
408 // 先写入铜焊盘，再写入供阻焊层引用的放大定义。
409 appendPad(baseName, pad, 0.0);
410 const QString maskName = solderMaskPadName(baseName);
411 appendPad(maskName, pad, kSolderMaskExpansionTh);
412
413 const QString stackName =
414 baseName + (pad.isThroughHole() ? QStringLiteral("_TH") : padstackSideSuffix(pad));
415 if (writtenStacks.contains(stackName))
416 continue;
417 output += QStringLiteral(".PADSTACK \"%1\"\n..PADSTACK_TYPE %2\n..TECHNOLOGY \"(Default)\"\n")
418 .arg(stackName, pad.isThroughHole() ? QStringLiteral("PIN_THROUGH") : QStringLiteral("PIN_SMD"));
419 output += QStringLiteral("...TECHNOLOGY_OPTIONS NONE\n");
420 if (pad.isThroughHole() || pad.layer == IR::LayerType::TopCopper)
421 output += QStringLiteral("...TOP_PAD \"%1\"\n").arg(baseName);
422 if (pad.isThroughHole() || pad.layer == IR::LayerType::BottomCopper)
423 output += QStringLiteral("...BOTTOM_PAD \"%1\"\n").arg(baseName);
424 if (pad.isThroughHole() || pad.layer == IR::LayerType::TopCopper)
425 output += QStringLiteral("...TOP_SOLDERMASK_PAD \"%1\"\n").arg(maskName);
426 if (pad.isThroughHole() || pad.layer == IR::LayerType::BottomCopper)
427 output += QStringLiteral("...BOTTOM_SOLDERMASK_PAD \"%1\"\n").arg(maskName);
428 if (!pad.isThroughHole() && pad.layer == IR::LayerType::TopCopper)
429 output += QStringLiteral("...TOP_SOLDERPASTE_PAD \"%1\"\n").arg(baseName);
430 if (!pad.isThroughHole() && pad.layer == IR::LayerType::BottomCopper)
431 output += QStringLiteral("...BOTTOM_SOLDERPASTE_PAD \"%1\"\n").arg(baseName);
432 if (pad.isThroughHole()) {
433 // 通孔焊盘需要独立的 Hole 定义，孔径以实际直径参与命名和写入。
434 const QString drillName = holeName(pad.holeSize, pad.holeLength, pad.isPlated);
435 output += QStringLiteral("...INTERNAL_PAD \"%1\"\n...HOLE_NAME \"%2\"\n").arg(baseName, drillName);
436 output += QStringLiteral(".Hole \"%1\"\n..POSITIVE_TOLERANCE 0\n..NEGATIVE_TOLERANCE 0\n").arg(drillName);
437 output += QStringLiteral("..HOLE_OPTIONS %1 DRILLED USER_GENERATED_NAME\n")
438 .arg(pad.isPlated ? QStringLiteral("PLATED") : QStringLiteral("NON_PLATED"));
439 if (pad.holeLength > pad.holeSize) {
440 // 槽孔长度和孔径分别写入目标格式的宽高字段，方向由 Cell 引脚旋转控制。
441 output += QStringLiteral("..SLOT\n...WIDTH %1\n...HEIGHT %2\n")
442 .arg(fmt(toTh(pad.holeLength)))
443 .arg(fmt(toTh(pad.holeSize)));
444 } else {
445 output += QStringLiteral("..ROUND\n...DIAMETER %1\n").arg(fmt(toTh(pad.holeSize)));
446 }
447 }
448 writtenStacks.insert(stackName);
449 }
450 for (const auto& hole : footprint.holes) {
451 // IR 中没有焊盘的独立孔仍需转换为可定位的安装孔引脚。
452 if (hole.radius <= 0.0)
453 continue;
454 const double diameterMm = hole.radius * 2.0;
455 const QString baseName = holePadName(diameterMm);
456 const QString stackName = standaloneHoleStackName(diameterMm);
457 if (!writtenPads.contains(baseName)) {
458 output += QStringLiteral(
459 ".PAD \"%1\"\n..PAD_OPTIONS USER_GENERATED_NAME\n..OFFSET (0, 0)\n..ROUND\n...DIAMETER %2\n")
460 .arg(baseName)
461 .arg(fmt(toTh(diameterMm)));
462 writtenPads.insert(baseName);
463 }
464 if (writtenStacks.contains(stackName))
465 continue;
466 const QString drillName = holeName(diameterMm, 0.0, false);
467 output += QStringLiteral(".PADSTACK \"%1\"\n..PADSTACK_TYPE PIN_THROUGH\n..TECHNOLOGY \"(Default)\"\n")
468 .arg(stackName);
469 output += QStringLiteral(
470 "...TECHNOLOGY_OPTIONS NONE\n...TOP_PAD \"%1\"\n...BOTTOM_PAD \"%1\"\n...INTERNAL_PAD "
471 "\"%1\"\n...HOLE_NAME \"%2\"\n")
472 .arg(baseName, drillName);
473 output += QStringLiteral(".Hole \"%1\"\n..POSITIVE_TOLERANCE 0\n..NEGATIVE_TOLERANCE 0\n").arg(drillName);
474 output += QStringLiteral("..HOLE_OPTIONS NON_PLATED DRILLED USER_GENERATED_NAME\n..ROUND\n...DIAMETER %1\n")
475 .arg(fmt(toTh(diameterMm)));
476 writtenStacks.insert(stackName);
477 }
478 return output.toUtf8();
479}

485QByteArray ExporterXpeditionFootprint::cellFile(const IR::FootprintComponentIR& footprint) const {
486 // Cell 的局部原点取包围盒中心，能让没有明确原点的封装保持几何居中。
487 const QRectF bounds = footprintBounds(footprint);
488 const QPointF origin = bounds.center();
489 QString output =
490 QStringLiteral(".FILETYPE CELL_LIBRARY\n.VERSION \"1.01.01\"\n.CREATOR \"EasyKiConverter\"\n\n.UNITS TH\n\n");
491 output += QStringLiteral(".PACKAGE_CELL \"%1\"\n ..NUMBER_LAYERS 2\n ..PACKAGE_GROUP General\n ..MOUNT_TYPE %2\n")
492 .arg(safeName(footprint.name),
493 footprint.hasThroughHolePads() || !footprint.holes.isEmpty() ? QStringLiteral("THROUGH")
494 : QStringLiteral("SURFACE"));
495
496 int pinIndex = 1;
497 for (const auto& pad : footprint.pads) {
498 const QString stackName =
499 padName(pad) + (pad.isThroughHole() ? QStringLiteral("_TH") : padstackSideSuffix(pad));
500 output +=
501 QStringLiteral(
502 " ..PIN \"%1\"\n ...XY (%2, %3)\n ...PADSTACK \"%4\"\n ...ROTATION %5\n ...PIN_OPTIONS NONE\n")
503 .arg(pad.number.isEmpty() ? QString::number(pinIndex) : pad.number)
504 .arg(fmt(toTh(pad.position.x() - origin.x())))
505 .arg(fmt(-toTh(pad.position.y() - origin.y())))
506 .arg(stackName)
507 .arg(fmt(pad.rotation));
508 ++pinIndex;
509 }
510
511 const auto appendPolyline =
512 [&](const LayerSection& layer, const QList<QPointF>& points, double width, bool filled) {
513 // 目标坐标系的 Y 轴方向与 IR 相反，这里集中完成平移和翻转。
514 if (layer.name.isEmpty() || points.isEmpty())
515 return;
516 QList<QPointF> normalized;
517 for (const QPointF& point : points)
518 normalized.append(QPointF(point.x() - origin.x(), -(point.y() - origin.y())));
519 output += QStringLiteral(" ..%1\n ...SIDE %2\n").arg(layer.name, layer.side);
520 output += shapeBlock(
521 filled ? QStringLiteral("POLYLINE_SHAPE") : QStringLiteral("POLYLINE_PATH"), normalized, width, filled);
522 };
523 for (const auto& outline : footprint.outlines)
524 appendPolyline(layerSection(outline.layer), outline.points, outline.strokeWidth, false);
525 for (const auto& track : footprint.tracks)
526 appendPolyline(layerSection(track.layer), track.points, track.width, false);
527 for (const auto& rect : footprint.rectangles)
528 appendPolyline(layerSection(rect.layer), rectanglePoints(rect), rect.strokeWidth, false);
529 for (const auto& circle : footprint.circles) {
530 const LayerSection layer = layerSection(circle.layer);
531 if (layer.name.isEmpty())
532 continue;
533 output += QStringLiteral(
534 " ..%1\n ...SIDE %2\n ...CIRCLE_PATH\n ....WIDTH %3\n ....XY (%4, %5)\n ....RADIUS %6\n")
535 .arg(layer.name, layer.side)
536 .arg(fmt(toTh(circle.strokeWidth)))
537 .arg(fmt(toTh(circle.center.x() - origin.x())))
538 .arg(fmt(-toTh(circle.center.y() - origin.y())))
539 .arg(fmt(toTh(circle.radius)));
540 }
541 for (const auto& arc : footprint.arcs) {
542 // 目标格式的封装图元统一使用折线，因此按最多 15 度的弦段近似圆弧。
543 const LayerSection layer = layerSection(arc.layer);
544 if (layer.name.isEmpty() || arc.radius <= 0.0)
545 continue;
546 double sweep = arc.endAngle - arc.startAngle;
547 if (qFuzzyIsNull(sweep))
548 sweep = 360.0;
549 const int segments = qMax(8, static_cast<int>(std::ceil(std::abs(sweep) / 15.0)));
550 QList<QPointF> points;
551 points.reserve(segments + 1);
552 for (int index = 0; index <= segments; ++index) {
553 const double angle = qDegreesToRadians(arc.startAngle + sweep * index / segments);
554 points.append(
555 QPointF(arc.center.x() + arc.radius * std::cos(angle), arc.center.y() + arc.radius * std::sin(angle)));
556 }
557 appendPolyline(layer, points, arc.width, false);
558 }
559 for (const auto& region : footprint.regions) {
560 // 填充区域必须闭合，否则目标工具会把它解释为开放轮廓。
561 const LayerSection layer = layerSection(region.layer);
562 if (layer.name.isEmpty() || region.vertices.size() < 3)
563 continue;
564 QList<QPointF> points = region.vertices;
565 if (points.first() != points.last())
566 points.append(points.first());
567 appendPolyline(layer, points, 0.0, true);
568 }
569 int holeIndex = 1;
570 for (const auto& hole : footprint.holes) {
571 // 独立孔通过 MH 前缀的无电气引脚表达，保持其位置可被 Cell 引用。
572 if (hole.radius <= 0.0)
573 continue;
574 output +=
575 QStringLiteral(
576 " ..PIN \"MH%1\"\n ...XY (%2, %3)\n ...PADSTACK \"%4\"\n ...ROTATION 0\n ...PIN_OPTIONS NONE\n")
577 .arg(holeIndex++)
578 .arg(fmt(toTh(hole.center.x() - origin.x())))
579 .arg(fmt(-toTh(hole.center.y() - origin.y())))
580 .arg(standaloneHoleStackName(hole.radius * 2.0));
581 }
582 for (const auto& text : footprint.texts) {
583 // 仅导出可见文本；隐藏文本不应改变目标封装的视觉结果。
584 if (!text.isDisplayed || text.text.isEmpty())
585 continue;
586 const LayerSection layer = layerSection(text.layer);
587 if (layer.name.isEmpty())
588 continue;
589 const double height = text.fontSize > 0.0 ? toTh(text.fontSize) : 50.0;
590 const double stroke = text.strokeWidth > 0.0 ? toTh(text.strokeWidth) : 3.0;
591 output += QStringLiteral(" ..TEXT \"%1\"\n ...TEXT_TYPE USER_DEFINED\n ...DISPLAY_ATTR\n")
592 .arg(escapedText(text.text));

631bool ExporterXpeditionFootprint::exportFootprintLibrary(const QList<IR::FootprintComponentIR>& footprints,
632 const QString&,
633 const QString& filePath,
634 bool,
635 bool,
636 const QString&,
637 const QString&,
638 bool,
639 const QString&) {
640 // 每次批量导出都清空旧诊断，避免上一次任务的消息污染当前结果。
641 m_diagnostics.clear();
642 XpeditionZipWriter archive;
643 QSet<QString> usedNames;
644 for (const auto& footprint : footprints) {
645 if (footprint.name.trimmed().isEmpty()) {
646 m_diagnostics.append(QStringLiteral("跳过名称为空的 Xpedition 封装"));
647 continue;
648 }
649 const QString baseName = safeName(footprint.name);
650 QString name = baseName;
651 int suffix = 2;
652 while (usedNames.contains(name))
653 name = QStringLiteral("%1_%2").arg(baseName).arg(suffix++);
654 if (name != baseName)
655 m_diagnostics.append(
656 QStringLiteral("Xpedition 封装名称重复，已重命名：%1 -> %2").arg(footprint.name, name));
657 for (const auto& pad : footprint.pads) {
658 if (padShape(pad).isEmpty()) {
659 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 的焊盘 %2 形状无法表达，导出已拒绝")
660 .arg(footprint.name, pad.number));
661 return false;
662 }
663 }
664 if (!footprint.arcs.isEmpty())
665 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 的 %2 个圆弧已使用折线近似")
666 .arg(footprint.name)
667 .arg(footprint.arcs.size()));
668 if (!footprint.texts.isEmpty())
669 m_diagnostics.append(
670 QStringLiteral("Xpedition 封装 %1 已写入 %2 个文本").arg(footprint.name).arg(footprint.texts.size()));
671 if (!footprint.regions.isEmpty())
672 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 已写入 %2 个填充区域")
673 .arg(footprint.name)
674 .arg(footprint.regions.size()));
675 if (!footprint.holes.isEmpty())
676 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 已将 %2 个独立孔写为安装孔引脚")
677 .arg(footprint.name)
678 .arg(footprint.holes.size()));
679 bool hasUnsupportedLayer = false;
680 for (const auto& track : footprint.tracks)
681 hasUnsupportedLayer |= layerSection(track.layer).name.isEmpty();
682 for (const auto& rectangle : footprint.rectangles)
683 hasUnsupportedLayer |= layerSection(rectangle.layer).name.isEmpty();
684 for (const auto& circle : footprint.circles)
685 hasUnsupportedLayer |= layerSection(circle.layer).name.isEmpty();
686 for (const auto& arc : footprint.arcs)
687 hasUnsupportedLayer |= layerSection(arc.layer).name.isEmpty();
688 for (const auto& region : footprint.regions)
689 hasUnsupportedLayer |= layerSection(region.layer).name.isEmpty();
690 for (const auto& outline : footprint.outlines)
691 hasUnsupportedLayer |= layerSection(outline.layer).name.isEmpty();
692 for (const auto& text : footprint.texts)
693 hasUnsupportedLayer |= layerSection(text.layer).name.isEmpty();
694 if (hasUnsupportedLayer)
695 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 包含无法映射的图层图元").arg(footprint.name));
696 if (std::any_of(footprint.texts.cbegin(), footprint.texts.cend(), [](const IR::FootprintTextIR& text) {
697 return text.mirror || !text.textPathPoints.isEmpty();
698 })) {
699 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 的文本镜像或路径字形未写入").arg(footprint.name));
700 }
701 if (std::any_of(footprint.regions.cbegin(), footprint.regions.cend(), [](const IR::FootprintRegionIR& region) {
702 return region.isKeepOut && region.layer != IR::LayerType::KeepOut;
703 })) {
704 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 的 KeepOut 标志与图层不一致").arg(footprint.name));
705 }
706 if (footprint.shouldGenerateCourtyard)
707 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 未自动生成 courtyard").arg(footprint.name));
708 usedNames.insert(name);
709 if (!archive.addFile(name + QStringLiteral("_Pads.hkp"), padstackFile(footprint)) ||
710 !archive.addFile(name + QStringLiteral("_Cell.hkp"), cellFile(footprint))) {
711 m_diagnostics.append(QStringLiteral("Xpedition 封装文件名重复：%1").arg(footprint.name));
712 return false;
713 }
714 if (!footprint.models3d.isEmpty())
715 m_diagnostics.append(QStringLiteral("Xpedition 封装 %1 未写入 3D 模型关联").arg(footprint.name));
716 }
717 if (archive.write(filePath))
718 return true;
719 m_diagnostics.append(QStringLiteral("无法写入 Xpedition 封装 ZIP：%1").arg(QFileInfo(filePath).fileName()));
720 return false;
721}
