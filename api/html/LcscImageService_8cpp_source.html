192bool LcscImageService::tryLoadCachedPreviewImages(const QString& componentId) {
193 // 访问统一缓存服务，查询已保存的预览图文件。
194 ComponentCacheService* cache = ComponentCacheService::instance();
195 if (!cache) {
196 return false;
197 }
198
199 // 统计当前元器件已有的预览图缓存数量。
200 int cachedCount = 0;
201 for (int i = 0; i < MAX_IMAGES_PER_COMPONENT; ++i) {
202 // 解析当前预览图序号对应的缓存路径。
203 QString path = cache->previewImagePath(componentId, i);
204 if (QFileInfo::exists(path)) {
205 cachedCount++;
206 }
207 }
208
209 if (cachedCount == 0) {
210 qDebug() << "LcscImageService: No cached preview images for" << componentId;
211 return false;
212 }
213
214 qDebug() << "LcscImageService: Found" << cachedCount << "cached preview images for" << componentId
215 << ", scheduling async cache load...";
216
217 return true;
218}

226void LcscImageService::loadCachedPreviewImagesAsync(const QString& componentId,
227 ComponentCacheService* cache,
228 quint64 requestToken) {
229 if (m_isCancelled || !isCurrentRequest(componentId, requestToken)) {
230 return;
231 }
232
233 // 保存需要在后台线程读取的预览图序号和路径。
234 QList<QPair<int, QString>> pathsToLoad;
235 for (int i = 0; i < MAX_IMAGES_PER_COMPONENT; ++i) {
236 // 解析当前预览图缓存路径。
237 QString path = cache->previewImagePath(componentId, i);
238 if (QFileInfo::exists(path)) {
239 pathsToLoad.append({i, path});
240 }
241 }
242
243 if (pathsToLoad.isEmpty()) {
244 qDebug() << "LcscImageService: No cached images found for" << componentId;
245 m_requestedComponents.remove(componentId);
246 m_requestTokens.remove(componentId);
247 checkDownloadCompletion(componentId, requestToken);
248 return;
249 }
250
251 m_expectedCounts[componentId] = pathsToLoad.size();
252 m_downloadCounts[componentId] = 0;
253
254 // 按图片序号保存后台读取到的图片数据。
255 auto componentImageData = QSharedPointer<QMap<int, QByteArray>>::create();
256 // 统计已完成读取的图片数量，并由回调共享更新。
257 auto loadedCount = QSharedPointer<int>::create(0);
258 // 保存本次需要读取的图片总数，用于判断批量读取是否结束。
259 const int totalImages = pathsToLoad.size();
260
261 for (const auto& [imageIndex, path] : pathsToLoad) {
262 // 在缓存专用线程池中异步读取当前图片文件。
263 QFuture<QByteArray> future = QtConcurrent::run(m_cacheThreadPool, [path]() {
264 QFile file(path);
265 if (file.open(QIODevice::ReadOnly)) {
266 return file.readAll();
267 }
268 return QByteArray();
269 });
270
271 // 监听当前图片读取任务，并在所属线程回调中汇总结果。
272 auto* watcher = new QFutureWatcher<QByteArray>(this);
273 connect(watcher,
274 &QFutureWatcher<QByteArray>::finished,
275 this,
276 [this, watcher, componentId, imageIndex, componentImageData, loadedCount, totalImages, requestToken]() {
277 // 获取当前异步读取任务返回的图片数据。
278 QByteArray imageData = watcher->result();
279 if (!isCurrentRequest(componentId, requestToken)) {
280 m_pendingImageWatchers.removeOne(watcher);
281 watcher->deleteLater();
282 return;
283 }
284 if (!imageData.isEmpty()) {
285 (*componentImageData)[imageIndex] = imageData;
286 }
287
288 ++(*loadedCount);
289 if (*loadedCount >= totalImages) {
290 m_downloadCounts[componentId] = totalImages;
291 checkDownloadCompletion(componentId, requestToken);
292 }
293
294 m_pendingImageWatchers.removeOne(watcher);
295 watcher->deleteLater();
296 });
297
298 m_pendingImageWatchers.append(watcher);
299 watcher->setFuture(future);
300 }
301}

337void LcscImageService::performApiSearch(const QString& componentId, quint64 requestToken) {
338 if (m_isCancelled) {
339 return;
340 }
341
342 // 保存产品搜索请求的表单编码数据。
343 QByteArray postData;
344 // 构造产品搜索请求的查询参数。
345 QUrlQuery query;
346 query.addQueryItem("keyword", componentId);
347 query.addQueryItem("currPage", "1");
348 query.addQueryItem("pageSize", "50");
349 query.addQueryItem("needAggs", "true");
350 postData = query.toString(QUrl::FullyEncoded).toUtf8();
351
352 // 使用产品搜索接口的默认网络策略，并覆盖本服务的超时和重试配置。
353 RequestProfile profile = RequestProfiles::productSearch();
354 profile.connectTimeoutMs = 30000;
355 profile.readTimeoutMs = 30000;
356 profile.maxRetries = MAX_RETRY_COUNT;
357
358 // 启动异步产品搜索请求，完成后解析精确匹配的产品信息。
359 AsyncNetworkRequest* request =
360 NetworkClient::instance().postAsync(QUrl("https://pro.lceda.cn/api/v2/eda/product/search"),
361 postData,
362 ResourceType::ProductSearch,
363 RetryPolicy::fromProfile(profile, isWeakNetworkEnabled()));
364 trackAsyncRequest(request);
365
366 QObject::connect(
367 request,
368 &AsyncNetworkRequest::finished,
369 this,
370 [this, request, componentId, requestToken](const NetworkResult& result) {
371 untrackAsyncRequest(request);
372
373 if (m_isCancelled) {
374 request->deleteLater();
375 return;
376 }
377
378 // 检查组件是否已被取消
379 if (!isCurrentRequest(componentId, requestToken)) {
380 request->deleteLater();
381 return;
382 }
383
384 if (!result.success) {
385 qWarning() << "LcscImageService: API search failed for component:" << componentId
386 << "error:" << result.error;
387 request->deleteLater();
388 m_requestedComponents.remove(componentId);
389 m_requestTokens.remove(componentId);
390 // 不再触发fallback，直接报告无预览图
391 emit error(componentId, "No preview image available from API");
392 return;
393 }
394
395 // 解析搜索响应中的精确匹配产品信息。
396 const std::optional<LcscProductInfo> productInfo = LcscProductParser::parse(componentId, result.data);
397 if (!productInfo.has_value()) {
398 qWarning() << "LcscImageService: No matching product found for component:" << componentId;
399 request->deleteLater();
400 m_requestedComponents.remove(componentId);
401 m_requestTokens.remove(componentId);
402 // 不再触发 fallback，直接报告无预览图。
403 emit error(componentId, "No preview image available from API");
404 return;
405 }
406
407 // 提取产品提供的预览图地址并通知调用方。
408 const QStringList imageUrls = productInfo->imageUrls;
409 emit lcscDataReady(componentId, productInfo->manufacturerPart, productInfo->datasheetUrl, imageUrls);
410
411 if (!imageUrls.isEmpty()) {
412 startPreviewImageDownloads(componentId, imageUrls, requestToken);
413 } else {
414 m_requestedComponents.remove(componentId);
415 m_requestTokens.remove(componentId);
416 emit error(componentId, "No images available");
417 }
418
419 request->deleteLater();
420 });
421}

430void LcscImageService::performDownload(const QString& componentId,
431 const QString& imageUrl,
432 int imageIndex,
433 quint64 requestToken) {
434 if (m_isCancelled) {
435 return;
436 }
437
438 // 捕获当前缓存代次，防止目录切换后旧响应写入新缓存。
439 const uint64_t gen = ComponentCacheService::instance()->currentGeneration();
440 // 启动当前预览图的异步下载请求。
441 AsyncNetworkRequest* request = NetworkClient::instance().getAsync(
442 QUrl(imageUrl),
443 ResourceType::PreviewImage,
444 RetryPolicy::fromProfile(RequestProfiles::previewImage(), isWeakNetworkEnabled()));
445 trackAsyncRequest(request);
446 connect(request,
447 &AsyncNetworkRequest::finished,
448 this,
449 [this, request, componentId, imageIndex, gen, requestToken](const NetworkResult& result) {
450 untrackAsyncRequest(request);
451
452 if (m_isCancelled) {
453 request->deleteLater();
454 return;
455 }
456
457 // 缓存代次变化表示该请求已经过期，不能再通知当前组件状态。
458 if (ComponentCacheService::instance()->currentGeneration() != gen) {
459 request->deleteLater();
460 return;
461 }
462
463 // 检查组件是否已被取消
464 if (!isCurrentRequest(componentId, requestToken)) {
465 request->deleteLater();
466 return;
467 }
468
469 if (!result.success) {
470 m_downloadCounts[componentId]++;
471 checkDownloadCompletion(componentId, requestToken);
472 request->deleteLater();
473 return;
474 }
475
476 // 读取网络响应内容，后续检查是否为有效图片而非错误页。
477 const QByteArray imageData = result.data;
478 if (!imageData.isEmpty() && !QString::fromUtf8(imageData).contains("<!DOCTYPE", Qt::CaseInsensitive)) {
479 ComponentCacheService::instance()->savePreviewImage(componentId, imageData, imageIndex, gen);
480 m_downloadCounts[componentId]++;
481 emit imageReady(componentId, imageData, imageIndex);
482 checkDownloadCompletion(componentId, requestToken);
483 } else {
484 qWarning() << "LcscImageService: Downloaded invalid data for" << componentId << "index"
485 << imageIndex << "- may be blocked (403), falling back to API";
486 m_downloadCounts[componentId]++;
487 checkDownloadCompletion(componentId, requestToken);
488 emit error(componentId, "Image blocked (403), please retry");
489 }
490
491 request->deleteLater();
492 });
493}

500void LcscImageService::checkDownloadCompletion(const QString& componentId, quint64 requestToken) {
501 if (!isCurrentRequest(componentId, requestToken)) {
502 return;
503 }
504 if (!m_expectedCounts.contains(componentId)) {
505 return;
506 }
507
508 // 获取当前组件预期下载的图片数量。
509 int expectedCount = m_expectedCounts[componentId];
510 // 获取当前组件已经完成下载的图片数量。
511 int downloadedCount = m_downloadCounts.value(componentId, 0);
512
513 if (downloadedCount >= expectedCount) {
514 emitAllImagesReady(componentId, requestToken);
515 }

523void LcscImageService::emitAllImagesReady(const QString& componentId, quint64 requestToken) {
524 if (!isCurrentRequest(componentId, requestToken)) {
525 return;
526 }
527 // 访问统一缓存服务，重新校验最终可用的图片文件。
528 ComponentCacheService* cache = ComponentCacheService::instance();
529 // 保存通过内容校验并将通知给调用方的图片路径。
530 QStringList imagePaths;
531 for (int i = 0; i < MAX_IMAGES_PER_COMPONENT; ++i) {
532 // 解析当前图片序号对应的缓存路径。
533 const QString path = cache->previewImagePath(componentId, i);
534 // 最终通知前重新读取并校验缓存，避免错误页或损坏图片仅因文件存在而被上层使用。
535 if (!cache->loadPreviewImage(componentId, i).isEmpty()) {
536 imagePaths.append(path);
537 }
538 }
539
540 if (imagePaths.isEmpty()) {
541 emit error(componentId, "No images downloaded");
542 } else {
543 emit allImagesReady(componentId, imagePaths);
544 }
545
546 m_downloadCounts.remove(componentId);
547 m_expectedCounts.remove(componentId);
548 m_requestedComponents.remove(componentId);
549 m_requestTokens.remove(componentId);
550}

572 trackAsyncRequest(request);

575 this,

577 untrackAsyncRequest(request);
578

580 request->deleteLater();
581 return;
582 }
583
584 // 缓存代次变化表示该请求已经过期，不能再通知当前组件状态。
585 if (ComponentCacheService::instance()->currentGeneration() != gen) {
586 request->deleteLater();
587 return;
588 }
589
590 if (!isCurrentDatasheetRequest(componentId, requestToken)) {
591 request->deleteLater();
592 return;
593 }
594

597 m_datasheetTokens.remove(componentId);

599 return;

601

603 const QByteArray datasheetData = result.data;
604 if (!datasheetData.isEmpty()) {
605 // 根据地址后缀推断缓存格式，供后续内容校验使用。
606 QString format = datasheetUrl.toLower().contains(".html") ? "html" : "pdf";
607 ComponentCacheService::instance()->saveDatasheet(componentId, datasheetData, format, gen);
608 // 从磁盘重新读取实际保存的数据，确保通知内容与缓存一致。
609 QByteArray savedData = ComponentCacheService::instance()->loadDatasheet(componentId);
610 emit datasheetReady(componentId, savedData);
611 m_datasheetTokens.remove(componentId);
612 } else {
613 emit error(componentId, "Failed to read datasheet data");
614 m_datasheetTokens.remove(componentId);
615 }
616
617 request->deleteLater();
618 });
