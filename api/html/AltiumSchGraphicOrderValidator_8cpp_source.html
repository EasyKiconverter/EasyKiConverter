16bool AltiumSchGraphicOrderValidator::validate(const AltiumSchComponent& component) {
17 QSet<QString> expected; // 按组件图元计算出的完整顺序键集合。
18 QSet<QString> actual; // 来源顺序中实际出现的顺序键集合。
19 QMap<QString, QSet<int>> pathSegments; // 按路径键收集的已出现片段索引。
20 QSet<QString> unsplitPaths; // 没有片段索引、需要整体写出的路径集合。
21 const auto key = [](const QString& type, int index, int partIndex) {
22 return QStringLiteral("%1:%2:%3").arg(type).arg(index).arg(partIndex);
23 };
24
25 QMap<int, int> partPinIndexes; // 各部件普通引脚的局部索引计数。
26 QMap<int, int> commonPinIndexes; // 公共引脚的局部索引计数。
27 for (const AltiumSchPin& pin : component.pins) {
28 const int partIndex = pin.sourcePartIndex; // 当前引脚的来源部件索引。
29 expected.insert(key(QStringLiteral("P"), partPinIndexes[partIndex]++, partIndex));
30 if (pin.ownerPartId == -1)
31 expected.insert(key(QStringLiteral("P"), commonPinIndexes[-1]++, -1));
32 }
33
34 const auto addUniqueIndexed = [&expected, &key](const QString& type, int index, int partIndex) {
35 if (index < 0)
36 return true;
37 const QString orderKey = key(type, index, partIndex); // 当前图元的唯一顺序键。
38 if (expected.contains(orderKey))
39 return false;
40 expected.insert(orderKey);
41 return true;
42 };
43 const auto addUniqueNonPathIndexed = [&addUniqueIndexed, &expected, &key](
44 const QString& type, int index, int partIndex) {
45 if (type == QStringLiteral("PT")) {
46 if (index >= 0)
47 expected.insert(key(type, index, partIndex));
48 return true;
49 }
50 return addUniqueIndexed(type, index, partIndex);
51 };
52 for (const AltiumSchRectangle& rect : component.rectangles) {
53 if (!addUniqueIndexed(QStringLiteral("R"), rect.sourceGraphicIndex, rect.sourcePartIndex))
54 return false;
55 }
56 for (const AltiumSchRoundRectangle& rect : component.roundRectangles) {
57 if (!addUniqueIndexed(QStringLiteral("R"), rect.sourceGraphicIndex, rect.sourcePartIndex))
58 return false;
59 }
60 for (const AltiumSchEllipse& ellipse : component.ellipses) {
61 if (!addUniqueNonPathIndexed(ellipse.sourceGraphicType, ellipse.sourceGraphicIndex, ellipse.sourcePartIndex))
62 return false;
63 }
64 for (const AltiumSchArc& arc : component.arcs) {
65 if (!addUniqueNonPathIndexed(arc.sourceGraphicType, arc.sourceGraphicIndex, arc.sourcePartIndex))
66 return false;
67 }
68 for (const AltiumSchPolyline& polyline : component.polylines) {
69 if (!addUniqueIndexed(QStringLiteral("PL"), polyline.sourceGraphicIndex, polyline.sourcePartIndex))
70 return false;
71 }
72 for (const AltiumSchPolygon& polygon : component.polygons) {
73 if (!addUniqueIndexed(QStringLiteral("PG"), polygon.sourceGraphicIndex, polygon.sourcePartIndex))
74 return false;
75 }
76 for (const AltiumSchText& text : component.texts) {
77 if (!text.isPinLabel && !addUniqueIndexed(QStringLiteral("T"), text.sourceGraphicIndex, text.sourcePartIndex))
78 return false;
79 }
80 const bool hasImageOrder = std::any_of(
81 component.graphicOrder.cbegin(), component.graphicOrder.cend(), [](const AltiumSchGraphicOrder& order) {
82 return order.type == QStringLiteral("I");
83 });
84 if (hasImageOrder) {
85 for (const AltiumSchImage& image : component.images) {
86 if (!addUniqueIndexed(QStringLiteral("I"), image.sourceGraphicIndex, image.sourcePartIndex))
87 return false;
88 }
89 }
90 const auto addPathSegment = [&pathSegments, &unsplitPaths, &key](
91 const QString& type, int index, int segmentIndex, int partIndex) {
92 if (type != QStringLiteral("PT") || index < 0)
93 return true;
94 const QString pathKey = key(type, index, partIndex); // 当前路径的唯一顺序键。
95 if (segmentIndex < 0)
96 unsplitPaths.insert(pathKey);
97 else {
98 QSet<int>& segments = pathSegments[pathKey]; // 当前路径已登记的片段索引集合。
99 if (segments.contains(segmentIndex))
100 return false;
101 segments.insert(segmentIndex);
102 }
103 return true;
104 };
105 for (const AltiumSchPath& path : component.paths) {
106 if (path.sourceGraphicIndex >= 0)
107 expected.insert(key(path.sourceGraphicType, path.sourceGraphicIndex, path.sourcePartIndex));
108 if (!addPathSegment(
109 path.sourceGraphicType, path.sourceGraphicIndex, path.sourceSegmentIndex, path.sourcePartIndex))
110 return false;
111 }
112 for (const AltiumSchBezier& bezier : component.beziers) {
113 if (bezier.controlPoints.size() == 4) {
114 if (bezier.sourceGraphicIndex >= 0)
115 expected.insert(key(bezier.sourceGraphicType, bezier.sourceGraphicIndex, bezier.sourcePartIndex));
116 if (!addPathSegment(bezier.sourceGraphicType,
117 bezier.sourceGraphicIndex,
118 bezier.sourceSegmentIndex,
119 bezier.sourcePartIndex))
120 return false;
121 }
122 }
123 for (const AltiumSchEllipticalArc& arc : component.ellipticalArcs) {
124 if (!addUniqueNonPathIndexed(arc.sourceGraphicType, arc.sourceGraphicIndex, arc.sourcePartIndex))
125 return false;
126 if (!addPathSegment(arc.sourceGraphicType, arc.sourceGraphicIndex, arc.sourceSegmentIndex, arc.sourcePartIndex))
127 return false;
128 }
129 for (const AltiumSchArc& arc : component.arcs) {
130 if (!addPathSegment(arc.sourceGraphicType, arc.sourceGraphicIndex, arc.sourceSegmentIndex, arc.sourcePartIndex))
131 return false;
132 }
133
134 // 这些图元只有在来源类型和索引同时有效时才能由 AltiumSchGraphicOrderWriter 写出。
135 // 否则应回退到默认顺序，避免来源类型非空但索引缺失的图元被静默丢弃。
136 const auto hasUnresolvedOrderedGraphic = [](const auto& graphics) {
137 for (const auto& graphic : graphics) {
138 if (!graphic.sourceGraphicType.isEmpty() && graphic.sourceGraphicIndex < 0)
139 return true;
140 }
141 return false;
142 };
143 if (hasUnresolvedOrderedGraphic(component.ellipses) || hasUnresolvedOrderedGraphic(component.arcs) ||
144 hasUnresolvedOrderedGraphic(component.paths) || hasUnresolvedOrderedGraphic(component.beziers) ||
145 hasUnresolvedOrderedGraphic(component.ellipticalArcs)) {
146 return false;
147 }
148
149 for (const AltiumSchGraphicOrder& order : component.graphicOrder) {
150 if (order.index < 0 || order.type.isEmpty())
151 return false;
152 const QString orderKey = key(order.type, order.index, order.partIndex); // 当前来源顺序项的唯一键。
153 if (actual.contains(orderKey))
154 return false;
155 actual.insert(orderKey);
156 }
157
158 QSet<int> emittedPins; // 已由来源顺序覆盖的组件引脚索引。
159 for (const AltiumSchGraphicOrder& order : component.graphicOrder) {
160 if (order.type != QStringLiteral("P"))
161 continue;
162 int localIndex = 0; // 当前部件范围内的引脚局部索引。
163 for (int pinIndex = 0; pinIndex < component.pins.size(); ++pinIndex) {
164 const AltiumSchPin& pin = component.pins.at(pinIndex); // 当前候选引脚。
165 const bool matchesPart =
166 order.partIndex < 0 ? pin.ownerPartId == -1 : pin.sourcePartIndex == order.partIndex;
167 if (!matchesPart)
168 continue;
169 if (localIndex == order.index) {
170 if (emittedPins.contains(pinIndex))
171 return false;
172 emittedPins.insert(pinIndex);
173 break;
174 }
175 ++localIndex;
176 }
177 }
178 if (emittedPins.size() != component.pins.size())
179 return false;
180
181 for (const QString& pathKey : unsplitPaths) {
182 if (pathSegments.contains(pathKey))
183 return false;
184 }
185 for (auto it = pathSegments.cbegin(); it != pathSegments.cend(); ++it) {
186 // 段索引允许出现缺口：前面的来源段可能因几何量化或参数无效被跳过，
187 // 但后续有效段仍应按实际索引写出，而不是迫使整个图元回退到分组顺序。
188 if (it.value().isEmpty())
189 return false;
190 }
191
192 for (const QString& expectedKey : expected) {
193 if (expectedKey.startsWith(QStringLiteral("P:")))
194 continue;
195 if (!actual.contains(expectedKey))
196 return false;
197 }
198 for (const QString& actualKey : actual) {
199 if (!expected.contains(actualKey))
200 return false;
201 }
202 return true;
203}
