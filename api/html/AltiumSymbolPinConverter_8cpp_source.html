154void applyPowerPinType(const IR::SymbolPinIR& pin, AltiumSchPin& altiumPin) {
155 if (altiumPin.electricalType == AltiumModels::PinElectricalType::Power)
156 return;

158 static const QSet<QString> powerPinNames = {

173AltiumSchPin AltiumSymbolPinConverter::convert(const IR::SymbolPinIR& pin) {
174 AltiumSchPin altiumPin; // 转换后的 Altium 引脚记录。
175 altiumPin.name = pin.name;
176 altiumPin.designator = pin.designator;
177 altiumPin.locationX = AltiumCoord::mmToRaw(pin.position.x());
178 altiumPin.locationY = AltiumCoord::mmToRaw(pin.position.y());
179 altiumPin.length = pin.length > 0.0 ? AltiumCoord::mmToRaw(pin.length) : 100000;
180 altiumPin.electricalType =
181 static_cast<AltiumModels::PinElectricalType>(AltiumLayerMap::toAltiumElectricalType(pin.electricalType));
182 altiumPin.orientation =
183 static_cast<AltiumModels::PinOrientation>(AltiumLayerMap::toAltiumPinOrientation(pin.direction));
184 // EasyEDA 的 pin name 显示标志在部分库中未设置，但名称字符串本身仍是符号的一部分。
185 altiumPin.showName =
186 !pin.hasNamePosition && (pin.display.showName || pin.showName || !pin.name.trimmed().isEmpty());
187 // 编号统一由独立的水平文本记录承载，禁止 Altium 按引脚方向旋转内置 designator。
188 altiumPin.showDesignator = false;
189 altiumPin.isHidden = !altiumPin.showName && !altiumPin.showDesignator;
190 altiumPin.color = AltiumSymbolConversionUtils::toAltiumColor(QColor(Qt::black));
191
192 if (pin.style.inverted || pin.hasDot)
193 altiumPin.symbolOuterEdge = 1;
194 if (pin.style.activeLow)
195 altiumPin.symbolOuterEdge = 4;
196 if (pin.style.clock || pin.hasClock)
197 altiumPin.symbolInnerEdge = 3;
198
199 applyPinDecoration(pin, altiumPin);
200 applyElectricalDecoration(pin, altiumPin);
201 altiumPin.ownerPartId = pin.commonToAllParts ? -1 : AltiumSymbolConversionUtils::toAltiumOwnerPartId(pin.partIndex);
202 altiumPin.sourcePartIndex = pin.partIndex;
203 applyPowerPinType(pin, altiumPin);
204 return altiumPin;
205}
