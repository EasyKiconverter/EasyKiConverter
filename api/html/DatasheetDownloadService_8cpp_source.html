35QByteArray DatasheetDownloadService::download(const QString& componentId,
36 const QString& datasheetUrl,
37 QString* format,
38 ComponentExportStatus::NetworkDiagnostics* diag,
39 QAtomicInt* cancelled,
40 bool weakNetwork,
41 uint64_t expectedGeneration) {
42 if (datasheetUrl.isEmpty()) {
43 return QByteArray();
44 }
45 // 固定本次请求允许写入的缓存代次，避免目录切换后旧请求污染新目录。
46 const uint64_t generation = expectedGeneration != 0 ? expectedGeneration : m_owner.currentGeneration();
47
48 // 记录从缓存读取或网络请求开始到返回结果的耗时。
49 QElapsedTimer timer;
50 timer.start();
51
52 // 根据 URL 初步判断格式，收到内容后仍会校验并允许降级为 HTML。
53 QString extension =
54 datasheetUrl.toLower().contains(QStringLiteral(".html")) ? QStringLiteral("html") : QStringLiteral("pdf");
55 if (format) {
56 *format = extension;
57 }
58
59 // 缓存目录路径、文件检查和读取必须与目录迁移串行化。
60 {
61 // 串行化缓存读取与目录迁移，确保路径和文件操作处于同一锁保护下。
62 QMutexLocker diskLocker(&m_owner.m_diskWriteMutex);
63 // 根据元器件编号和候选格式解析数据手册缓存文件路径。
64 const QString fullPath = m_owner.resolveDatasheetPath(componentId, extension, false);
65 if (QFileInfo::exists(fullPath)) {
66 // 读取已存在的缓存文件，后续仍需经过内容校验。
67 QFile file(fullPath);
68 if (file.open(QIODevice::ReadOnly)) {
69 // 保存缓存文件的原始内容，供格式校验和调用方返回。
70 const QByteArray cachedData = file.readAll();
71 file.close();
72 // 根据缓存文件后缀确定实际数据手册格式。
73 const QString cachedFormat = fullPath.endsWith(QStringLiteral(".pdf"), Qt::CaseInsensitive)
74 ? QStringLiteral("pdf")
75 : QStringLiteral("html");
76 if (CacheDataValidator::isValidDatasheet(cachedData, cachedFormat)) {
77 LOG_DEBUG(LogModule::Core, "Datasheet loaded from disk cache: {}", fullPath);
78 if (diag) {
79 diag->url = datasheetUrl;
80 diag->statusCode = 200;
81 diag->errorString.clear();
82 diag->responseContentType.clear();
83 diag->retryAfter.clear();
84 diag->rateLimitRemaining.clear();
85 diag->rateLimitReset.clear();
86 diag->responseSummary.clear();
87 diag->retryCount = 0;
88 diag->latencyMs = timer.elapsed();
89 diag->wasRateLimited = false;
90 diag->hasRateLimitHint = false;
91 }
92 if (format) {
93 *format = cachedFormat;
94 }
95 return cachedData;
96 }
97 // 记录无效缓存移入回收站时的诊断信息，不影响后续网络下载。
98 QString trashError;
99 if (CacheSafety::isOwnedComponentFile(m_owner.cacheDir(), fullPath))
100 CacheSafety::moveToTrash(fullPath, &trashError);
101 }
102 }
103 }
104
105 // 在发起网络请求前检查取消标志，避免不必要的外部请求。
106 if (cancelled && cancelled->loadRelaxed()) {
107 LOG_DEBUG(LogModule::Core, "Datasheet download cancelled for {} before start", componentId);
108 return QByteArray();
109 }
110
111 // 根据网络配置生成数据手册请求的重试策略。
112 const RetryPolicy policy = RetryPolicy::fromProfile(RequestProfiles::datasheet(), weakNetwork);
113 // 执行数据手册网络请求并收集响应、取消和诊断结果。
114 const NetworkResult result = NetworkClient::instance().get(QUrl(datasheetUrl), ResourceType::Datasheet, policy);
115
116 // 保存经过校验、准备写入缓存并返回给调用方的内容。
117 QByteArray data;
118 // 保存网络层返回的 HTTP 状态码。
119 const int statusCode = result.statusCode;
120 // 保存取消、校验或网络失败的可读错误信息。
121 QString errorString;
122 // 保存网络层实际使用的重试次数。
123 const int retryCount = result.retryCount;
124 // 保存网络层是否触发限流，以便回写诊断信息。
125 const bool wasRateLimited = result.diagnostic.wasRateLimited;
126
127 // 取消优先于响应成功状态，防止取消后的响应继续写入缓存。
128 if (cancelled && cancelled->loadRelaxed()) {
129 errorString = QStringLiteral("Cancelled");
130 } else if (result.wasCancelled) {
131 errorString = QStringLiteral("Cancelled");
132 } else if (result.success) {
133 data = result.data;
134 if (format && extension == QStringLiteral("pdf") && data.size() >= 5 && !data.startsWith("%PDF-")) {
135 extension = QStringLiteral("html");
136 *format = extension;
137 }
138 if (!CacheDataValidator::isValidDatasheet(data, extension)) {
139 data.clear();
140 errorString = QStringLiteral("Invalid datasheet data");
141 }
142 } else {
143 errorString = result.error;
144 }
145
146 // 将网络层诊断信息完整传递给调用方，便于导出报告分析失败原因。
147 if (diag) {
148 diag->url = datasheetUrl;
149 diag->statusCode = statusCode;
150 diag->errorString = errorString;
151 diag->responseContentType = result.diagnostic.responseContentType;
152 diag->retryAfter = result.diagnostic.retryAfter;
153 diag->rateLimitRemaining = result.diagnostic.rateLimitRemaining;
154 diag->rateLimitReset = result.diagnostic.rateLimitReset;
155 diag->responseSummary = result.diagnostic.responseSummary;
156 diag->retryCount = retryCount;
157 diag->latencyMs = timer.elapsed();
158 diag->wasRateLimited = wasRateLimited;
159 diag->hasRateLimitHint = result.diagnostic.hasRateLimitHint;
160 }
161
162 // 只有有效数据且未被取消时才写入缓存，使用请求创建时的代次阻止旧任务污染新目录。
163 if (errorString.isEmpty() && !data.isEmpty()) {
164 m_owner.saveDatasheet(componentId, data, extension, generation);
165 } else if (!errorString.isEmpty() && errorString != QStringLiteral("Cancelled")) {
166 LOG_WARN(LogModule::Core, "Datasheet download failed for {}: {}", componentId, errorString);
167 }
168
169 return data;
170}
