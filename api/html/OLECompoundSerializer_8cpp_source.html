17void OLECompoundSerializer::serializeDirectoryEntry(const OLECompoundWriter::DirectoryEntry& entry,
18 QByteArray& buffer) const {
19 buffer.resize(OLECompoundWriter::DIR_ENTRY_SIZE);
20 buffer.fill(0);
21 int offset = 0; // 当前写入位置。
22
23 // 名称（64 字节，UTF-16LE）。
24 for (int i = 0; i < 32; ++i) {
25 buffer[offset++] = static_cast<char>(entry.name[i] & 0xFF);
26 buffer[offset++] = static_cast<char>((entry.name[i] >> 8) & 0xFF);
27 }
28
29 // 名称大小（2 字节，包含结尾空字符）。
30 buffer[offset++] = static_cast<char>(entry.nameSize & 0xFF);
31 buffer[offset++] = static_cast<char>((entry.nameSize >> 8) & 0xFF);
32
33 // 对象类型和红黑树颜色标志。
34 buffer[offset++] = static_cast<char>(entry.objectType);
35 buffer[offset++] = static_cast<char>(entry.colorFlag);
36
37 // 左子节点、右子节点和子节点索引。
38 const uint32_t childLinks[] = {entry.leftChild, entry.rightChild, entry.child}; // 三个目录树链接索引。
39 for (uint32_t link : childLinks) {
40 buffer[offset++] = static_cast<char>(link & 0xFF);
41 buffer[offset++] = static_cast<char>((link >> 8) & 0xFF);
42 buffer[offset++] = static_cast<char>((link >> 16) & 0xFF);
43 buffer[offset++] = static_cast<char>((link >> 24) & 0xFF);
44 }
45
46 // CLSID（16 字节）。
47 for (uint8_t value : entry.clsid) {
48 buffer[offset++] = static_cast<char>(value);
49 }
50
51 // StateBits（4 字节）。
52 for (int i = 0; i < 4; ++i) {
53 buffer[offset++] = static_cast<char>((entry.stateBits >> (i * 8)) & 0xFF);
54 }
55
56 // 创建时间和修改时间（各 8 字节）。
57 for (int i = 0; i < 8; ++i) {
58 buffer[offset++] = static_cast<char>((entry.creationTime >> (i * 8)) & 0xFF);
59 }
60 for (int i = 0; i < 8; ++i) {
61 buffer[offset++] = static_cast<char>((entry.modifiedTime >> (i * 8)) & 0xFF);
62 }
63
64 // 起始扇区（4 字节）。
65 buffer[offset++] = static_cast<char>(entry.startSector & 0xFF);
66 buffer[offset++] = static_cast<char>((entry.startSector >> 8) & 0xFF);
67 buffer[offset++] = static_cast<char>((entry.startSector >> 16) & 0xFF);
68 buffer[offset++] = static_cast<char>((entry.startSector >> 24) & 0xFF);
69
70 // 流大小（8 字节）。
71 for (int i = 0; i < 8; ++i) {
72 buffer[offset++] = static_cast<char>((entry.streamSize >> (i * 8)) & 0xFF);
73 }
74}

80void OLECompoundSerializer::serializeFileHeader(QByteArray& header) const {
81 header.resize(OLECompoundWriter::SECTOR_SIZE);
82 header.fill(0);
83 int offset = 0; // 当前文件头写入位置。
84
85 // Compound File Binary Format 魔数（8 字节）。
86 const char signature[] = {static_cast<char>(0xD0),
87 static_cast<char>(0xCF),
88 static_cast<char>(0x11),
89 static_cast<char>(0xE0),
90 static_cast<char>(0xA1),
91 static_cast<char>(0xB1),
92 static_cast<char>(0x1A),
93 static_cast<char>(0xE1)};
94 for (char value : signature) {
95 header[offset++] = value;
96 }
97
98 // CLSID（16 字节）保留为零。
99 offset += 16;
100
101 // V3 版本、小端字节序、512 字节扇区和 64 字节 mini sector。
102 header[offset++] = 0x3E;
103 header[offset++] = 0x00;
104 header[offset++] = 0x03;
105 header[offset++] = 0x00;
106 header[offset++] = static_cast<char>(0xFE);
107 header[offset++] = static_cast<char>(0xFF);
108 header[offset++] = 0x09;
109 header[offset++] = 0x00;
110 header[offset++] = 0x06;
111 header[offset++] = 0x00;
112
113 // 保留字段和 V3 不使用的目录扇区数。
114 offset += 6;
115 offset += 4;
116
117 const uint32_t fatSectorCount =
118 static_cast<uint32_t>(m_owner.m_fat.size()) / (OLECompoundWriter::SECTOR_SIZE / 4); // FAT 扇区数量。
119 writeUInt32(header, offset, fatSectorCount);
120
121 // FAT 和 DIFAT 位于文件头之后，目录紧随其后。
122 const uint32_t difatSectorCount = static_cast<uint32_t>(m_owner.m_difatSectors.size()); // DIFAT 扇区数量。
123 writeUInt32(header, offset, fatSectorCount + difatSectorCount);
124
125 // Transaction signature 保留为零。
126 offset += 4;
127
128 // 小流阈值为 4096 字节。
129 writeUInt32(header, offset, 4096);
130
131 uint32_t firstMiniFatSector = OLECompoundWriter::ENDOFCHAIN; // 第一个 MiniFAT 扇区索引。
132 if (!m_owner.m_miniFat.isEmpty()) {
133 const uint32_t directorySectors =
134 (static_cast<uint32_t>(m_owner.m_directory.size()) * OLECompoundWriter::DIR_ENTRY_SIZE +
135 OLECompoundWriter::SECTOR_SIZE - 1) /
136 OLECompoundWriter::SECTOR_SIZE; // 目录占用的扇区数量。
137 firstMiniFatSector = fatSectorCount + difatSectorCount + directorySectors;
138 }
139 writeUInt32(header, offset, firstMiniFatSector);
140
141 const uint32_t miniFatSectorCount =
142 static_cast<uint32_t>((m_owner.m_miniFat.size() * 4 + OLECompoundWriter::SECTOR_SIZE - 1) /
143 OLECompoundWriter::SECTOR_SIZE); // MiniFAT 占用的扇区数量。
144 writeUInt32(header, offset, miniFatSectorCount);
145
146 const uint32_t firstDifatSector =
147 difatSectorCount == 0 ? OLECompoundWriter::ENDOFCHAIN : fatSectorCount; // 第一个 DIFAT 扇区索引。
148 writeUInt32(header, offset, firstDifatSector);
149 writeUInt32(header, offset, difatSectorCount);
150
151 // 头部最多直接保存 109 个 FAT 扇区索引。
152 for (uint32_t i = 0; i < 109; ++i) {
153 writeUInt32(header, offset, i < fatSectorCount ? i : OLECompoundWriter::FREESECT);
154 }
155}
