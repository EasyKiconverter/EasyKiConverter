49inline SvgArcResult parseSvgArcPath(const QString& pathStr, double scaleFactor = EASYEDA_PX_TO_MM) {
50 SvgArcResult result; // SVG 弧线解析结果。
51 if (pathStr.isEmpty())
52 return result;
53
54 // 提取所有 token（保留命令字母）
55 QStringList tokens; // 按命令和数值拆分后的 SVG token 列表。
56 QString current; // 当前正在收集的数值 token。
57 for (const QChar& ch : pathStr) {
58 if (ch.isLetter() || ch == '-') {
59 if (!current.isEmpty()) {
60 tokens.append(current.trimmed());
61 current.clear();
62 }
63 if (ch == '-') {
64 current = ch;
65 } else {
66 tokens.append(QString(ch));
67 }
68 } else if (ch == ',' || ch.isSpace()) {
69 if (!current.isEmpty()) {
70 tokens.append(current.trimmed());
71 current.clear();
72 }
73 } else {
74 current += ch;
75 }
76 }
77 if (!current.isEmpty()) {
78 tokens.append(current.trimmed());
79 }
80
81 // 找到 M 和 A 命令
82 double startX = 0, startY = 0; // M 命令的起点坐标。
83 double rx = 0, ry = 0, xRot = 0; // A 命令的椭圆半径和旋转角度。
84 int largeArc = 0, sweep = 0; // SVG 大弧标志和扫掠方向。
85 double endX = 0, endY = 0; // A 命令的终点坐标。
86 bool hasM = false, hasA = false; // 是否成功解析 M/A 命令。
87
88 for (int i = 0; i < tokens.size(); ++i) {
89 QChar cmd = tokens[i][0].toUpper(); // 当前 SVG 命令字母。
90 if (cmd == 'M' && i + 2 < tokens.size()) {
91 startX = tokens[i + 1].toDouble();
92 startY = tokens[i + 2].toDouble();
93 hasM = true;
94 i += 2;
95 } else if (cmd == 'A' && i + 7 < tokens.size()) {
96 rx = tokens[i + 1].toDouble();
97 ry = tokens[i + 2].toDouble();
98 xRot = tokens[i + 3].toDouble();
99 largeArc = tokens[i + 4].toInt();
100 sweep = tokens[i + 5].toInt();
101 endX = tokens[i + 6].toDouble();
102 endY = tokens[i + 7].toDouble();
103 hasA = true;
104 i += 7;
105 }
106 }
107
108 if (!hasM || !hasA || rx <= 0 || ry <= 0)
109 return result;
110
111 // SVG 弧线 → 圆心转换
112 double phi = xRot * M_PI / 180.0; // 弧线旋转角度，单位为弧度。
113 double dx = (startX - endX) / 2.0; // 端点中点坐标系的 X 差值。
114 double dy = (startY - endY) / 2.0; // 端点中点坐标系的 Y 差值。
115
116 double x1 = std::cos(phi) * dx + std::sin(phi) * dy; // 旋转坐标系中的起点 X。
117 double y1 = -std::sin(phi) * dx + std::cos(phi) * dy; // 旋转坐标系中的起点 Y。
118
119 double rxSq = rx * rx; // X 半径平方。
120 double rySq = ry * ry; // Y 半径平方。
121 double x1Sq = x1 * x1; // 旋转起点 X 平方。
122 double y1Sq = y1 * y1; // 旋转起点 Y 平方。
123
124 double temp = (rxSq * rySq - rxSq * y1Sq - rySq * x1Sq) / (rxSq * y1Sq + rySq * x1Sq); // 圆心计算中间量。
125 if (temp < 0)
126 temp = 0;
127 temp = std::sqrt(temp);
128
129 double factor = (largeArc == sweep) ? -1.0 : 1.0; // 圆心候选象限因子。
130 double cx1 = factor * temp * rx * y1 / ry; // 旋转坐标系中的圆心 X。
131 double cy1 = -factor * temp * ry * x1 / rx; // 旋转坐标系中的圆心 Y。
132
133 double cx = std::cos(phi) * cx1 - std::sin(phi) * cy1 + (startX + endX) / 2.0; // 弧线圆心 X。
134 double cy = std::sin(phi) * cx1 + std::cos(phi) * cy1 + (startY + endY) / 2.0; // 弧线圆心 Y。
135
136 // 计算起始角度和终止角度
137 auto getAngle = [](double ux, double uy, double vx, double vy) -> double {
138 double dot = ux * vx + uy * vy; // 两个向量的点积。
139 double mag = std::sqrt(ux * ux + uy * uy) * std::sqrt(vx * vx + vy * vy); // 向量长度乘积。
140 if (mag < 1e-12)
141 return 0.0;
142 double cosAngle = std::clamp(dot / mag, -1.0, 1.0); // 限幅后的夹角余弦。
143 double angle = std::acos(cosAngle); // 向量夹角，单位为弧度。
144 if (ux * vy - uy * vx < 0)
145 angle = -angle;
146 return angle;
147 };
148
149 double startAngle = getAngle(1.0, 0.0, (x1 - cx1) / rx, (y1 - cy1) / ry); // 弧线起始角度。
150 double deltaAngle = getAngle((x1 - cx1) / rx, (y1 - cy1) / ry, (-x1 - cx1) / rx, (-y1 - cy1) / ry); // 扫掠角度。
151
152 // 规范化
153 while (startAngle < 0)
154 startAngle += 2 * M_PI;
155 while (startAngle >= 2 * M_PI)
156 startAngle -= 2 * M_PI;
157
158 if (sweep) {
159 if (deltaAngle < 0)
160 deltaAngle += 2 * M_PI;
161 } else {
162 if (deltaAngle > 0)
163 deltaAngle -= 2 * M_PI;
164 }
165
166 double startDeg = startAngle * 180.0 / M_PI; // 起始角度，单位为度。
167 double endDeg = (startAngle + deltaAngle) * 180.0 / M_PI; // 终止角度，单位为度。
168
169 result.center = QPointF(cx * scaleFactor, cy * scaleFactor);
170 result.radius = rx * scaleFactor; // 简化：使用 rx 作为半径
171 result.startAngle = startDeg;
172 result.endAngle = endDeg;
173
174 return result;
175}

182inline FootprintComponentIR toFootprintIR(const FootprintData& data) {
183 FootprintComponentIR ir; // 待填充的封装 IR。
184
185 const auto layerExpansion = [&data](int layerId) {
186 for (const LayerDefinition& layer : data.layers()) {
187 if (layer.layerId == layerId)
188 return layer.expansion;
189 }
190 return 0.0;
191 };
192
193 // 通用元数据
194 ir.name = data.info().name;
195 ir.description = data.info().description;
196
197 // 转换焊盘
198 for (const auto& pad : data.pads()) {
199 FootprintPadIR pir; // 当前焊盘的封装 IR。
200 pir.number = pad.number;
201 pir.position = QPointF(pad.centerX * EASYEDA_PX_TO_MM, pad.centerY * EASYEDA_PX_TO_MM);
202 pir.shape = EasyedaPadShapeMap::toPadShape(pad.shape);
203 pir.size = QSizeF(pad.width * EASYEDA_PX_TO_MM, pad.height * EASYEDA_PX_TO_MM);
204 pir.layer = EasyedaLayerMap::toLayerType(pad.layerId);
205 pir.rotation = pad.rotation;
206 pir.holeSize = pad.holeRadius * 2.0 * EASYEDA_PX_TO_MM; // radius -> diameter
207 pir.holeLength = pad.holeLength * EASYEDA_PX_TO_MM;
208 pir.netName = pad.net;
209 pir.isPlated = pad.isPlated;
210 pir.isLocked = pad.isLocked;
211 pir.padType = (pad.holeRadius > 0) ? PadType::ThroughHole : PadType::Smd;
212 if (pir.isSmd()) {
213 const bool bottom = pir.layer == LayerType::BottomCopper; // 是否位于底层铜层。
214 pir.solderMaskExpansionMm = layerExpansion(bottom ? 8 : 7);
215 // EasyEDA layer records expose paste expansion separately when present;
216 // absent values intentionally remain zero instead of being inferred.
217 pir.pasteMaskContractionMm = layerExpansion(bottom ? 6 : 5);
218 }
219
220 // 异形焊盘自定义形状
221 if (pir.shape == PadShape::Polygon && !pad.points.isEmpty()) {
222 pir.customShapePoints = parseFlatPointString(pad.points);
223 // 注意：自定义形状点相对于焊盘中心，但不翻转 Y（与旧代码一致）
224 for (auto& pt : pir.customShapePoints) {
225 pt.setX(pt.x() - pad.centerX * EASYEDA_PX_TO_MM);
226 pt.setY(pt.y() - pad.centerY * EASYEDA_PX_TO_MM);
227 }
228 }
229
230 ir.pads.append(pir);
231 }
232
233 // 转换走线
234 for (const auto& track : data.tracks()) {
235 FootprintTrackIR tir; // 当前走线的封装 IR。
236 tir.points = parseFlatPointString(track.points);
237 tir.width = track.strokeWidth * EASYEDA_PX_TO_MM;
238 tir.layer = EasyedaLayerMap::toLayerType(track.layerId);
239 tir.netName = track.net;
240 tir.isLocked = track.isLocked;
241 ir.tracks.append(tir);
242 }
243
244 // 转换安装孔
245 for (const auto& hole : data.holes()) {
246 FootprintHoleIR hir; // 当前安装孔的封装 IR。
247 hir.center = QPointF(hole.centerX * EASYEDA_PX_TO_MM, hole.centerY * EASYEDA_PX_TO_MM);
248 hir.radius = hole.radius * EASYEDA_PX_TO_MM;
249 hir.isLocked = hole.isLocked;
250 ir.holes.append(hir);
251 }
252
253 // 转换圆
254 for (const auto& circle : data.circles()) {
255 FootprintCircleIR cir; // 当前圆形图元的封装 IR。
256 cir.center = QPointF(circle.cx * EASYEDA_PX_TO_MM, circle.cy * EASYEDA_PX_TO_MM);
257 cir.radius = circle.radius * EASYEDA_PX_TO_MM;
258 cir.strokeWidth = circle.strokeWidth * EASYEDA_PX_TO_MM;
259 cir.layer = EasyedaLayerMap::toLayerType(circle.layerId);
260 cir.isLocked = circle.isLocked;
261 ir.circles.append(cir);
262 }
263
264 // 转换矩形
265 for (const auto& rect : data.rectangles()) {
266 FootprintRectangleIR rir; // 当前矩形图元的封装 IR。
267 rir.bounds = QRectF(rect.x * EASYEDA_PX_TO_MM,
268 rect.y * EASYEDA_PX_TO_MM,
269 rect.width * EASYEDA_PX_TO_MM,
270 rect.height * EASYEDA_PX_TO_MM);
271 rir.strokeWidth = rect.strokeWidth * EASYEDA_PX_TO_MM;
272 rir.layer = EasyedaLayerMap::toLayerType(rect.layerId);
273 rir.isLocked = rect.isLocked;
274 ir.rectangles.append(rir);
275 }
276
277 // 转换圆弧
278 for (const auto& arc : data.arcs()) {
279 FootprintArcIR air; // 当前圆弧图元的封装 IR。
280 // 从 SVG 弧线路径解析圆心、半径和角度
281 SvgArcResult arcResult = parseSvgArcPath(arc.path); // 从 SVG 路径得到的弧线几何。
282 if (arcResult.radius > 0) {
283 air.center = arcResult.center;
284 air.radius = arcResult.radius;
285 air.startAngle = arcResult.startAngle;
286 air.endAngle = arcResult.endAngle;
287 } else {
288 // 使用 parseSimpleSvgPath 提取坐标点
289 const QList<QPointF> arcPoints = parseSimpleSvgPath(arc.path); // 简化路径解析出的备用点集。
290 if (arcPoints.size() >= 3) {
291 air.center = arcPoints[2];
292 const double dx = arcPoints[0].x() - air.center.x(); // 起点到圆心的 X 差值。
293 const double dy = arcPoints[0].y() - air.center.y(); // 起点到圆心的 Y 差值。
294 air.radius = qSqrt(dx * dx + dy * dy);
295 }
296 }
297 air.width = arc.strokeWidth * EASYEDA_PX_TO_MM;
298 air.layer = EasyedaLayerMap::toLayerType(arc.layerId);
299 air.netName = arc.net;
300 air.isLocked = arc.isLocked;
301 ir.arcs.append(air);
302 }
303
304 // 转换文本
305 for (const auto& text : data.texts()) {
306 FootprintTextIR tir; // 当前文本的封装 IR。
307 tir.text = text.text;
308 tir.position = QPointF(text.centerX * EASYEDA_PX_TO_MM, text.centerY * EASYEDA_PX_TO_MM);
309 tir.rotation = text.rotation;
310 tir.mirror = (text.mirror == "1" || text.mirror.toLower() == "true");
311 tir.strokeWidth = text.strokeWidth * EASYEDA_PX_TO_MM;
312 tir.fontSize = text.fontSize * EASYEDA_PX_TO_MM;
313 tir.layer = EasyedaLayerMap::toLayerType(text.layerId);
314 tir.isDisplayed = text.isDisplayed;
315 tir.isFabrication = (text.type == "N");
316 tir.isLocked = text.isLocked;
317
318 // 非 ASCII 文本路径
319 if (!text.textPath.isEmpty()) {
320 tir.textPathPoints = parseSimpleSvgPath(text.textPath);
321 }
322
323 ir.texts.append(tir);
324 }
325
326 // 转换实心区域
327 for (const auto& region : data.solidRegions()) {
328 FootprintRegionIR rir; // 当前实心区域的封装 IR。
329 rir.vertices = parseSimpleSvgPath(region.path);
330 rir.layer = EasyedaLayerMap::toLayerType(region.layerId);
331 rir.isKeepOut = region.isKeepOut || (region.layerId == 99);
332 rir.isLocked = region.isLocked;
333 ir.regions.append(rir);
334 }
335
336 // 转换板框轮廓
337 for (const auto& outline : data.outlines()) {
338 // EasyEDA 的 layer 19 是 SVGNODE outline3D 的模型辅助轮廓，
339 // 不是封装的二维 PCB 图元。若将其转成 Track，Altium 中会在
340 // 未映射层上显示成黑色矩形/大面积黑色轮廓；3D 模型边界由
341 // ExporterAltiumFootprint::generateComponentBody() 单独处理。
342 if (outline.layerId == 19)
343 continue;
344 FootprintOutlineIR oir; // 当前板框轮廓的封装 IR。
345 oir.points = parseSimpleSvgPath(outline.path);
346 oir.strokeWidth = outline.strokeWidth * EASYEDA_PX_TO_MM;
347 oir.layer = EasyedaLayerMap::toLayerType(outline.layerId);
348 oir.isLocked = outline.isLocked;
349 ir.outlines.append(oir);
350 }
351
352 // 仅转换带有模型标识的有效三维数据，避免空模型污染目标格式的诊断和统计。
353 const Model3DData model3D = data.model3D();
354 if (model3D.isValid())
355 ir.models3d.append(toModel3DIR(model3D));
356
357 // 设置 courtyard 生成标志：当原始 bbox 有有效尺寸时生成
358 const auto& bbox = data.bbox();
359 ir.shouldGenerateCourtyard = (bbox.width > 0 && bbox.height > 0);
360
361 return ir;
362}
