30XpeditionHkpType parseType(const QString& value) {
31 const QString type = unquote(value).toUpper(); // 规范化后的文件类型名称。
32 if (type == QStringLiteral("PADSTACK_LIBRARY"))

47XpeditionHkpDocument XpeditionHkpReader::parse(const QString& content, const QString& filePath) {
48 XpeditionHkpDocument document; // 当前 HKP 文档及其诊断。
49 document.diagnostics.setFilePath(filePath);
50 if (content.trimmed().isEmpty()) {
51 document.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("HKP 文件为空"));
52 return document;
53 }
54
55 document.sections = IndentedSectionParser::parse(content, &document.diagnostics);
56 for (const SectionNode& section : document.sections) { // 读取显式文件头字段。
57 if (section.keyword.toUpper() == QStringLiteral("FILETYPE"))
58 document.type = parseType(section.value);
59 else if (section.keyword.toUpper() == QStringLiteral("UNITS"))
60 document.unit = UnitConverter::parseUnit(unquote(section.value), &document.diagnostics);
61 }
62 if (document.type == XpeditionHkpType::Unknown) {
63 for (const SectionNode& section : document.sections) {
64 const QString keyword = section.keyword.toUpper(); // 当前分段关键字。
65 if (keyword == QStringLiteral("PAD") || keyword == QStringLiteral("PADSTACK") ||
66 keyword == QStringLiteral("HOLE")) {
67 document.type = XpeditionHkpType::PadstackLibrary;
68 break;
69 }
70 if (keyword == QStringLiteral("PACKAGE_CELL")) {
71 document.type = XpeditionHkpType::CellLibrary;
72 break;
73 }
74 if (keyword == QStringLiteral("NUMBER")) {
75 document.type = XpeditionHkpType::PartsDatabase;
76 break;
77 }
78 }
79 }
80 if (!document.isRecognized())
81 document.diagnostics.add(
82 ParseSeverity::Error, ParseScope::File, QStringLiteral("未识别的 Xpedition HKP 文件类型"));
83 if (document.unit == LengthUnit::Unknown)
84 document.diagnostics.add(
85 ParseSeverity::Warning, ParseScope::File, QStringLiteral("HKP 文件未声明可识别的单位"));
86 document.model = XpeditionHkpModelParser::parse(document.sections, document.unit, &document.diagnostics);
87 return document;
88}

113QString mergedName(const QString& name, const QSet<QString>& usedNames) {
114 if (!usedNames.contains(name))
115 return name;
116 int suffix = 2; // 名称冲突时使用的起始后缀。
117 QString candidate; // 当前候选名称。
118 do {
119 candidate = QStringLiteral("%1_%2").arg(name).arg(suffix++);
120 } while (usedNames.contains(candidate));
121 return candidate;
122}

148template <typename Definition>
149void appendDefinition(const Definition& source,
150 QList<Definition>& target,
151 QMap<QString, QStringList>& variants,
152 QSet<QString>& usedNames,
153 const QMap<QString, QStringList>& sourceVariants,
154 ParseDiagnostics* diagnostics,
155 ParseScope scope) {
156 Definition merged = source; // 当前定义的合并副本。
157 const QString rawName = originalName(sourceVariants, source.name); // 源文件中的原始名称。
158 merged.name = mergedName(source.name, usedNames);
159 if (merged.name != source.name && diagnostics)
160 diagnostics->add(ParseSeverity::Warning,
161 scope,
162 QStringLiteral("跨文件定义名称重复，已重命名：%1 -> %2").arg(source.name, merged.name),
163 rawName,
164 source.line);
165 usedNames.insert(merged.name);
166 target.append(merged);
167 variants[rawName].append(merged.name);
168}

198XpeditionHkpDocument XpeditionHkpMerger::merge(const QList<XpeditionHkpDocument>& documents, const QString& filePath) {
199 XpeditionHkpDocument merged; // 合并后的 HKP 文档。
200 merged.diagnostics.setFilePath(filePath);
201 if (documents.isEmpty()) {
202 merged.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("没有可合并的 HKP 文档"));
203 return merged;
204 }
205
206 QSet<QString> padNames; // 已使用的 Pad 名称。
207 QSet<QString> holeNames; // 已使用的孔名称。
208 QSet<QString> padstackNames; // 已使用的 Padstack 名称。
209 QSet<QString> cellNames; // 已使用的 Cell 名称。
210 for (const XpeditionHkpDocument& document : documents) { // 逐个合并输入文档。
211 merged.diagnostics.append(document.diagnostics);
212 if (!document.isRecognized())
213 continue;
214 if (merged.type == XpeditionHkpType::Unknown)
215 merged.type = document.type;
216 else if (merged.type != document.type)
217 merged.diagnostics.add(
218 ParseSeverity::Error, ParseScope::File, QStringLiteral("HKP 文档类型不一致，无法安全合并"), filePath);
219 if (merged.unit == LengthUnit::Unknown)
220 merged.unit = document.unit;
221 else if (document.unit != LengthUnit::Unknown && merged.unit != document.unit)
222 merged.diagnostics.add(
223 ParseSeverity::Error, ParseScope::File, QStringLiteral("HKP 文档单位不一致，无法安全合并"), filePath);
224 merged.sections.append(document.sections);
225 for (const XpeditionPadDefinition& item : document.model.pads)
226 appendDefinition(item,
227 merged.model.pads,
228 merged.model.padNameVariants,
229 padNames,
230 document.model.padNameVariants,
231 &merged.diagnostics,
232 ParseScope::Footprint);
233 for (const XpeditionHoleDefinition& item : document.model.holes)
234 appendDefinition(item,
235 merged.model.holes,
236 merged.model.holeNameVariants,
237 holeNames,
238 document.model.holeNameVariants,
239 &merged.diagnostics,
240 ParseScope::Footprint);
241 for (const XpeditionPadstackDefinition& item : document.model.padstacks)
242 appendDefinition(item,
243 merged.model.padstacks,
244 merged.model.padstackNameVariants,
245 padstackNames,
246 document.model.padstackNameVariants,
247 &merged.diagnostics,
248 ParseScope::Footprint);
249 for (const XpeditionCellDefinition& item : document.model.cells)
250 appendDefinition(item,
251 merged.model.cells,
252 merged.model.cellNameVariants,
253 cellNames,
254 document.model.cellNameVariants,
255 &merged.diagnostics,
256 ParseScope::Footprint);
257 for (const XpeditionPartDefinition& item : document.model.parts) {
258 for (const XpeditionPartDefinition& existing : merged.model.parts) {
259 if (existing.number == item.number)
260 merged.diagnostics.add(ParseSeverity::Error,
261 ParseScope::Component,
262 QStringLiteral("跨文件器件编号重复：%1").arg(item.number),
263 item.number,
264 item.line);
265 }
266 merged.model.parts.append(item);
267 }
268 }
269 if (merged.type == XpeditionHkpType::Unknown)
270 merged.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("没有可识别的 HKP 文档"));
271 reportAmbiguousVariants(
272 merged.model.padNameVariants, &merged.diagnostics, ParseScope::Footprint, QStringLiteral("Pad"));
273 reportAmbiguousVariants(
274 merged.model.holeNameVariants, &merged.diagnostics, ParseScope::Footprint, QStringLiteral("孔"));
275 reportAmbiguousVariants(
276 merged.model.padstackNameVariants, &merged.diagnostics, ParseScope::Footprint, QStringLiteral("Padstack"));
277 reportAmbiguousVariants(
278 merged.model.cellNameVariants, &merged.diagnostics, ParseScope::Footprint, QStringLiteral("Cell"));
279 merged.model.parts.squeeze();
280 return merged;
281}
