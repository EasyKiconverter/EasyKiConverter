30void ExportReportGenerator::writeDetailedReport(const QString& reason,
31 const ExportOptions& options,
32 const ExportOverallProgress& progress) {
33 // 只在调试模式下生成详细报告
34 if (!options.debugMode) {
35 return;
36 }
37
38 if (options.outputPath.trimmed().isEmpty()) {
39 return;
40 }
41
42 QDir outputDir(options.outputPath);
43 if (!outputDir.exists() && !outputDir.mkpath(QStringLiteral("."))) {
44 qWarning() << "ParallelExportService: Failed to create diagnostics report directory:" << outputDir.path();
45 return;
46 }
47
48 const QString reportPath =
49 outputDir.filePath(QStringLiteral("easykiconverter_export_detailed_report.md"));
50 QSaveFile file(reportPath);
51 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
52 qWarning() << "ParallelExportService: Failed to open export detailed report for writing:" << reportPath;
53 return;
54 }
55
56 QTextStream out(&file);
57 out.setEncoding(QStringConverter::Utf8);
58 out << "# EasyKiConverter Export Detailed Report\n\n";
59 out << "- Generated: " << QDateTime::currentDateTime().toString(Qt::ISODate) << "\n";
60 out << "- Reason: " << reason << "\n";
61 out << "- Weak network support: " << (options.weakNetworkSupport ? "enabled" : "disabled") << "\n";
62 out << "- Output path: " << options.outputPath << "\n";
63 out << "- Current stage: " << static_cast<int>(progress.currentStage) << "\n";
64 out << "- Start time: " << progress.startTime.toString(Qt::ISODate) << "\n";
65 out << "- End time: " << progress.endTime.toString(Qt::ISODate) << "\n";
66 out << "- Total components: " << progress.totalComponents << "\n\n";
67
68 out << "## Export Options\n\n";
69 out << "- Library name: " << options.libName << "\n";
70 out << "- Export symbol: " << (options.exportSymbol ? "yes" : "no") << "\n";
71 out << "- Export footprint: " << (options.exportFootprint ? "yes" : "no") << "\n";
72 out << "- Export 3D model: " << (options.exportModel3D ? "yes" : "no") << "\n";
73 out << "- Export preview images: " << (options.exportPreviewImages ? "yes" : "no") << "\n";
74 out << "- Export datasheet: " << (options.exportDatasheet ? "yes" : "no") << "\n";
75 out << "- Overwrite existing files: " << (options.overwriteExistingFiles ? "yes" : "no") << "\n";
76 out << "- Update mode: " << (options.updateMode ? "yes" : "no") << "\n";
77 out << "- Debug mode: " << (options.debugMode ? "yes" : "no") << "\n\n";
78
79 out << "## Preload\n\n";
80 out << "- Completed: " << progress.preloadProgress.completedCount << "/" << progress.preloadProgress.totalCount
81 << "\n";
82 out << "- Success: " << progress.preloadProgress.successCount << "\n";
83 out << "- Failed: " << progress.preloadProgress.failedCount << "\n";
84 out << "- In progress: " << progress.preloadProgress.inProgressCount << "\n";
85 if (!progress.preloadProgress.failedComponents.isEmpty()) {
86 out << "\n### Preload Failures\n\n";
87 for (auto it = progress.preloadProgress.failedComponents.cbegin();
88 it != progress.preloadProgress.failedComponents.cend();
89 ++it) {
90 out << "- `" << it.key() << "`: " << it.value() << "\n";
91 }
92 }
93
94 out << "\n## Export Type Progress\n\n";
95 for (auto it = progress.exportTypeProgress.cbegin(); it != progress.exportTypeProgress.cend(); ++it) {
96 const ExportTypeProgress& typeProgress = it.value();
97 out << "### " << it.key() << "\n\n";
98 out << "- Completed: " << typeProgress.completedCount << "/" << typeProgress.totalCount << "\n";
99 out << "- Success: " << typeProgress.successCount << "\n";
100 out << "- Failed: " << typeProgress.failedCount << "\n";
101 out << "- Skipped: " << typeProgress.skippedCount << "\n";
102 out << "- In progress: " << typeProgress.inProgressCount << "\n";
103 if (!typeProgress.diagnostics.isEmpty()) {
104 out << "\n#### Exporter Diagnostics\n\n";
105 for (const QString& diagnostic : typeProgress.diagnostics)
106 out << "- " << diagnostic << "\n";
107 }
108 bool hasNonSuccessItems = false;
109 for (auto statusIt = typeProgress.itemStatus.cbegin(); statusIt != typeProgress.itemStatus.cend();
110 ++statusIt) {
111 const auto status = statusIt.value().status;
112 if (status == ExportItemStatus::Status::Failed || status == ExportItemStatus::Status::Skipped) {
113 hasNonSuccessItems = true;
114 break;
115 }
116 }
117 if (hasNonSuccessItems) {
118 out << "\n#### Failed or Skipped Items\n\n";
119 for (auto statusIt = typeProgress.itemStatus.cbegin(); statusIt != typeProgress.itemStatus.cend();
120 ++statusIt) {
121 const ExportItemStatus& status = statusIt.value();
122 QString statusName;
123 if (status.status == ExportItemStatus::Status::Failed)
124 statusName = QStringLiteral("failed");
125 else if (status.status == ExportItemStatus::Status::Skipped)
126 statusName = QStringLiteral("skipped");
127 else
128 continue;
129
130 // 文本。
131 const QString reasonText = status.errorMessage.isEmpty()
132 ? QStringLiteral("未提供原因")
133 : status.errorMessage;
134 out << "- `" << statusIt.key() << "`: " << statusName << " — " << reasonText << "\n";
135 }
136 }
137 bool hasDiagnostics = false;
138 for (auto statusIt = typeProgress.itemStatus.cbegin(); statusIt != typeProgress.itemStatus.cend(); ++statusIt) {
139 if (!statusIt.value().diagnostics.isEmpty()) {
140 hasDiagnostics = true;
141 break;
142 }
143 }
144 if (hasDiagnostics) {
145 out << "\n#### Input Diagnostics\n\n";
146 for (auto statusIt = typeProgress.itemStatus.cbegin(); statusIt != typeProgress.itemStatus.cend();
147 ++statusIt) {
148 if (statusIt.value().diagnostics.isEmpty())
149 continue;
150 out << "- `" << statusIt.key() << "`:\n";
151 for (const QString& diagnostic : statusIt.value().diagnostics)
152 out << " - " << diagnostic << "\n";
153 }
154 }
155 }
156
157 out << "\n## Weak Network Diagnostics\n\n";
158 out << "### Network Runtime Stats\n\n```\n";
159 out << NetworkClient::instance().formatRuntimeStats() << "\n";
160 out << "```\n";
161
162 if (!file.commit()) {
163 qWarning() << "ParallelExportService: Failed to commit export detailed report:" << reportPath;
164 return;
165 }
166
167 qInfo() << "ParallelExportService: Export detailed report written to" << reportPath;
168}
