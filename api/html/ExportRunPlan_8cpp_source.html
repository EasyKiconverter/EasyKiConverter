47ExportRunPlan buildExportRunPlan(const ExportOptions& options,
48 const QStringList& componentIds,
49 const QMap<QString, QSharedPointer<ComponentData>>& cachedData) {
50 ExportRunPlan plan;
51 const bool combinedTarget =
52 options.targetFormat == TargetEdaFormat::Eagle || options.targetFormat == TargetEdaFormat::Cadstar ||
53 options.targetFormat == TargetEdaFormat::Allegro || options.targetFormat == TargetEdaFormat::LibrePcb ||
54 options.targetFormat == TargetEdaFormat::Horizon;
55 const bool embeddedModelTarget =
56 options.targetFormat == TargetEdaFormat::Altium || options.targetFormat == TargetEdaFormat::Allegro ||
57 options.targetFormat == TargetEdaFormat::LibrePcb ||
58 options.targetFormat == TargetEdaFormat::Horizon;
59 // 仅符号模式仍由组合库写入阶段负责，但进度需要展示为 Symbol。
60 plan.symbolOnlyCombinedLibrary = combinedTarget && options.exportSymbol && !options.exportFootprint;
61 // Eagle/CADSTAR 的组合库由封装阶段统一写入，符号单独导出也复用同一文件事务。
62 plan.enableSymbol =
63 options.exportSymbol && options.targetFormat != TargetEdaFormat::Allegro &&
64 options.targetFormat != TargetEdaFormat::Eagle && options.targetFormat != TargetEdaFormat::Cadstar &&
65 options.targetFormat != TargetEdaFormat::LibrePcb && options.targetFormat != TargetEdaFormat::Horizon;
66 // OrCAD Capture XML 只承载符号和封装名称属性，PCB 封装几何由其他目标库负责。
67 // 独立三维导出不应顺带生成封装库；只有目标格式需要把模型关联写入库时，才保留封装阶段。
68 plan.enableFootprint = (options.exportFootprint && options.targetFormat != TargetEdaFormat::Orcad) ||
69 (combinedTarget && options.exportSymbol) || (embeddedModelTarget && options.exportModel3D);
70 // 目标格式没有经过本项目验证的原生模型关联时，仍输出独立模型文件并保留诊断。
71 plan.enableModel3D = options.exportModel3D;
72 plan.runExternalModel3DStage = plan.enableModel3D && options.targetFormat != TargetEdaFormat::Altium &&
73 options.targetFormat != TargetEdaFormat::Allegro &&
74 options.targetFormat != TargetEdaFormat::LibrePcb &&
75 options.targetFormat != TargetEdaFormat::Horizon;
76 plan.enablePreview = options.exportPreviewImages;
77 plan.enableDatasheet = options.exportDatasheet;
78
79 // Eagle 和 CADSTAR 在同时选择符号、封装时才写入完整组合库；仅封装导出不应无条件要求符号缓存。
80 const bool needsSymbolData =
81 plan.enableSymbol || (combinedTarget && options.exportSymbol);
82 // 组合目标的仅符号路径直接调用符号 writer，不应因为没有封装缓存而被预加载阶段拦截。
83 const bool needsFootprintData = (options.exportFootprint && options.targetFormat != TargetEdaFormat::Orcad) ||
84 (embeddedModelTarget && options.exportModel3D);
85 for (const QString& componentId : componentIds) {
86 const auto it = cachedData.constFind(componentId);
87 const auto& component =
88 it == cachedData.cend() ? QSharedPointer<ComponentData>() : it.value();
89 if (component && component->isValid() && (!needsSymbolData || component->symbolData()) &&
90 (!needsFootprintData || component->footprintData())) {
91 plan.exportableComponentIds.append(componentId);
92 } else {
93 plan.missingDataComponentIds.append(componentId);
94 }
95 }
96 return plan;
97}
