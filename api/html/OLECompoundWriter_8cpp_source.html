66int OLECompoundWriter::createDirectoryEntry(const QString& name, ObjectType type) {
67 DirectoryEntry entry; // 待写入的目录条目。
68 entry.objectType = type;
69
70 // 写入 UTF-16LE 名称
71 QString utf16Name = name; // 待编码为 UTF-16LE 的名称。
72 int charCount = qMin(utf16Name.length(), 31); // OLE 目录名称允许的字符数。
73 for (int i = 0; i < charCount; ++i) {
74 entry.name[i] = static_cast<uint16_t>(utf16Name.at(i).unicode());
75 }
76 entry.nameSize = static_cast<uint16_t>((charCount + 1) * 2); // 含 null 终止符
77
78 int index = m_directory.size(); // 新条目在目录数组中的索引。
79 m_directory.append(entry);
80
81 return index;
82}

90bool OLECompoundWriter::addStorage(const QString& parentPath, const QString& name) {
91 if (!m_initialized || m_finalized || !isValidEntryName(name) || !m_pathToNode.contains(parentPath)) {
92 setError(QStringLiteral("无法创建存储: %1/%2").arg(parentPath, name));
93 return false;
94 }
95
96 QString fullPath = parentPath.isEmpty() ? name : parentPath + "/" + name; // 新存储的完整路径。
97
98 const QString foldedPath = fullPath.toCaseFolded(); // 用于大小写无关冲突检测的路径。
99 if (m_entryPaths.contains(foldedPath)) {
100 setError(QStringLiteral("存储路径重复: %1").arg(fullPath));
101 return false;
102 }
103
104 int dirIndex = createDirectoryEntry(name, ObjectType::Storage); // 新存储对应的目录索引。
105
106 // 创建节点
107 StorageNode node; // 新建的存储节点。
108 node.name = name;
109 node.fullPath = fullPath;
110 node.dirIndex = dirIndex;
111 int nodeIndex = m_nodes.size(); // 新节点在节点数组中的索引。
112 m_nodes.append(node);
113 m_pathToNode[fullPath] = nodeIndex;
114 m_entryPaths.insert(foldedPath);
115
116 // 添加到父节点的子列表
117 if (parentPath.isEmpty()) {
118 // 根存储的子节点
119 if (m_pathToNode.contains("")) {
120 m_nodes[m_pathToNode[""]].children.append(nodeIndex);
121 }
122 } else {
123 if (m_pathToNode.contains(parentPath)) {
124 m_nodes[m_pathToNode[parentPath]].children.append(nodeIndex);
125 }
126 }
127
128 return true;
129}

214uint32_t OLECompoundWriter::buildChildTree(const QVector<int>& children) {
215 if (children.isEmpty()) {
216 return NOSTREAM;
217 }
218
219 // CFB 的 sibling tree 不只是二叉搜索树，还必须满足红黑树不变量。
220 // children 已按 CFB 名称顺序排列；逐项插入并进行标准红黑树修复，
221 // 可以同时保持目录顺序和所有根到叶路径相同的黑高。
222 QVector<int> parent(m_directory.size(), -1); // 红黑树节点的父节点索引。
223 int root = -1; // 当前红黑树根节点索引。
224
225 auto left = [this](int index) {
226 const uint32_t child = m_directory[index].leftChild; // 左子节点目录索引。
227 return child == NOSTREAM ? -1 : static_cast<int>(child);
228 };
229 auto right = [this](int index) {
230 const uint32_t child = m_directory[index].rightChild; // 右子节点目录索引。
231 return child == NOSTREAM ? -1 : static_cast<int>(child);
232 };
233 auto setLeft = [this, &parent](int index, int child) {
234 m_directory[index].leftChild = child < 0 ? NOSTREAM : static_cast<uint32_t>(child);
235 if (child >= 0) {
236 parent[child] = index;
237 }
238 };
239 auto setRight = [this, &parent](int index, int child) {
240 m_directory[index].rightChild = child < 0 ? NOSTREAM : static_cast<uint32_t>(child);
241 if (child >= 0) {
242 parent[child] = index;
243 }
244 };
245
246 auto rotateLeft = [&](int pivot) {
247 const int promoted = right(pivot); // 左旋后提升的右子节点。
248 const int oldParent = parent[pivot]; // 旋转前枢轴的父节点。
249 setRight(pivot, left(promoted));
250 parent[promoted] = oldParent;
251 if (oldParent < 0) {
252 root = promoted;
253 } else if (left(oldParent) == pivot) {
254 m_directory[oldParent].leftChild = static_cast<uint32_t>(promoted);
255 } else {
256 m_directory[oldParent].rightChild = static_cast<uint32_t>(promoted);
257 }
258 setLeft(promoted, pivot);
259 };
260 auto rotateRight = [&](int pivot) {
261 const int promoted = left(pivot); // 右旋后提升的左子节点。
262 const int oldParent = parent[pivot]; // 旋转前枢轴的父节点。
263 setLeft(pivot, right(promoted));
264 parent[promoted] = oldParent;
265 if (oldParent < 0) {
266 root = promoted;
267 } else if (left(oldParent) == pivot) {
268 m_directory[oldParent].leftChild = static_cast<uint32_t>(promoted);
269 } else {
270 m_directory[oldParent].rightChild = static_cast<uint32_t>(promoted);
271 }
272 setRight(promoted, pivot);
273 };
274
275 for (int nodeIndex : children) {
276 const int inserted = m_nodes[nodeIndex].dirIndex; // 新插入节点的目录索引。
277 m_directory[inserted].leftChild = NOSTREAM;
278 m_directory[inserted].rightChild = NOSTREAM;
279 m_directory[inserted].colorFlag = 0; // 新节点为红色
280
281 // 排序后的节点总是插入当前最大值的右侧。旋转只改变树形，
282 // 不改变中序顺序，因此该性质在后续插入中仍成立。
283 int insertionParent = -1; // 新节点按名称顺序插入的父节点。
284 int cursor = root; // 当前查找位置。
285 while (cursor >= 0) {
286 insertionParent = cursor;
287 cursor = right(cursor);
288 }
289 parent[inserted] = insertionParent;
290 if (insertionParent < 0) {
291 root = inserted;
292 } else {
293 m_directory[insertionParent].rightChild = static_cast<uint32_t>(inserted);
294 }
295
296 int current = inserted; // 红黑树修复当前处理的节点。
297 while (parent[current] >= 0 && m_directory[parent[current]].colorFlag == 0) {
298 const int currentParent = parent[current]; // 当前节点的父节点。
299 const int grandParent = parent[currentParent]; // 当前节点的祖父节点。
300 if (currentParent == left(grandParent)) {
301 const int uncle = right(grandParent); // 当前节点的叔节点。
302 if (uncle >= 0 && m_directory[uncle].colorFlag == 0) {
303 m_directory[currentParent].colorFlag = 1;
304 m_directory[uncle].colorFlag = 1;
305 m_directory[grandParent].colorFlag = 0;
306 current = grandParent;
307 } else {
308 if (current == right(currentParent)) {
309 current = currentParent;
310 rotateLeft(current);
311 }
312 m_directory[parent[current]].colorFlag = 1;
313 m_directory[parent[parent[current]]].colorFlag = 0;
314 rotateRight(parent[parent[current]]);
315 }
316 } else {
317 const int uncle = left(grandParent); // 当前节点的叔节点。
318 if (uncle >= 0 && m_directory[uncle].colorFlag == 0) {
319 m_directory[currentParent].colorFlag = 1;
320 m_directory[uncle].colorFlag = 1;
321 m_directory[grandParent].colorFlag = 0;
322 current = grandParent;
323 } else {
324 if (current == left(currentParent)) {
325 current = currentParent;
326 rotateRight(current);
327 }
328 m_directory[parent[current]].colorFlag = 1;
329 m_directory[parent[parent[current]]].colorFlag = 0;
330 rotateLeft(parent[parent[current]]);
331 }
332 }
333 }
334 m_directory[root].colorFlag = 1;
335 }
336
337 return static_cast<uint32_t>(root);
338}

344void OLECompoundWriter::buildSubTree(int nodeIdx) {
345 StorageNode& node = m_nodes[nodeIdx]; // 当前待构建子树的存储节点。
346 if (node.children.isEmpty()) {
347 return;
348 }
349
350 QVector<int> sorted = node.children; // 按 CFB 名称规则排序后的子节点索引。
351 std::sort(sorted.begin(), sorted.end(), [this](int a, int b) {
352 const QString& nameA = m_nodes[a].name; // 第一个节点名称。
353 const QString& nameB = m_nodes[b].name; // 第二个节点名称。
354 if (nameA.size() != nameB.size()) {
355 return nameA.size() < nameB.size();
356 }
357 return nameA.compare(nameB, Qt::CaseInsensitive) < 0;
358 });
359
360 m_directory[node.dirIndex].child = buildChildTree(sorted);
361
362 // 递归处理子存储节点
363 for (int childIdx : node.children) {
364 if (m_nodes[childIdx].dirIndex >= 0 &&
365 m_directory[m_nodes[childIdx].dirIndex].objectType == ObjectType::Storage) {
366 buildSubTree(childIdx);
367 }
368 }
369}

397void OLECompoundWriter::finalize() {
398 // 根节点已在 create() 中创建并注册到 m_pathToNode[""]
399 // 此处无需再创建，避免 prepend 导致索引偏移
400
401 // 为所有流创建目录条目并分配数据
402 for (const StreamData& stream : m_streams) {
403 int dirIndex = createDirectoryEntry(stream.streamName, ObjectType::Stream); // 流目录索引。
404 uint64_t dataSize = static_cast<uint64_t>(stream.data.size()); // 流数据字节数。
405
406 if (dataSize == 0) {
407 // CFB 零长度流不占用普通扇区或 mini sector。
408 m_directory[dirIndex].startSector = ENDOFCHAIN;
409 m_directory[dirIndex].streamSize = 0;
410 } else if (dataSize < MINI_STREAM_CUTOFF) {
411 // 小数据写入 Mini-Stream
412 uint32_t miniSectorStart =
413 static_cast<uint32_t>(m_miniStream.size() / MINI_SECTOR_SIZE); // Mini-Stream 起始扇区。
414 m_miniStream.append(stream.data);
415 // 对齐到 mini 扇区边界
416 int remainder = stream.data.size() % MINI_SECTOR_SIZE; // 未对齐的尾部字节数。
417 if (remainder > 0) {
418 m_miniStream.append(QByteArray(MINI_SECTOR_SIZE - remainder, 0));
419 }
420 m_directory[dirIndex].startSector = miniSectorStart;
421 m_directory[dirIndex].streamSize = dataSize;
422
423 // 更新 Mini-FAT
424 uint32_t miniSectorsNeeded =
425 (dataSize + MINI_SECTOR_SIZE - 1) / MINI_SECTOR_SIZE; // 所需 Mini-FAT 扇区数。
426 for (uint32_t i = 0; i < miniSectorsNeeded; ++i) {
427 if (i < miniSectorsNeeded - 1) {
428 m_miniFat.append(miniSectorStart + i + 1);
429 } else {
430 m_miniFat.append(ENDOFCHAIN);
431 }
432 }
433 } else {
434 // 大数据写入普通扇区
435 uint32_t sectorStart = static_cast<uint32_t>(m_dataSectors.size()); // 普通数据起始扇区。
436 int remaining = stream.data.size(); // 尚未写入的字节数。
437 int srcOffset = 0; // 当前源数据偏移。
438 while (remaining > 0) {
439 QByteArray sectorData(SECTOR_SIZE, 0); // 当前普通扇区缓冲区。
440 int chunkSize = qMin(remaining, static_cast<int>(SECTOR_SIZE)); // 当前扇区写入字节数。
441 memcpy(sectorData.data(), stream.data.constData() + srcOffset, chunkSize);
442 m_dataSectors.append(sectorData);
443 srcOffset += chunkSize;
444 remaining -= chunkSize;
445 }
446 m_directory[dirIndex].startSector = sectorStart;
447 m_directory[dirIndex].streamSize = dataSize;
448
449 // 更新 FAT
450 uint32_t sectorsNeeded = (dataSize + SECTOR_SIZE - 1) / SECTOR_SIZE;
451 for (uint32_t i = 0; i < sectorsNeeded; ++i) {
452 if (i < sectorsNeeded - 1) {
453 m_fat.append(sectorStart + i + 1);
454 } else {
455 m_fat.append(ENDOFCHAIN);
456 }
457 }
458 }
459
460 // 创建流节点并添加到正确的父节点
461 StorageNode streamNode; // 当前流对应的节点。
462 streamNode.name = stream.streamName;
463 streamNode.fullPath =
464 stream.storagePath.isEmpty() ? stream.streamName : stream.storagePath + "/" + stream.streamName;
465 streamNode.dirIndex = dirIndex;
466 int streamNodeIdx = m_nodes.size(); // 流节点在节点数组中的索引。
467 m_nodes.append(streamNode);
468
469 if (stream.storagePath.isEmpty()) {
470 // 根流：添加到 Root Entry 的子列表
471 if (m_pathToNode.contains("")) {
472 m_nodes[m_pathToNode[""]].children.append(streamNodeIdx);
473 }
474 } else {
475 // 子流：添加到父 Storage 的子列表
476 if (m_pathToNode.contains(stream.storagePath)) {
477 m_nodes[m_pathToNode[stream.storagePath]].children.append(streamNodeIdx);
478 }
479 }
480 }
481
482 // 为 Root Entry 分配 Mini-Stream 的存储扇区
483 if (!m_miniStream.isEmpty()) {
484 // miniStreamSectorStart 是大数据扇区数量（数据区内的相对偏移）
485 uint32_t miniStreamSectorStart = static_cast<uint32_t>(m_dataSectors.size()); // Mini-Stream 起始扇区。
486 int remaining = m_miniStream.size(); // 尚未写入的 Mini-Stream 字节数。
487 int srcOffset = 0; // 当前 Mini-Stream 源偏移。
488 while (remaining > 0) {
489 QByteArray sectorData(SECTOR_SIZE, 0); // 当前 Mini-Stream 扇区缓冲区。
490 int chunkSize = qMin(remaining, static_cast<int>(SECTOR_SIZE)); // 当前扇区写入字节数。
491 memcpy(sectorData.data(), m_miniStream.constData() + srcOffset, chunkSize);
492 m_dataSectors.append(sectorData);
493 srcOffset += chunkSize;
494 remaining -= chunkSize;
495 }
496 // startSector 在 post-processing 中会加 dataOffset 转为绝对扇区号
497 m_directory[0].startSector = miniStreamSectorStart;
498 m_directory[0].streamSize = static_cast<uint64_t>(m_miniStream.size());
499
500 // 更新 FAT（链式指针使用数据区相对值，post-processing 会加 dataOffset）
501 uint32_t miniStreamSectors =
502 (static_cast<uint32_t>(m_miniStream.size()) + SECTOR_SIZE - 1) / SECTOR_SIZE; // Mini-Stream 扇区数。
503 for (uint32_t i = 0; i < miniStreamSectors; ++i) {
504 if (i < miniStreamSectors - 1) {
505 m_fat.append(miniStreamSectorStart + i + 1);
506 } else {
507 m_fat.append(ENDOFCHAIN);
508 }
509 }
510 }
511
512 // 构建红黑树
513 buildRedBlackTree();
514
515 // 确保目录大小是扇区对齐的
516 uint32_t dirSectors = (static_cast<uint32_t>(m_directory.size()) * DIR_ENTRY_SIZE + SECTOR_SIZE - 1) /
517 SECTOR_SIZE; // 目录占用的扇区数。
518 while (m_directory.size() < static_cast<int>(dirSectors * (SECTOR_SIZE / DIR_ENTRY_SIZE))) {
519 DirectoryEntry empty; // 用于填充目录扇区的空条目。
520 empty.objectType = ObjectType::Unknown;
521 empty.startSector = FREESECT;
522 m_directory.append(empty);
523 }
524
525 // 确保 Mini-FAT 大小是扇区对齐的
526 while (!m_miniFat.isEmpty() && m_miniFat.size() % (SECTOR_SIZE / 4) != 0) {
527 m_miniFat.append(FREESECT);
528 }
529 uint32_t miniFatSectors =
530 static_cast<uint32_t>((m_miniFat.size() * 4 + SECTOR_SIZE - 1) / SECTOR_SIZE); // Mini-FAT 扇区数。
531
532 // ---- 构建完整的 FAT（包含所有扇区的条目）----
533 // 文件布局：FAT | DIFAT | Directory | MiniFAT | Data。
534 // m_fat 目前只有数据扇区的链式条目（每个条目=一个数据扇区的后继指针）
535 // 需要补充 FAT/dir/MiniFAT 的条目，并为未使用的数据扇区填 FREESECT
536 uint32_t dataSectorCount = static_cast<uint32_t>(m_fat.size()); // 当前数据区扇区条目数。
537 uint32_t entriesPerSector = SECTOR_SIZE / 4; // 每个 FAT 扇区可容纳的条目数。
538
539 // 迭代计算 FAT 扇区数（FAT 扇区自身也占条目）
540 uint32_t fatSectors = 0; // FAT 扇区数量。
541 uint32_t difatSectors = 0; // DIFAT 扇区数量。
542 uint32_t totalEntries = 0; // 完整 FAT 所需条目总数。
543 for (int iter = 0; iter < 32; ++iter) {
544 uint32_t newTotal =
545 fatSectors + difatSectors + dirSectors + miniFatSectors + dataSectorCount; // 本轮估算总条目数。
546 newTotal = ((newTotal + entriesPerSector - 1) / entriesPerSector) * entriesPerSector;
547 uint32_t newFatSectors = newTotal / entriesPerSector; // 本轮估算 FAT 扇区数。
548 uint32_t newDifatSectors =
549 newFatSectors <= 109 ? 0 : (newFatSectors - 109 + (entriesPerSector - 2)) / (entriesPerSector - 1);
550 if (newFatSectors == fatSectors && newDifatSectors == difatSectors && newTotal == totalEntries) {
551 break;
552 }
553 fatSectors = newFatSectors;
554 difatSectors = newDifatSectors;
555 totalEntries = newTotal;
556 }
557
558 // 构建完整的 FAT 向量
559 QVector<uint32_t> fullFat;
560 fullFat.reserve(totalEntries);
561
562 // [0..fatSectors-1] FAT sectors → 标记为 FATSECT
563 for (uint32_t i = 0; i < fatSectors; ++i) {
564 fullFat.append(FATSECT);
565 }
566
567 // DIFAT sectors are allocation-table metadata and have their own marker.
568 for (uint32_t i = 0; i < difatSectors; ++i) {
569 fullFat.append(DIFSECT);
570 }
571
572 const uint32_t directoryOffset = fatSectors + difatSectors; // 目录区在文件扇区布局中的起始偏移。
573 // Directory sectors → 链式
574 for (uint32_t i = 0; i < dirSectors; ++i) {
575 if (i < dirSectors - 1) {
576 fullFat.append(directoryOffset + i + 1);
577 } else {
578 fullFat.append(ENDOFCHAIN);
579 }
580 }
581
582 // MiniFAT sectors 也必须通过 FAT 串成链，不能只依赖物理连续性。
583 const uint32_t firstMiniFatOffset = directoryOffset + dirSectors;
584 for (uint32_t i = 0; i < miniFatSectors; ++i) {
585 fullFat.append(i + 1 < miniFatSectors ? firstMiniFatOffset + i + 1 : ENDOFCHAIN);
586 }
587
588 // Data sectors → 来自 m_fat，需要加偏移量
589 // m_fat 中的链式指针是从 0 开始的数据扇区索引，需要偏移到文件中的实际位置
590 uint32_t dataOffset = fatSectors + difatSectors + dirSectors + miniFatSectors;
591 for (uint32_t val : m_fat) {
592 if (val == ENDOFCHAIN) {
593 fullFat.append(ENDOFCHAIN);
594 } else {
595 fullFat.append(val + dataOffset);
596 }
597 }
598
599 // 填充 FREESECT 到刚好 fatSectors 个扇区（不多不少，保证 header 和实际一致）
600 uint32_t fatCapacity = fatSectors * entriesPerSector;
601 while (static_cast<uint32_t>(fullFat.size()) < fatCapacity) {
602 fullFat.append(FREESECT);
603 }
604
605 m_fat = fullFat;
606
607 // 头部只能容纳 109 个 FAT sector id，其余 id 以每扇区 127 项串成 DIFAT 链。
608 m_difatSectors.clear();
609 for (uint32_t difatIndex = 0; difatIndex < difatSectors; ++difatIndex) {
610 QByteArray sector(SECTOR_SIZE, static_cast<char>(0xFF)); // 当前 DIFAT 扇区缓冲区。
611 const uint32_t firstFatIndex =
612 109 + difatIndex * (entriesPerSector - 1); // 当前 DIFAT 扇区引用的第一个 FAT 索引。
613 for (uint32_t item = 0; item < entriesPerSector - 1; ++item) {
614 const uint32_t fatIndex = firstFatIndex + item; // 当前 FAT 扇区索引。
615 if (fatIndex >= fatSectors) {
616 break;
617 }
618 const int byteOffset = static_cast<int>(item * 4); // FAT 索引在 DIFAT 扇区中的字节偏移。
619 sector[byteOffset] = static_cast<char>(fatIndex & 0xFF);
620 sector[byteOffset + 1] = static_cast<char>((fatIndex >> 8) & 0xFF);
621 sector[byteOffset + 2] = static_cast<char>((fatIndex >> 16) & 0xFF);
622 sector[byteOffset + 3] = static_cast<char>((fatIndex >> 24) & 0xFF);
623 }
624 const uint32_t next =
625 difatIndex + 1 < difatSectors ? fatSectors + difatIndex + 1 : ENDOFCHAIN; // 下一个 DIFAT 扇区索引。
626 const int nextOffset = static_cast<int>((entriesPerSector - 1) * 4); // 链接字段的字节偏移。
627 sector[nextOffset] = static_cast<char>(next & 0xFF);
628 sector[nextOffset + 1] = static_cast<char>((next >> 8) & 0xFF);
629 sector[nextOffset + 2] = static_cast<char>((next >> 16) & 0xFF);
630 sector[nextOffset + 3] = static_cast<char>((next >> 24) & 0xFF);
631 m_difatSectors.append(sector);
632 }
633
634 // Directory stream start sectors must be absolute CFB sector numbers.
635 // Mini-stream entries keep their offset inside the root mini stream.
636 for (DirectoryEntry& entry : m_directory) {
637 if (entry.objectType == ObjectType::Root && entry.streamSize > 0) {
638 entry.startSector += dataOffset;
639 } else if (entry.objectType == ObjectType::Stream && entry.streamSize >= MINI_STREAM_CUTOFF &&
640 entry.startSector != ENDOFCHAIN) {
641 entry.startSector += dataOffset;
642 }
643 }
644}

651bool OLECompoundWriter::saveToFile(const QString& filePath) {
652 if (!m_initialized) {
653 qWarning() << "OLECompoundWriter::saveToFile: Not initialized";
654 return false;
655 }
656
657 if (hasError()) {
658 qWarning() << "OLECompoundWriter::saveToFile: Aborted due to structure error:" << errorString();
659 return false;
660 }
661
662 if (!m_finalized) {
663 finalize();
664 m_finalized = true;
665 }
666
667 qDebug() << "OLECompoundWriter::saveToFile: Writing to" << filePath << "streams:" << m_streams.size()
668 << "directory:" << m_directory.size() << "fat:" << m_fat.size();
669
670 // 确保父目录存在
671 QFileInfo fileInfo(filePath);
672 if (!fileInfo.dir().exists()) {
673 qDebug() << "OLECompoundWriter::saveToFile: Creating parent directory:" << fileInfo.path();
674 QDir().mkpath(fileInfo.path());
675 }
676
677 QSaveFile file(filePath);
678 if (!file.open(QIODevice::WriteOnly)) {
679 qWarning() << "OLECompoundWriter::saveToFile: Failed to open file:" << filePath
680 << "error:" << file.errorString() << "exists:" << QFile::exists(filePath)
681 << "isDir:" << QFileInfo(filePath).isDir();
682 return false;
683 }
684
685 // 写入头部（512 字节）
686 QByteArray header;
687 serializeFileHeader(header);
688 auto writeAll = [&file, &filePath](const QByteArray& bytes) {
689 if (file.write(bytes) == bytes.size()) {
690 return true;
691 }
692 qWarning() << "OLECompoundWriter::saveToFile: Short write:" << filePath << file.errorString();
693 return false;
694 };
695 if (!writeAll(header)) {
696 file.cancelWriting();
697 return false;
698 }
699
700 // 写入 FAT 扇区（每个扇区 128 个条目）
701 uint32_t fatSectorCount = static_cast<uint32_t>(m_fat.size()) / (SECTOR_SIZE / 4); // FAT 扇区数量。
702 uint32_t entriesPerSector = SECTOR_SIZE / 4; // 每个 FAT 扇区的条目数。
703 for (uint32_t i = 0; i < fatSectorCount; ++i) {
704 QByteArray sectorData(SECTOR_SIZE, 0); // 当前 FAT 扇区缓冲区。
705 uint32_t startIdx = i * entriesPerSector; // 当前 FAT 扇区的首条目索引。
706 for (uint32_t j = 0; j < entriesPerSector && (startIdx + j) < static_cast<uint32_t>(m_fat.size()); ++j) {
707 uint32_t val = m_fat[startIdx + j]; // 当前 FAT 条目值。
708 int off = j * 4; // 条目在扇区中的字节偏移。
709 sectorData[off] = static_cast<char>(val & 0xFF);
710 sectorData[off + 1] = static_cast<char>((val >> 8) & 0xFF);
711 sectorData[off + 2] = static_cast<char>((val >> 16) & 0xFF);
712 sectorData[off + 3] = static_cast<char>((val >> 24) & 0xFF);
713 }
714 if (!writeAll(sectorData)) {
715 file.cancelWriting();
716 return false;
717 }
718 }
719
720 for (const QByteArray& sector : m_difatSectors) {
721 if (!writeAll(sector)) {
722 file.cancelWriting();
723 return false;
724 }
725 }
726
727 // 写入目录扇区
728 QByteArray dirData;
729 for (const DirectoryEntry& entry : m_directory) {
730 QByteArray entryData; // 当前目录条目的序列化数据。
731 serializeDirectoryEntry(entry, entryData);
732 dirData.append(entryData);
733 }
734 if (!writeAll(dirData)) {
735 file.cancelWriting();
736 return false;
737 }
738
739 // 写入 Mini-FAT 扇区
740 if (!m_miniFat.isEmpty()) {
741 QByteArray miniFatData; // Mini-FAT 序列化数据。
742 for (uint32_t val : m_miniFat) {
743 miniFatData.append(static_cast<char>(val & 0xFF));
744 miniFatData.append(static_cast<char>((val >> 8) & 0xFF));
745 miniFatData.append(static_cast<char>((val >> 16) & 0xFF));
746 miniFatData.append(static_cast<char>((val >> 24) & 0xFF));
747 }
748 if (!writeAll(miniFatData)) {
749 file.cancelWriting();
750 return false;
751 }
752 }
753
754 // 写入数据扇区
755 for (const QByteArray& sector : m_dataSectors) {
756 if (!writeAll(sector)) {
757 file.cancelWriting();
758 return false;
759 }
760 }
761
762 if (!file.commit()) {
763 qWarning() << "OLECompoundWriter::saveToFile: Commit failed:" << filePath << file.errorString();
764 return false;
765 }
766 return true;
767}
