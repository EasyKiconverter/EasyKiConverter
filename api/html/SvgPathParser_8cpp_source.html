59SvgArcGeometry calculateArcGeometry(const QPointF& start,
60 double radiusX,
61 double radiusY,
62 double xRotation,
63 bool largeArc,
64 bool sweep,
65 const QPointF& end) {
66 SvgArcGeometry geometry; // 待计算的圆弧几何结果。
67 radiusX = std::abs(radiusX);
68 radiusY = std::abs(radiusY);
69 if (radiusX <= 0.0 || radiusY <= 0.0 || start == end)
70 return geometry;
71
72 const double phi = xRotation * PI / 180.0; // 椭圆旋转角度，单位为弧度。
73 const double cosPhi = std::cos(phi); // 旋转角度余弦。
74 const double sinPhi = std::sin(phi); // 旋转角度正弦。
75 const double dx = (start.x() - end.x()) / 2.0; // 端点中点坐标的 X 偏移。
76 const double dy = (start.y() - end.y()) / 2.0; // 端点中点坐标的 Y 偏移。
77 const double xPrime = cosPhi * dx + sinPhi * dy; // 旋转坐标系中的 X 坐标。
78 const double yPrime = -sinPhi * dx + cosPhi * dy; // 旋转坐标系中的 Y 坐标。
79 double rxSquared = radiusX * radiusX; // X 半径平方。
80 double rySquared = radiusY * radiusY; // Y 半径平方。
81 const double lambda = xPrime * xPrime / rxSquared + yPrime * yPrime / rySquared; // 半径修正比例。
82 if (lambda > 1.0) {
83 const double scale = std::sqrt(lambda); // 保证端点落在椭圆上的半径缩放因子。
84 radiusX *= scale;
85 radiusY *= scale;
86 rxSquared = radiusX * radiusX;
87 rySquared = radiusY * radiusY;
88 }
89
90 const double denominator = rxSquared * yPrime * yPrime + rySquared * xPrime * xPrime; // 圆心公式分母。
91 if (qFuzzyIsNull(denominator))
92 return geometry;
93 const double numerator = // 圆心公式中经过非负裁剪的根式分子。
94 std::max(0.0, rxSquared * rySquared - rxSquared * yPrime * yPrime - rySquared * xPrime * xPrime);
95 const double factor = // 决定大弧和扫掠方向的圆心符号因子。
96 (largeArc == sweep ? -1.0 : 1.0) * std::sqrt(numerator / denominator);
97 const double centerPrimeX = factor * radiusX * yPrime / radiusY; // 旋转坐标系中的圆心 X 坐标。
98 const double centerPrimeY = -factor * radiusY * xPrime / radiusX; // 旋转坐标系中的圆心 Y 坐标。
99 const QPointF center( // 原始坐标系中的圆心。
100 cosPhi * centerPrimeX - sinPhi * centerPrimeY + (start.x() + end.x()) / 2.0,
101 sinPhi * centerPrimeX + cosPhi * centerPrimeY + (start.y() + end.y()) / 2.0);
102 const double ux = (xPrime - centerPrimeX) / radiusX; // 起点单位向量 X 分量。
103 const double uy = (yPrime - centerPrimeY) / radiusY; // 起点单位向量 Y 分量。
104 const double vx = (-xPrime - centerPrimeX) / radiusX; // 终点单位向量 X 分量。
105 const double vy = (-yPrime - centerPrimeY) / radiusY; // 终点单位向量 Y 分量。
106 double deltaAngle = std::atan2(ux * vy - uy * vx, ux * vx + uy * vy); // 初始扫掠角度。
107 if (sweep && deltaAngle < 0.0)
108 deltaAngle += 2.0 * PI;
109 else if (!sweep && deltaAngle > 0.0)
110 deltaAngle -= 2.0 * PI;
111
112 geometry.valid = true;
113 geometry.center = center;
114 geometry.radiusX = radiusX;
115 geometry.radiusY = radiusY;
116 geometry.startAngle = std::atan2(uy, ux);
117 geometry.deltaAngle = deltaAngle;
118 return geometry;
119}

127QList<SvgPathSegment> approximateRotatedEllipse(const SvgArcGeometry& geometry, double xRotation) {
128 QList<SvgPathSegment> segments; // 生成的贝塞尔线段列表。
129 const double segmentAngle = PI / 2.0; // 单段最大扫掠角度。
130 const int segmentCount = // 圆弧需要拆分的线段数量。
131 qMax(1, static_cast<int>(std::ceil(std::abs(geometry.deltaAngle) / segmentAngle)));
132 const double phi = xRotation * PI / 180.0; // 椭圆旋转角度，单位为弧度。
133 const double cosPhi = std::cos(phi); // 旋转角度余弦。
134 const double sinPhi = std::sin(phi); // 旋转角度正弦。
135 const auto pointAt = [&](double angle) {
136 return QPointF(geometry.center.x() + cosPhi * geometry.radiusX * std::cos(angle) -
137 sinPhi * geometry.radiusY * std::sin(angle),
138 geometry.center.y() + sinPhi * geometry.radiusX * std::cos(angle) +
139 cosPhi * geometry.radiusY * std::sin(angle));
140 };
141 const auto derivativeAt = [&](double angle) {
142 return QPointF(-cosPhi * geometry.radiusX * std::sin(angle) - sinPhi * geometry.radiusY * std::cos(angle),
143 -sinPhi * geometry.radiusX * std::sin(angle) + cosPhi * geometry.radiusY * std::cos(angle));
144 };
145
146 for (int i = 0; i < segmentCount; ++i) {
147 const double startAngle = geometry.startAngle + geometry.deltaAngle * i / segmentCount; // 当前段起始角度。
148 const double endAngle = geometry.startAngle + geometry.deltaAngle * (i + 1) / segmentCount; // 当前段终止角度。
149 const double deltaAngle = endAngle - startAngle; // 当前段扫掠角度。
150 const double controlFactor = 4.0 / 3.0 * std::tan(deltaAngle / 4.0); // 贝塞尔控制点比例。
151 const QPointF start = pointAt(startAngle); // 当前段起点。
152 const QPointF end = pointAt(endAngle); // 当前段终点。
153 const QPointF startDerivative = derivativeAt(startAngle); // 起点切线方向。
154 const QPointF endDerivative = derivativeAt(endAngle); // 终点切线方向。
155 SvgPathSegment segment; // 当前生成的贝塞尔线段。
156 segment.type = SvgPathSegment::Type::CubicBezier;
157 segment.start = i == 0 ? start : segments.last().end;
158 segment.control1 = start + controlFactor * startDerivative;
159 segment.control2 = end - controlFactor * endDerivative;
160 segment.end = i == segmentCount - 1 ? pointAt(geometry.startAngle + geometry.deltaAngle) : end;
161 segments.append(segment);
162 }
163 return segments;
164}

173QList<QPointF> SvgPathParser::parsePath(const QString& path) {
174 QList<QPointF> points; // 解析得到的路径点集合。
175 if (path.isEmpty()) {
176 return points;
177 }
178
179 QStringList tokens = splitPath(path); // 分词后的 SVG 命令和参数。
180 for (const QString& token : tokens) {
181 if (token.isEmpty() || token.at(0).isLetter())
182 continue;
183 bool ok = false; // 参数转换是否成功。
184 const double value = token.toDouble(&ok); // 当前参数数值。
185 if (!ok || !std::isfinite(value))
186 return {};
187 }
188 double currentX = 0.0; // 当前路径位置的 X 坐标。
189 double currentY = 0.0; // 当前路径位置的 Y 坐标。
190 QPointF lastCubicControl; // 上一个三次贝塞尔控制点。
191 QPointF lastQuadraticControl; // 上一个二次贝塞尔控制点。
192 QChar previousCommand; // 上一个 SVG 路径命令。
193
194 int i = 0; // 当前命令或参数在分词列表中的索引。
195 while (i < tokens.size()) {
196 QString cmd = tokens[i]; // 当前 SVG 命令词。
197 if (cmd.isEmpty()) {
198 i++;
199 continue;
200 }
201 // 参数只能由前面的命令消费；遇到残留数字时跳过，避免把参数误报为命令。
202 if (!cmd.at(0).isLetter()) {
203 i++;
204 continue;
205 }
206
207 QChar command = cmd[0].toUpper(); // 规范化后的命令字符。
208
209 // 处理M/m（MoveTo）命
210 if (command == 'M') {
211 bool relative = (cmd[0] == 'm'); // 当前 MoveTo 是否使用相对坐标。
212 i++;
213 while (i < tokens.size()) {
214 bool okX = false; // X 参数转换是否成功。
215 bool okY = false; // Y 参数转换是否成功。
216 double x = tokens[i].toDouble(&okX); // 当前点的 X 参数。
217 if (!okX)
218 break;
219 i++;
220 if (i >= tokens.size())
221 break;
222 double y = tokens[i].toDouble(&okY); // 当前点的 Y 参数。
223 if (!okY)
224 break;
225 i++;
226
227 QPointF pt = createPoint(x, y, relative, currentX, currentY); // 转换后的路径点。
228 points.append(pt);
229 }
230 previousCommand = 'M';
231 }
232 // 处理L/l（LineTo）命
233 else if (command == 'L') {
234 bool relative = (cmd[0] == 'l'); // 当前 LineTo 是否使用相对坐标。
235 i++;
236 while (i < tokens.size()) {
237 bool okX = false; // X 参数转换是否成功。
238 bool okY = false; // Y 参数转换是否成功。
239 double x = tokens[i].toDouble(&okX); // 当前点的 X 参数。
240 if (!okX)
241 break;
242 i++;
243 if (i >= tokens.size())
244 break;
245 double y = tokens[i].toDouble(&okY); // 当前点的 Y 参数。
246 if (!okY)
247 break;
248 i++;
249
250 QPointF pt = createPoint(x, y, relative, currentX, currentY); // 转换后的路径点。
251 points.append(pt);
252 }
253 previousCommand = 'L';
254 }
255 // 处理H/h（Horizontal LineTo）命
256 else if (command == 'H') {
257 bool relative = (cmd[0] == 'h'); // 当前 Horizontal LineTo 是否使用相对坐标。
258 i++;
259 while (i < tokens.size()) {
260 bool ok = false; // 参数转换是否成功。
261 double dx = tokens[i].toDouble(&ok); // 当前水平位移或绝对 X 坐标。
262 if (!ok)
263 break;
264 i++;
265
266 if (relative) {
267 currentX += dx;
268 } else {
269 currentX = dx;
270 }

272 }
273 previousCommand = 'H';
274 }
275 // 处理V/v（Vertical LineTo）命
276 else if (command == 'V') {
277 bool relative = (cmd[0] == 'v'); // 当前 Vertical LineTo 是否使用相对坐标。
278 i++;
279 while (i < tokens.size()) {
280 bool ok = false; // 参数转换是否成功。
281 double dy = tokens[i].toDouble(&ok); // 当前垂直位移或绝对 Y 坐标。
282 if (!ok)
283 break;
284 i++;

286 if (relative) {
287 currentY += dy;
288 } else {
289 currentY = dy;

556QList<SvgPathSegment> SvgPathParser::parseSegments(const QString& path) {
557 QList<SvgPathSegment> segments; // 解析得到的结构化路径线段。
558 if (path.isEmpty())
559 return segments;
560
561 const QStringList tokens = splitPath(path); // 分词后的 SVG 命令和参数。
562 QPointF current; // 当前路径位置。
563 QPointF subpathStart; // 当前子路径起点。
564 QPointF lastCubicControl; // 上一个三次贝塞尔控制点。
565 QPointF lastQuadraticControl; // 上一个二次贝塞尔控制点。
566 QChar previousCommand; // 上一个路径命令。
567 bool hasCurrent = false; // 是否已有当前路径位置。
568 bool hasSubpath = false; // 是否已建立子路径。
569
570 auto isNumber = [&](int index) {
571 if (index >= tokens.size())
572 return false;
573 bool ok = false; // 数值转换是否成功。
574 const double value = tokens.at(index).toDouble(&ok); // 当前 SVG 数值标记的解析结果。
575 return ok && std::isfinite(value);
576 };
577 auto readNumbers = [&](int& index, int count, QList<double>& values) {
578 if (index + count > tokens.size())
579 return false;
580 values.clear();
581 values.reserve(count);
582 for (int n = 0; n < count; ++n) {
583 bool ok = false; // 当前参数转换是否成功。
584 const double value = tokens.at(index++).toDouble(&ok); // 当前参数数值。
585 if (!ok || !std::isfinite(value))
586 return false;
587 values.append(value);
588 }
589 return true;
590 };
591 auto addLine = [&](const QPointF& end) {
592 if (!hasCurrent) {
593 current = end;
594 return;
595 }
596 if (current == end)
597 return;
598 SvgPathSegment segment; // 当前新增的直线路径线段。
599 segment.type = SvgPathSegment::Type::Line;
600 segment.start = current;
601 segment.end = end;
602 segments.append(segment);
603 current = end;
604 };
605 auto addCubic = [&](const QPointF& control1, const QPointF& control2, const QPointF& end) {
606 if (!hasCurrent)
607 return;
608 SvgPathSegment segment; // 当前新增的三次贝塞尔线段。
609 segment.type = SvgPathSegment::Type::CubicBezier;
610 segment.start = current;
611 segment.control1 = control1;
612 segment.control2 = control2;
613 segment.end = end;
614 segments.append(segment);
615 current = end;
616 };
617 auto addQuadratic = [&](const QPointF& control, const QPointF& end) {
618 if (!hasCurrent)
619 return;
620 SvgPathSegment segment; // 当前新增的二次贝塞尔线段。
621 segment.type = SvgPathSegment::Type::QuadraticBezier;
622 segment.start = current;
623 segment.control1 = control;
624 segment.end = end;
625 segments.append(segment);
626 current = end;
627 };
628
629 int index = 0; // 当前命令或参数索引。
630 while (index < tokens.size()) {
631 const QString commandToken = tokens.at(index++); // 当前 SVG 命令词。
632 if (commandToken.isEmpty() || commandToken.at(0).isDigit() || commandToken.at(0) == '.' ||
633 commandToken.at(0) == '+' || commandToken.at(0) == '-') {
634 continue;
635 }
636 const QChar command = commandToken.at(0).toUpper(); // 规范化后的命令字符。
637 const bool relative = commandToken.at(0).isLower(); // 当前命令是否使用相对坐标。
638
639 if (command == 'M') {
640 bool first = true; // 当前 MoveTo 参数组是否为子路径首点。
641 while (isNumber(index) && isNumber(index + 1)) {
642 QList<double> values; // 当前命令参数组。
643 if (!readNumbers(index, 2, values))
644 break;
645 QPointF point(values[0], values[1]); // 当前 MoveTo 目标点。
646 if (relative && hasCurrent)
647 point += current;
648 if (first) {
649 current = point;
650 subpathStart = point;
651 hasCurrent = true;
652 hasSubpath = true;
653 first = false;
654 } else {
655 addLine(point);
656 }
657 }
658 previousCommand = 'M';
659 continue;
660 }
661
662 if (command == 'Z') {
663 if (hasCurrent && hasSubpath)
664 addLine(subpathStart);
665 previousCommand = 'Z';
666 continue;
667 }
668
669 if (command == 'L') {
670 while (isNumber(index) && isNumber(index + 1)) {
671 QList<double> values; // 当前 LineTo 参数组。
672 if (!readNumbers(index, 2, values))
673 break;
674 QPointF point(values[0], values[1]); // 当前 LineTo 目标点。
675 if (relative)
676 point += current;
677 addLine(point);
678 }
679 previousCommand = 'L';
680 continue;
681 }
682
683 if (command == 'H' || command == 'V') {
684 while (isNumber(index)) {
685 QList<double> values; // 当前水平或垂直线参数组。
686 if (!readNumbers(index, 1, values))
687 break;
688 QPointF point = current; // 当前水平或垂直线目标点。
689 if (command == 'H')
690 point.setX(relative ? current.x() + values[0] : values[0]);
691 else
692 point.setY(relative ? current.y() + values[0] : values[0]);
693 addLine(point);
694 }
695 previousCommand = command;
696 continue;
697 }
698
699 const int groupSize = // 当前命令的一组参数数量。
700 command == 'A' ? 7
701 : command == 'C' ? 6
702 : (command == 'S' || command == 'Q') ? 4
703 : command == 'T' ? 2
704 : 0;
705 if (groupSize == 0) {
706 qWarning() << "SVG: Unknown segment command:" << command;
707 continue;
708 }
709
710 while (isNumber(index)) {
711 QList<double> values; // 当前曲线或圆弧参数组。
712 if (!readNumbers(index, groupSize, values)) {
713 qWarning() << "SVG: Invalid segment parameters for command:" << command;
714 break;
715 }
716 if (!hasCurrent) {
717 qWarning() << "SVG: Segment command without origin point:" << command;
718 continue;
719 }
720 const QPointF start = current; // 当前曲线或圆弧起点。
721 if (command == 'A') {
722 QPointF end(values[5], values[6]); // 圆弧终点。
723 if (relative)
724 end += start;
725 const QList<QPointF> arcPoints = // 圆弧离散点或近似折线点。
726 parseArc(start, values[0], values[1], values[2], values[3] != 0, values[4] != 0, end);
727 const bool isCircular = values[0] > 0.0 && qFuzzyCompare(values[0], values[1]); // 是否为圆形圆弧。
728 if (isCircular && arcPoints.size() >= 3 && start != end) {
729 SvgPathSegment segment; // 当前圆形圆弧线段。
730 segment.type = SvgPathSegment::Type::CircularArc;
731 segment.start = start;
732 segment.arcMid = arcPoints.at(arcPoints.size() / 2);
733 segment.end = end;
734 segments.append(segment);
735 current = end;
736 } else if (!isCircular && values[0] > 0.0 && values[1] > 0.0) {
737 const SvgArcGeometry geometry = // 当前椭圆弧几何参数。
738 calculateArcGeometry(
739 start, values[0], values[1], values[2], values[3] != 0, values[4] != 0, end);
740 if (geometry.valid) {
741 if (qFuzzyIsNull(std::sin(values[2] * PI / 180.0))) {
742 SvgPathSegment segment; // 当前椭圆弧线段。
743 segment.type = SvgPathSegment::Type::EllipticalArc;
744 segment.start = start;
745 segment.arcCenter = geometry.center;
746 segment.radiusX = geometry.radiusX;
747 segment.radiusY = geometry.radiusY;
748 segment.arcStartAngle = geometry.startAngle * 180.0 / PI;
749 segment.arcEndAngle = (geometry.startAngle + geometry.deltaAngle) * 180.0 / PI;
750 segment.end = end;
751 segments.append(segment);
752 } else {
753 segments.append(approximateRotatedEllipse(geometry, values[2]));
754 }
755 current = end;
756 } else {
757 for (const QPointF& point : arcPoints)
758 addLine(point);
759 }
760 } else {
761 for (const QPointF& point : arcPoints)
762 addLine(point);
763 }
764 } else if (command == 'C' || command == 'S' || command == 'Q' || command == 'T') {
765 QPointF control1; // 三次贝塞尔第一个控制点。
766 QPointF control2; // 三次贝塞尔第二个控制点。
767 QPointF end; // 当前曲线终点。
768 if (command == 'C') {
769 control1 = QPointF(values[0], values[1]);
770 control2 = QPointF(values[2], values[3]);
771 end = QPointF(values[4], values[5]);
772 if (relative) {
773 control1 += start;
774 control2 += start;
775 end += start;
776 }
777 } else if (command == 'S') {
778 control1 =
779 (previousCommand == 'C' || previousCommand == 'S')
780 ? QPointF(2.0 * start.x() - lastCubicControl.x(), 2.0 * start.y() - lastCubicControl.y())
781 : start;
782 control2 = QPointF(values[0], values[1]);
783 end = QPointF(values[2], values[3]);
784 if (relative) {
785 control2 += start;
786 end += start;
787 }
788 } else if (command == 'Q' || command == 'T') {
789 QPointF quadraticControl; // 二次贝塞尔控制点。
790 if (command == 'Q') {
791 quadraticControl = QPointF(values[0], values[1]);
792 end = QPointF(values[2], values[3]);
793 if (relative) {
794 quadraticControl += start;
795 end += start;
796 }
797 } else {
798 quadraticControl = (previousCommand == 'Q' || previousCommand == 'T')
799 ? QPointF(2.0 * start.x() - lastQuadraticControl.x(),
800 2.0 * start.y() - lastQuadraticControl.y())
801 : start;
802 end = QPointF(values[0], values[1]);
803 if (relative)
804 end += start;
805 }
806 addQuadratic(quadraticControl, end);
807 lastQuadraticControl = quadraticControl;
808 }
809 if (command == 'C' || command == 'S') {
810 addCubic(control1, control2, end);
811 }
812 if (command == 'C' || command == 'S')
813 lastCubicControl = control2;
814 }
815 previousCommand = command;
816 }
817 }
818 for (const SvgPathSegment& segment : segments) {
819 if (!isFiniteSegment(segment))
820 return {};
821 }
822 return segments;
823}

830QStringList SvgPathParser::splitPath(const QString& path) {
831 // 将命令字母前后添加空格，然后按空格分
832 QString processed = path; // 经过分隔符规范化的路径文本。
833 processed.replace(QRegularExpression("([MmZzLlHhVvCcSsQqTtAa])"), " \\1 ");
834 processed.replace(QRegularExpression("(?<=[0-9.])(?=[+-])"), " ");
835 const QStringList rawTokens = // 初步分词结果。
836 processed.split(QRegularExpression("[\\s,]+"), Qt::SkipEmptyParts);
837
838 // SVG 允许一个命令后连续跟随多组参数。当前主解析循环按一次命令消费一组
839 // 曲线/圆弧参数，因此在分词阶段把后续参数组展开成同类型的重复命令。
840 QStringList tokens; // 展开重复参数组后的词元列表。
841 int index = 0; // 当前原始词元索引。
842 while (index < rawTokens.size()) {
843 const QString command = rawTokens.at(index++); // 当前命令或参数词元。
844 tokens.append(command);
845 if (command.size() != 1 || !command.at(0).isLetter())
846 continue;
847
848 const QChar upper = command.at(0).toUpper(); // 规范化后的命令字符。
849 int groupSize = 0; // 当前命令每组参数数量。
850 if (upper == 'A')
851 groupSize = 7;
852 else if (upper == 'C')
853 groupSize = 6;
854 else if (upper == 'S' || upper == 'Q')
855 groupSize = 4;
856 else if (upper == 'T')
857 groupSize = 2;
858 if (groupSize == 0)
859 continue;
860
861 int parameterCount = 0; // 当前参数组已收集的数量。
862 while (index < rawTokens.size() && !rawTokens.at(index).at(0).isLetter()) {
863 if (parameterCount == groupSize) {
864 tokens.append(command);
865 parameterCount = 0;
866 }
867 tokens.append(rawTokens.at(index++));
868 ++parameterCount;
869 }
870 }
871 return tokens;
872}

905QList<QPointF> SvgPathParser::parseArc(const QPointF& startPoint,
906 double rx,
907 double ry,
908 double xRotation,
909 bool largeArcFlag,
910 bool sweepFlag,
911 const QPointF& endPoint) {
912 QList<QPointF> points; // 圆弧离散后的路径点。
913
914 // SVG 规范规定起点和终点相同时不绘制弧线，避免后续圆心计算出现 0/0。
915 if (startPoint == endPoint) {
916 points.append(startPoint);
917 return points;
918 }
919
920 // 如果半径，直接返回起点和终点
921 if (rx <= 0 || ry <= 0) {
922 points.append(startPoint);
923 points.append(endPoint);
924 return points;
925 }
926
927 // 将角度转换为弧度
928 double phi = xRotation * PI / 180.0; // 椭圆旋转角度，单位为弧度。
929
930 // 计算中点
931 double dx = (startPoint.x() - endPoint.x()) / 2.0; // 端点中点 X 偏移。
932 double dy = (startPoint.y() - endPoint.y()) / 2.0; // 端点中点 Y 偏移。
933
934 // 旋转坐标
935 double x1 = cos(phi) * dx + sin(phi) * dy; // 旋转坐标系中的起点 X 坐标。
936 double y1 = -sin(phi) * dx + cos(phi) * dy; // 旋转坐标系中的起点 Y 坐标。
937
938 // 计算临时变量
939 double rxSq = rx * rx; // X 半径平方。
940 double rySq = ry * ry; // Y 半径平方。
941 double x1Sq = x1 * x1; // 旋转坐标 X 平方。
942 double y1Sq = y1 * y1; // 旋转坐标 Y 平方。
943
944 // 计算校正因子
945 double temp = // 圆心计算使用的半径修正因子。
946 (rxSq * rySq - rxSq * y1Sq - rySq * x1Sq) / (rxSq * y1Sq + rySq * x1Sq);
947 if (temp < 0)
948 temp = 0;
949 temp = sqrt(temp);
950
951 // 根据largeArcFlag和sweepFlag确定符号
952 double factor = (largeArcFlag == sweepFlag) ? -1.0 : 1.0; // 大弧和扫掠方向对应的圆心符号。
953 double cx1 = factor * temp * rx * y1 / ry; // 旋转坐标系中的圆心 X 坐标。
954 double cy1 = -factor * temp * ry * x1 / rx; // 旋转坐标系中的圆心 Y 坐标。
955
956 // 计算圆心
957 double cx = cos(phi) * cx1 - sin(phi) * cy1 + (startPoint.x() + endPoint.x()) / 2.0; // 圆心 X 坐标。
958 double cy = sin(phi) * cx1 + cos(phi) * cy1 + (startPoint.y() + endPoint.y()) / 2.0; // 圆心 Y 坐标。
959
960 // 计算起始角度和角度增
961 double startAngle = getAngle(1.0, 0.0, (x1 - cx1) / rx, (y1 - cy1) / ry); // 起始角度，单位为弧度。
962 double deltaAngle = // 扫掠角度，单位为弧度。
963 getAngle((x1 - cx1) / rx, (y1 - cy1) / ry, (-x1 - cx1) / rx, (-y1 - cy1) / ry);
964
965 // 规范化角
966 while (startAngle < 0)
967 startAngle += 2 * PI;
968 while (startAngle >= 2 * PI)
969 startAngle -= 2 * PI;
970
971 // 根据sweepFlag调整角度增量
972 if (sweepFlag) {
973 if (deltaAngle < 0)
974 deltaAngle += 2 * PI;
975 } else {
976 if (deltaAngle > 0)
977 deltaAngle -= 2 * PI;
978 }
979
980 // 将弧度转换为角度
981 double startAngleDeg = startAngle * 180.0 / PI; // 起始角度，单位为度。
982 double deltaAngleDeg = deltaAngle * 180.0 / PI; // 扫掠角度，单位为度。
983
984 // 计算圆弧上的
985 points = calcArcPoints(cx, cy, rx, ry, startAngleDeg, deltaAngleDeg, xRotation);
986
987 return points;
988}

1001QList<QPointF> SvgPathParser::calcArcPoints(double cx,
1002 double cy,
1003 double rx,
1004 double ry,
1005 double startAngle,
1006 double deltaAngle,
1007 double xRotation) {
1008 QList<QPointF> points; // 圆弧离散后的点列表。
1009 const int splitCount = 32; // 分割32段
1010 double step = deltaAngle / splitCount; // 每个离散区间的角度步长。
1011
1012 double phi = xRotation * PI / 180.0; // 椭圆旋转角度，单位为弧度。
1013
1014 for (int i = 0; i <= splitCount; i++) {
1015 double theta = (startAngle + i * step) * PI / 180.0; // 当前离散点角度，单位为弧度。
1016 double cosTheta = cos(theta); // 当前角度余弦。
1017 double sinTheta = sin(theta); // 当前角度正弦。
1018
1019 double x = cos(phi) * rx * cosTheta - sin(phi) * ry * sinTheta + cx; // 当前点 X 坐标。
1020 double y = sin(phi) * rx * cosTheta + cos(phi) * ry * sinTheta + cy; // 当前点 Y 坐标。
1021
1022 points.append(QPointF(x, y));
1023 }
1024
1025 return points;
1026}

1036double SvgPathParser::getAngle(double x1, double y1, double x2, double y2) {
1037 // 计算向量点积和叉
1038 double dot = x1 * x2 + y1 * y2; // 两个向量的点积。
1039 double cross = x1 * y2 - y1 * x2; // 两个向量的二维叉积。
1040
1041 // 计算角度
1042 double angle = atan2(cross, dot); // 根据点积和叉积得到的有向角度。
1043 return angle;
1044}

1059QList<QPointF> SvgPathParser::bezierToPolyline(double startX,
1060 double startY,
1061 double cp1X,
1062 double cp1Y,
1063 double cp2X,
1064 double cp2Y,
1065 double endX,
1066 double endY,
1067 int segments) {
1068 QList<QPointF> res; // 三次贝塞尔曲线离散后的点列表。
1069
1070 // 三次贝塞尔曲线公式
1071 // B(t) = (1-t)^3 * P0 + 3*(1-t)^2*t * P1 + 3*(1-t)*t^2 * P2 + t^3 * P3
1072
1073 for (int i = 0; i <= segments; ++i) {
1074 double t = static_cast<double>(i) / segments; // 当前曲线参数。
1075 double t2 = t * t; // 曲线参数平方。
1076 double t3 = t2 * t; // 曲线参数立方。
1077 double mt = 1.0 - t; // 曲线参数补值。
1078 double mt2 = mt * mt; // 曲线参数补值平方。
1079 double mt3 = mt2 * mt; // 曲线参数补值立方。
1080
1081 double x = mt3 * startX + 3.0 * mt2 * t * cp1X + 3.0 * mt * t2 * cp2X + t3 * endX; // 当前点 X 坐标。
1082 double y = mt3 * startY + 3.0 * mt2 * t * cp1Y + 3.0 * mt * t2 * cp2Y + t3 * endY; // 当前点 Y 坐标。
1083
1084 res.append(QPointF(x, y));
1085 }
1086
1087 return res;
1088}
