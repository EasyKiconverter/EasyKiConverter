327void ComponentService::fetchMultipleComponentsData(const QStringList& componentIds, bool fetch3DModel) {
328 m_maxConcurrentRequests = ConfigService::instance()->getValidationConcurrentCount();
329 if (m_queueManager != nullptr) {
330 m_queueManager->setMaxConcurrentRequests(m_maxConcurrentRequests);
331 }
332
333 QStringList normalizedComponentIds; // 去重并统一大小写后的元器件编号。
334 QSet<QString> seenComponentIds; // 已加入批量请求的元器件编号集合。
335 for (const QString& componentId : componentIds) {
336 const QString normalizedId = componentId.toUpper(); // 当前编号的规范化形式。
337 if (!normalizedId.isEmpty() && !seenComponentIds.contains(normalizedId)) {
338 seenComponentIds.insert(normalizedId);
339 normalizedComponentIds.append(normalizedId);
340 }
341 }
342
343 qDebug() << "Fetching data for" << normalizedComponentIds.size()
344 << "components with async queue (max concurrent:" << m_maxConcurrentRequests << ")";
345
346 // 防止重复启动批量处理，避免队列状态混乱
347 if (m_parallelContext != nullptr) {
348 qWarning() << "Batch processing already in progress, ignoring new request";
349 return;
350 }
351
352 // 初始化并行获取状态
353 m_parallelContext = new ParallelFetchContext(this); // 当前批量请求的并行上下文。
354 connect(m_parallelContext, &ParallelFetchContext::allCompleted, this, [this](const QList<ComponentData>& data) {
355 const QMap<QString, QString> failedComponents =
356 m_parallelContext->failedComponents(); // 批量请求失败的元器件及其错误信息。
357 emit allComponentsDataCollected(data);
358 emit allComponentsDataCollectedWithErrors(data, failedComponents);
359 m_activeRequestCount = 0;
360 resetQueueState();
361 });
362 m_parallelContext->start(normalizedComponentIds.size());
363 m_activeRequestCount = 0;
364 m_batchFetch3DModel = fetch3DModel;
365
366 // 使用 ComponentQueueManager 管理队列
367 m_queueManager->start(normalizedComponentIds);
368}
