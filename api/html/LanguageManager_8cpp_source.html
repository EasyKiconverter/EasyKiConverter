47void LanguageManager::setLanguage(const QString& languageCode, bool force) {
48 qInfo() << "[LanguageManager] setLanguage() called with:" << languageCode << " (current:" << m_currentLanguage
49 << ", initializing:" << m_isInitializing << ", loadedFromConfig:" << m_loadedFromConfig
50 << ", force:" << force << ")";
51
52 if (m_currentLanguage == languageCode) {
53 qInfo() << "[LanguageManager] Language unchanged, returning";
54
55 return;
56 }
57
58 // 如果正在初始化且已经从配置加载了语言，则拒绝覆盖（除非 force=true）
59 // 这样可以防止 QML 组件在初始化时覆盖配置文件的语言设置
60 if (m_isInitializing && m_loadedFromConfig && !force) {
61 qWarning() << "[LanguageManager] Refusing to override language during initialization after loading from "
62 "config. Requested:"
63 << languageCode << ", Keeping:" << m_currentLanguage;
64
65 return;
66 }
67
68 m_currentLanguage = languageCode;
69
70 qInfo() << "[LanguageManager] Language changed to" << m_currentLanguage;
71
72 // 只在非初始化期间保存设置
73
74 if (!m_isInitializing) {
75 saveLanguageSettings();
76
77 } else {
78 qInfo() << "[LanguageManager] Skipping save during initialization";
79 }
80
81 // 安装新的翻译器
82
83 installTranslator(languageCode);
84
85 emit languageChanged(languageCode);
86
87 emit refreshRequired();
88
89 qInfo() << "[LanguageManager] Language changed signal emitted:" << languageCode;
90}

112void LanguageManager::loadLanguageSettings() {
113 // 从 ConfigService 加载语言设置
114 auto* configService = ConfigService::instance();
115 if (configService) {
116 m_currentLanguage = configService->getLanguage();
117 qInfo() << "[LanguageManager] Loaded language settings from ConfigService:" << m_currentLanguage;
118 } else {
119 qWarning() << "[LanguageManager] ConfigService not available, using default language: en";
120 m_currentLanguage = "en";
121 }
122
123 m_loadedFromConfig = true;
124
125 // 安装翻译器
126 qInfo() << "[LanguageManager] Installing translator for language:" << m_currentLanguage;
127 installTranslator(m_currentLanguage);
128}

148void LanguageManager::installTranslator(const QString& languageCode) {
149 QCoreApplication* app = QCoreApplication::instance(); // 当前 Qt 应用实例。
150 if (!app) {
151 qWarning() << "[LanguageManager] QCoreApplication instance not available";
152 return;
153 }
154
155 // 移除旧翻译器
156 if (m_translator) {
157 app->removeTranslator(m_translator);
158 m_translator->deleteLater();
159 m_translator = nullptr;
160 }
161
162 // Qt 资源系统中的翻译文件路径
163 // 由 qt_add_translations 自动生成并注册到资源系统
164 // qt_add_translations 默认使用 /i18n 前缀
165 QString resourcePath = QString(":/i18n/translations_easykiconverter_%1.qm").arg(languageCode);
166
167 m_translator = new QTranslator(this);
168
169 // 尝试从 Qt 资源系统加载
170 if (m_translator->load(resourcePath)) {
171 app->installTranslator(m_translator);
172 qInfo() << "[LanguageManager] [OK] Loaded translator from resource:" << resourcePath;
173 return;
174 }
175
176 // 如果资源文件不存在，尝试从文件系统加载（开发时可能有用）
177 QString localPath = QString("resources/translations/translations_easykiconverter_%1.qm").arg(languageCode);
178 if (m_translator->load(localPath)) {
179 app->installTranslator(m_translator);
180 qInfo() << "[LanguageManager] [OK] Loaded translator from filesystem:" << localPath;
181 return;
182 }
183
184 qWarning() << "[LanguageManager] [FAIL] Cannot load translation:" << languageCode;
185 qWarning() << "[LanguageManager] Tried paths:";
186 qWarning() << " 1. Resource path:" << resourcePath;
187 qWarning() << " 2. File system path:" << localPath;
188 m_translator->deleteLater();
189 m_translator = nullptr;
190}
