52 QFileInfo urlInfo(datasheetUrl); // URL 路径信息。
53 if (!urlInfo.suffix().isEmpty()) {

55 }

57 QString fileName = QString("%1.%2").arg(componentName, fileExtension); // 输出文件名。
58 QString datasheetPath = QString("%1/%2").arg(datasheetDirPath, fileName); // 输出文件完整路径。
59
60 if (!m_options.overwriteExistingFiles && QFile::exists(datasheetPath)) {
61 qWarning() << "Datasheet already exists:" << datasheetPath;
62 return true;
63 }
64
65 // 使用统一网络客户端的默认重试策略下载数据手册。
66 NetworkResult result = NetworkClient::instance().get(QUrl(datasheetUrl), ResourceType::Datasheet); // 下载结果。
67
68 if (result.success) {
69 QFile file(datasheetPath); // 数据手册输出文件。
70 if (file.open(QIODevice::WriteOnly)) {
71 file.write(result.data);
72 file.close();
73 qDebug() << "Datasheet exported successfully:" << datasheetPath;
74 return true;
75 } else {
76 qWarning() << "Failed to open file for writing:" << datasheetPath;
77 return false;
78 }
79 } else {
80 qWarning() << "Failed to download datasheet:" << datasheetUrl << result.error;
81 return false;
