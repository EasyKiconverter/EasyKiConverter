31CadstarPadShape parsePadShape(const QString& value, ParseDiagnostics* diagnostics, int line) {
32 // 规范化大小写，兼容不同 Cadstar 导出器的命名风格。
33 const QString shape = value.trimmed().toUpper();
34 if (shape == QStringLiteral("ROUND") || shape == QStringLiteral("CIRCLE"))
35 return CadstarPadShape::Round;
36 if (shape == QStringLiteral("SQUARE"))
37 return CadstarPadShape::Square;
38 if (shape == QStringLiteral("RECT") || shape == QStringLiteral("RECTANGLE"))
39 return CadstarPadShape::Rectangle;
40 if (shape == QStringLiteral("OBLONG"))
41 return CadstarPadShape::Oblong;
42 if (shape == QStringLiteral("OCTAGON"))
43 return CadstarPadShape::Octagon;
44 if (shape == QStringLiteral("POLYGON") || shape == QStringLiteral("CUSTOM"))
45 return CadstarPadShape::Custom;
46 if (diagnostics)
47 diagnostics->add(ParseSeverity::Warning,
48 ParseScope::Field,
49 QStringLiteral("未知 Cadstar 焊盘形状：%1").arg(value),
50 value,
51 line);
52 return CadstarPadShape::Unknown;
53}

76QPointF point(const QStringList& values, int start, ParseDiagnostics* diagnostics, const QString& field, int line) {
77 if (start >= values.size()) {
78 if (diagnostics)
79 diagnostics->add(
80 ParseSeverity::Error, ParseScope::Field, QStringLiteral("Cadstar 坐标字段缺失"), field, line);
81 return {};
82 }
83
84 // 读取并清理首个坐标令牌，支持括号坐标和分离坐标两种写法。
85 const QString token = values.at(start).trimmed();
86 if (token.startsWith(QChar('(')) && token.endsWith(QChar(')'))) {
87 const QStringList pair =
88 token.mid(1, token.size() - 2).split(QRegularExpression(QStringLiteral("\\s+|,")), Qt::SkipEmptyParts);
89 if (pair.size() == 2)
90 return {number(pair.at(0), diagnostics, field + QStringLiteral(".X"), line),
91 number(pair.at(1), diagnostics, field + QStringLiteral(".Y"), line)};
92 } else if (start + 1 < values.size()) {
93 return {number(values.at(start), diagnostics, field + QStringLiteral(".X"), line),
94 number(values.at(start + 1), diagnostics, field + QStringLiteral(".Y"), line)};
95 }
96
97 if (diagnostics)
98 diagnostics->add(ParseSeverity::Error,
99 ParseScope::Field,
100 QStringLiteral("Cadstar 坐标格式错误：%1").arg(token),
101 field,
102 line);
103 return {};
104}

153QStringList expandNumbers(const QString& value, ParseDiagnostics* diagnostics, int line) {
154 // 匹配 [起始:结束:步长] 形式的编号范围。
155 static const QRegularExpression range(QStringLiteral(R"(^\[(\d+)\s*:\s*(\d+)(?:\s*:\s*(\d+))?\]$)"));
156 // 保存范围匹配结果。
157 const QRegularExpressionMatch match = range.match(value.trimmed());
158 if (!match.hasMatch())
159 return value.split(QRegularExpression(QStringLiteral("\\s*,\\s*")), Qt::SkipEmptyParts);
160
161 // 记录范围各部分的整数转换结果。
162 bool startOk = false; // 记录起始编号的整数转换结果。
163 bool endOk = false; // 记录结束编号的整数转换结果。
164 bool stepOk = true; // 记录步长的整数转换结果。
165 // 解析范围起点、终点和步长。
166 const int start = match.captured(1).toInt(&startOk); // 编号范围起点。
167 const int end = match.captured(2).toInt(&endOk); // 编号范围终点。
168 const int step = match.captured(3).isEmpty() ? 1 : match.captured(3).toInt(&stepOk); // 编号步长。
169 if (!startOk || !endOk || !stepOk || step <= 0 || start > end) {
170 if (diagnostics)
171 diagnostics->add(ParseSeverity::Error,
172 ParseScope::Field,
173 QStringLiteral("Cadstar 引脚编号范围非法：%1").arg(value),
174 QStringLiteral("NUMBERS"),
175 line);
176 return {};
177 }
178
179 // 按步长展开闭区间编号。
180 QStringList result;
181 for (int numberValue = start; numberValue <= end; numberValue += step)
182 result.append(QString::number(numberValue));
183 return result;
184}

193bool parseGraphic(const Node& node, CadstarGraphic& graphic, ParseDiagnostics* diagnostics) {
194 // 统一图形关键字大小写后进行类型匹配。
195 const QString keyword = node.keyword.toUpper();
196 if (keyword == QStringLiteral("LINE"))
197 graphic.type = CadstarGraphicType::Line;
198 else if (keyword == QStringLiteral("RECT"))
199 graphic.type = CadstarGraphicType::Rectangle;
200 else if (keyword == QStringLiteral("CIRCLE"))
201 graphic.type = CadstarGraphicType::Circle;
202 else if (keyword == QStringLiteral("ARC"))
203 graphic.type = CadstarGraphicType::Arc;
204 else if (keyword == QStringLiteral("POLYLINE"))
205 graphic.type = CadstarGraphicType::Polyline;
206 else if (keyword == QStringLiteral("POLYGON"))
207 graphic.type = CadstarGraphicType::Polygon;
208 else
209 return false;
210
211 graphic.points = points(node.arguments, diagnostics, keyword, node.line);
212 if ((graphic.type == CadstarGraphicType::Line || graphic.type == CadstarGraphicType::Rectangle ||
213 graphic.type == CadstarGraphicType::Arc) &&
214 graphic.points.size() < 2) {
215 if (diagnostics)
216 diagnostics->add(ParseSeverity::Error,
217 ParseScope::Field,
218 QStringLiteral("Cadstar 图形点数量不足：%1").arg(keyword),
219 keyword,
220 node.line);
221 return false;
222 }
223 if ((graphic.type == CadstarGraphicType::Polyline || graphic.type == CadstarGraphicType::Polygon) &&
224 graphic.points.size() < (graphic.type == CadstarGraphicType::Polygon ? 3 : 2)) {
225 if (diagnostics)
226 diagnostics->add(ParseSeverity::Error,
227 ParseScope::Field,
228 QStringLiteral("Cadstar 多点图形点数量不足：%1").arg(keyword),
229 keyword,
230 node.line);
231 return false;
232 }
233 if (graphic.type == CadstarGraphicType::Circle)
234 graphic.radius =
235 namedNumber(node.arguments, QStringLiteral("RADIUS"), diagnostics, QStringLiteral("RADIUS"), node.line);
236 graphic.width =
237 namedNumber(node.arguments, QStringLiteral("WIDTH"), diagnostics, QStringLiteral("WIDTH"), node.line);
238 return true;
239}

295QString mergedName(const QString& name, const QSet<QString>& usedNames) {
296 if (!usedNames.contains(name))
297 return name;
298 // 从 2 开始递增，保持同一输入的重命名结果稳定。
299 int suffix = 2;
300 // 保存当前尝试的候选名称。
301 QString candidate;
302 do {
303 candidate = QStringLiteral("%1_%2").arg(name).arg(suffix++);
304 } while (usedNames.contains(candidate));
305 return candidate;
306}

333template <typename Definition>
334void appendMergedDefinition(const Definition& source,
335 QList<Definition>& target,
336 QMap<QString, QStringList>& variants,
337 QSet<QString>& usedNames,
338 const QMap<QString, QStringList>& sourceVariants,
339 ParseDiagnostics* diagnostics,
340 ParseScope scope,
341 const QString& type) {
342 // 复制定义后在目标库命名空间中调整名称。
343 Definition merged = source;
344 // 保留跨文件合并前的原始名称，供诊断和关联使用。
345 const QString rawName = originalName(sourceVariants, source.name);
346 merged.name = mergedName(source.name, usedNames);
347 if (merged.name != source.name && diagnostics)
348 diagnostics->add(ParseSeverity::Warning,
349 scope,
350 QStringLiteral("跨文件%1名称重复，已重命名：%2 -> %3").arg(type, source.name, merged.name),
351 rawName);
352 usedNames.insert(merged.name);
353 target.append(merged);
354 variants[rawName].append(merged.name);
355}

429CadstarLibrary CadstarParser::parse(const QString& content, const QString& filePath) {
430 // 保存当前文件的所有定义、索引和解析诊断。
431 CadstarLibrary library;
432 library.diagnostics.setFilePath(filePath);
433 // 将文本解析为 Cadstar 顶层分段节点。
434 const QList<DelimitedSectionNode> sections =
435 DelimitedSectionParser::parse(content, kCadstarLeafKeywords, &library.diagnostics);
436
437 for (const Node& root : sections) {
438 if (root.keyword == QStringLiteral("UNITS")) {
439 library.unit = UnitConverter::parseUnit(firstArgument(root), &library.diagnostics);
440 continue;
441 }
442 if (root.keyword == QStringLiteral("PAD")) {
443 // 保存当前焊盘定义及其几何、钻孔参数。
444 CadstarPad pad;
445 pad.name = firstArgument(root);
446 for (const Node& child : root.children) {
447 if (child.keyword == QStringLiteral("SHAPE"))
448 pad.shape = parsePadShape(firstArgument(child), &library.diagnostics, child.line);
449 else if (child.keyword == QStringLiteral("DIAMETER"))
450 pad.diameter =
451 number(firstArgument(child), &library.diagnostics, QStringLiteral("PAD.DIAMETER"), child.line);
452 else if (child.keyword == QStringLiteral("WIDTH"))
453 pad.width =
454 number(firstArgument(child), &library.diagnostics, QStringLiteral("PAD.WIDTH"), child.line);
455 else if (child.keyword == QStringLiteral("HEIGHT"))
456 pad.height =
457 number(firstArgument(child), &library.diagnostics, QStringLiteral("PAD.HEIGHT"), child.line);
458 // 读取焊盘中心偏移，保持来源坐标语义。
459 else if (child.keyword == QStringLiteral("OFFSET")) {
460 pad.offsetX = number(
461 child.arguments.value(0), &library.diagnostics, QStringLiteral("PAD.OFFSET.X"), child.line);
462 pad.offsetY = number(
463 child.arguments.value(1), &library.diagnostics, QStringLiteral("PAD.OFFSET.Y"), child.line);
464 } else if (child.keyword == QStringLiteral("POLYGON") || child.keyword == QStringLiteral("POINTS"))
465 pad.polygon =
466 points(child.arguments, &library.diagnostics, QStringLiteral("PAD.POLYGON"), child.line);
467 // 读取圆孔或槽孔参数，非法数值由严格解析器诊断。
468 else if (child.keyword == QStringLiteral("HOLE")) {
469 // 读取钻孔形状，用于区分圆孔和槽孔。
470 const QString shape = child.arguments.value(0).toUpper();
471 if (shape == QStringLiteral("SLOT")) {
472 pad.holeWidth = number(child.arguments.value(1),
473 &library.diagnostics,
474 QStringLiteral("PAD.HOLE.WIDTH"),
475 child.line);
476 pad.holeHeight = number(child.arguments.value(2),
477 &library.diagnostics,
478 QStringLiteral("PAD.HOLE.HEIGHT"),
479 child.line);
480 } else {
481 pad.holeDiameter = number(child.arguments.value(1),
482 &library.diagnostics,
483 QStringLiteral("PAD.HOLE.DIAMETER"),
484 child.line);
485 }
486 }
487 }
488 if (pad.name.isEmpty())
489 library.diagnostics.add(
490 ParseSeverity::Error, ParseScope::Field, QStringLiteral("Cadstar 焊盘缺少名称"), {}, root.line);
491 library.pads.append(pad);
492 library.padNameVariants[pad.name].append(pad.name);
493 reportDuplicate(library.pads, pad.name, &library.diagnostics, QStringLiteral("焊盘"), root.line);
494 continue;
495 }
496 if (root.keyword == QStringLiteral("PACKAGE")) {
497 // 保存当前封装定义及其引脚和图形。
498 CadstarPackage packageModel;
499 packageModel.name = firstArgument(root);
500 for (const Node& child : root.children) {
501 if (child.keyword == QStringLiteral("DESCRIPTION"))
502 packageModel.description = child.arguments.join(QStringLiteral(" "));
503 else if (child.keyword == QStringLiteral("PROPERTY"))
504 packageModel.properties.insert(child.arguments.value(0),
505 child.arguments.mid(1).join(QStringLiteral(" ")));
506 // 解析封装引脚及其焊盘引用。
507 else if (child.keyword == QStringLiteral("PIN")) {
508 // 保存当前封装引脚及其焊盘关联。
509 CadstarPackagePin pin;
510 pin.number = firstArgument(child);
511 for (const Node& pinChild : child.children) {
512 if (pinChild.keyword == QStringLiteral("POSITION"))
513 pin.position = point(pinChild.arguments,
514 0,
515 &library.diagnostics,
516 QStringLiteral("PACKAGE.PIN.POSITION"),
517 pinChild.line);
518 else if (pinChild.keyword == QStringLiteral("ROTATION"))
519 pin.rotation = number(firstArgument(pinChild),
520 &library.diagnostics,
521 QStringLiteral("PACKAGE.PIN.ROTATION"),
522 pinChild.line);
523 else if (pinChild.keyword == QStringLiteral("PAD"))
524 pin.padName = firstArgument(pinChild);
525 }
526 if (pin.number.isEmpty())
527 library.diagnostics.add(ParseSeverity::Error,
528 ParseScope::Footprint,
529 QStringLiteral("Cadstar 封装引脚缺少编号"),
530 {},
531 child.line);
532 packageModel.pins.append(pin);
533 } else {
534 // 尝试将未被专门处理的子节点解析为封装图形。
535 CadstarGraphic graphic;
536 if (parseGraphic(child, graphic, &library.diagnostics))
537 packageModel.graphics.append(graphic);
538 }
539 }
540 if (packageModel.name.isEmpty())
541 library.diagnostics.add(
542 ParseSeverity::Error, ParseScope::Footprint, QStringLiteral("Cadstar 封装缺少名称"), {}, root.line);
543 library.packages.append(packageModel);
544 library.packageNameVariants[packageModel.name].append(packageModel.name);
545 reportDuplicate(
546 library.packages, packageModel.name, &library.diagnostics, QStringLiteral("封装"), root.line);
547 continue;
548 }
549 if (root.keyword == QStringLiteral("COMPONENT")) {
550 // 保存当前符号定义及其引脚和图形。
551 CadstarComponent component;
552 component.name = firstArgument(root);
553 for (const Node& child : root.children) {
554 if (child.keyword == QStringLiteral("VERSION"))
555 component.version = static_cast<int>(number(
556 firstArgument(child), &library.diagnostics, QStringLiteral("COMPONENT.VERSION"), child.line));
557 else if (child.keyword == QStringLiteral("PROPERTY"))
558 component.properties.insert(child.arguments.value(0),
559 child.arguments.mid(1).join(QStringLiteral(" ")));
560 // 解析符号引脚、编号范围和显示属性。
561 else if (child.keyword == QStringLiteral("PIN")) {
562 // 保存当前符号引脚及其编号、类型和标签。
563 CadstarComponentPin pin;
564 pin.id = firstArgument(child);
565 for (const Node& pinChild : child.children) {
566 if (pinChild.keyword == QStringLiteral("START"))
567 pin.start = point(pinChild.arguments,
568 0,
569 &library.diagnostics,
570 QStringLiteral("COMPONENT.PIN.START"),
571 pinChild.line);
572 else if (pinChild.keyword == QStringLiteral("END"))
573 pin.end = point(pinChild.arguments,
574 0,
575 &library.diagnostics,
576 QStringLiteral("COMPONENT.PIN.END"),
577 pinChild.line);
578 else if (pinChild.keyword == QStringLiteral("ROTATION"))
579 pin.rotation = number(firstArgument(pinChild),
580 &library.diagnostics,
581 QStringLiteral("COMPONENT.PIN.ROTATION"),
582 pinChild.line);
583 else if (pinChild.keyword == QStringLiteral("INVERTED"))
584 pin.inverted = true;
585 else if (pinChild.keyword == QStringLiteral("PINTYPE"))
586 pin.pinType = pinChild.arguments.join(QStringLiteral(" "));
587 else if (pinChild.keyword == QStringLiteral("NUMBERS"))
588 pin.numbers = expandNumbers(
589 pinChild.arguments.join(QStringLiteral(" ")), &library.diagnostics, pinChild.line);
590 // 保留引脚编号的显示状态。
591 else if (pinChild.keyword == QStringLiteral("NUMBER_VISIBLE"))
592 pin.numberVisible = number(firstArgument(pinChild),
593 &library.diagnostics,
594 QStringLiteral("COMPONENT.PIN.NUMBER_VISIBLE"),
595 pinChild.line) != 0.0;
596 // 读取引脚标签及其位置和可见性属性。
597 else if (pinChild.keyword == QStringLiteral("LABEL")) {
598 pin.label = firstArgument(pinChild);
599 pin.labelPosition = point(pinChild.arguments,
600 1,
601 &library.diagnostics,
602 QStringLiteral("COMPONENT.PIN.LABEL"),
603 pinChild.line);
604 if (pinChild.arguments.size() > 2)
605 pin.labelVisible = number(pinChild.arguments.at(2),
606 &library.diagnostics,
607 QStringLiteral("COMPONENT.PIN.LABEL_VISIBLE"),
608 pinChild.line) != 0.0;
609 }
610 }
611 component.pins.append(pin);
612 } else {
613 // 尝试将未被专门处理的子节点解析为符号图形。
614 CadstarGraphic graphic;
615 if (parseGraphic(child, graphic, &library.diagnostics))
616 component.graphics.append(graphic);
617 }
618 }
619 if (component.name.isEmpty())
620 library.diagnostics.add(
621 ParseSeverity::Error, ParseScope::Symbol, QStringLiteral("Cadstar 符号缺少名称"), {}, root.line);
622 library.components.append(component);
623 library.componentNameVariants[component.name].append(component.name);
624 reportDuplicate(
625 library.components, component.name, &library.diagnostics, QStringLiteral("符号"), root.line);
626 continue;
627 }
628 if (root.keyword == QStringLiteral("PART")) {
629 // 保存当前器件定义及其符号、封装关联。
630 CadstarPart part;
631 part.name = firstArgument(root);
632 for (const Node& child : root.children) {
633 if (child.keyword == QStringLiteral("COMPONENT"))
634 part.componentName = firstArgument(child);
635 else if (child.keyword == QStringLiteral("PACKAGE"))
636 part.packageName = firstArgument(child);
637 else if (child.keyword == QStringLiteral("DESCRIPTION"))
638 part.description = child.arguments.join(QStringLiteral(" "));
639 else if (child.keyword == QStringLiteral("PROPERTY"))
640 part.properties.insert(child.arguments.value(0), child.arguments.mid(1).join(QStringLiteral(" ")));
641 }
642 if (part.name.isEmpty() || part.componentName.isEmpty() || part.packageName.isEmpty())
643 library.diagnostics.add(ParseSeverity::Error,
644 ParseScope::Component,
645 QStringLiteral("Cadstar Part 缺少名称或关联"),
646 part.name,
647 root.line);
648 library.parts.append(part);
649 library.partNameVariants[part.name].append(part.name);
650 }
651 }
652
653 if (!library.isRecognized())
654 library.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("未识别的 Cadstar ASCII 库"));
655 return library;
656}

682CadstarLibrary CadstarMerger::merge(const QList<CadstarLibrary>& libraries, const QString& filePath) {
683 // 保存合并后的定义、名称索引和诊断。
684 CadstarLibrary merged;
685 merged.diagnostics.setFilePath(filePath);
686 if (libraries.isEmpty()) {
687 merged.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("没有可合并的 Cadstar 库"));
688 return merged;
689 }
690
691 // 分别记录各类定义已占用的目标名称。
692 QSet<QString> padNames; // 已占用的焊盘名称。
693 QSet<QString> packageNames; // 已占用的封装名称。
694 QSet<QString> componentNames; // 已占用的符号名称。
695 QSet<QString> partNames; // 已占用的器件名称。
696 for (const CadstarLibrary& library : libraries) {
697 merged.diagnostics.append(library.diagnostics);
698 if (merged.unit != LengthUnit::Unknown && library.unit != LengthUnit::Unknown && merged.unit != library.unit)
699 merged.diagnostics.add(
700 ParseSeverity::Error, ParseScope::File, QStringLiteral("Cadstar 库单位不一致，无法安全合并"), filePath);
701 else if (merged.unit == LengthUnit::Unknown)
702 merged.unit = library.unit;
703

743 merged.partNameVariants[rawName].append(mergedPart.name);

747 reportAmbiguousNames(

750 reportAmbiguousNames(merged.partNameVariants, &merged.diagnostics, ParseScope::Component, QStringLiteral("器件"));
751 if (!merged.isRecognized())
752 merged.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("合并结果没有 Cadstar 定义"));
