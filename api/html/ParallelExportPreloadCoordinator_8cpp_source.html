17void ParallelExportPreloadCoordinator::processNextBatch(ParallelExportService& service) {
18 // 网络服务缺失时按批次读取磁盘缓存，避免一次性阻塞事件循环。
19 if (service.m_cancelRequested || service.m_progress.currentStage != ExportOverallProgress::Stage::Preloading) {
20 return;
21 }
22
23 constexpr int kBatchSize = 8;
24 int processedInBatch = 0;
25 while (processedInBatch < kBatchSize && service.m_nextPreloadIndex < service.m_componentIds.size() &&
26 !service.m_cancelRequested) {
27 const QString componentId = service.m_componentIds.at(service.m_nextPreloadIndex++);
28 {
29 QMutexLocker locker(&service.m_progressMutex);
30 service.m_progress.preloadProgress.currentComponentId = componentId;
31 service.m_progress.preloadProgress.inProgressCount = 1;
32 }
33
34 ComponentData data;
35 if (service.m_componentService) {
36 data = service.m_componentService->getComponentData(componentId);
37 const QSharedPointer<ComponentData> diskCachedData = // 磁盘缓存中的组件数据。
38 ExportWorkerHelpers::loadDiskCachedComponentData(componentId);
39 ExportWorkerHelpers::mergeComponentData(data, diskCachedData);
40 } else {
41 const QSharedPointer<ComponentData> diskCachedData = // 磁盘缓存中的组件数据。
42 ExportWorkerHelpers::loadDiskCachedComponentData(componentId);
43 if (diskCachedData)
44 data = *diskCachedData;
45 }
46
47 {
48 QMutexLocker locker(&service.m_progressMutex);
49 // 数据。
50 const bool hasValidData = !data.lcscId().isEmpty() && data.isValid() && data.symbolData() != nullptr &&
51 data.footprintData() != nullptr;
52 if (hasValidData) {
53 service.m_cachedData[componentId] = QSharedPointer<ComponentData>::create(data);
54 service.m_progress.preloadProgress.successCount++;
55 qDebug() << "ParallelExportService: Loaded data for" << componentId;
56 } else {
57 service.m_progress.preloadProgress.failedCount++;
58 service.m_progress.preloadProgress.failedComponents[componentId] =
59 QStringLiteral("Incomplete component data");
60 qWarning() << "ParallelExportService: No valid data found for component:" << componentId;
61 }
62 service.m_progress.preloadProgress.completedCount++;
63 service.m_progress.preloadProgress.inProgressCount = 0;
64 }
65 ++processedInBatch;
66 }
67
68 const bool finished = service.m_nextPreloadIndex >= service.m_componentIds.size();
69 if (finished) {
70 QMutexLocker locker(&service.m_progressMutex);
71 service.m_progress.preloadProgress.currentComponentId.clear();
72 service.m_progress.currentStage = ExportOverallProgress::Stage::Idle;
73 service.m_progress.endTime = QDateTime::currentDateTime();
74 service.m_preloadCompleted = true;
75 }
76
77 qDebug() << "ParallelExportService: Preload batch completed. Success:"
78 << service.m_progress.preloadProgress.successCount
79 << "Failed:" << service.m_progress.preloadProgress.failedCount;
80 emit service.preloadProgressChanged(service.m_progress.preloadProgress);
81 service.updateOverallProgress();
82
83 if (finished) {
84 emit service.preloadCompleted(service.m_progress.preloadProgress.successCount,
85 service.m_progress.preloadProgress.failedCount);
86 return;
87 }
88 QTimer::singleShot(0, &service, [&service]() { processNextBatch(service); });
89}

97void ParallelExportPreloadCoordinator::handleCollectedData(ParallelExportService& service,
98 const QList<ComponentData>& componentDataList,
99 const QMap<QString, QString>& failedComponents) {
100 // 回调只处理当前预加载，取消后的迟到结果必须被忽略。
101 if (service.m_componentService) {
102 QObject::disconnect(
103 service.m_componentService, &ComponentService::allComponentsDataCollectedWithErrors, &service, nullptr);
104 }
105 {
106 QMutexLocker locker(&service.m_progressMutex);
107 if (service.m_cancelRequested) {
108 qDebug() << "ParallelExportService: Ignoring late callback after cancel requested";
109 return;
110 }
111 }
112
113 int successCount = 0;
114 int failedCount = failedComponents.size();
115 for (auto it = failedComponents.cbegin(); it != failedComponents.cend(); ++it) // 当前失败组件记录。
116 service.m_progress.preloadProgress.failedComponents[it.key()] = it.value();
117
118 for (const ComponentData& data : componentDataList) { // 当前成功获取的组件数据。
119 const QString componentId = data.lcscId();
120 if (componentId.isEmpty()) {
121 ++failedCount;
122 continue;
123 }
124
125 // 数据。
126 const bool hasValidData = data.isValid() && data.symbolData() != nullptr &&
127 data.footprintData() != nullptr;
128 if (hasValidData) {
129 service.m_cachedData[componentId] = QSharedPointer<ComponentData>::create(data);
130 ++successCount;
131 qDebug() << "ParallelExportService: Cached data for" << componentId;
132 } else {
133 ++failedCount;
134 service.m_progress.preloadProgress.failedComponents[componentId] = QStringLiteral("No valid data found");
135 qWarning() << "ParallelExportService: No valid data for component:" << componentId;
136 }
137 }
138
139 {
140 QMutexLocker locker(&service.m_progressMutex);
141 service.m_progress.preloadProgress.successCount = successCount;
142 service.m_progress.preloadProgress.failedCount = failedCount;
143 service.m_progress.preloadProgress.completedCount = successCount + failedCount;
144 service.m_progress.preloadProgress.inProgressCount = 0;
145 service.m_progress.preloadProgress.currentComponentId.clear();
146 service.m_progress.currentStage = ExportOverallProgress::Stage::Idle;
147 service.m_progress.endTime = QDateTime::currentDateTime();
148 }
149
150 qDebug() << "ParallelExportService: Parallel preload completed. Success:" << successCount
151 << "Failed:" << failedCount;
152 service.logNetworkRuntimeStats(QStringLiteral("preload-completed"));
153 service.writeExportDetailedReport(QStringLiteral("preload-completed"));
154 service.m_preloadCompleted = true;
155 emit service.preloadProgressChanged(service.m_progress.preloadProgress);
156 emit service.preloadCompleted(successCount, failedCount);
157}
