91QStringList expandPinNumbers(const QString& value, ParseDiagnostics* diagnostics, int line) {
92 QStringList numbers; // 展开后的编号列表。
93 const QStringList entries = value.split(QChar(','), Qt::KeepEmptyParts); // 逗号分隔的原始项。
94 const QRegularExpression rangeExpression(
95 QStringLiteral(R"(^\[\s*(\d+)\s*:\s*(\d+)(?:\s*:\s*(\d+))?\s*\]$)")); // 范围表达式。
96 for (const QString& rawEntry : entries) {
97 const QString entry = rawEntry.trimmed(); // 当前编号项。
98 if (entry.isEmpty()) {
99 if (diagnostics)
100 diagnostics->add(ParseSeverity::Error,
101 ParseScope::Symbol,
102 QStringLiteral("Xpedition 引脚编号范围包含空项"),
103 value,
104 line);
105 continue;
106 }
107 const QRegularExpressionMatch match = rangeExpression.match(entry); // 当前项的范围匹配结果。
108 if (!match.hasMatch()) {
109 if (entry.startsWith(QChar('[')) || entry.endsWith(QChar(']'))) {
110 if (diagnostics)
111 diagnostics->add(ParseSeverity::Error,
112 ParseScope::Symbol,
113 QStringLiteral("Xpedition 引脚编号范围格式非法"),
114 entry,
115 line);
116 } else {
117 numbers.append(entry);
118 }
119 continue;
120 }
121
122 bool startOk = false; // 起始值解析状态。
123 bool endOk = false; // 结束值解析状态。
124 bool stepOk = true; // 步长解析状态。
125 const int start = match.captured(1).toInt(&startOk); // 范围起点。
126 const int end = match.captured(2).toInt(&endOk); // 范围终点。
127 const int step = match.captured(3).isEmpty() ? 1 : match.captured(3).toInt(&stepOk); // 范围步长。
128 if (!startOk || !endOk || !stepOk || step <= 0 || end < start) {
129 if (diagnostics)
130 diagnostics->add(ParseSeverity::Error,
131 ParseScope::Symbol,
132 QStringLiteral("Xpedition 引脚编号范围边界非法"),
133 entry,
134 line);
135 continue;
136 }
137 for (int number = start; number <= end; number += step)
138 numbers.append(QString::number(number));
139 }
140 return numbers;
141}

175XpeditionSymbolDocument XpeditionSymbolParser::parse(const QString& content, const QString& filePath) {
176 XpeditionSymbolDocument document; // 当前符号文档及其诊断。
177 document.diagnostics.setFilePath(filePath);
178 XpeditionSymbolPin* currentPin = nullptr; // 当前正在读取属性的引脚。
179 int currentPart = 0; // 当前符号单元索引。
180 bool closeNextPolyline = false; // 下一条折线是否需要闭合。
181
182 const QStringList lines = content.split(QRegularExpression(QStringLiteral("\\r?\\n"))); // 按行拆分的源文本。
183 for (int index = 0; index < lines.size(); ++index) {
184 const int line = index + 1; // 当前源文件行号。
185 const QString raw = lines.at(index).trimmed(); // 去除外围空白后的命令行。
186 if (raw.isEmpty())
187 continue;
188 if (raw.startsWith(QChar('+'))) {
189 if (closeNextPolyline && !document.model.polylines.isEmpty())
190 document.model.polylines.last().closed = true;
191 closeNextPolyline = false;
192 continue;
193 }
194 if (raw.startsWith(QChar('|')))
195 continue;
196
197 const QString command = raw.left(1); // Xpedition 命令字符。
198 const QStringList values = fields(raw.mid(1)); // 命令参数字段。
199 if (command == QStringLiteral("V")) {
200 if (values.isEmpty() ||
201 !integer(values.first(), &document.diagnostics, QStringLiteral("V"), line, document.model.version))
202 diagnoseMalformed(&document.diagnostics, command, line);
203 } else if (command == QStringLiteral("K")) {
204 if (values.size() >= 2)
205 document.model.name = values.mid(1).join(QStringLiteral("_"));
206 else
207 diagnoseMalformed(&document.diagnostics, command, line);
208 } else if (command == QStringLiteral("F")) {
209 if (!values.isEmpty())
210 document.model.footprint = values.first();
211 } else if (command == QStringLiteral("Y")) {
212 if (values.isEmpty() ||
213 !integer(values.first(), &document.diagnostics, QStringLiteral("Y"), line, document.model.symbolType))
214 diagnoseMalformed(&document.diagnostics, command, line);
215 } else if (command == QStringLiteral("D")) {
216 if (values.size() < 4)
217 diagnoseMalformed(&document.diagnostics, command, line);
218 else
219 document.model.bounds = QRectF(point(values, 0, 1, &document.diagnostics, QStringLiteral("D"), line),
220 point(values, 2, 3, &document.diagnostics, QStringLiteral("D"), line));
221 } else if (command == QStringLiteral("Z")) {
222 if (values.isEmpty())
223 diagnoseMalformed(&document.diagnostics, command, line);
224 else
225 document.model.zoomLevel = number(values.first(), &document.diagnostics, QStringLiteral("Z"), line);
226 } else if (command == QStringLiteral("U")) {
227 const QString property = propertyValue(values); // U 命令的键值属性。
228 const int equal = property.indexOf(QChar('=')); // 键值分隔符位置。
229 if (equal > 0) {
230 const QString key = property.left(equal); // 属性名称。
231 const QString value = property.mid(equal + 1); // 属性值。
232 document.model.properties.insert(key, value);
233 if (key == QStringLiteral("PARTS"))
234 integer(value, &document.diagnostics, key, line, document.model.partCount);
235 else if (key == QStringLiteral("PART"))
236 integer(value, &document.diagnostics, key, line, currentPart);
237 else if (key == QStringLiteral("HETERO"))
238 document.model.heterogeneousParts =
239 value.split(QRegularExpression(QStringLiteral("[,()]")), Qt::SkipEmptyParts);
240 }
241 } else if (command == QStringLiteral("P")) {
242 if (values.size() < 8) {
243 diagnoseMalformed(&document.diagnostics, command, line);
244 currentPin = nullptr;
245 continue;
246 }
247 XpeditionSymbolPin pin; // 当前命令解析出的引脚。
248 pin.id = static_cast<int>(number(values.at(0), &document.diagnostics, QStringLiteral("P.ID"), line));
249 pin.start = point(values, 1, 2, &document.diagnostics, QStringLiteral("P"), line);
250 pin.end = point(values, 3, 4, &document.diagnostics, QStringLiteral("P"), line);
251 integer(values.at(6), &document.diagnostics, QStringLiteral("P.ROTATION"), line, pin.rotation);
252 int inverted = 0; // 反相标记数值。
253 integer(values.at(7), &document.diagnostics, QStringLiteral("P.INVERTED"), line, inverted);
254 pin.inverted = inverted != 0;
255 pin.partIndex = currentPart;
256 document.model.pins.append(pin);
257 currentPin = &document.model.pins.last();
258 } else if (command == QStringLiteral("A")) {
259 const QString property = propertyValue(values); // A 命令的键值属性。
260 const int equal = property.indexOf(QChar('=')); // 键值分隔符位置。
261 if (currentPin != nullptr && equal > 0) {
262 const QString key = property.left(equal); // 属性名称。
263 const QString value = property.mid(equal + 1); // 属性值。
264 if (key == QStringLiteral("PINTYPE"))
265 currentPin->pinType = value;
266 else if (key == QStringLiteral("#"))
267 currentPin->numbers = expandPinNumbers(value, &document.diagnostics, line);
268 else if (key == QStringLiteral("PART"))
269 integer(value, &document.diagnostics, key, line, currentPart);
270 }
271 if (currentPin != nullptr && values.size() > 5 && property.startsWith(QStringLiteral("#=")))
272 currentPin->numbersVisible = values.at(5) != QStringLiteral("0");
273 } else if (command == QStringLiteral("L")) {
274 if (currentPin == nullptr || values.size() < 7) {
275 diagnoseMalformed(&document.diagnostics, command, line);
276 continue;
277 }
278 currentPin->namePosition = point(values, 0, 1, &document.diagnostics, QStringLiteral("L"), line);
279 currentPin->nameSize = number(values.at(2), &document.diagnostics, QStringLiteral("L.SIZE"), line);
280 currentPin->nameRotation = number(values.at(3), &document.diagnostics, QStringLiteral("L.ROTATION"), line);
281 const int nameStart = values.size() >= 9 ? 8 : 6; // 名称文本起始字段。
282 const int visibilityIndex = values.size() >= 9 ? 6 : 5; // 可见性字段索引。
283 currentPin->nameVisible = values.at(visibilityIndex) != QStringLiteral("0");
284 currentPin->name = values.mid(nameStart).join(QStringLiteral(" "));
285 } else if (command == QStringLiteral("l")) {
286 if (values.size() < 5) {
287 diagnoseMalformed(&document.diagnostics, command, line);
288 continue;
289 }
290 int count = 0; // 折线顶点数量。
291 integer(values.first(), &document.diagnostics, QStringLiteral("l.COUNT"), line, count);
292 if (values.size() < 1 + count * 2) {
293 diagnoseMalformed(&document.diagnostics, command, line);
294 continue;
295 }
296 XpeditionSymbolPolyline polyline; // 当前折线图形。
297 polyline.partIndex = currentPart;
298 for (int valueIndex = 1; valueIndex < 1 + count * 2; valueIndex += 2)
299 polyline.points.append(
300 point(values, valueIndex, valueIndex + 1, &document.diagnostics, QStringLiteral("l"), line));
301 document.model.polylines.append(polyline);
302 closeNextPolyline = count >= 3;
303 } else if (command == QStringLiteral("b")) {
304 if (values.size() < 4) {
305 diagnoseMalformed(&document.diagnostics, command, line);
306 continue;
307 }
308 document.model.rectangles.append({point(values, 0, 1, &document.diagnostics, QStringLiteral("b"), line),
309 point(values, 2, 3, &document.diagnostics, QStringLiteral("b"), line),
310 currentPart});
311 } else if (command == QStringLiteral("c")) {
312 if (values.size() < 3) {
313 diagnoseMalformed(&document.diagnostics, command, line);
314 continue;
315 }
316 document.model.circles.append(
317 {point(values, 0, 1, &document.diagnostics, QStringLiteral("c"), line),
318 number(values.at(2), &document.diagnostics, QStringLiteral("c.RADIUS"), line),
319 currentPart});
320 } else if (command == QStringLiteral("a")) {
321 if (values.size() < 6) {
322 diagnoseMalformed(&document.diagnostics, command, line);
323 continue;
324 }
325 document.model.arcs.append({point(values, 0, 1, &document.diagnostics, QStringLiteral("a"), line),
326 point(values, 2, 3, &document.diagnostics, QStringLiteral("a"), line),
327 point(values, 4, 5, &document.diagnostics, QStringLiteral("a"), line),
328 currentPart});
329 } else if (command == QStringLiteral("T")) {
330 if (values.size() < 6) {
331 diagnoseMalformed(&document.diagnostics, command, line);
332 continue;
333 }
334 document.model.texts.append(
335 {point(values, 0, 1, &document.diagnostics, QStringLiteral("T"), line),
336 number(values.at(2), &document.diagnostics, QStringLiteral("T.SIZE"), line),
337 number(values.at(3), &document.diagnostics, QStringLiteral("T.ROTATION"), line),
338 static_cast<int>(number(values.at(4), &document.diagnostics, QStringLiteral("T.ORIGIN"), line)),
339 values.mid(5).join(QStringLiteral(" ")),
340 currentPart});
341 } else if (command == QStringLiteral("E")) {
342 currentPin = nullptr;
343 } else {
344 document.diagnostics.add(ParseSeverity::Skipped,
345 ParseScope::Symbol,
346 QStringLiteral("跳过不支持的 Xpedition 符号命令：%1").arg(command),
347 command,
348 line);
349 }
350 }
351 if (!document.isRecognized())
352 document.diagnostics.add(ParseSeverity::Error, ParseScope::File, QStringLiteral("未识别的 Xpedition 符号文件"));
353 return document;
354}
