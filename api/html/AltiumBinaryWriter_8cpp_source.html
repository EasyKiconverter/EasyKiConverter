115void AltiumBinaryWriter::endBlock() {
116 if (m_blockPositions.isEmpty()) {
117 return;
118 }
119
120 int blockStart = m_blockPositions.takeLast(); // 当前块头起始偏移。
121 int payloadSize = m_buffer.size() - blockStart - 4; // 块载荷字节数。
122 uint8_t flags = static_cast<uint8_t>(m_buffer[blockStart]); // 块标志位。
123
124 // 回填大小：(flags << 24) | payload_size
125 uint32_t sizeHeader = (static_cast<uint32_t>(flags) << 24) | (static_cast<uint32_t>(payloadSize) & 0x00FFFFFF);
126 m_buffer[blockStart] = static_cast<char>(sizeHeader & 0xFF);
127 m_buffer[blockStart + 1] = static_cast<char>((sizeHeader >> 8) & 0xFF);
128 m_buffer[blockStart + 2] = static_cast<char>((sizeHeader >> 16) & 0xFF);
129 m_buffer[blockStart + 3] = static_cast<char>((sizeHeader >> 24) & 0xFF);
130}

171void AltiumBinaryWriter::writeCStringParameterBlock(const QMap<QString, QString>& params) {
172 QString paramStr; // 拼接后的参数字符串。
173 QList<QString> orderedKeys; // 按协议优先级排列的参数键。
174 for (const QString& preferred : {QStringLiteral("HEADER"),
175 QStringLiteral("RECORD"),
176 QStringLiteral("PATTERN"),
177 QStringLiteral("ISNOTACCESIBLE"),
178 QStringLiteral("IndexInSheet"),
179 QStringLiteral("OWNERPARTID"),
180 QStringLiteral("OwnerPartId")}) {
181 if (params.contains(preferred))
182 orderedKeys.append(preferred);
183 }
184 for (auto it = params.constBegin(); it != params.constEnd(); ++it) {
185 if (!orderedKeys.contains(it.key()))
186 orderedKeys.append(it.key());
187 }
188 for (const QString& key : orderedKeys) {
189 auto it = params.constFind(key); // 当前参数迭代器。
190 if (it == params.constEnd())
191 continue;
192 QString value = it.value(); // 当前字段对应的文本值。
193 value.replace('|', ' ');
194 value.replace(QChar::Null, ' ');
195 paramStr += "|" + it.key() + "=" + value;
196 }
197 paramStr += "|";
198 writeCStringParameterBlockRaw(paramStr);
199}

205void AltiumBinaryWriter::writeCStringParameterBlockUtf8(const QMap<QString, QString>& params) {
206 QByteArray encoded; // 已编码的参数字节。
207 // Altium 的参数块虽然语义上是键值集合，但记录识别器会读取首个
208 // 参数来判断记录类型。QMap 的字典序会把 ALLPINCOUNT/AREACOLOR
209 // 放在 RECORD/HEADER 前面，从而使合法数据被当作未知记录。保持少量
210 // 协议首字段的稳定顺序，其余字段仍使用 QMap 的确定性顺序。
211 QList<QString> orderedKeys;
212 for (const QString& preferred : {QStringLiteral("HEADER"),
213 QStringLiteral("RECORD"),
214 QStringLiteral("PATTERN"),
215 QStringLiteral("ISNOTACCESIBLE"),
216 QStringLiteral("IndexInSheet"),
217 QStringLiteral("OWNERPARTID"),
218 QStringLiteral("OwnerPartId")}) {
219 if (params.contains(preferred))
220 orderedKeys.append(preferred);
221 }
222 for (auto it = params.constBegin(); it != params.constEnd(); ++it) {
223 if (!orderedKeys.contains(it.key()))
224 orderedKeys.append(it.key());
225 }
226
227 for (const QString& keyName : orderedKeys) {
228 auto it = params.constFind(keyName); // 当前参数迭代器。
229 if (it == params.constEnd())
230 continue;
231 const QByteArray ansiKey = it.key().toLatin1(); // ANSI 参数键。
232 const QByteArray utf8Key = it.key().toUtf8(); // UTF-8 参数键。
233 QString value = it.value(); // 当前字段对应的文本值。
234 value.replace('|', ' ');
235 value.replace(QChar::Null, ' ');
236 const QByteArray ansiValue = value.toLatin1(); // 当前字段转换后的 ANSI 字节序列。
237 encoded += '|';
238 encoded += ansiKey;
239 encoded += '=';
240 encoded += ansiValue;
241
242 // %UTF8% 参数的键和值都必须使用原始 UTF-8 字节，而不是再次 Latin-1 转码。
243 if (QString::fromLatin1(ansiValue) != value) {
244 encoded += "|%UTF8%";
245 encoded += utf8Key;
246 encoded += '=';
247 encoded += value.toUtf8();
248 }
249 }
250 encoded += '|';
251 writeCStringParameterBlockRaw(encoded);
252}
