16static QString resolveOutputPath(const QString& configuredPath, const QString& libName) {
17 // 保存待解析的输出路径。
18 QString outputPath = configuredPath;
19 // 获取用户文档目录作为相对路径的默认根目录。
20 const QString documentsPath = QStandardPaths::writableLocation(QStandardPaths::DocumentsLocation);
21 QDir exportRoot(documentsPath);
22 if (!exportRoot.exists(QStringLiteral("EasyKiConverter")))
23 exportRoot.mkdir(QStringLiteral("EasyKiConverter"));
24 exportRoot.cd(QStringLiteral("EasyKiConverter"));
25
26 if (outputPath.isEmpty())
27 return exportRoot.absoluteFilePath(libName);
28
29 QDir outputDir(outputPath);
30 if (outputDir.isAbsolute())
31 return outputDir.cleanPath(outputPath);
32 return exportRoot.absoluteFilePath(outputPath);
33}

40static const char* targetFormatName(TargetEdaFormat format) {

69ExportOptions ExportOptionsBuilder::build(const ExportSettingsViewModel& viewModel) {
70 // 保存组装后的导出选项。
71 ExportOptions options;
72 options.outputPath = resolveOutputPath(viewModel.m_outputPath, viewModel.m_libName);
73 if (options.outputPath.isEmpty()) {
74 // 获取桌面目录作为输出路径的最终回退位置。
75 const QString desktopPath = QStandardPaths::writableLocation(QStandardPaths::DesktopLocation);
76 options.outputPath = QDir(desktopPath).absoluteFilePath(viewModel.m_libName);
77 }
78
79 options.libName = viewModel.m_libName;
80 options.exportSymbol = viewModel.m_exportSymbol;
81 options.exportFootprint = viewModel.m_exportFootprint;
82 options.exportModel3D = viewModel.m_exportModel3D;
83 options.exportModel3DFormat = viewModel.m_exportModel3DFormat;
84 options.exportModel3DPathMode = viewModel.m_exportModel3DPathMode;
85 options.exportPreviewImages = viewModel.m_exportPreviewImages;
86 options.exportDatasheet = viewModel.m_exportDatasheet;
87 options.weakNetworkSupport = viewModel.m_weakNetworkSupport;
88 options.debugMode = viewModel.m_debugMode;
89 options.exportSymbolDescription = viewModel.m_exportSymbolDescription;
90 options.exportFootprintDescription = viewModel.m_exportFootprintDescription;
91 options.symbolLibraryDescription = viewModel.m_symbolLibraryDescription;
92 options.footprintLibraryDescription = viewModel.m_footprintLibraryDescription;
93 options.footprintLibraryKeywords = viewModel.m_footprintLibraryKeywords;
94 options.targetFormat = viewModel.m_targetModel
95 ? static_cast<TargetEdaFormat>(viewModel.m_targetModel->currentTargetFormat())
96 : TargetEdaFormat::KiCad;
97 // Xpedition ZIP 只能完整替换，界面的“覆盖”选项不能被误传成更新模式。
98 // 标记各类目标格式是否需要特殊的完整替换策略。
99 const bool xpeditionTarget = options.targetFormat == TargetEdaFormat::Xpedition;
100 // 标记是否选择 Allegro 目标格式。
101 const bool allegroTarget = options.targetFormat == TargetEdaFormat::Allegro;
102 // 标记是否选择 PADS 目标格式。
103 const bool padsTarget = options.targetFormat == TargetEdaFormat::Pads;
104 // 标记是否选择 Eagle 目标格式。
105 const bool eagleTarget = options.targetFormat == TargetEdaFormat::Eagle;
106 // 标记是否选择 P-CAD 目标格式。
107 const bool pcadTarget = options.targetFormat == TargetEdaFormat::Pcad;
108 // 标记是否选择 OrCAD 目标格式。
109 const bool orcadTarget = options.targetFormat == TargetEdaFormat::Orcad;
110 // 标记是否选择 LibrePCB 目标格式。
111 const bool librePcbTarget = options.targetFormat == TargetEdaFormat::LibrePcb;
112 // 汇总所有仅支持完整替换的目标格式。
113 const bool requiresFullReplacement = xpeditionTarget || allegroTarget || padsTarget || eagleTarget || pcadTarget ||
114 orcadTarget || librePcbTarget ||
115 options.targetFormat == TargetEdaFormat::Cadstar;
116 if (allegroTarget)
117 options.exportSymbol = false;
118 if (orcadTarget)
119 options.exportFootprint = false;
120 options.overwriteExistingFiles = viewModel.m_overwriteExistingFiles || viewModel.m_exportMode == 2;
121 options.updateMode = viewModel.m_exportMode == 1 && !requiresFullReplacement;
122
123 qInfo() << "Export options:" << "OutputPath:" << options.outputPath << "LibName:" << options.libName
124 << "TargetFormat:" << targetFormatName(options.targetFormat) << "Symbol:" << options.exportSymbol
125 << "Footprint:" << options.exportFootprint << "3D Model:" << options.exportModel3D
126 << "3D Model Format:" << options.exportModel3DFormat << "(1=WRL, 2=STEP, 3=Both)"
127 << "3D Model Path Mode:" << options.exportModel3DPathMode << "(0=Relative, 1=Absolute)"
128 << "Preview Images:" << options.exportPreviewImages << "Datasheet:" << options.exportDatasheet
129 << "Client Weak Network Adaptation:" << options.weakNetworkSupport << "Update Mode:" << options.updateMode
130 << "Debug Mode:" << options.debugMode << "Symbol Description:" << options.exportSymbolDescription
131 << "Footprint Description:" << options.exportFootprintDescription
132 << "Symbol Library Description:" << options.symbolLibraryDescription
133 << "Footprint Library Description:" << options.footprintLibraryDescription;
134 return options;
135}
