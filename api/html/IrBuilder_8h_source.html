76 auto convertPins = [&](const QList<SymbolPin>& pins,
77 double originX,
78 double originY,
79 int partIdx = 0,
80 bool commonToAllParts = false) {
81 for (const auto& pin : pins) {
82 SymbolPinIR pir; // 当前引脚的符号 IR。
83 pir.name = pin.name.text;
84 pir.designator = pin.settings.spicePinNumber;
85 pir.position = GeometryNormalizer::transformPoint(pin.settings.posX, pin.settings.posY, originX, originY);
86 // 只有显示且坐标有限的名称文本才使用独立位置。
87 const bool hasNamePosition = pin.name.isDisplayed &&
88 !pin.name.text.isEmpty() && // 是否有有效的引脚名称坐标。
89 std::isfinite(pin.name.posX) && std::isfinite(pin.name.posY);
90 pir.namePosition = hasNamePosition
91 ? GeometryNormalizer::transformPoint(pin.name.posX, pin.name.posY, originX, originY)
92 : pir.position;
93 pir.nameRotation = pin.name.rotation;
94 pir.nameAnchor = pin.name.textAnchor;
95 pir.nameFontSizeMm = pin.name.fontSize * EASYEDA_PX_TO_MM;
96 pir.hasNamePosition = hasNamePosition;
97 // 只有显示且坐标有限的编号文本才使用独立位置。
98 const bool hasNumberPosition = pin.number.isDisplayed &&
99 !pin.number.text.isEmpty() && // 是否有有效的引脚编号坐标。
100 std::isfinite(pin.number.posX) && std::isfinite(pin.number.posY);
101 pir.numberPosition = hasNumberPosition ? GeometryNormalizer::transformPoint(
102 pin.number.posX, pin.number.posY, originX, originY)
103 : pir.position;
104 pir.numberRotation = pin.number.rotation;
105 pir.numberAnchor = pin.number.textAnchor;
106 pir.numberFontSizeMm = pin.number.fontSize * EASYEDA_PX_TO_MM;
107 pir.hasNumberPosition = hasNumberPosition;
108
109 // 从 pinPath SVG 解析引脚长度
110 const QString pathStr = pin.pinPath.path; // 引脚 SVG 路径文本。
111 static const QRegularExpression lengthExpression(
112 QStringLiteral(R"([hHvV]\s*([-+]?(?:\d+\.?\d*|\.\d+)))")); // 引脚长度匹配表达式。
113 const QRegularExpressionMatch lengthMatch = lengthExpression.match(pathStr); // 匹配到的长度字段。
114 if (lengthMatch.hasMatch()) {
115 bool ok = false; // 长度字段是否解析成功。
116 const double len = lengthMatch.captured(1).toDouble(&ok); // SVG 路径中的原始长度。
117 if (ok)
118 pir.length = qAbs(len) * EASYEDA_PX_TO_MM;
119 }
120
121 pir.direction = EasyedaPinTypeMap::toPinDirection(pin.settings.rotation);
122 pir.electricalType = EasyedaPinTypeMap::toPinElectricalType(static_cast<int>(pin.settings.type));
123
124 // 显示控制
125 pir.showName = pin.name.isDisplayed;
126 pir.showDesignator = pin.settings.isDisplayed;
127 pir.display.showName = pin.name.isDisplayed;
128 pir.display.showDesignator = pin.settings.isDisplayed;
129
130 // 引脚样式（语义层）
131 pir.hasDot = pin.dot.isDisplayed;
132 pir.hasClock = pin.clock.isDisplayed;
133 pir.style.inverted = pin.dot.isDisplayed;
134 pir.style.clock = pin.clock.isDisplayed;
135 if (pin.dot.isDisplayed && pin.clock.isDisplayed) {
136 pir.style.decoration = PinDecoration::InvertedClock;
137 } else if (pin.dot.isDisplayed) {
138 pir.style.decoration = PinDecoration::Dot;
139 } else if (pin.clock.isDisplayed) {
140 pir.style.decoration = PinDecoration::Clock;
141 }
142
143 pir.partIndex = commonToAllParts ? -1 : partIdx;
144 pir.commonToAllParts = commonToAllParts;
145 ir.pins.append(pir);
146 }

236 // 转换辅助 lambda：处理路径
237 auto convertPaths = [&](const QList<SymbolPath>& paths, double originX, double originY, int partIdx = 0) {
238 for (const auto& path : paths) {
239 SymbolPathIR pathIR; // 当前路径的符号 IR。
240 pathIR.points = GeometryNormalizer::parseSimpleSvgPath(path.paths);
241 QPointF originMm(originX * EASYEDA_PX_TO_MM, originY * EASYEDA_PX_TO_MM); // 已缩放的局部原点。
242 pathIR.points = GeometryNormalizer::transformPoints(pathIR.points, originMm);
243 const auto transformSegmentPoint = [&](const QPointF& point) {
244 return GeometryNormalizer::transformPoints({point * EASYEDA_PX_TO_MM}, originMm).value(0);
245 };
246 for (const SvgPathSegment& sourceSegment : SvgPathParser::parseSegments(path.paths)) {
247 SymbolPathSegmentIR segment; // 当前路径片段的符号 IR。
248 if (sourceSegment.type == SvgPathSegment::Type::QuadraticBezier)
249 segment.type = SymbolPathSegmentIR::Type::QuadraticBezier;
250 else if (sourceSegment.type == SvgPathSegment::Type::CubicBezier)
251 segment.type = SymbolPathSegmentIR::Type::CubicBezier;
252 else if (sourceSegment.type == SvgPathSegment::Type::CircularArc)
253 segment.type = SymbolPathSegmentIR::Type::CircularArc;
254 else if (sourceSegment.type == SvgPathSegment::Type::EllipticalArc)
255 segment.type = SymbolPathSegmentIR::Type::EllipticalArc;
256 else
257 segment.type = SymbolPathSegmentIR::Type::Line;
258 segment.start = transformSegmentPoint(sourceSegment.start);
259 segment.end = transformSegmentPoint(sourceSegment.end);
260 if (segment.type == SymbolPathSegmentIR::Type::QuadraticBezier ||
261 segment.type == SymbolPathSegmentIR::Type::CubicBezier) {
262 segment.control1 = transformSegmentPoint(sourceSegment.control1);
263 }
264 if (segment.type == SymbolPathSegmentIR::Type::CubicBezier) {
265 segment.control2 = transformSegmentPoint(sourceSegment.control2);
266 } else if (segment.type == SymbolPathSegmentIR::Type::CircularArc) {
267 segment.arcMid = transformSegmentPoint(sourceSegment.arcMid);
268 } else if (segment.type == SymbolPathSegmentIR::Type::EllipticalArc) {
269 segment.arcCenter = transformSegmentPoint(sourceSegment.arcCenter);
270 segment.radiusX = sourceSegment.radiusX * EASYEDA_PX_TO_MM;
271 segment.radiusY = sourceSegment.radiusY * EASYEDA_PX_TO_MM;
272 segment.arcStartAngle = -sourceSegment.arcStartAngle;
273 segment.arcEndAngle = -sourceSegment.arcEndAngle;
274 }
275 pathIR.segments.append(segment);
276 }
277 // SVG Z 命令闭合路径
278 if (path.paths.contains('Z', Qt::CaseInsensitive) && pathIR.points.size() >= 2) {
279 if (pathIR.points.first() != pathIR.points.last()) {
280 pathIR.points.append(pathIR.points.first());
281 }
282 }
283 pathIR.strokeColor = parseColor(path.strokeColor);
284 pathIR.strokeWidth = path.strokeWidth * EASYEDA_PX_TO_MM;
285 pathIR.strokeStyle = GeometryNormalizer::parseStrokeStyle(path.strokeStyle);
286 pathIR.isFilled = path.fillColor;
287 pathIR.partIndex = partIdx;
288 ir.paths.append(pathIR);
289 }
