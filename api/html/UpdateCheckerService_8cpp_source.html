34SemanticVersion parseVersion(const QString& value) {
35 const QRegularExpression expression(
36 QStringLiteral("^v?(\\d+)(?:\\.(\\d+))?(?:\\.(\\d+))?(?:-([0-9A-Za-z-]+(?:\\.[0-9A-Za-z-]+)*))?(?:\\+[0-9A-Za-"
37 "z-]+(?:\\.[0-9A-Za-z-]+)*)?$"));
38 const QRegularExpressionMatch match = expression.match(value.trimmed());
39 if (!match.hasMatch())
40 return {};
41
42 SemanticVersion result;
43 result.major = match.captured(1).toInt();
44 result.minor = match.captured(2).isEmpty() ? 0 : match.captured(2).toInt();
45 result.patch = match.captured(3).isEmpty() ? 0 : match.captured(3).toInt();
46 result.prerelease = match.captured(4).isEmpty() ? QStringList() : match.captured(4).split('.');
47 result.valid = true;
48 return result;
49}

56int compareVersions(const SemanticVersion& left, const SemanticVersion& right) {
57 if (left.major != right.major)
58 return left.major < right.major ? -1 : 1;
59 if (left.minor != right.minor)
60 return left.minor < right.minor ? -1 : 1;
61 if (left.patch != right.patch)
62 return left.patch < right.patch ? -1 : 1;
63 if (left.prerelease.isEmpty() && right.prerelease.isEmpty())
64 return 0;
65 if (left.prerelease.isEmpty())
66 return 1;
67 if (right.prerelease.isEmpty())
68 return -1;
69
70 const QRegularExpression numeric(QStringLiteral("^[0-9]+$"));
71 const int count = qMax(left.prerelease.size(), right.prerelease.size());
72 for (int index = 0; index < count; ++index) {
73 if (index >= left.prerelease.size())
74 return -1;
75 if (index >= right.prerelease.size())
76 return 1;
77 const QString& leftPart = left.prerelease.at(index);
78 const QString& rightPart = right.prerelease.at(index);
79 const bool leftNumeric = numeric.match(leftPart).hasMatch();
80 const bool rightNumeric = numeric.match(rightPart).hasMatch();
81 if (leftNumeric && rightNumeric && leftPart.toLongLong() != rightPart.toLongLong())
82 return leftPart.toLongLong() < rightPart.toLongLong() ? -1 : 1;
83 if (leftNumeric != rightNumeric)
84 return leftNumeric ? -1 : 1;
85 if (leftPart != rightPart)
86 return leftPart < rightPart ? -1 : 1;
87 }
88 return 0;
89}

133QString chooseAsset(const QJsonArray& assets) {

136 for (const QJsonValue& value : assets) {

241void UpdateCheckerService::checkForUpdates(bool force) {
242 if (m_checking)
243 return;
244
245 ConfigService* config = ConfigService::instance();
246 const qint64 now = QDateTime::currentSecsSinceEpoch();
247 const qint64 interval = static_cast<qint64>(config->getUpdateCheckIntervalHours()) * 3600;
248 if (!force && !config->getUpdateAutoCheck())
249 return;
250 if (!force && config->getUpdateLastCheckTime() > 0 && now - config->getUpdateLastCheckTime() < interval)
251 return;
252
253 config->setUpdateLastCheckTime(now);
254 emit metadataChanged();
255 setError(QString());
256 setStatus(Status::Checking);
257 setChecking(true);
258
259 const RetryPolicy policy =
260 RetryPolicy::fromProfile(RequestProfiles::updateCheck(), config->getWeakNetworkSupport());
261 m_activeRequest =
262 m_networkClient->getAsync(QUrl(QString::fromLatin1(RELEASES_URL)), ResourceType::UpdateCheck, policy);
263 connect(m_activeRequest, &AsyncNetworkRequest::finished, this, [this](const NetworkResult& result) {
264 if (m_activeRequest) {
265 m_activeRequest->deleteLater();
266 m_activeRequest = nullptr;
267 }
268 setChecking(false);
269 if (result.wasCancelled) {
270 m_rateLimited = false;
271 setError(QStringLiteral("Update check was cancelled"));
272 setStatus(Status::NotChecked);
273 return;
274 }
275
276 if (!result.success) {
277 loadCachedRelease();
278 m_rateLimited = result.statusCode == 403 || result.statusCode == 429 ||
279 result.diagnostic.errorType == NetworkErrorType::Forbidden ||
280 result.diagnostic.errorType == NetworkErrorType::RateLimited ||
281 result.diagnostic.wasRateLimited;
282 setError(result.error.isEmpty() ? QStringLiteral("Update check failed") : result.error);
283 setStatus(Status::Failed);
284 return;
285 }
286
287 m_rateLimited = false;
288 QJsonParseError parseError;
289 const QJsonDocument document = QJsonDocument::fromJson(result.data, &parseError);
290 if (parseError.error != QJsonParseError::NoError || !document.isObject()) {
291 setError(QStringLiteral("Invalid release response"));
292 setStatus(Status::Failed);
293 return;
294 }
295 applyRelease(document.object(), false);
296 });
297}

357QString UpdateCheckerService::normalizeVersion(const QString& version) {
358 const SemanticVersion parsed = parseVersion(version);
359 if (!parsed.valid)
360 return QString();
361 // 规范化后的 SemVer 文本。
362 QString normalized = QStringLiteral("%1.%2.%3").arg(parsed.major).arg(parsed.minor).arg(parsed.patch);
363 if (!parsed.prerelease.isEmpty())
364 normalized += QStringLiteral("-") + parsed.prerelease.join('.');
365 const int buildMarker = version.indexOf('+');
366 if (buildMarker >= 0)
367 normalized += version.mid(buildMarker).trimmed();
368 return normalized;
369}

387bool UpdateCheckerService::validateRelease(const QJsonObject& release, QString* reason) const {
388 const QString tag = release.value(QStringLiteral("tag_name")).toString();
389 const QString name = release.value(QStringLiteral("name")).toString();
390 const QUrl url(release.value(QStringLiteral("html_url")).toString());
391 if (release.value(QStringLiteral("draft")).toBool(false) ||
392 release.value(QStringLiteral("prerelease")).toBool(false)) {
393 if (reason)
394 *reason = QStringLiteral("Release is not a stable published release");
395 return false;
396 }
397 if (normalizeVersion(tag).isEmpty() || name.isEmpty() || !url.isValid() ||
398 url.scheme() != QStringLiteral("https")) {
399 if (reason)
400 *reason = QStringLiteral("Incomplete release metadata");
401 return false;
402 }
403 if (!release.value(QStringLiteral("assets")).isArray()) {
404 if (reason)
405 *reason = QStringLiteral("Release assets are missing");
406 return false;
407 }
408 return true;
409}

416bool UpdateCheckerService::applyRelease(const QJsonObject& release, bool fromCache) {
417 QString reason;
418 if (!validateRelease(release, &reason)) {
419 setError(reason);
420 setStatus(Status::Failed);
421 return false;
422 }
423 const QString version = normalizeVersion(release.value(QStringLiteral("tag_name")).toString());
424 ConfigService* config = ConfigService::instance();
425 const QString ignored = config->getUpdateIgnoredVersion();
426 const QString reminded = config->getUpdateRemindedVersion();
427 config->beginBatchUpdate();
428 if (!fromCache) {
429 config->setUpdateCachedRelease(release);
430 config->setUpdateLastSuccessfulCheckTime(QDateTime::currentSecsSinceEpoch());
431 }
432 if (!ignored.isEmpty() && isRemoteVersionNewer(ignored, version))
433 config->setUpdateIgnoredVersion(QString());
434 if (!reminded.isEmpty() && isRemoteVersionNewer(reminded, version))
435 config->setUpdateRemindedVersion(QString());
436 config->endBatchUpdate();
437
438 applyUpdateInfo(version,
439 release.value(QStringLiteral("name")).toString(),
440 release.value(QStringLiteral("html_url")).toString(),
441 chooseAsset(release.value(QStringLiteral("assets")).toArray()));
442 setError(QString());
443 setStatus(m_hasUpdate ? (m_dismissed ? Status::Ignored : Status::UpdateAvailable) : Status::UpToDate);
444 emit metadataChanged();
445 return true;
446}

477void UpdateCheckerService::applyUpdateInfo(const QString& version,
478 const QString& name,
479 const QString& url,
480 const QString& asset) {
481 const QString ignoredVersion = ConfigService::instance()->getUpdateIgnoredVersion();
482 const QString remindedVersion = ConfigService::instance()->getUpdateRemindedVersion();
483 const bool oldState = m_hasUpdate;
484 const QString oldVersion = m_latestVersion;
485 const bool oldDismissed = m_dismissed;
486 m_latestVersion = version;
487 m_releaseName = name;
488 m_releaseUrl = url;
489 m_assetUrl = asset;
490 m_hasUpdate = isRemoteVersionNewer(currentVersion(), version);
491 m_dismissed = m_hasUpdate && ((!ignoredVersion.isEmpty() && ignoredVersion == version) ||
492 (!remindedVersion.isEmpty() && remindedVersion == version));
493 if (oldState != m_hasUpdate || oldVersion != m_latestVersion || oldDismissed != m_dismissed)
494 emit updateStateChanged();
