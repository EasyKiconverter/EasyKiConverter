18QString CachePathResolver::componentDir(const QString& cacheRoot, const QString& componentId) {
19 if (!BomParser::validateId(componentId)) {
20 return QString();
21 }
22 const QString normalizedId = componentId.toUpper(); // 统一大小写后的元件编号。
23 const QString normalizedPath = QDir::cleanPath(cacheRoot + "/" + normalizedId); // 规范化后的默认缓存路径。
24 if (QFileInfo::exists(normalizedPath)) {
25 return normalizedPath;
26 }
27
28 const QDir rootDir(cacheRoot);
29 for (const QString& entry : rootDir.entryList(QDir::Dirs | QDir::NoDotAndDotDot)) {
30 if (entry.compare(normalizedId, Qt::CaseInsensitive) == 0) {
31 return QDir::cleanPath(rootDir.absoluteFilePath(entry));
32 }
33 }
34 return normalizedPath;
35}
