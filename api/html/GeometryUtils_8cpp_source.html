46QPointF GeometryUtils::getArcCenter(double startX,
47 double startY,
48 double endX,
49 double endY,
50 int rotationDirection,
51 double radius) {
52 double arcDistance = distance(startX, startY, endX, endY); // 端点间距离。
53
54 // 记录端点之间的距离，用于归一化圆弧方向向量。
55 if (arcDistance < std::numeric_limits<double>::epsilon()) {
56 qWarning() << "Arc distance is too small, cannot calculate center";
57 return QPointF(startX, startY);
58 }
59
60 double mX = (startX + endX) / 2.0; // 端点中点 X 坐标。
61 double mY = (startY + endY) / 2.0; // 端点中点 Y 坐标。
62
63 double u = (endX - startX) / arcDistance; // 端点方向的 X 单位分量。
64 double v = (endY - startY) / arcDistance; // 端点方向的 Y 单位分量。
65
66 double hSquared = radius * radius - (arcDistance * arcDistance) / 4.0; // 圆心到弦中点距离的平方。
67
68 // 检查半径是否足以连接两个端点。
69 if (hSquared < 0) {
70 qWarning() << "Radius is too small for the given arc distance";
71 return QPointF(mX, mY);
72 }
73
74 double h = std::sqrt(hSquared); // 圆心到弦中点的距离。
75
76 double centerX = mX - rotationDirection * h * v; // 圆心 X 坐标。
77 double centerY = mY + rotationDirection * h * u; // 圆心 Y 坐标。
78
79 return QPointF(centerX, centerY);
80}

217void GeometryUtils::computeArc(double startX,
218 double startY,
219 double radiusX,
220 double radiusY,
221 double angle,
222 bool largeArcFlag,
223 bool sweepFlag,
224 double endX,
225 double endY,
226 double& centerX,
227 double& centerY,
228 double& angleExtent) {
229 // 计算起点和终点中点坐标系中的半弦长分量。
230 double dx2 = (startX - endX) / 2.0; // 半弦向量 X 分量。
231 double dy2 = (startY - endY) / 2.0; // 半弦向量 Y 分量。
232
233 // 将椭圆旋转角从度转换为弧度。
234 angle = degreesToRadians(std::fmod(angle, 360.0));
235 double cosAngle = std::cos(angle); // 旋转角余弦。
236 double sinAngle = std::sin(angle); // 旋转角正弦。
237
238 // 第一步：将端点变换到未旋转的椭圆坐标系。
239 double x1 = cosAngle * dx2 + sinAngle * dy2; // 局部坐标 X 分量。
240 double y1 = -sinAngle * dx2 + cosAngle * dy2; // 局部坐标 Y 分量。
241
242 // 取半径绝对值，保证后续计算使用非负半径。
243 radiusX = std::abs(radiusX);
244 radiusY = std::abs(radiusY);
245 double PradiusX = radiusX * radiusX; // X 半径平方。
246 double PradiusY = radiusY * radiusY; // Y 半径平方。
247 double Px1 = x1 * x1; // 局部 X 分量平方。
248 double Py1 = y1 * y1; // 局部 Y 分量平方。
249
250 // 检查当前半径是否足以覆盖端点。
251 double radiiCheck = 0.0;
252 if (PradiusX != 0.0 && PradiusY != 0.0) {
253 radiiCheck = Px1 / PradiusX + Py1 / PradiusY;
254 }
255
256 if (radiiCheck > 1.0) {
257 radiusX = std::sqrt(radiiCheck) * radiusX;
258 radiusY = std::sqrt(radiiCheck) * radiusY;
259 PradiusX = radiusX * radiusX;
260 PradiusY = radiusY * radiusY;
261 }
262
263 // 第二步：计算未旋转坐标系中的圆心偏移。
264 int sign = (largeArcFlag == sweepFlag) ? -1 : 1; // 圆心选择方向符号。
265 double sq = 0.0; // 圆心系数平方。
266 if (PradiusX * Py1 + PradiusY * Px1 > 0.0) {
267 sq = (PradiusX * PradiusY - PradiusX * Py1 - PradiusY * Px1) / (PradiusX * Py1 + PradiusY * Px1);
268 }
269 sq = std::max(sq, 0.0);
270 double coef = sign * std::sqrt(sq); // 圆心系数。
271 double cx1 = coef * ((radiusX * y1) / radiusY); // 局部圆心 X 偏移。
272 double cy1 = (radiusX != 0.0) ? coef * -((radiusY * x1) / radiusX) : 0.0; // 局部圆心 Y 偏移。
273
274 // 第三步：将局部圆心变换回原坐标系。
275 double sx2 = (startX + endX) / 2.0; // 端点中点 X 坐标。
276 double sy2 = (startY + endY) / 2.0; // 端点中点 Y 坐标。
277 centerX = sx2 + (cosAngle * cx1 - sinAngle * cy1);
278 centerY = sy2 + (sinAngle * cx1 + cosAngle * cy1);
279
280 // 第四步：根据起止向量计算扫过角度。
281 double ux = (radiusX != 0.0) ? (x1 - cx1) / radiusX : 0.0; // 起点单位向量 X 分量。
282 double uy = (radiusY != 0.0) ? (y1 - cy1) / radiusY : 0.0; // 起点单位向量 Y 分量。
283 double vx = (radiusX != 0.0) ? (-x1 - cx1) / radiusX : 0.0; // 终点单位向量 X 分量。
284 double vy = (radiusY != 0.0) ? (-y1 - cy1) / radiusY : 0.0; // 终点单位向量 Y 分量。
285
286 // 计算两向量之间的夹角。
287 double n = std::sqrt((ux * ux + uy * uy) * (vx * vx + vy * vy)); // 向量模长乘积。
288 double p = ux * vx + uy * vy; // 向量点积。
289 sign = ((ux * vy - uy * vx) < 0.0) ? -1 : 1;
290
291 if (n != 0.0) {
292 double ratio = p / n; // 夹角余弦。
293 if (std::abs(ratio) < 1.0) {
294 angleExtent = radiansToDegrees(sign * std::acos(ratio));
295 } else {
296 angleExtent = 360.0 + 359.0;
297 }
298 } else {
299 angleExtent = 360.0 + 359.0;
300 }
301
302 if (!sweepFlag && angleExtent > 0.0) {
303 angleExtent -= 360.0;
304 } else if (sweepFlag && angleExtent < 0.0) {
305 angleExtent += 360.0;
306 }
307
308 int angleExtentSign = (angleExtent < 0.0) ? 1 : -1; // 保留扫掠方向的符号。
309 angleExtent = (std::abs(angleExtent) / 360.0) * 360.0 * angleExtentSign;
310}

317GeometryUtils::SvgArcResult GeometryUtils::solveSvgArc(const QString& param) {
318 SvgArcResult res; // SVG 椭圆弧解析结果。
319
320 // 解析 SVG 弧参数字符串
321 // 格式: "M x1 y1 A rx ry phi fa fs x2 y2"
322 QStringList args = param.split(QRegularExpression("\\s+"), Qt::SkipEmptyParts); // 分隔后的弧参数。
323
324 if (args.size() < 10) {
325 qWarning() << "Invalid SVG arc parameter:" << param;
326 return res;
327 }
328
329 if (args[0] != "M" && args[0] != "m") {
330 qWarning() << "SVG arc must start with M command:" << param;
331 return res;
332 }
333
334 if (args[3] != "A" && args[3] != "a") {
335 qWarning() << "Expected A command in SVG arc:" << param;
336 return res;
337 }
338
339 double x1 = args[1].toDouble(); // 起点 X 坐标。
340 double y1 = args[2].toDouble(); // 起点 Y 坐标。
341 double rx = args[4].toDouble(); // 椭圆 X 半径。
342 double ry = args[5].toDouble(); // 椭圆 Y 半径。
343 double phiAngle = args[6].toDouble(); // 椭圆旋转角度。
344 int fa = args[7].toInt(); // 大弧标志。
345 int fs = args[8].toInt(); // 扫掠方向标志。
346 double x2 = args[9].toDouble(); // 终点 X 坐标。
347 double y2 = (args.size() > 10) ? args[10].toDouble() : 0.0; // 终点 Y 坐标。
348
349 res.rx = rx;
350 res.ry = ry;
351 res.startPt.x = x1;
352 res.startPt.y = y1;
353 res.endPt.x = x2;
354 res.endPt.y = y2;
355 res.xRotate = phiAngle;
356
357 // 如果半径，退化为直线
358 if (rx == 0 || ry == 0) {
359 res.cx = (x1 + x2) / 2.0; // 退化弧的中心 X 坐标。
360 res.cy = (y1 + y2) / 2.0; // 退化弧的中心 Y 坐标。
361 res.deltaAngle = 360.0; // 退化弧使用整圆范围标记。
362 return res;
363 }
364
365 // 将椭圆旋转角度转换为弧度。
366 double phi = degreesToRadians(phiAngle); // 椭圆旋转角（弧度）。
367 double cos_phi = std::cos(phi); // 椭圆旋转角余弦。
368 double sin_phi = std::sin(phi); // 椭圆旋转角正弦。
369
370 // 计算端点中点坐标系中的半弦向量。
371 double dx = (x1 - x2) / 2.0; // 半弦向量 X 分量。
372 double dy = (y1 - y2) / 2.0; // 半弦向量 Y 分量。
373
374 // 将半弦向量变换到未旋转坐标系。
375 double x1_ = cos_phi * dx + sin_phi * dy; // 未旋转坐标系 X 分量。
376 double y1_ = -sin_phi * dx + cos_phi * dy; // 未旋转坐标系 Y 分量。
377
378 // 按端点位置放大不足以覆盖端点的半径。
379 double rx_sq = rx * rx; // X 半径平方。
380 double ry_sq = ry * ry; // Y 半径平方。
381 double x1_sq = x1_ * x1_; // 局部 X 分量平方。
382 double y1_sq = y1_ * y1_; // 局部 Y 分量平方。
383
384 double lambda = x1_sq / rx_sq + y1_sq / ry_sq; // 端点覆盖比例。
385 if (lambda > 1.0) {
386 double sqrtLambda = std::sqrt(lambda); // 半径放大比例。
387 rx *= sqrtLambda;
388 ry *= sqrtLambda;
389 rx_sq = rx * rx;
390 ry_sq = ry * ry;
391 }
392
393 // 根据 SVG 标志计算局部圆心。
394 double temp = // 局部圆心系数。
395 std::sqrt(std::max(0.0, (rx_sq * ry_sq - rx_sq * y1_sq - ry_sq * x1_sq) / (rx_sq * y1_sq + ry_sq * x1_sq)));
396
397 int sign = (fa == fs) ? -1 : 1; // 圆心选择方向符号。
398 double cx_ = sign * temp * rx * y1_ / ry; // 局部圆心 X 坐标。
399 double cy_ = -sign * temp * ry * x1_ / rx; // 局部圆心 Y 坐标。
400
401 // 将局部圆心变换回原坐标系。
402 res.cx = cos_phi * cx_ - sin_phi * cy_ + (x1 + x2) / 2.0;
403 res.cy = sin_phi * cx_ + cos_phi * cy_ + (y1 + y2) / 2.0;
404
405 // 计算椭圆弧的起始角度和扫过角度。
406 double angle1 = getAngle(1.0, 0.0, (x1_ - cx_) / rx, (y1_ - cy_) / ry); // 起始角度。
407 double angle2 = getAngle((x1_ - cx_) / rx, (y1_ - cy_) / ry, (-x1_ - cx_) / rx, (-y1_ - cy_) / ry); // 扫过角度。
408
409 res.startAngle = radiansToDegrees(angle1);
410 res.deltaAngle = radiansToDegrees(angle2);
411
412 // 将起始角度规范化到一整圆范围内。
413 while (res.startAngle < 0)
414 res.startAngle += 360.0;
415 while (res.startAngle >= 360.0)
416 res.startAngle -= 360.0;
417 res.startAngle = std::abs(res.startAngle);
418
419 // 根据扫掠方向调整扫过角度符号。
420 if (fs != 0) {
421 if (res.deltaAngle < 0)
422 res.deltaAngle += 360.0;
423 } else {
424 if (res.deltaAngle > 0)
425 res.deltaAngle -= 360.0;
426 }
427
428 return res;
429}

439double GeometryUtils::getAngle(double x1, double y1, double x2, double y2) {
440 // angle = arccos( (U dot V) / (|U|*|V|) )
441 double factor = (x1 * y2 - y1 * x2 > 0) ? 1.0 : -1.0; // 叉积确定的方向符号。
442 double dot = x1 * x2 + y1 * y2; // 两向量点积。
443 double mag1 = std::sqrt(x1 * x1 + y1 * y1); // 第一个向量模长。
444 double mag2 = std::sqrt(x2 * x2 + y2 * y2); // 第二个向量模长。
445
446 if (mag1 == 0 || mag2 == 0)
447 return 0.0;
448
449 double ratio = dot / (mag1 * mag2); // 夹角余弦。
450 ratio = std::clamp(ratio, -1.0, 1.0); // 防止数值误
451
452 return factor * std::acos(ratio);
453}

460GeometryUtils::SvgArcEndpoints GeometryUtils::calcSvgArc(const SvgArcResult& arc) {
461 SvgArcEndpoints res; // 椭圆弧端点结果。
462
463 double phi = degreesToRadians(arc.xRotate); // 椭圆旋转角（弧度）。
464 double theta = -degreesToRadians(arc.startAngle); // 起始参数角度。
465 double dTheta = -degreesToRadians(arc.deltaAngle); // 参数角度增量。
466
467 // 计算起点
468 res.x1 = std::cos(phi) * std::cos(theta) * arc.rx - std::sin(phi) * std::sin(theta) * arc.rx + arc.cx;
469 res.y1 = std::sin(phi) * std::cos(theta) * arc.ry - std::cos(phi) * std::sin(theta) * arc.ry + arc.cy;
470
471 // 计算终点
472 theta += dTheta;
473 res.x2 = std::cos(phi) * std::cos(theta) * arc.rx - std::sin(phi) * std::sin(theta) * arc.rx + arc.cx;
474 res.y2 = std::sin(phi) * std::cos(theta) * arc.ry - std::cos(phi) * std::sin(theta) * arc.ry + arc.cy;
475
476 return res;
477}

485QList<GeometryUtils::SvgPoint> GeometryUtils::arcToPath(const SvgArcResult& arc, bool includeStart) {
486 QList<SvgPoint> res; // 离散后的路径点。
487
488 const int splitCount = 32; // 将整圆划分为 32 段以控制圆弧采样精度。
489 double step = 360.0 / splitCount; // 当前采样步长。
490
491 double startAngle = arc.startAngle; // 当前采样起始角度。
492 double deltaAngle = arc.deltaAngle; // 尚未采样的角度范围。
493
494 if (deltaAngle < 0)
495 step = -step;
496
497 while (std::abs(deltaAngle) > 0.1) {
498 if (std::abs(deltaAngle) < std::abs(step)) {
499 step = deltaAngle;
500 }
501
502 SvgArcResult tempArc = arc; // 当前采样段的弧参数副本。
503 tempArc.startAngle = startAngle; // 当前采样段起始角度。
504 tempArc.deltaAngle = step; // 当前采样段扫过角度。
505
506 SvgArcEndpoints pt = calcSvgArc(tempArc); // 当前采样段的端点。
507
508 if (res.isEmpty() && includeStart) {
509 res.append(SvgPoint(pt.x1, pt.y1));
510 }
511 res.append(SvgPoint(pt.x2, pt.y2));
512
513 deltaAngle -= step;
514 startAngle += step;
515 }
516
517 return res;
518}
