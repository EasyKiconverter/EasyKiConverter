148 const auto parameterCount = [](QChar command) {

169 int tokenIndex = 0;
170 while (!hasInvalidSyntax && tokenIndex < tokens.size()) {
171 if (!isCommandToken(tokens.at(tokenIndex))) {
172 hasInvalidSyntax = true;
173 break;
174 }
175 const QChar command = tokens.at(tokenIndex++).at(0);
176 const int groupSize = parameterCount(command);
177 if (command.toUpper() == QLatin1Char('Z'))
178 continue;
179
180 int groupCount = 0;
181 while (tokenIndex < tokens.size() && !isCommandToken(tokens.at(tokenIndex))) {
182 if (tokenIndex + groupSize > tokens.size()) {
183 hasInvalidSyntax = true;
184 break;
185 }
186 for (int parameterIndex = 0; parameterIndex < groupSize; ++parameterIndex) {
187 if (isCommandToken(tokens.at(tokenIndex + parameterIndex))) {
188 hasInvalidSyntax = true;
189 break;
190 }
191 }
192 if (hasInvalidSyntax)
193 break;
194 tokenIndex += groupSize;
195 ++groupCount;

197 if (command.toUpper() == QLatin1Char('M') && groupCount > 1)
