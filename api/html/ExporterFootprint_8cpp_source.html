24static void computeBBox(const IR::FootprintComponentIR& footprint,
25 double& outMinX,
26 double& outMinY,
27 double& outMaxX,
28 double& outMaxY) {
29 outMinX = std::numeric_limits<double>::max();
30 outMinY = std::numeric_limits<double>::max();
31 outMaxX = std::numeric_limits<double>::lowest();
32 outMaxY = std::numeric_limits<double>::lowest();
33
34 for (const auto& pad : footprint.pads) {
35 outMinX = qMin(outMinX, pad.position.x() - pad.size.width() / 2.0);
36 outMinY = qMin(outMinY, pad.position.y() - pad.size.height() / 2.0);
37 outMaxX = qMax(outMaxX, pad.position.x() + pad.size.width() / 2.0);
38 outMaxY = qMax(outMaxY, pad.position.y() + pad.size.height() / 2.0);
39 }
40 for (const auto& track : footprint.tracks) {
41 for (const QPointF& pt : track.points) {
42 outMinX = qMin(outMinX, pt.x());
43 outMinY = qMin(outMinY, pt.y());
44 outMaxX = qMax(outMaxX, pt.x());
45 outMaxY = qMax(outMaxY, pt.y());
46 }
47 }
48 for (const auto& hole : footprint.holes) {
49 outMinX = qMin(outMinX, hole.center.x() - hole.radius);
50 outMinY = qMin(outMinY, hole.center.y() - hole.radius);
51 outMaxX = qMax(outMaxX, hole.center.x() + hole.radius);
52 outMaxY = qMax(outMaxY, hole.center.y() + hole.radius);
53 }
54 for (const auto& circle : footprint.circles) {
55 outMinX = qMin(outMinX, circle.center.x() - circle.radius);
56 outMinY = qMin(outMinY, circle.center.y() - circle.radius);
57 outMaxX = qMax(outMaxX, circle.center.x() + circle.radius);
58 outMaxY = qMax(outMaxY, circle.center.y() + circle.radius);
59 }
60 for (const auto& rect : footprint.rectangles) {
61 outMinX = qMin(outMinX, rect.bounds.left());
62 outMinY = qMin(outMinY, rect.bounds.top());
63 outMaxX = qMax(outMaxX, rect.bounds.right());
64 outMaxY = qMax(outMaxY, rect.bounds.bottom());
65 }
66 for (const auto& arc : footprint.arcs) {
67 outMinX = qMin(outMinX, arc.center.x() - arc.radius);
68 outMinY = qMin(outMinY, arc.center.y() - arc.radius);
69 outMaxX = qMax(outMaxX, arc.center.x() + arc.radius);
70 outMaxY = qMax(outMaxY, arc.center.y() + arc.radius);
71 }
72 for (const auto& text : footprint.texts) {
73 outMinX = qMin(outMinX, text.position.x());
74 outMinY = qMin(outMinY, text.position.y());
75 outMaxX = qMax(outMaxX, text.position.x());
76 outMaxY = qMax(outMaxY, text.position.y());
77 }
78 for (const auto& region : footprint.regions) {
79 for (const QPointF& pt : region.vertices) {
80 outMinX = qMin(outMinX, pt.x());
81 outMinY = qMin(outMinY, pt.y());
82 outMaxX = qMax(outMaxX, pt.x());
83 outMaxY = qMax(outMaxY, pt.y());
84 }
85 }

171bool ExporterFootprint::exportFootprintLibrary(const QList<IR::FootprintComponentIR>& footprints,
172 const QString& libName,
173 const QString& filePath,
174 bool preferWrl,
175 bool exportStep,
176 const QString& libraryDescription,
177 const QString& libraryKeywords,
178 bool useAbsolutePaths,
179 const QString& model3DBaseDir) {
180 qDebug() << "=== Export Footprint Library ===";
181 qDebug() << "Library name:" << libName;
182 qDebug() << "Output path:" << filePath;
183 qDebug() << "Footprint count:" << footprints.count();
184 qDebug() << "Library description:" << libraryDescription;
185 qDebug() << "Library keywords:" << libraryKeywords;
186
187 QDir libDir(filePath); // 封装库输出目录。
188
189 if (!libDir.exists()) {
190 qDebug() << "Creating footprint library directory:" << filePath;
191 if (!libDir.mkpath(".")) {
192 qWarning() << "Failed to create footprint library directory:" << filePath;
193 return false;
194 }
195 }
196
197 QSet<QString> existingFootprintNames; // 已存在的封装名称。
198 QStringList existingFiles = libDir.entryList(QStringList("*.kicad_mod"), QDir::Files); // 已存在封装文件。
199 for (const QString& fileName : existingFiles) {
200 QString fpName = fileName; // 去除扩展名后的封装名称。
201 fpName.remove(".kicad_mod");
202 existingFootprintNames.insert(fpName);
203 qDebug() << "Found existing footprint:" << fpName;
204 }
205
206 qDebug() << "Existing footprints count:" << existingFootprintNames.count();
207
208 const QString safeLibName = PathSecurity::sanitizeFilename(libName); // 安全化后的库名称。
209 const QString resolvedBaseDir =
210 model3DBaseDir.isEmpty() ? QFileInfo(filePath).absolutePath() : model3DBaseDir; // 解析后的模型根目录。
211
212 int exportedCount = 0; // 新增封装数量。
213 int overwrittenCount = 0; // 覆盖封装数量。
214 for (const IR::FootprintComponentIR& footprint : footprints) {
215 QString fpName = footprint.name; // 当前封装名称。
216 QString fileName = fpName + ".kicad_mod"; // 当前封装文件名。
217 QString fullPath = libDir.filePath(fileName); // 当前封装完整路径。
218
219 bool exists = existingFootprintNames.contains(fpName); // 当前封装是否已存在。
220 if (exists) {
221 qDebug() << "Overwriting existing footprint:" << fpName;
222 overwrittenCount++;
223 } else {
224 qDebug() << "Exporting new footprint:" << fpName;
225 exportedCount++;
226 }
227
228 // 确定 3D 模型路径
229 // 只要 models3d 非空且有名称即视为有 3D 模型引用（不依赖 isValid()，
230 // 因为 isValid() 要求有实际 OBJ/STEP 数据，而路径生成仅需名称）
231 bool hasModel3D =
232 !footprint.models3d.isEmpty() && !footprint.models3d.first().name().isEmpty(); // 是否存在有效模型引用。
233 QString modelName; // 当前 3D 模型名称。
234 if (hasModel3D) {
235 modelName = footprint.models3d.first().name();
236 if (modelName.isEmpty()) {
237 modelName = fpName;
238 }
239 modelName = PathSecurity::sanitizeFilename(modelName);
240 }
241
242 QString content; // 当前封装生成内容。
243 const bool useWrl = preferWrl && hasModel3D; // 是否生成 WRL 模型引用。
244 const bool useStep = exportStep && hasModel3D; // 是否生成 STEP 模型引用。
245
246 if (useWrl && useStep) {
247 QString wrlPath = // WRL 模型输出路径。
248 buildModel3DPath(safeLibName, modelName, QStringLiteral("wrl"), useAbsolutePaths, resolvedBaseDir);
249 QString stepPath = // STEP 模型输出路径。
250 buildModel3DPath(safeLibName, modelName, QStringLiteral("step"), useAbsolutePaths, resolvedBaseDir);
251 content = generateFootprintContent(footprint, wrlPath, stepPath, libraryDescription, libraryKeywords);
252 } else if (useWrl) {
253 QString wrlPath = // WRL 模型输出路径。
254 buildModel3DPath(safeLibName, modelName, QStringLiteral("wrl"), useAbsolutePaths, resolvedBaseDir);
255 content = generateFootprintContent(footprint, wrlPath, libraryDescription, libraryKeywords);
256 } else if (useStep) {
257 QString stepPath = // STEP 模型输出路径。
258 buildModel3DPath(safeLibName, modelName, QStringLiteral("step"), useAbsolutePaths, resolvedBaseDir);
259 content = generateFootprintContent(footprint, stepPath, libraryDescription, libraryKeywords);
260 } else {
261 content = generateFootprintContent(footprint, QString(), libraryDescription, libraryKeywords);
262 }
263
264 QFile file(fullPath); // 当前封装输出文件。
265 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
266 qWarning() << "Failed to open file for writing:" << fullPath;
267 continue;
268 }
269
270 QTextStream out(&file); // UTF-8 输出流。
271 out.setEncoding(QStringConverter::Utf8);
272 out << content;
273 file.close();
274
275 qDebug() << " Exported to:" << fullPath;
276 }
277
278 qDebug() << "Export complete - New:" << exportedCount << "Overwritten:" << overwrittenCount;
279 qDebug() << "Footprint library exported to:" << filePath;
280
281 return true;
282}

331void ExporterFootprint::generateFootprintBaseContent(const IR::FootprintComponentIR& footprint,
332 QString& content,
333 double& outOriginX,
334 double& outOriginY,
335 const QString& libraryDescription,
336 const QString& libraryKeywords) const {
337 content += QString("(footprint easykiconverter:%1\n").arg(footprint.name);
338 content += " (version 20221018)\n";
339
340 // 导出封装描述 (descr)
341 const QString fpDescription = footprint.description.trimmed();
342 if (!fpDescription.isEmpty()) {
343 content += QString(" (descr \"%1\")\n").arg(fpDescription);
344 }
345
346 // 导出封装标签 (tags)
347 if (!libraryKeywords.isEmpty()) {
348 content += QString(" (tags \"%1\")\n").arg(libraryKeywords);
349 }
350
351 // 判断是否有通孔焊盘
352 if (footprint.hasThroughHolePads()) {
353 content += " (attr through_hole)\n";
354 } else {
355 content += " (attr smd)\n";
356 }
357
358 // 计算边界框（mm 单位）
359 double bboxMinX = 0.0, bboxMinY = 0.0, bboxMaxX = 0.0, bboxMaxY = 0.0; // 封装边界框坐标。
360 computeBBox(footprint, bboxMinX, bboxMinY, bboxMaxX, bboxMaxY);
361
362 // 如果没有有效的边界框，使用默认值
363 bool hasValidBBox = (bboxMinX <= bboxMaxX && bboxMinY <= bboxMaxY); // 边界框是否有效。
364 if (!hasValidBBox) {
365 bboxMinX = bboxMinY = 0.0;
366 bboxMaxX = bboxMaxY = 1.0;
367 }
368
369 // 计算边界框中心作为原点
370 double originX = (bboxMinX + bboxMaxX) / 2.0; // 封装原点 X 坐标。
371 double originY = (bboxMinY + bboxMaxY) / 2.0; // 封装原点 Y 坐标。
372
373 // 计算 pad Y 范围（相对于原点）
374 double yLow = 0; // Pad 的最小相对 Y 坐标。
375 double yHigh = 0; // Pad 的最大相对 Y 坐标。
376 if (!footprint.pads.isEmpty()) {
377 yLow = footprint.pads.first().position.y() - originY;
378 yHigh = yLow;
379 for (const IR::FootprintPadIR& pad : footprint.pads) {
380 double padY = pad.position.y() - originY; // 当前 Pad 的相对 Y 坐标。
381 if (padY < yLow)
382 yLow = padY;
383 if (padY > yHigh)
384 yHigh = padY;
385 }
386 }
387
388 content += QString(" (fp_text reference REF** (at 0 %1) (layer F.SilkS)\n").arg(yLow - 4 * IR::EASYEDA_PX_TO_MM);
389 content += " (effects (font (size 1 1) (thickness 0.15)))\n";
390 content += " )\n";
391
392 content += QString(" (fp_text value %1 (at 0 %2) (layer F.Fab)\n")
393 .arg(footprint.name)
394 .arg(yHigh + 4 * IR::EASYEDA_PX_TO_MM);
395 content += " (effects (font (size 1 1) (thickness 0.15)))\n";
396 content += " )\n";
397
398 for (const IR::FootprintTrackIR& track : footprint.tracks) {
399 content += m_graphicsGenerator.generateTrack(track, originX, originY);
400 }
401 for (const IR::FootprintRectangleIR& rect : footprint.rectangles) {
402 content += m_graphicsGenerator.generateRectangle(rect, originX, originY);
403 }
404
405 for (const IR::FootprintPadIR& pad : footprint.pads) {
406 content += m_graphicsGenerator.generatePad(pad, originX, originY);
407 }
408
409 for (const IR::FootprintHoleIR& hole : footprint.holes) {
410 content += m_graphicsGenerator.generateHole(hole, originX, originY);
411 }
412
413 for (const IR::FootprintCircleIR& circle : footprint.circles) {
414 content += m_graphicsGenerator.generateCircle(circle, originX, originY);
415 }
416
417 for (const IR::FootprintArcIR& arc : footprint.arcs) {
418 content += m_graphicsGenerator.generateArc(arc, originX, originY);
419 }
420
421 for (const IR::FootprintTextIR& text : footprint.texts) {
422 content += m_graphicsGenerator.generateText(text, originX, originY);
423 }
424
425 bool hasCourtYard = false; // 是否已有禁止区/边界区域可作为 courtyard。
426 for (const IR::FootprintRegionIR& region : footprint.regions) {
427 QString regionContent = m_graphicsGenerator.generateSolidRegion(region, originX, originY); // 区域生成内容。
428 content += regionContent;
429 if (region.layer == IR::LayerType::KeepOut) {
430 hasCourtYard = true;
431 }
432 }
433
434 if (!hasCourtYard && footprint.shouldGenerateCourtyard && hasValidBBox) {
435 double x1 = std::floor((bboxMinX - originX) * 100.0) / 100.0; // courtyard 左边界。
436 double y1 = std::floor((bboxMinY - originY) * 100.0) / 100.0; // courtyard 上边界。
437 double x2 = std::floor((bboxMaxX - originX) * 100.0) / 100.0; // courtyard 右边界。
438 double y2 = std::floor((bboxMaxY - originY) * 100.0) / 100.0; // courtyard 下边界。
439 content += m_graphicsGenerator.generateCourtyardFromBBox(x1, y1, x2, y2);
440 qWarning() << "Warning: No courtyard found, generated from BBox";
441 }
442
443 outOriginX = originX;
444 outOriginY = originY;
445}
