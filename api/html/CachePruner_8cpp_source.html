74qint64 CachePruner::pruneTo(qint64 targetSizeBytes) {
75 struct CacheEntry {
76 QString name;
77 QDateTime lastModified;
78 qint64 size;
79 };
80
81 // 第一阶段：按文件收集可验证的缓存条目（排除 model3d 和未知文件）。
82 QList<CacheEntry> cacheList; // 按文件记录的可验证缓存条目。
83 qint64 currentSize = 0; // 当前可维护缓存总大小。
84
85 for (const QString& componentPath : CacheSafety::ownedComponentDirectories(m_cacheRoot)) {
86 for (const QString& path : CacheSafety::ownedComponentFiles(m_cacheRoot, componentPath)) {
87 const QFileInfo info(path); // 当前缓存文件信息。
88 const qint64 entrySize = info.size(); // 当前缓存文件大小。
89 cacheList.append({path, info.lastModified(), entrySize});
90 currentSize += entrySize;
91 }
92 }
93
94 if (currentSize <= targetSizeBytes) {
95 return currentSize;
96 }
97
98 // 按最后修改时间排序，最早修改的条目优先裁剪。
99 std::sort(cacheList.begin(), cacheList.end(), [](const CacheEntry& a, const CacheEntry& b) {
100 return a.lastModified < b.lastModified;
101 });
102
103 // 第二阶段：将经过所有权验证的条目移入回收站。
104 for (const auto& entry : cacheList) {
105 if (currentSize <= targetSizeBytes) {
106 break;
107 }
108
109 QString error; // 当前回收站操作的错误信息。
110 if (CacheSafety::moveToTrash(entry.name, &error, m_trash)) {
111 currentSize -= entry.size;
112 }
113 }
114
115 return currentSize;
116}
