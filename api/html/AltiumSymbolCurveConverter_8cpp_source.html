29AltiumSchArc AltiumSymbolCurveConverter::convertArc(const IR::SymbolArcIR& arc) {
30 AltiumSchArc altiumArc; // 转换后的 Altium 圆弧记录。
31 const double ax = arc.startPoint.x(); // 起点 X 坐标。
32 const double ay = arc.startPoint.y(); // 起点 Y 坐标。
33 const double bx = arc.midPoint.x(); // 中点 X 坐标。
34 const double by = arc.midPoint.y(); // 中点 Y 坐标。
35 const double cx = arc.endPoint.x(); // 终点 X 坐标。
36 const double cy = arc.endPoint.y(); // 终点 Y 坐标。
37 const double determinant = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by)); // 三点圆心方程行列式。
38 QPointF center = (arc.startPoint + arc.endPoint) / 2.0; // 退化情况下使用的首尾中点。
39 if (std::abs(determinant) > 1e-12) {
40 const double a2 = ax * ax + ay * ay; // 起点坐标平方和。
41 const double b2 = bx * bx + by * by; // 中点坐标平方和。
42 const double c2 = cx * cx + cy * cy; // 终点坐标平方和。
43 center.setX((a2 * (by - cy) + b2 * (cy - ay) + c2 * (ay - by)) / determinant);
44 center.setY((a2 * (cx - bx) + b2 * (ax - cx) + c2 * (bx - ax)) / determinant);
45 }
46 const double dx = arc.startPoint.x() - center.x(); // 起点相对圆心的 X 偏移。
47 const double dy = arc.startPoint.y() - center.y(); // 起点相对圆心的 Y 偏移。
48 const double radius = std::sqrt(dx * dx + dy * dy); // 圆弧半径。
49 altiumArc.centerX = AltiumCoord::mmToRaw(center.x());
50 altiumArc.centerY = AltiumCoord::mmToRaw(center.y());
51 altiumArc.radius = AltiumCoord::mmToRaw(radius);
52 altiumArc.startAngle = std::atan2(arc.startPoint.y() - center.y(), arc.startPoint.x() - center.x()) * 180.0 / kPi;
53 altiumArc.endAngle = std::atan2(arc.endPoint.y() - center.y(), arc.endPoint.x() - center.x()) * 180.0 / kPi;
54 altiumArc.lineWidth = AltiumCoord::lineWidthMmToIndex(arc.strokeWidth);
55 altiumArc.lineStyle = toAltiumLineStyle(arc.strokeStyle);
56 altiumArc.color = toAltiumColor(arc.strokeColor);
57 altiumArc.ownerPartId = toAltiumOwnerPartId(arc.partIndex);
58 return altiumArc;
59}
