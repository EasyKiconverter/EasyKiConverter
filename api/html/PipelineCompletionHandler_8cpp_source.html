34void PipelineCompletionHandler::exportPreviewImages(const QMap<QString, QSharedPointer<ComponentData>>& preloadedData,
35 const ExportOptions& options) {
36 if (!options.exportPreviewImages) {
37 return;
38 }
39
40 int successCount = 0; // 成功导出预览图片的元器件数量。

99bool PipelineCompletionHandler::exportPreviewImagesFromMemory(const QList<QByteArray>& imageDataList,
100 const QString& outputPath,
101 const QString& componentName) {
102 QDir outputDir(outputPath); // 导出根目录对象。
103 QString imagesDir = outputDir.filePath("images"); // 预览图片输出目录。
104 if (!outputDir.exists(imagesDir)) {
105 if (!outputDir.mkpath(imagesDir)) {
106 qWarning() << "Failed to create images directory:" << imagesDir;
107 return false;
108 }
109 }
110
111 QString safeName = componentName; // 替换非法字符后的文件名前缀。
112 safeName.replace(INVALID_FILENAME_CHARS, "_");
113
114 bool allSuccess = true; // 所有图片是否均写入成功。
115 int exportedCount = 0; // 已写入的图片数量。
116
117 for (int i = 0; i < imageDataList.size(); ++i) {
118 if (imageDataList[i].isEmpty()) {
119 continue;
120 }
121
122 QString filename = QString("%1_%2.jpg").arg(safeName).arg(i); // 当前图片文件名。
123 QString filePath = QDir(imagesDir).filePath(filename); // 当前图片完整路径。
124
125 QFile file(filePath); // 当前图片输出文件。
126 if (file.open(QIODevice::WriteOnly)) {
127 file.write(imageDataList[i]);
128 file.close();
129 exportedCount++;
130 } else {
131 allSuccess = false;
132 }
133 }
134
135 return allSuccess;
136}

146bool PipelineCompletionHandler::exportDatasheetFromMemory(const QByteArray& datasheetData,
147 const QString& outputPath,
148 const QString& componentName,
149 const QString& format) {
150 QDir outputDir(outputPath); // 导出根目录对象。
151 QString datasheetsDir = outputDir.filePath("datasheets"); // 数据手册输出目录。
152 if (!outputDir.exists(datasheetsDir)) {
153 if (!outputDir.mkpath(datasheetsDir)) {
154 qWarning() << "Failed to create datasheets directory:" << datasheetsDir;
155 return false;
156 }
157 }
158
159 QString safeName = componentName; // 替换非法字符后的文件名前缀。
160 safeName.replace(INVALID_FILENAME_CHARS, "_");
161
162 QString extension = format.toLower(); // 规范化后的文件扩展名。
163 if (extension != "html") {
164 extension = "pdf";
165 }
166
167 QString filename = QString("%1.%2").arg(safeName).arg(extension); // 当前数据手册文件名。
168 QString filePath = QDir(datasheetsDir).filePath(filename); // 当前数据手册完整路径。
169
170 QFile file(filePath); // 当前数据手册输出文件。
171 if (file.open(QIODevice::WriteOnly)) {
172 file.write(datasheetData);
173 file.close();
174 return true;
175 }
176 return false;
177}

185void PipelineCompletionHandler::exportPreviewImagesFromViewModel(ComponentListViewModel* componentListViewModel,
186 ExportProgressViewModel* progressViewModel,
187 const ExportOptions& options) {
188 if (!options.exportPreviewImages || !componentListViewModel) {
189 return;
190 }
191
192 QMap<QString, QSharedPointer<ComponentData>> allData = // 从 ViewModel 获取最新的预加载数据。
193 componentListViewModel->getAllPreloadedData();
194 if (allData.isEmpty()) {
195 qDebug() << "No preloaded data available from ViewModel for preview images export";
196 emit previewImagesExported(0);
197 return;
198 }
199
200 int successCount = 0; // 成功导出预览图片的元器件数量。
201 for (auto it = allData.constBegin(); it != allData.constEnd(); ++it) {
202 const QSharedPointer<ComponentData>& componentData = it.value(); // 当前元器件数据。
203 QString componentId = componentData->lcscId(); // 当前元器件编号。
204
205 QList<QByteArray> previewImageDataList = componentData->previewImageData(); // 当前预览图片数据。
206
207 // 1. 如果内存中有数据，直接使用
208 // 2. 如果内存中没有但有 URL，尝试从磁盘缓存加载
209 if (previewImageDataList.isEmpty() && !componentData->previewImages().isEmpty()) {
210 const QStringList& imageUrls = componentData->previewImages(); // 当前元器件的预览图片地址。
211 for (int i = 0; i < imageUrls.size() && i < 3; ++i) {
212 QByteArray imageData = // 当前预览图片的缓存或下载数据。
213 ComponentCacheService::instance()->loadPreviewImage(componentId, i);
214 if (!imageData.isEmpty()) {
215 previewImageDataList.append(imageData);
216 qDebug() << "Preview image loaded from disk cache for component:" << componentId << "index:" << i;
217 } else {
218 // 3. 磁盘缓存也没有，从 URL 下载并缓存
219 qDebug() << "Preview image not in cache, downloading for component:" << componentId
220 << "index:" << i;
221 imageData = ComponentCacheService::instance()->downloadPreviewImage(
222 componentId, imageUrls[i], i, nullptr, nullptr, options.weakNetworkSupport);
223 if (!imageData.isEmpty()) {
224 previewImageDataList.append(imageData);
225 }
226 }
227 }
228 }
229
230 bool exportSuccess = false; // 当前元器件的预览图片是否导出成功。
231 if (!previewImageDataList.isEmpty()) {
232 // 用于生成文件名的元器件名称。
233 QString componentName = componentData->name().isEmpty() ? componentId : componentData->name();
234
235 if (exportPreviewImagesFromMemory(previewImageDataList, options.outputPath, componentName)) {
236 successCount++;
237 exportSuccess = true;
238 } else {
239 qWarning() << "Failed to export preview images for component:" << componentId;
240 }
241 } else {
242 // 没有预览图数据但不需要导出时，也标记为成功（没有预览图不代表失败）
243 if (!options.exportPreviewImages || componentData->previewImages().isEmpty()) {
244 exportSuccess = true;
245 }
246 }
247
248 // 更新预览图导出状态。
249 componentListViewModel->updateExportStatus(componentId, exportSuccess ? 1 : 0, -1);
250 if (progressViewModel) {
251 progressViewModel->updateComponentExportStatus(componentId, exportSuccess ? 1 : 0, -1);
252 }
253 }
254 qDebug() << "Preview images export from ViewModel completed:" << successCount << "components";
255 emit previewImagesExported(successCount);
256}
