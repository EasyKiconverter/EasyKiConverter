25QString canonicalOrCleanPath(const QString& path) {
26 QFileInfo info(QDir::cleanPath(QFileInfo(path).absoluteFilePath()));
27 QStringList missingComponents;
28 while (!info.exists() && info.absoluteFilePath() != info.absolutePath()) {
29 missingComponents.prepend(info.fileName());
30 info = QFileInfo(info.absolutePath());
31 }
32 if (info.exists()) {
33 QString canonical = info.canonicalFilePath();
34 if (canonical.isEmpty())
35 canonical = info.absoluteFilePath();
36 for (const QString& component : missingComponents)
37 canonical = QDir(canonical).filePath(component);
38 return QDir::cleanPath(canonical);
39 }
40 return QDir::cleanPath(QFileInfo(path).absoluteFilePath());
41}

48bool containsSymlinkComponent(const QString& path) {

200bool CacheSafety::isSafePath(const QString& path, QString* normalizedPath, QString* error) {
201 const QString trimmed = path.trimmed();
202 if (trimmed.isEmpty()) {
203 if (error)
204 *error = QStringLiteral("缓存目录不能为空");
205 return false;
206 }
207 if (!QDir::isAbsolutePath(trimmed)) {
208 if (error)
209 *error = QStringLiteral("缓存目录必须使用绝对路径");
210 return false;
211 }
212
213 const QFileInfo info(trimmed);
214 if (containsSymlinkComponent(trimmed)) {
215 if (error)
216 *error = QStringLiteral("缓存目录不能包含符号链接路径分量");
217 return false;
218 }
219 const QString normalized = canonicalOrCleanPath(trimmed);
220 const QString root = QDir::cleanPath(QDir::rootPath());
221 const QString home = canonicalOrCleanPath(QDir::homePath());
222 if (normalized == root || normalized == home) {
223 if (error)
224 *error = QStringLiteral("不能选择文件系统根目录或用户主目录");
225 return false;
226 }
227 if (info.exists() && !info.isDir()) {
228 if (error)
229 *error = QStringLiteral("缓存路径必须是目录");
230 return false;
231 }
232 if (normalizedPath)
233 *normalizedPath = normalized;
234 return true;
235}

248 if (error)

250 return false;
251 }
252 return true;

261bool CacheSafety::hasValidMarker(const QString& path, const QString& format) {
262 const QString markerPath =
263 QDir(path).filePath(format == QStringLiteral("root") ? ownershipMarkerName() : model3DMarkerName());
264 const QFileInfo markerInfo(markerPath);
265 if (!markerInfo.exists() || markerInfo.isSymLink() || !markerInfo.isFile())
266 return false;
267 QFile file(markerPath);
268 if (!file.open(QIODevice::ReadOnly | QIODevice::Text))
269 return false;
270 QJsonParseError parseError;
271 const QJsonObject marker = QJsonDocument::fromJson(file.readAll(), &parseError).object();
272 return parseError.error == QJsonParseError::NoError &&
273 marker.value(QStringLiteral("format")).toString() == format &&
274 marker.value(QStringLiteral("owner")).toString() == QString::fromLatin1(kOwner) &&
275 marker.value(QStringLiteral("version")).toInt() == kMarkerVersion;
276}

310bool CacheSafety::ensureOwnedRoot(const QString& path, QString* error) {
311 QString normalized;
312 if (!isSafePath(path, &normalized, error))
313 return false;
314 QDir dir;
315 if (!dir.exists(normalized) && !dir.mkpath(normalized)) {
316 if (error)
317 *error = QStringLiteral("无法创建缓存目录：%1").arg(normalized);
318 return false;
319 }
320 if (hasValidMarker(normalized, QStringLiteral("root")))
321 return true;
322 if (!QDir(normalized).entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot).isEmpty() &&
323 !canAdoptLegacyRoot(normalized)) {
324 if (error)
325 *error = QStringLiteral("缓存目录不是空目录且无法验证所有权：%1").arg(normalized);
326 return false;
327 }
328 const bool adoptLegacy = canAdoptLegacyRoot(normalized);
329 if (!writeMarker(normalized, QStringLiteral("root"), error))
330 return false;
331 if (adoptLegacy) {
332 for (const QString& componentPath : legacyComponentDirectories(normalized)) {
333 if (!upgradeLegacyComponentMetadata(componentPath)) {
334 if (error)
335 *error = QStringLiteral("无法升级旧缓存元数据：%1").arg(componentPath);
336 return false;
337 }
338 }
339 }
340 return true;
341}

572bool CacheSafety::moveToTrash(const QString& path, QString* error, const TrashFunction& trash) {
573 const QFileInfo info(path);
574 if (!info.exists() || info.isSymLink()) {
575 if (error)
576 *error = QStringLiteral("拒绝处理不存在或符号链接条目：%1").arg(path);
577 return false;
578 }
579 const TrashFunction operation =
580 trash ? trash : [](const QString& target, QString* reason) { // 默认系统回收站操作。
581 if (QFile::moveToTrash(target))
582 return true;
583 if (reason)
584 *reason = QStringLiteral("系统回收站不支持或移动失败");
585 return false;
586 };
587 if (!operation(path, error))
588 return false;
589 if (QFileInfo::exists(path)) {
590 if (error)
591 *error = QStringLiteral("回收站操作未移除原路径：%1").arg(path);
592 return false;
593 }
594 return true;
595}
