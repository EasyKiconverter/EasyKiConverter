31QString registerScript() {
32 return QStringLiteral(
33 "import json, os, sys\n"
34 "import horizon\n"
35 "pool = os.path.abspath(sys.argv[1])\n"
36 "with open(os.path.join(pool, 'pool.json'), encoding='utf-8') as f:\n"
37 " pool_uuid = json.load(f)['uuid']\n"
38 "registered = horizon.PoolManager.get_pools()\n"
39 "if pool in registered and registered[pool] != pool_uuid:\n"
40 " raise RuntimeError('Horizon Pool at this path has a different UUID: ' + pool)\n"
41 "for registered_path, registered_uuid in registered.items():\n"
42 " if os.path.abspath(registered_path) != pool and registered_uuid == pool_uuid:\n"
43 " raise RuntimeError('Horizon Pool UUID is already registered at another path: ' + registered_path)\n"
44 "horizon.PoolManager.add_pool(pool)\n"
45 "registered_after = horizon.PoolManager.get_pools()\n"
46 "if pool not in {os.path.abspath(p) for p in registered_after}:\n"
47 " raise RuntimeError('Horizon Pool was not registered: ' + pool)\n");
48}

71bool runOfficialScript(const QString& poolPath,
72 const QString& script,
73 const QString& operation,
74 QStringList& diagnostics) {
75 const QFileInfo poolInfo(QDir(poolPath).filePath(QStringLiteral("pool.json")));
76 if (!poolInfo.isFile()) {
77 diagnostics.append(
78 QStringLiteral("Horizon: %1 前缺少 pool.json：%2").arg(operation, poolInfo.absoluteFilePath()));
79 return false;
80 }
81
82 QProcess process;
83 QProcessEnvironment environment = QProcessEnvironment::systemEnvironment();
84 // 可选的 Horizon Python 模块路径。
85 QString pythonPath = environment.value(QStringLiteral("EASYKICONVERTER_HORIZON_PYTHONPATH")).trimmed();
86 if (!pythonPath.isEmpty()) {
87 const QFileInfo modulePathInfo(pythonPath);
88 if (!modulePathInfo.isAbsolute())
89 pythonPath = QDir::cleanPath(QDir::current().absoluteFilePath(pythonPath));
90 const QString currentPythonPath =
91 environment.value(QStringLiteral("PYTHONPATH"));
92 environment.insert(
93 QStringLiteral("PYTHONPATH"),
94 currentPythonPath.isEmpty() ? pythonPath : pythonPath + QDir::listSeparator() + currentPythonPath);
95 }
96 process.setProcessEnvironment(environment);
97 const QString interpreter =
98 environment.value(QStringLiteral("EASYKICONVERTER_HORIZON_PYTHON"), QStringLiteral("python3"));
99 process.start(interpreter, {QStringLiteral("-c"), script, poolInfo.absolutePath()});
100 if (!process.waitForStarted(5000)) {
101 diagnostics.append(QStringLiteral("Horizon: %1 无法启动官方 Python binding（%2）：%3")
102 .arg(operation, interpreter, process.errorString()));
103 return false;
104 }
105 if (!process.waitForFinished(kProcessTimeoutMs)) {
106 process.kill();
107 process.waitForFinished();
108 diagnostics.append(QStringLiteral("Horizon: 官方 %1 超时：%2").arg(operation, poolInfo.absolutePath()));
109 return false;
110 }
111 if (process.exitStatus() != QProcess::NormalExit || process.exitCode() != 0) {
112 QString detail = processOutput(process);
113 if (detail.isEmpty())
114 detail = process.errorString();
115 diagnostics.append(QStringLiteral("Horizon: 官方 %1 失败：%2").arg(operation, detail));
116 return false;
117 }
118 return true;
119}
