125uint8_t AltiumLayerMap::toAltiumPinOrientation(int easyedaRotation) {
126 // EasyEDA 角度（度）→ Altium 方向索引
127 int normalized = ((easyedaRotation % 360) + 360) % 360; // 归一化到 0 至 359 度。
128 if (normalized < 45 || normalized >= 315)
129 return 0; // Right
130 if (normalized >= 45 && normalized < 135)
131 return 1; // Up (90°)
132 if (normalized >= 135 && normalized < 225)
133 return 2; // Left (180°)
134 return 3; // Down (270°)
135}
