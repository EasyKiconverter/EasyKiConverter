132QJsonObject SymbolShapeSerializer::toJson(const SymbolArc& arc) {
133 QJsonObject json; // 保存圆弧字段的 JSON 对象。
134 QJsonArray pathArray; // 保存圆弧路径点的 JSON 数组。
135 for (const QPointF& point : arc.path) {
136 QJsonObject pointObj; // 当前路径点的 JSON 对象。
137 pointObj["x"] = point.x();
138 pointObj["y"] = point.y();
139 pathArray.append(pointObj);
140 }
141 json["path"] = pathArray;
142 json["helper_dots"] = arc.helperDots;
143 json["stroke_color"] = arc.strokeColor;
144 json["stroke_width"] = arc.strokeWidth;
145 json["stroke_style"] = arc.strokeStyle;
146 json["fill_color"] = arc.fillColor;
147 json["id"] = arc.id;
148 json["is_locked"] = arc.isLocked;
149 return json;
150}

152// 从 JSON 恢复圆弧路径、辅助点和图形样式属性。
158bool SymbolShapeSerializer::fromJson(SymbolArc& arc, const QJsonObject& json) {
159 if (json.contains("path") && json["path"].isArray()) {
160 QJsonArray pathArray = json["path"].toArray(); // 圆弧路径 JSON 数组。
161 arc.path.clear();
162 for (const QJsonValue& value : pathArray) {
163 if (value.isObject()) {
164 QJsonObject pointObj = value.toObject(); // 当前路径点 JSON 对象。
165 QPointF point(pointObj["x"].toDouble(0.0), pointObj["y"].toDouble(0.0)); // 当前路径点。
166 arc.path.append(point);
167 }
168 }
169 }
170 arc.helperDots = json["helper_dots"].toString();
171 arc.strokeColor = json["stroke_color"].toString();
172 arc.strokeWidth = json["stroke_width"].toDouble(0.0);
173 arc.strokeStyle = json["stroke_style"].toString();
174 arc.fillColor = json["fill_color"].toBool(false);
175 arc.id = json["id"].toString();
176 arc.isLocked = json["is_locked"].toBool(false);
177 return true;
178}
