38bool PreviewImageExporter::exportFromUrls(const QStringList& imageUrls,
39 const QString& outputPath,
40 const QString& componentName) {
41 if (imageUrls.isEmpty()) {
42 return true;
43 }
44
45 QString imagesDirPath = // 预览图片输出目录。
46 QString("%1/%2.preview").arg(outputPath, m_options.libName);
47 if (!createPreviewDirectory(imagesDirPath)) {
48 qWarning() << "Failed to create preview images directory:" << imagesDirPath;
49 return false;
50 }
51
52 bool allSuccess = true; // 所有图片是否均导出成功。
53 for (int i = 0; i < imageUrls.size(); ++i) {
54 QString imageUrl = imageUrls[i]; // 当前预览图片地址。
55 QString fileExtension = QFileInfo(imageUrl).suffix(); // 当前图片文件扩展名。
56 if (fileExtension.isEmpty()) {
57 fileExtension = "jpg";
58 }
59
60 QString fileName = generateFileName(componentName, fileExtension, i + 1, imageUrls.size()); // 输出文件名。
61 QString imagePath = QString("%1/%2").arg(imagesDirPath, fileName); // 输出文件完整路径。
62
63 if (!m_options.overwriteExistingFiles && QFile::exists(imagePath)) {
64 qWarning() << "Preview image already exists:" << imagePath;
65 continue;
66 }
67
68 // Use NetworkClient with default retry policy (3 retries, 30s timeout)
69 NetworkResult result = // 预览图片下载结果。
70 NetworkClient::instance().get(QUrl(imageUrl), ResourceType::PreviewImage);
71
72 if (result.success) {
73 QFile file(imagePath); // 当前预览图片输出文件。
74 if (file.open(QIODevice::WriteOnly)) {
75 file.write(result.data);
76 file.close();
77 qDebug() << "Preview image exported successfully:" << imagePath;
78 } else {
79 qWarning() << "Failed to open file for writing:" << imagePath;
80 allSuccess = false;
81 }
82 } else {
83 qWarning() << "Failed to download preview image:" << imageUrl << result.error;
84 allSuccess = false;
85 }
86 }

98bool PreviewImageExporter::exportFromCache(const QStringList& imagePaths,
99 const QString& outputPath,
100 const QString& componentName) {
101 if (imagePaths.isEmpty()) {
102 return true;
103 }
104
105 // exportFromCache 只进行文件复制操作，不需要网络管理器
106 QString imagesDirPath = // 预览图片输出目录。
107 QString("%1/%2.preview").arg(outputPath, m_options.libName);
108 if (!createPreviewDirectory(imagesDirPath)) {
109 qWarning() << "Failed to create preview images directory:" << imagesDirPath;
110 return false;
111 }
112
113 bool allSuccess = true; // 所有图片是否均复制成功。
114 for (int i = 0; i < imagePaths.size(); ++i) {
115 QString sourcePath = imagePaths[i]; // 当前缓存图片路径。
116
117 if (!QFile::exists(sourcePath)) {
118 qWarning() << "Source preview image file not found:" << sourcePath;
119 allSuccess = false;
120 continue;
121 }
122
123 QString fileExtension = QFileInfo(sourcePath).suffix(); // 当前图片文件扩展名。
124 if (fileExtension.isEmpty()) {
125 fileExtension = "jpg";
126 }
127
128 QString fileName = generateFileName(componentName, fileExtension, i + 1, imagePaths.size()); // 输出文件名。
129 QString destPath = QString("%1/%2").arg(imagesDirPath, fileName); // 输出文件完整路径。
130
131 if (!m_options.overwriteExistingFiles && QFile::exists(destPath)) {
132 qWarning() << "Preview image already exists:" << destPath;
133 continue;
134 }
135
136 if (QFile::copy(sourcePath, destPath)) {
137 qDebug() << "Preview image copied from cache:" << destPath;
138 } else {
139 qWarning() << "Failed to copy preview image from cache:" << sourcePath << "to" << destPath;
140 allSuccess = false;
141 }
142 }
143
144 return allSuccess;
