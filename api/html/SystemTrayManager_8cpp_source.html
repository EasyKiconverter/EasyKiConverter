146void SystemTrayManager::showExportNotification(int successCount,
147 int failureCount,
148 int symbolCount,
149 int footprintCount,
150 int model3DCount,
151 const QString& duration) {
152 if (!m_trayIcon) {
153 qDebug() << "System tray icon not initialized";
154 return;
155 }
156
157 if (!QSystemTrayIcon::supportsMessages()) {
158 qDebug() << "System tray does not support messages on this platform";
159 return;
160 }
161
162 QString title; // 通知标题。
163 QString message; // 通知正文。
164 QSystemTrayIcon::MessageIcon iconType = QSystemTrayIcon::Information; // 通知图标类型。
165
166 if (failureCount > 0) {
167 title = tr("导出完成");
168
169 double successRate = 0.0; // 导出成功率百分比。
170 if (successCount + failureCount > 0) {
171 successRate = (static_cast<double>(successCount) * 100.0) / (successCount + failureCount);
172 }
173
174 if (successCount == 0) {
175 iconType = QSystemTrayIcon::Critical;
176 message = tr("导出失败：%1 个元器件全部失败").arg(successCount + failureCount);
177 } else if (successRate < 50.0) {
178 iconType = QSystemTrayIcon::Warning;
179 message = tr("成功 %1 个，失败 %2 个").arg(successCount).arg(failureCount);
180 } else {
181 message = tr("成功 %1 个，失败 %2 个").arg(successCount).arg(failureCount);
182 }
183
184 if (successCount > 0) {
185 message += tr("\n输出：符号 %1 · 封装 %2 · 3D %3").arg(symbolCount).arg(footprintCount).arg(model3DCount);
186 }
187 } else {
188 iconType = QSystemTrayIcon::Information;
189 title = tr("导出完成");
190
191 if (successCount == 1) {
192 message = tr("成功导出 1 个元器件");
193 } else {
194 message = tr("成功导出 %1 个元器件").arg(successCount);
195 }
196
197 QStringList stats; // 按行组织的导出统计信息。
198 stats << tr("输出：符号 %1 · 封装 %2 · 3D %3").arg(symbolCount).arg(footprintCount).arg(model3DCount);
199
200 if (!duration.isEmpty()) {
201 stats << tr("耗时：%1").arg(duration);
202 }
203
204 message += "\n" + stats.join("\n");
205 }
206
207 qDebug() << "Attempting to show notification:" << title << message;
208 m_trayIcon->showMessage(title, message, iconType, 12000);
209 qDebug() << "Notification shown successfully";
210}
