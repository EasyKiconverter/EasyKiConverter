42bool ComponentCacheReadCoordinator::isCacheValid(const ComponentCacheService& owner, const QString& componentId) {
43 QMutexLocker diskLocker(&owner.m_diskWriteMutex); // 保护目录迁移期间的校验。
44 const QJsonObject metadata = CacheMetadataStore::read(owner.metadataPath(componentId)); // 元数据对象。
45 if (metadata.isEmpty()) {
46 return false;
47 }
48
49 if (metadata.contains(QStringLiteral("lcscId")) &&
50 (!metadata.value(QStringLiteral("lcscId")).isString() ||
51 metadata.value(QStringLiteral("lcscId")).toString().compare(componentId, Qt::CaseInsensitive) != 0)) {
52 return false;
53 }
54
55 if (!CacheMetadataStore::hasValidModel3D(metadata)) {
56 return false;
57 }
58
59 const bool hasCadJson = // CAD JSON 是否有效。
60 hasValidCadDataFile(CacheFileLayout::cadDataFile(owner.componentCacheDir(componentId)));
61 const bool hasBasicIdentity = // 元数据是否包含基本身份信息。
62 !metadata.value(QStringLiteral("lcscId")).toString().isEmpty() ||
63 !metadata.value(QStringLiteral("name")).toString().isEmpty();
64 return hasCadJson || hasBasicIdentity;

73QSharedPointer<ComponentData> ComponentCacheReadCoordinator::loadComponentData(const ComponentCacheService& owner,
74 const QString& componentId) {
75 QMutexLocker diskLocker(&owner.m_diskWriteMutex); // 保护目录迁移期间的读取。
76 const QString metaPath = owner.metadataPath(componentId); // 元数据文件路径。
77 if (!QFileInfo::exists(metaPath)) {
78 return nullptr;
79 }
80
81 QMutexLocker locker(&owner.m_mutex); // 保护一级缓存相关状态。
82 const QJsonObject metadata = CacheMetadataStore::read(metaPath); // 元数据对象。
83 if (metadata.isEmpty()) {
84 return nullptr;
85 }
86
87 if (!CacheMetadataStore::hasValidModel3D(metadata)) {
88 LOG_WARN(LogModule::Core, "Rejected component cache with invalid 3D metadata: {}", componentId);
89 return nullptr;
90 }
91

93 if (!componentData) {
94 return nullptr;
95 }
96
97 LOG_DEBUG(LogModule::Core, "Loaded component data from disk cache: {}", componentId);
98 return componentData;

107bool ComponentCacheReadCoordinator::hasSymbolFootprintCache(const ComponentCacheService& owner,
108 const QString& componentId) {
109 QMutexLocker diskLocker(&owner.m_diskWriteMutex); // 保护目录迁移期间的读取。
110 const QString cadDataPath = // CAD JSON 文件路径。
111 CacheFileLayout::cadDataFile(owner.componentCacheDir(componentId));
112 const bool hasMemoryMetadata = owner.m_memoryCache.containsMetadata(componentId); // 一级缓存是否命中。

114 if (hasMemoryMetadata) {
115 LOG_DEBUG(LogModule::Core,
116 "hasSymbolFootprintCache: memory hit for {}, cad_data.json exists: {}",
117 componentId,
118 valid);
119 } else if (!valid) {
120 LOG_DEBUG(LogModule::Core, "hasSymbolFootprintCache: no cad_data.json for {}", componentId);
121 } else {
122 LOG_DEBUG(LogModule::Core, "hasSymbolFootprintCache: valid CAD data for {}", componentId);
123 }
124 return valid;
125}
