26bool Model3DBase::fromJson(const QJsonObject& json) {
27 const auto readCoordinate = [&json](const QString& name, double current, double* value) {
28 if (!json.contains(name)) {
29 *value = current;
30 return true;
31 }
32 const QJsonValue jsonValue = json.value(name);
33 if (!jsonValue.isDouble() || !std::isfinite(jsonValue.toDouble()))
34 return false;
35 *value = jsonValue.toDouble();
36 return true;
37 };
38 double parsedX = x;
39 double parsedY = y;

41 if (!readCoordinate(QStringLiteral("x"), x, &parsedX) || !readCoordinate(QStringLiteral("y"), y, &parsedY) ||
42 !readCoordinate(QStringLiteral("z"), z, &parsedZ))
43 return false;
44 x = parsedX;
45 y = parsedY;
46 z = parsedZ;
47 return true;

84bool Model3DData::fromJson(const QJsonObject& json) {
85 QString parsedName = m_name;
86 QString parsedUuid = m_uuid;
87 QString parsedRawObj = m_rawObj;
88 Model3DBase parsedTranslation = m_translation;
89 Model3DBase parsedRotation = m_rotation;
90 Model3DBase parsedStepOffsetMm = m_stepOffsetMm;
91 QByteArray parsedStep = m_step;
92
93 const auto readOptionalString = [&json](const QString& name, QString* value) {
94 if (!json.contains(name))
95 return true;
96 const QJsonValue jsonValue = json.value(name);
97 if (!jsonValue.isString())
98 return false;
99 *value = jsonValue.toString();
100 return true;
101 };
102 if (!readOptionalString(QStringLiteral("name"), &parsedName) ||
103 !readOptionalString(QStringLiteral("uuid"), &parsedUuid) ||
104 !readOptionalString(QStringLiteral("raw_obj"), &parsedRawObj))
105 return false;
106
107 const auto readOptionalVector = [&json](const QString& name, Model3DBase* value) {
108 if (!json.contains(name))
109 return true;
110 const QJsonValue jsonValue = json.value(name);
111 if (!jsonValue.isObject())
112 return false;
113 return value->fromJson(jsonValue.toObject());
114 };
115 if (!readOptionalVector(QStringLiteral("translation"), &parsedTranslation)) {
116 qWarning() << "Failed to parse 3D model translation";
117 return false;
118 }
119
120 if (!readOptionalVector(QStringLiteral("rotation"), &parsedRotation)) {
121 qWarning() << "Failed to parse 3D model rotation";
122 return false;
123 }
124
125 if (!readOptionalVector(QStringLiteral("step_offset_mm"), &parsedStepOffsetMm)) {
126 qWarning() << "Failed to parse 3D model STEP offset";
127 return false;
128 }
129
130 // 解析 Base64 编码STEP 数据
131 if (json.contains(QStringLiteral("step"))) {
132 const QJsonValue jsonValue = json.value(QStringLiteral("step"));
133 if (!jsonValue.isString())
134 return false;
135 parsedStep = QByteArray::fromBase64(jsonValue.toString().toLatin1());
136 }
137
138 m_name = parsedName;
139 m_uuid = parsedUuid;
140 m_rawObj = parsedRawObj;
141 m_translation = parsedTranslation;
142 m_rotation = parsedRotation;
143 m_stepOffsetMm = parsedStepOffsetMm;
144 m_step = parsedStep;
145
146 return true;
147}
