77 // 根据筛选模式和搜索文本更新委托的显示状态。
78 function updateFilter() {
79 // 移除所有空格，实现更宽容的搜索 (例如 "C 2040" -> "c2040")
80 var searchTerm = componentListView.searchText.toLowerCase().replace(/\s+/g, '');
81 var filterMode = componentListView.componentListController ? componentListView.componentListController.filterMode : "all";
82 // 遍历所有项进行处理
83 for (var i = 0; i < items.count; i++) {
84 var item = items.get(i);
85 // 获取数据对象
86 var dataObj = item.model.itemData;
87 var idStr = dataObj && dataObj.componentId !== undefined ? dataObj.componentId : "";
88 var validationPhase = dataObj && dataObj.validationPhase !== undefined ? dataObj.validationPhase : "idle";
89 // 验证状态筛选（使用 validationPhase）
90 var passFilter = false;
91 if (filterMode === "all") {
92 passFilter = true;
93 } else if (filterMode === "validating") {
94 // 验证中：仅表示 CAD 验证尚未完成
95 passFilter = (validationPhase === "validating");
96 } else if (filterMode === "valid") {
97 // 有效：验证已完成或正在获取预览图的项目都属于"有效"
98 passFilter = (validationPhase === "completed" || validationPhase === "fetching_preview");
99 } else if (filterMode === "invalid") {
100 passFilter = (validationPhase === "failed");
101 }
102
103 // 搜索词筛选
104 var passSearch = true;
105 if (searchTerm !== "" && idStr.toLowerCase().indexOf(searchTerm) === -1) {
106 passSearch = false;
107 }
108
109 // 同时满足筛选和搜索条件才显示
110 item.inDisplay = passFilter && passSearch;
111 }
112 }

148 // 请求当前可见区域及缓冲区域内元件的预览图。
149 function requestVisiblePreviewImages() {
150 if (!componentListView.componentListController) {
151 return;
152 }
153
154 var ids = [];
155 var top = componentList.contentY;
156 var bottom = top + componentList.height;
157 var prefetchMargin = componentList.cellHeight;
158 for (var i = 0; i < componentList.count; ++i) {
159 var delegate = componentList.itemAtIndex(i);
160 if (!delegate || !delegate.itemData || !delegate.itemData.componentId) {
161 continue;
162 }
163
164 var itemTop = delegate.y;
165 var itemBottom = delegate.y + delegate.height;
166 if (itemBottom < top - prefetchMargin || itemTop > bottom + prefetchMargin) {
167 continue;
168 }
169
170 if (!delegate.itemData.previewImageCount || delegate.itemData.previewImageCount === 0) {
171 ids.push(delegate.itemData.componentId);
172 }
173 }
174
175 if (ids.length > 0) {
176 componentListView.componentListController.fetchPreviewImages(ids);
177 }
178 }
