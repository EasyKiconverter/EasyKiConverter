62QString FileUtils::toAbsolutePath(const QString& path) {
63 if (path.isEmpty()) {
64 return "";
65 }
66
67 QDir dir(path);
68 if (dir.isAbsolute()) {
69 // 已经是绝对路径，规范化路径分隔符
70 return dir.cleanPath(path);
71 }
72
73 // 相对路径，转换为绝对路径
74 QString absolutePath = dir.absolutePath();
75 qDebug() << "FileUtils::toAbsolutePath: Converting relative path" << path << "to" << absolutePath;
76 return absolutePath;
77}

122bool FileUtils::openFolderOnWindows(const QString& absolutePath) {
123#ifdef Q_OS_WIN
124 // 使用 explorer 命令直接打开文件夹
125 QStringList arguments;
126 arguments << QDir::toNativeSeparators(absolutePath);
127
128 qDebug() << "FileUtils::openFolderOnWindows: Running explorer with arguments:" << arguments;
129
130 QProcess process;
131 bool success = process.startDetached("explorer", arguments);
132
133 if (!success) {
134 qWarning() << "FileUtils::openFolderOnWindows: Failed to start explorer, trying QDesktopServices fallback";
135 // 降级方案：使用 QDesktopServices
136 QUrl url = QUrl::fromLocalFile(absolutePath);
137 return QDesktopServices::openUrl(url);
138 }
139
140 return success;
141#else
142 Q_UNUSED(absolutePath);
143 return false;
144#endif
145}
