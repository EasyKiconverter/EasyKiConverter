120void EasyedaApi::fetchWithNetworkClient(const QString& id, const QUrl& url, ResourceType resourceType, bool isBinary) {
121 if (!m_networkClient) {
122 emit fetchError(id, "NetworkClient not available");
123 return;
124 }
125
126 RequestProfile profile = RequestProfiles::fromType(resourceType); // 资源对应的请求配置。
127 RetryPolicy policy = RetryPolicy::fromProfile(profile, m_weakNetworkSupport); // 当前网络策略。
128 AsyncNetworkRequest* request = m_networkClient->getAsync(url, resourceType, policy); // 创建的异步请求。
129
130 {
131 QMutexLocker locker(&m_requestsMutex);
132 m_activeRequests.append({QPointer<AsyncNetworkRequest>(request), id});
133 }
134
135 connect(request,
136 &AsyncNetworkRequest::finished,
137 this,
138 [this, request, id, resourceType, isBinary](const NetworkResult&) {
139 handleAsyncRequestFinished(request, id, resourceType, isBinary);
140 });
141}

150void EasyedaApi::handleAsyncRequestFinished(AsyncNetworkRequest* request,
151 const QString& id,
152 ResourceType resourceType,
153 bool isBinary) {
154 const NetworkResult result = request->result(); // 请求完成后的网络结果。
155
156 {
157 QMutexLocker locker(&m_requestsMutex);
158 const auto it = std::find_if(m_activeRequests.cbegin(), // 活动请求列表中的匹配项。
159 m_activeRequests.cend(),
160 [request](const ActiveRequest& active) { return active.request == request; });
161 if (it == m_activeRequests.cend()) {
162 qDebug() << "EasyedaApi: Discarding callback for cancelled request:" << id;
163 request->deleteLater();
164 return;
165 }
166 m_activeRequests.erase(it);
167 }
168
169 if (result.wasCancelled) {
170 if (resourceType == ResourceType::ComponentInfo) {
171 m_isFetching = false;
172 }
173 emit fetchError(id, "Request cancelled");
174 request->deleteLater();
175 return;
176 }
177
178 if (!result.success) {
179 if (resourceType == ResourceType::ComponentInfo) {
180 m_isFetching = false;
181 }
182 emit fetchError(id, result.error);
183 request->deleteLater();
184 return;
185 }
186
187 if (isBinary) {
188 emit model3DFetched(id, result.data);
189 request->deleteLater();
190 return;
191 }
192
193 QJsonParseError parseError; // JSON 响应解析错误。
194 QJsonDocument jsonDoc = QJsonDocument::fromJson(result.data, &parseError); // 解析后的 JSON 文档。
195 if (parseError.error != QJsonParseError::NoError) {
196 emit fetchError(id, QString("Failed to parse JSON response: %1").arg(parseError.errorString()));
197 request->deleteLater();
198 return;
199 }
200
201 if (!jsonDoc.isObject()) {
202 emit fetchError(id, "JSON response is not an object");
203 request->deleteLater();
204 return;
205 }
206
207 const QJsonObject jsonObject = jsonDoc.object(); // 结构化的 JSON 响应对象。
208 // 按资源类型分发结构化响应。
209 switch (resourceType) {
210 case ResourceType::ComponentInfo:
211 m_isFetching = false;
212 handleComponentInfoResponse(id, jsonObject);
213 break;
214 case ResourceType::CadData:
215 handleCadDataResponse(id, jsonObject);
216 break;
217 default:
218 emit fetchError(id, "Unsupported JSON resource type");
219 break;
220 }
221
222 request->deleteLater();
223}

228void EasyedaApi::cancelRequest() {
229 QList<QPointer<AsyncNetworkRequest>> requestsToCancel; // 待取消的请求快照。
230 {
231 QMutexLocker locker(&m_requestsMutex);
232 for (const ActiveRequest& active : std::as_const(m_activeRequests)) {
233 requestsToCancel.append(active.request);
234 }
235 m_activeRequests.clear();
236 }
237
238 for (const auto& req : requestsToCancel) {
239 if (req && !req.isNull()) {
240 req->cancel();
241 }
242 }
243 m_isFetching = false;
244}

250void EasyedaApi::cancelRequestForId(const QString& id) {
251 QList<QPointer<AsyncNetworkRequest>> requestsToCancel; // 待取消的目标请求列表。
252 {
253 QMutexLocker locker(&m_requestsMutex);
254 for (int index = m_activeRequests.size() - 1; index >= 0; --index) {
255 const ActiveRequest& active = m_activeRequests.at(index); // 当前倒序检查的活动请求。
256 if (active.id.compare(id, Qt::CaseInsensitive) == 0) {
257 requestsToCancel.append(active.request);
258 m_activeRequests.removeAt(index);
259 }
260 }
261 }
262
263 for (const auto& request : requestsToCancel) {
264 if (request && !request.isNull()) {
265 request->cancel();
266 }
267 }
268 if (id.compare(m_currentLcscId, Qt::CaseInsensitive) == 0) {
269 m_isFetching = false;
270 }
271}
