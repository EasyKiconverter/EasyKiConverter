68QString SymbolGraphicsGenerator::generatePin(const IR::SymbolPinIR& pin) const {
69 QString content; // 单个引脚的序列化文本。
70
71 // 坐标已为 mm，直接计算相对位置
72 double x = pin.position.x() - m_originX; // 引脚相对原点的 X 坐标。
73 double y = pin.position.y() - m_originY; // 引脚相对原点的 Y 坐标。
74
75 // 引脚长度（已为 mm）
76 double length = pin.length;
77
78 // 确保引脚长度为正数（KiCad 引脚长度必须是正数）
79 length = std::abs(length);
80
81 // 确保引脚长度不为 0
82 if (length < 0.01) {
83 length = 2.54; // 默认引脚长度（100mil）
84 }
85
86 // 使用 IR 映射获取 KiCad 引脚电气类型
87 const char* kicadPinType = KiCadTypeMap::toKicadPinType(pin.electricalType);
88
89 // 引脚样式：从 IR 语义层获取
90 QString kicadPinStyle = "line";
91 if (pin.style.inverted && pin.style.clock) {
92 kicadPinStyle = "inverted_clock";
93 } else if (pin.style.inverted) {
94 kicadPinStyle = "inverted";
95 } else if (pin.style.clock) {
96 kicadPinStyle = "clock";
97 }
98
99 // 处理引脚名称和编号
100 QString pinName = pin.name;
101 pinName.replace(" ", "");
102 QString pinNumber = pin.designator; // 去除空格前的引脚编号。
103 pinNumber.replace(" ", "");
104
105 // 如果引脚名称为空，使用引脚编号
106 if (pinName.isEmpty()) {
107 pinName = pinNumber;
108 }
109
110 const bool useOriginalNamePosition = pin.hasNamePosition && !pinName.isEmpty(); // 是否保留原始名称位置。
111 const QString nameHide = useOriginalNamePosition ? QStringLiteral(" hide") : QString(); // 名称字段隐藏标记。
112
113 // 使用 IR 方向枚举获取 KiCad 角度
114 double kicadOrientation = KiCadTypeMap::toKicadAngle(pin.direction);
115
116 content += QString(" (pin %1 %2\n").arg(kicadPinType, kicadPinStyle);
117 content += QString(" (at %1 %2 %3)\n").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2).arg(kicadOrientation, 0, 'f', 0);
118 content += QString(" (length %1)\n").arg(length, 0, 'f', 2);
119 if (useOriginalNamePosition) {
120 content +=
121 QString(" (name \"%1\" (effects (font (size 1.27 1.27) (thickness 0))%2))\n").arg(pinName, nameHide);
122 } else {
123 content += QString(" (name \"%1\" (effects (font (size 1.27 1.27) (thickness 0) )))\n").arg(pinName);
124 }
125 // 引脚编号使用 KiCad 原生字段，避免把原始编号再绘制成倾斜的普通文本图元。
126 content += QString(" (number \"%1\" (effects (font (size 1.27 1.27) (thickness 0) )))\n").arg(pinNumber);
127 content += " )\n";
128
129 auto kicadJustification = [](const QString& anchor) {
130 const QString normalized = anchor.trimmed().toLower(); // 统一后的文本锚点名称。
131 if (normalized == QStringLiteral("start") || normalized == QStringLiteral("left"))
132 return QStringLiteral(" (justify left)");
133 if (normalized == QStringLiteral("end") || normalized == QStringLiteral("right"))
134 return QStringLiteral(" (justify right)");
135 if (normalized == QStringLiteral("top"))
136 return QStringLiteral(" (justify top)");
137 if (normalized == QStringLiteral("bottom"))
138 return QStringLiteral(" (justify bottom)");
139 return QString();
140 };
141 auto appendOriginalText =
142 [&](const QString& text, const QPointF& position, double rotation, const QString& anchor, double fontSizeMm) {
143 double textRotation = rotation; // KiCad 文本旋转角度。
144 if (textRotation != 0.0)
145 textRotation = 360.0 - textRotation;
146 const double size = fontSizeMm > 0.0 ? fontSizeMm : 1.27; // 文本字号，单位为毫米。
147 content += " (text\n";
148 content += QString(" \"%1\"\n").arg(text);
149 content += QString(" (at %1 %2 %3)\n")
150 .arg(position.x() - m_originX, 0, 'f', 2)
151 .arg(position.y() - m_originY, 0, 'f', 2)
152 .arg(textRotation, 0, 'f', 0);
153 content += QString(" (effects (font (size %1 %1) (thickness 0.1))%2)\n")
154 .arg(size, 0, 'f', 2)
155 .arg(kicadJustification(anchor));
156 content += " )\n";
157 };
158 if (useOriginalNamePosition)
159 appendOriginalText(pinName, pin.namePosition, pin.nameRotation, pin.nameAnchor, pin.nameFontSizeMm);
160 return content;
161}

228
229 content += " (arc\n";

231 content += QString(" (mid %1 %2)\n").arg(midX, 0, 'f', 2).arg(midY, 0, 'f', 2);
232 content += QString(" (end %1 %2)\n").arg(endX, 0, 'f', 2).arg(endY, 0, 'f', 2);
233 content += QString(" (stroke (width %1) (type default))\n").arg(strokeWidth, 0, 'f', 3);

251QString SymbolGraphicsGenerator::generateEllipse(const IR::SymbolEllipseIR& ellipse) const {
252 QString content; // 椭圆的序列化文本。
253
254 // 坐标已为 mm，计算相对位置
255 double cx = ellipse.center.x() - m_originX; // 椭圆中心 X 坐标。
256 double cy = ellipse.center.y() - m_originY; // 椭圆中心 Y 坐标。
257 double radiusX = ellipse.radiusX; // 椭圆 X 方向半径。
258 double radiusY = ellipse.radiusY; // 椭圆 Y 方向半径。
259 double strokeWidth = ellipse.strokeWidth; // 椭圆描边宽度。
260
261 // 如果是圆形（radiusX 等于 radiusY），使用 circle 元素
262 if (qAbs(radiusX - radiusY) < 0.01) {
263 content += " (circle\n";
264 content += QString(" (center %1 %2)\n").arg(cx, 0, 'f', 2).arg(cy, 0, 'f', 2);
265 content += QString(" (radius %1)\n").arg(radiusX, 0, 'f', 2);
266 content += QString(" (stroke (width %1) (type default))\n").arg(strokeWidth, 0, 'f', 3);
267 content += " (fill (type none))\n";
268 content += " )\n";
269 } else {
270 // 椭圆：使用 32 段折线近似
271 content += " (polyline\n";
272 content += " (pts";
273
274 const int segments = 32; // 椭圆折线近似使用的分段数。
275 for (int i = 0; i <= segments; ++i) {
276 double angle = 2.0 * M_PI * i / segments; // 当前近似点的角度。
277 double x = cx + radiusX * std::cos(angle); // 近似点 X 坐标。
278 double y = cy + radiusY * std::sin(angle); // 近似点 Y 坐标。
279 content += QString(" (xy %1 %2)").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2);
280 }
281
282 content += ")\n";
283 content += QString(" (stroke (width %1) (type default))\n").arg(strokeWidth, 0, 'f', 3);
284
285 if (ellipse.isFilled) {
286 content += " (fill (type background))\n";
287 } else {
288 content += " (fill (type none))\n";
289 }
290 content += " )\n";
291 }
292
293 return content;
294}

301QString SymbolGraphicsGenerator::generatePolygon(const IR::SymbolPolygonIR& polygon) const {
302 QString content; // 多边形的序列化文本。
303 double strokeWidth = polygon.strokeWidth; // 多边形描边宽度。
304
305 // IR 中坐标已解析为 QList<QPointF>
306 if (polygon.points.size() >= 2) {
307 // KiCad V6 不支持 polygon 元素，使用 polyline 代替
308 content += " (polyline\n";
309 content += " (pts";
310 QString firstPoint; // 首个去重后的顶点文本。
311 QString lastPoint; // 最近一次写入的顶点文本。
312 for (const QPointF& pt : polygon.points) {
313 double x = pt.x() - m_originX; // 顶点相对原点的 X 坐标。
314 double y = pt.y() - m_originY; // 顶点相对原点的 Y 坐标。
315 QString point = QString(" (xy %1 %2)").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2); // KiCad 顶点文本。
316
317 // 避免重复点
318 if (point != lastPoint) {
319 content += point;
320 if (firstPoint.isEmpty()) {
321 firstPoint = point;
322 }
323 lastPoint = point;
324 }
325 }
326 // 多边形总是重复第一个点以闭合
327 if (!firstPoint.isEmpty() && firstPoint != lastPoint) {
328 content += firstPoint;
329 }
330 content += ")\n";
331 content += QString(" (stroke (width %1) (type default))\n").arg(strokeWidth, 0, 'f', 3);
332 if (polygon.isFilled) {
333 content += " (fill (type background))\n";
334 } else {
335 content += " (fill (type none))\n";
336 }
337 content += " )\n";
338 }
339
340 return content;
341}

443QString SymbolGraphicsGenerator::generateText(const IR::SymbolTextIR& text) const {
444 QString content; // 文本图元的序列化文本。
445
446 // 坐标已为 mm，计算相对位置
447 double x = text.position.x() - m_originX; // 文本相对原点的 X 坐标。
448 double y = text.position.y() - m_originY; // 文本相对原点的 Y 坐标。
449
450 // IR 中 fontSizeMm 已从 pt 转换；部分 EasyEDA 文本记录未提供字号，
451 // 使用与平台属性一致的默认字号，避免生成 0.00 的不可见文本。
452 double fontSize = text.fontSizeMm > 0.0 ? text.fontSizeMm : 1.27; // 文本字号，单位为毫米。
453
454 // 处理粗体和斜体（IR 中 italic 已为 bool）
455 QString fontStyle = "";
456 if (text.bold) {
457 fontStyle += "bold ";
458 }
459 if (text.italic) {
460 fontStyle += "italic ";
461 }
462 if (fontStyle.isEmpty()) {
463 fontStyle = "normal";
464 } else {
465 fontStyle = fontStyle.trimmed();
466 }
467
468 // 处理旋转角度
469 double rotation = text.rotation;
470 if (rotation != 0) {
471 rotation = 360 - rotation;
472 }
473
474 // 处理可见性
475 QString hide = text.visible ? "" : "hide";
476
477 // 转义文本内容
478 QString textContent = text.text;
479 textContent.replace(" ", "");
480 if (textContent.isEmpty()) {
481 textContent = "~";
482 }
483
484 // 生成文本元素
485 content += " (text\n";
486 content += QString(" \"%1\"\n").arg(textContent);
487 content += QString(" (at %1 %2 %3)\n").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2).arg(rotation, 0, 'f', 0);
488 content += QString(" (effects (font (size %1 %2) (thickness 0.1) %3) %4)\n")
489 .arg(fontSize, 0, 'f', 2)
490 .arg(fontSize, 0, 'f', 2)
491 .arg(fontStyle)
492 .arg(hide);
493 content += " )\n";
494
495 return content;
496}
