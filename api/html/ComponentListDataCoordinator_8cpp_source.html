41void ComponentListDataCoordinator::handleCadData(ComponentListViewModel& viewModel,
42 const QString& componentId,
43 const ComponentData& data) {
44 // 查找需要更新 CAD 数据的列表项。
45 auto item = viewModel.findItemData(componentId);
46 if (!item) {
47 qWarning() << "handleCadDataReady: item not found for componentId:" << componentId
48 << ", m_componentIdIndex size:" << viewModel.m_componentIdIndex.size();
49 return;
50 }
51
52 // 已经验证通过的项目只更新数据，避免重复触发验证完成和队列推进。
53 if (item->isValid()) {
54 item->setComponentData(QSharedPointer<ComponentData>::create(data));
55 return;
56 }
57
58 item->setComponentData(QSharedPointer<ComponentData>::create(data));
59 item->setFetching(false);
60 item->setValid(true);
61 item->setRetryable(true);
62 item->setValidationPhase("completed");
63 item->setErrorMessage("");
64 viewModel.scheduleListUpdate();
65
66 viewModel.m_validationStateManager->onComponentValidated(componentId);
67 QTimer::singleShot(0, &viewModel, [&viewModel, componentId]() { viewModel.onValidationComplete(componentId); });
68}

76void ComponentListDataCoordinator::handleFetchError(ComponentListViewModel& viewModel,
77 const QString& componentId,
78 const QString& error) {
79 qWarning() << "Fetch error for:" << componentId << "-" << error;
80
81 // 查找需要记录错误的列表项。
82 auto item = viewModel.findItemData(componentId);
83 if (!item) {
84 return;
85 }
86
87 // 预览图失败不应改变已经完成的 CAD 验证状态。
88 // 记录 CAD 验证是否已经完成。
89 const bool wasAlreadyValid = item->isValid();
90 item->setFetching(false);
91
92 if (!wasAlreadyValid) {
93 // 判断错误是否属于 CAD 数据失败。
94 const bool isCadDataFailure = ComponentValidationErrorPolicy::isCadDataFailure(error);
95 if (isCadDataFailure) {
96 item->setValid(false);
97 item->setValidationPhase("failed");
98 // 判断该错误是否不应重试。
99 const bool nonRetryable = ComponentValidationErrorPolicy::isNonRetryable(error);
100 item->setRetryable(!nonRetryable);
101 if (ComponentValidationErrorPolicy::isNotFound(error)) {
102 item->setErrorMessage(QCoreApplication::translate("ComponentListViewModel", "元器件不存在（404）"));
103 } else {
104 item->setErrorMessage(error);
105 }
106 viewModel.m_validationStateManager->onComponentFailed(componentId);
107 viewModel.scheduleListUpdate();
108 QTimer::singleShot(
109 0, &viewModel, [&viewModel, componentId]() { viewModel.onValidationComplete(componentId); });
110 } else {
111 // 预览图失败只更新提示文本，不改变 CAD 验证结果。
112 switch (ComponentValidationErrorPolicy::classifyPreviewError(error)) {
113 case ComponentValidationErrorPolicy::PreviewErrorKind::Timeout:
114 item->setErrorMessage(
115 QCoreApplication::translate("ComponentListViewModel", "预览图获取超时（网络不稳定）"));
116 break;
117 case ComponentValidationErrorPolicy::PreviewErrorKind::NotFound:
118 item->setErrorMessage(QCoreApplication::translate("ComponentListViewModel", "预览图不存在"));
119 break;
120 case ComponentValidationErrorPolicy::PreviewErrorKind::Forbidden:
121 item->setErrorMessage(QCoreApplication::translate("ComponentListViewModel", "预览图获取被拒绝"));
122 break;
123 case ComponentValidationErrorPolicy::PreviewErrorKind::Other:
124 item->setErrorMessage(QCoreApplication::translate("ComponentListViewModel", "预览图获取失败"));
125 break;
126 }
127 }
128 }
129
130 viewModel.scheduleListUpdate();
131}

141void ComponentListDataCoordinator::handleLcscData(ComponentListViewModel& viewModel,
142 const QString& componentId,
143 const QString& manufacturerPart,
144 const QString& datasheetUrl,
145 const QStringList& imageUrls) {
146 // 查找需要更新 LCSC 元数据的列表项。
147 auto item = viewModel.findItemData(componentId);
148 if (!item) {
149 qWarning() << "Component" << componentId << "not found in list, cannot update LCSC data";
150 return;
151 }
152
153 if (item->componentData()) {
154 // 获取列表项中的共享元器件数据。
155 auto data = item->componentData();
156 if (!manufacturerPart.isEmpty()) {
157 data->setManufacturerPart(manufacturerPart);
158 }
159 if (!datasheetUrl.isEmpty()) {
160 data->setDatasheet(datasheetUrl);
161 data->setDatasheetFormat(datasheetUrl.toLower().contains(".html") ? "html" : "pdf");
162 }
163 if (!imageUrls.isEmpty()) {
164 data->setPreviewImages(imageUrls);
165 }
166 }
167
168 if (!manufacturerPart.isEmpty()) {
169 item->setNameSilent(manufacturerPart);
170 viewModel.m_batchUpdateItems.add(item);
171 if (!viewModel.m_batchUpdateTimer->isActive()) {
172 viewModel.m_batchUpdateTimer->start();
173 }
174 }
