24void ComponentRequestCoordinator::start(ComponentService& service, const QString& componentId, bool fetch3DModel) {
25 // 统一编号大小写，确保请求状态和缓存键使用同一表示。
26 const QString normalizedId = componentId.toUpper(); // 规范化后的元件编号。
27 ComponentData reusableData; // 可复用的已有组件数据。
28 uint64_t generation = 0; // 当前请求捕获的缓存代次。
29 bool reuseExistingData = false; // 是否直接复用已有数据。
30 bool shouldSkipAsDuplicate = false; // 是否因重复活动请求而跳过。
31
32 // 先登记请求状态，避免并发入口为同一个编号创建多个后台任务。
33 {
34 QMutexLocker locker(&service.m_fetchingComponentsMutex);
35 if (service.m_fetchingComponents.contains(normalizedId)) {
36 const ComponentService::FetchingComponent& existing =
37 service.m_fetchingComponents[normalizedId]; // 已有请求状态。
38 const bool hasReusableData = // 已有数据是否可复用。
39 !existing.data.lcscId().isEmpty() && (existing.hasCadData || existing.data.symbolData() != nullptr ||
40 existing.data.footprintData() != nullptr);
41
42 if (hasReusableData) {
43 reusableData = existing.data;
44 reuseExistingData = true;
45 } else if (service.m_parallelContext != nullptr || !existing.requestActive) {
46 qDebug() << "ComponentService: Removing stale fetching state for" << normalizedId << "before retry";
47 service.m_fetchingComponents.remove(normalizedId);
48 } else {
49 shouldSkipAsDuplicate = true;
50 }
51 }
52
53 if (!reuseExistingData && !shouldSkipAsDuplicate && !service.m_fetchingComponents.contains(normalizedId)) {
54 // 创建占位条目并捕获当前缓存代次，旧回调不能覆盖后续请求。
55 ComponentService::FetchingComponent& fetching =
56 service.m_fetchingComponents[normalizedId]; // 新请求占位状态。
57 service.initializeFetchingComponent(fetching, normalizedId, fetch3DModel);
58 ComponentCacheService::instance()->clearTombstone(normalizedId);
59 ComponentCacheService::instance()->clearGlobalTombstone();
60 }
61 }
62
63 if (reuseExistingData) {
64 qDebug() << "ComponentService: Reusing existing fetched data for:" << normalizedId;
65 service.updateComponentCache(normalizedId, reusableData);
66 emit service.cadDataReady(normalizedId, reusableData);
67 if (service.m_parallelContext != nullptr) {
68 service.handleParallelDataCollected(normalizedId, reusableData);
69 }
70 return;
71 }
72
73 if (shouldSkipAsDuplicate) {
74 qDebug() << "ComponentService: Already fetching for" << normalizedId << ", skipping duplicate request";
75 return;
76 }
77
78 ComponentCacheService* cache = ComponentCacheService::instance(); // 全局缓存服务。
79 if (service.m_api) {
80 service.m_api->setWeakNetworkSupport(ConfigService::instance()->getWeakNetworkSupport());
81 }
82 if (cache->hasSymbolFootprintCache(normalizedId)) {
83 qDebug() << "ComponentService: Symbol/Footprint cache hit for" << normalizedId
84 << ", loading from cache (component exists)";
85 QTimer::singleShot(0, &service, [&service, normalizedId, fetch3DModel, cache]() {
86 service.loadComponentDataFromCacheAsync(normalizedId, fetch3DModel, cache);
87 });
88 return;
89 }
90
91 // 缓存未命中时在后台线程解析 CAD，避免请求入口阻塞主线程。
92 {
93 QMutexLocker locker(&service.m_fetchingComponentsMutex);
94 const auto it = service.m_fetchingComponents.find(normalizedId); // 当前请求状态。
95 if (it != service.m_fetchingComponents.end()) {
96 generation = it->cacheGeneration;
97 }
98 }
99 if (generation == 0) {
100 qWarning() << "ComponentRequestCoordinator: No fetching state for" << normalizedId << ", aborting";
101 return;
102 }
103
104 const QFuture<CadFetchTaskResult> future = // 后台 CAD 获取任务。
105 QtConcurrent::run([normalizedId]() { return CadDataLoader::fetchAndParseCadData(normalizedId); });
106 auto* watcher = new QFutureWatcher<CadFetchTaskResult>(&service); // CAD 获取完成观察器。
107 QObject::connect(
108 watcher, &QFutureWatcher<CadFetchTaskResult>::finished, &service, [&service, watcher, generation]() {
109 const CadFetchTaskResult result = watcher->result(); // CAD 获取结果。
110 watcher->deleteLater();
111 service.handleCadFetchResult(result, generation);
112 });
113 watcher->setFuture(future);
114}
