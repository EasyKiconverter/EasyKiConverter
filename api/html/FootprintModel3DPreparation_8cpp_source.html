33bool calculateStepGeometryCenter(const QByteArray& stepData, Model3DBase* center) {
34 if (stepData.isEmpty() || center == nullptr) {
35 return false;
36 }
37
38 const QString content = QString::fromLatin1(stepData);
39 static const QRegularExpression pointRegex( // STEP 笛卡尔点定义匹配器。
40 QStringLiteral("#(\\d+)\\s*=\\s*CARTESIAN_POINT\\s*\\(\\s*('[^']*'|\\$)\\s*,\\s*\\(([^()]*)\\)\\s*\\)"),
41 QRegularExpression::CaseInsensitiveOption);
42 static const QRegularExpression vertexPointRegex(QStringLiteral("VERTEX_POINT\\s*\\([^,]*,\\s*#(\\d+)\\s*\\)"),
43 QRegularExpression::CaseInsensitiveOption);
44
45 struct Vec3 {
46 double x = 0;
47 double y = 0;
48 double z = 0;
49 };
50
51 QHash<int, Vec3> allPoints;
52 QRegularExpressionMatchIterator pointMatches = pointRegex.globalMatch(content);
53 while (pointMatches.hasNext()) {
54 const QRegularExpressionMatch match = pointMatches.next();
55 bool idOk = false;
56 const int pointId = match.captured(1).toInt(&idOk);
57 if (!idOk) {
58 continue;
59 }
60
61 const QStringList coordinateParts = match.captured(3).split(',', Qt::SkipEmptyParts);
62 if (coordinateParts.size() < 3) {
63 continue;
64 }
65
66 bool okX = false;
67 bool okY = false;
68 bool okZ = false;
69 const double x = coordinateParts.at(0).trimmed().toDouble(&okX);
70 const double y = coordinateParts.at(1).trimmed().toDouble(&okY);
71 const double z = coordinateParts.at(2).trimmed().toDouble(&okZ);
72 if (okX && okY && okZ) {
73 allPoints.insert(pointId, {x, y, z});
74 }
75 }
76
77 double minX = std::numeric_limits<double>::max();
78 double maxX = std::numeric_limits<double>::lowest();
79 double minY = std::numeric_limits<double>::max();
80 double maxY = std::numeric_limits<double>::lowest();
81 double minZ = std::numeric_limits<double>::max();
82 bool hasGeometryPoint = false;
83 QRegularExpressionMatchIterator vertexMatches = vertexPointRegex.globalMatch(content);
84 while (vertexMatches.hasNext()) {
85 bool ok = false;
86 const int pointId = vertexMatches.next().captured(1).toInt(&ok);
87 if (!ok || !allPoints.contains(pointId)) {
88 continue;
89 }
90
91 const Vec3 point = allPoints.value(pointId);
92 minX = qMin(minX, point.x);
93 maxX = qMax(maxX, point.x);
94 minY = qMin(minY, point.y);
95 maxY = qMax(maxY, point.y);
96 minZ = qMin(minZ, point.z);
97 hasGeometryPoint = true;
98 }
99
100 if (!hasGeometryPoint) {
101 return false;
102 }
103
104 center->x = (minX + maxX) / 2.0;
105 center->y = (minY + maxY) / 2.0;
106 center->z = minZ;
107 return true;
108}

115double calculateStepZOffset(double wrlDisplayMinZ, double stepMinZ) {
116 if (wrlDisplayMinZ == std::numeric_limits<double>::max()) {
117 return stepMinZ > kModelZOffsetEpsilonMm ? -stepMinZ : 0.0;
118 }
119 const double offset = wrlDisplayMinZ - stepMinZ;
120 return qAbs(offset) < kModelZOffsetEpsilonMm ? 0.0 : offset;
121}

129bool readModelSourceZMm(const QByteArray& cadJsonRaw, const QString& modelUuid, double* zMm) {
130 if (cadJsonRaw.isEmpty() || modelUuid.isEmpty() || zMm == nullptr) {
131 return false;
132 }
133
134 QJsonParseError parseError;
135 const QJsonDocument cadDoc = QJsonDocument::fromJson(cadJsonRaw, &parseError);
136 if (parseError.error != QJsonParseError::NoError || !cadDoc.isObject()) {
137 return false;
138 }
139
140 const QJsonObject packageData = // 封装详情中的原始数据对象。
141 cadDoc.object().value(QStringLiteral("packageDetail")).toObject().value(QStringLiteral("dataStr")).toObject();
142 const QJsonArray shapes = packageData.value(QStringLiteral("shape")).toArray();
143 for (int pass = 0; pass < 2; ++pass) { // 当前按 UUID 或类型的匹配轮次。
144 for (const QJsonValue& shapeValue : shapes) { // 当前图形 JSON 值。
145 const QString shape = shapeValue.toString();
146 if (!shape.startsWith(QStringLiteral("SVGNODE~"))) {
147 continue;
148 }
149
150 QJsonParseError svgParseError;
151 const QJsonDocument svgDoc = // 解析后的 SVG 节点文档。
152 QJsonDocument::fromJson(shape.mid(QStringLiteral("SVGNODE~").size()).toUtf8(), &svgParseError);
153 if (svgParseError.error != QJsonParseError::NoError || !svgDoc.isObject()) {
154 continue;
155 }
156
157 const QJsonObject attrs = svgDoc.object().value(QStringLiteral("attrs")).toObject();
158 if (pass == 0 && attrs.value(QStringLiteral("uuid")).toString() != modelUuid) {
159 continue;
160 }
161 if (pass == 1 && attrs.value(QStringLiteral("c_etype")).toString() != QLatin1String("outline3D")) {
162 continue;
163 }
164
165 bool ok = false;
166 const QJsonValue zValue = attrs.value(QStringLiteral("z"));
167 const double z = zValue.isString() ? zValue.toString().toDouble(&ok) : zValue.toDouble();
168 ok = ok || zValue.isDouble();
169 if (ok) {
170 *zMm = GeometryUtils::convertToMm(z);
171 }
172 return ok;
173 }
174 }
175 return false;
176}

184double calculateWrlBaseZOffset(const QString& footprintType, double wrlDisplayMinZ, double sourceZMm) {
185 if (footprintType != QLatin1String(kSmdFootprintType) || wrlDisplayMinZ == std::numeric_limits<double>::max()) {
186 return 0.0;
187 }
188 const double offset = -wrlDisplayMinZ + sourceZMm;
189 return offset > kModelZOffsetEpsilonMm ? offset : 0.0;
190}

196double zOffsetMmToEasyEdaUnits(double zOffsetMm) {
197 const double roundedOffset = std::round(zOffsetMm * 100.0) / 100.0;
198 return -(roundedOffset - kEasyEdaZOffsetBias) / kEasyEdaUnitToMm;
199}

210void prepare(FootprintData& footprint,
211 const QSharedPointer<ComponentData>& componentData,
212 const QString& componentId,
213 uint64_t generation) {
214 Model3DData model3D = footprint.model3D();
215 if (componentData->model3DData() && !componentData->model3DData()->uuid().isEmpty()) {
216 model3D.setUuid(componentData->model3DData()->uuid());
217 }
218 if (model3D.name().isEmpty() && componentData->model3DData()) {
219 model3D.setName(componentData->model3DData()->name());
220 }
221 if (model3D.name().isEmpty()) {
222 model3D.setName(footprint.info().model3DName);
223 }
224 if (model3D.uuid().isEmpty()) {
225 return;
226 }
227
228 QByteArray stepData =
229 componentData->model3DData() ? componentData->model3DData()->step() : QByteArray();
230 if (stepData.isEmpty()) {
231 stepData = ComponentCacheService::instance()->loadModel3D(model3D.uuid(), QStringLiteral("step"));
232 }
233 if (!stepData.isEmpty() && !Exporter3DModel::hasUsableStepData(stepData)) {
234 qWarning() << "FootprintModel3DPreparation: Ignoring malformed cached STEP for" << componentId << "uuid"
235 << model3D.uuid();
236 stepData.clear();
237 }
238 if (!stepData.isEmpty()) {
239 ComponentCacheService::instance()->saveModel3D(model3D.uuid(), stepData, QStringLiteral("step"), generation);
240 }
241 if (stepData.isEmpty()) {
242 Exporter3DModel exporter;
243 if (exporter.downloadStepDataSync(model3D.uuid(), &stepData) && Exporter3DModel::hasUsableStepData(stepData)) {
244 ComponentCacheService::instance()->saveModel3D(
245 model3D.uuid(), stepData, QStringLiteral("step"), generation);
246 }
247 }
248 if (!stepData.isEmpty() && !Exporter3DModel::hasUsableStepData(stepData)) {
249 qWarning() << "FootprintModel3DPreparation: Rejecting invalid downloaded STEP for" << componentId << "uuid"
250 << model3D.uuid();
251 stepData.clear();
252 }
253 if (stepData.isEmpty()) {
254 return;
255 }
256
257 Model3DBase geometryCenter;
258 if (!calculateStepGeometryCenter(stepData, &geometryCenter)) {
259 qWarning() << "FootprintModel3DPreparation: STEP geometry center unavailable for" << componentId;
260 }
261
262 QByteArray objData = componentData->model3DObjRaw();
263 if (objData.isEmpty() && componentData->model3DData()) {
264 objData = componentData->model3DData()->rawObj().toUtf8();
265 }
266 if (!objData.isEmpty() && !Exporter3DModel::hasUsableObjGeometry(objData)) {
267 qWarning() << "FootprintModel3DPreparation: Ignoring malformed OBJ data for" << componentId;
268 objData.clear();
269 }
270 if (!objData.isEmpty()) {
271 ComponentCacheService::instance()->saveModel3D(model3D.uuid(), objData, QStringLiteral("obj"), generation);
272 }
273 if (objData.isEmpty()) {
274 objData = ComponentCacheService::instance()->loadModel3D(model3D.uuid(), QStringLiteral("obj"));
275 }
276 if (!objData.isEmpty() && !Exporter3DModel::hasUsableObjGeometry(objData)) {
277 objData.clear();
278 }
279
280 QByteArray wrlData;
281 if (objData.isEmpty()) {
282 wrlData = ComponentCacheService::instance()->loadModel3D(model3D.uuid(), QStringLiteral("wrl"));
283 }
284 if (!wrlData.isEmpty() && !Exporter3DModel::hasUsableWrlGeometry(wrlData)) {
285 wrlData.clear();
286 }
287 if (objData.isEmpty() && wrlData.isEmpty()) {
288 Exporter3DModel exporter;
289 if (exporter.downloadObjDataSync(model3D.uuid(), &objData) && Exporter3DModel::hasUsableObjGeometry(objData)) {
290 ComponentCacheService::instance()->saveModel3D(model3D.uuid(), objData, QStringLiteral("obj"), generation);
291 }
292 }
293
294 double wrlDisplayMinZ = std::numeric_limits<double>::max();
295 if (!objData.isEmpty()) {
296 const double rawMinZ = Exporter3DModel::calculateObjMinZ(objData);
297 wrlDisplayMinZ = rawMinZ > 0.0 ? 0.0 : rawMinZ;
298 } else if (!wrlData.isEmpty()) {
299 wrlDisplayMinZ = Exporter3DModel::calculateWrlDisplayMinZ(wrlData);
300 }
301
302 QByteArray cadJsonRaw = componentData->cadJsonRaw();
303 if (cadJsonRaw.isEmpty()) {
304 cadJsonRaw = ComponentCacheService::instance()->loadCadDataJson(componentId);
305 }
306 double sourceZMm = GeometryUtils::convertToMm(model3D.translation().z);
307 (void)readModelSourceZMm(cadJsonRaw, model3D.uuid(), &sourceZMm);
308 const double wrlBaseZOffset =
309 calculateWrlBaseZOffset(footprint.info().type, wrlDisplayMinZ, sourceZMm);
310 if (wrlBaseZOffset > 0.0) {
311 Model3DBase translation = model3D.translation();
312 translation.z = zOffsetMmToEasyEdaUnits(wrlBaseZOffset);
313 model3D.setTranslation(translation);
314 }
315
316 Model3DBase stepOffset;
317 stepOffset.x = -geometryCenter.x;
318 stepOffset.y = -geometryCenter.y;
319 stepOffset.z = calculateStepZOffset(wrlDisplayMinZ, geometryCenter.z);
320 model3D.setStepOffsetMm(stepOffset);
321 model3D.setStep(stepData);
322 footprint.setModel3D(model3D);
323 qDebug() << "FootprintModel3DPreparation: Prepared model for" << componentId << "uuid" << model3D.uuid()
324 << "offset:" << stepOffset.x << stepOffset.y << stepOffset.z;
325}
