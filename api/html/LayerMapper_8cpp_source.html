26void LayerMapper::initializeLayerMapping() {
27 // 初始KiCad 图层名称
28 m_kicadLayerNames[F_Cu] = "F.Cu";
29 m_kicadLayerNames[B_Cu] = "B.Cu";
30 for (int i = 1; i <= 30; i++) {
31 m_kicadLayerNames[i] = QString("In%1.Cu").arg(i);
32 }
33 m_kicadLayerNames[F_SilkS] = "F.SilkS";
34 m_kicadLayerNames[B_SilkS] = "B.SilkS";
35 m_kicadLayerNames[F_Mask] = "F.Mask";
36 m_kicadLayerNames[B_Mask] = "B.Mask";
37 m_kicadLayerNames[F_Paste] = "F.Paste";
38 m_kicadLayerNames[B_Paste] = "B.Paste";
39 m_kicadLayerNames[F_Adhes] = "F.Adhes";
40 m_kicadLayerNames[B_Adhes] = "B.Adhes";
41 m_kicadLayerNames[Edge_Cuts] = "Edge.Cuts";
42 m_kicadLayerNames[F_CrtYd] = "F.CrtYd";
43 m_kicadLayerNames[B_CrtYd] = "B.CrtYd";
44 m_kicadLayerNames[F_Fab] = "F.Fab";
45 m_kicadLayerNames[B_Fab] = "B.Fab";
46 m_kicadLayerNames[Dwgs_User] = "Dwgs.User";
47 m_kicadLayerNames[Cmts_User] = "Cmts.User";
48 m_kicadLayerNames[Eco1_User] = "Eco1.User";
49 m_kicadLayerNames[Eco2_User] = "Eco2.User";
50 m_kicadLayerNames[Margin] = "Margin";
51 for (int i = 1; i <= 9; i++) {
52 m_kicadLayerNames[User_1 + i - 1] = QString("User.%1").arg(i);
53 }
54
55 // 初始化嘉立创图层 ID -> KiCad 图层 ID 映射
56 // 一、电气信号层
57 m_layerIdMapping[1] = F_Cu; // TopLayer -> F.Cu
58 m_layerIdMapping[2] = B_Cu; // BottomLayer -> B.Cu
59
60 // 内电Inner1~Inner32 (ID 21-52) -> In1.Cu~In32.Cu
61 for (int i = 0; i < 32; i++) {
62 m_layerIdMapping[21 + i] = In1_Cu + i;
63 }
64
65 // 二、丝印层
66 m_layerIdMapping[3] = F_SilkS; // TopSilkLayer -> F.SilkS
67 m_layerIdMapping[4] = B_SilkS; // BottomSilkLayer -> B.SilkS
68
69 // 三、阻焊层
70 m_layerIdMapping[7] = F_Mask; // TopSolderMaskLayer -> F.Mask
71 m_layerIdMapping[8] = B_Mask; // BottomSolderMaskLayer -> B.Mask
72
73 // 四、助焊层
74 m_layerIdMapping[5] = F_Paste; // TopPasteMaskLayer -> F.Paste
75 m_layerIdMapping[6] = B_Paste; // BottomPasteMaskLayer -> B.Paste
76
77 // 五、机械与结构
78 m_layerIdMapping[10] = Edge_Cuts; // BoardOutLine -> Edge.Cuts
79 m_layerIdMapping[15] = F_Fab; // Mechanical -> F.Fab (顶层装配)
80 m_layerIdMapping[99] = F_CrtYd; // ComponentShapeLayer -> F.CrtYd (元件占位)
81 m_layerIdMapping[100] = F_Fab; // LeadShapeLayer -> F.Fab (引脚形状)
82 m_layerIdMapping[101] = F_SilkS; // ComponentPolarityLayer -> F.SilkS (极性标
83
84 // 六、装配与文档
85 m_layerIdMapping[13] = F_Fab; // TopAssembly -> F.Fab
86 m_layerIdMapping[14] = B_Fab; // BottomAssembly -> B.Fab
87 m_layerIdMapping[12] = Dwgs_User; // Document -> Dwgs.User
88
89 // 七、特殊功能层
90 m_layerIdMapping[9] = Dwgs_User; // Ratlines -> Dwgs.User (飞线)
91 m_layerIdMapping[11] = F_Cu; // Multi-Layer -> F.Cu (通孔焊盘，实际在所有层)
92 m_layerIdMapping[19] = Dwgs_User; // 3DModel -> Dwgs.User (3D 模型引用)
93 m_layerIdMapping[102] = Dwgs_User; // Hole -> Dwgs.User (非金属化
94 m_layerIdMapping[103] = Cmts_User; // DRCError -> Cmts.User (DRC 错误)
95
96 // 初始化嘉立创图层名称 -> KiCad 图层 ID 映射
97 // 内层索引计算：In.(n-14).Cu
98 // 例如：Inner1 -> In1.Cu, Inner2 -> In2.Cu...
99 m_layerNameMapping["TopLayer"] = F_Cu;
100 m_layerNameMapping["BottomLayer"] = B_Cu;
101 for (int i = 1; i <= 32; i++) {
102 // 使用清晰的公式：In1_Cu + (i - 1)
103 // 使用从 In1.Cu 开始的连续枚举值映射内部铜层。
104 m_layerNameMapping[QString("Inner%1").arg(i)] = In1_Cu + (i - 1);
105 }
106 m_layerNameMapping["TopSilkLayer"] = F_SilkS;
107 m_layerNameMapping["BottomSilkLayer"] = B_SilkS;
108 m_layerNameMapping["TopSolderMaskLayer"] = F_Mask;
109 m_layerNameMapping["BottomSolderMaskLayer"] = B_Mask;
110 m_layerNameMapping["TopPasteMaskLayer"] = F_Paste;
111 m_layerNameMapping["BottomPasteMaskLayer"] = B_Paste;
112 m_layerNameMapping["BoardOutLine"] = Edge_Cuts;
113 m_layerNameMapping["Mechanical"] = F_Fab;
114 m_layerNameMapping["ComponentShapeLayer"] = F_CrtYd;
115 m_layerNameMapping["LeadShapeLayer"] = F_Fab;
116 m_layerNameMapping["ComponentPolarityLayer"] = F_SilkS;
117 m_layerNameMapping["TopAssembly"] = F_Fab;
118 m_layerNameMapping["BottomAssembly"] = B_Fab;
119 m_layerNameMapping["Document"] = Dwgs_User;
120 m_layerNameMapping["Ratlines"] = Dwgs_User;
121 m_layerNameMapping["Multi-Layer"] = F_Cu;
122 m_layerNameMapping["3DModel"] = Dwgs_User;
123 m_layerNameMapping["Hole"] = Dwgs_User;
124 m_layerNameMapping["DRCError"] = Cmts_User;
125
126 qDebug() << "LayerMapper initialized with" << m_layerIdMapping.size() << "layer mappings";
127}

134int LayerMapper::mapToKiCadLayer(int easyedaLayerId) const {
135 if (m_layerIdMapping.contains(easyedaLayerId)) {
136 return m_layerIdMapping.value(easyedaLayerId);

233bool LayerMapper::isMechanicalLayer(int kicadLayerId) {
234 // 机械层：Edge.Cuts (44), F.CrtYd (45), B.CrtYd (46), F.Fab (47), B.Fab (48)
235 // Dwgs.User (49), Cmts.User (50), Eco1.User (51), Eco2.User (52), Margin (53)
236 return (kicadLayerId >= Edge_Cuts && kicadLayerId <= Margin) || (kicadLayerId >= User_1 && kicadLayerId <= User_9);
