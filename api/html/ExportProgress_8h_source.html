269 Stage currentStage = Stage::Idle;

272 QMap<QString, ExportTypeProgress> exportTypeProgress;
273 QDateTime startTime;
274 QDateTime endTime;
275
280 int totalSuccessCount() const {
281 int count = 0;
282 for (const auto& p : exportTypeProgress) {
283 count += p.successCount;
284 }
285 return count;
286 }
287
292 int totalFailedCount() const {
293 int count = 0;
294 for (const auto& p : exportTypeProgress) {
295 count += p.failedCount;
296 }
297 return count;
298 }
299
304 bool isComplete() const {
305 if (currentStage != Stage::Exporting && currentStage != Stage::Preloading) {
306 return currentStage == Stage::Completed || currentStage == Stage::Cancelled ||
307 currentStage == Stage::Failed;
308 }
309 if (currentStage == Stage::Preloading) {
310 return preloadProgress.isComplete();
311 }
312 for (const auto& p : exportTypeProgress) {
313 if (!p.isComplete())
314 return false;
315 }
316 return true;
317 }
318
330 int overallPercentage() const {
331 if (currentStage == Stage::Idle)
332 return 0;
333 if (currentStage == Stage::Preloading)
334 return preloadProgress.percentage();
335 if (currentStage == Stage::Exporting) {
336 int total = 0;
337 int completed = 0;
338 for (const auto& p : exportTypeProgress) {
339 total += p.totalCount;
340 completed += p.completedCount;
341 }
342 return total > 0 ? (completed * 100 / total) : 0;
343 }
344 if (currentStage == Stage::Completed)
345 return 100;
346 return 0;
347 }
