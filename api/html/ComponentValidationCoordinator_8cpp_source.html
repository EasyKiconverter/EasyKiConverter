12void ComponentValidationCoordinator::start(ComponentListViewModel& viewModel) {
13 // 读取配置的验证并发数上限。
14 const int concurrentWorkers = ConfigService::instance()->getValidationConcurrentCount();
15
16 // 只调度仍处于验证状态的元件，避免延迟完成回调导致重复请求。
17 for (ComponentListItemData* item : viewModel.m_componentList) {
18 if (item && item->isFetching() && !item->isValid()) {
19 viewModel.m_validationQueue.enqueue(item->componentId());
20 }
21 }
22
23 if (viewModel.m_validationQueue.isEmpty()) {
24 return;
25 }
26
27 // 首次启动时记录队列总数，后续追加元件由 ValidationStateManager 单独维护。
28 if (viewModel.m_validationTotalCount == 0) {
29 viewModel.m_validationTotalCount = viewModel.m_validationQueue.size();
30 }
31
32 // 计算本轮首次启动的验证请求数量。
33 const int initialCount = qMin(concurrentWorkers, viewModel.m_validationQueue.size());
34 for (int i = 0; i < initialCount; ++i) {
35 // 取出下一个待验证的元器件编号。
36 const QString componentId = viewModel.m_validationQueue.takeNext();
37 viewModel.m_service->fetchComponentData(componentId, false);
38 ++viewModel.m_validationPendingCount;
39 }

62void ComponentValidationCoordinator::complete(ComponentListViewModel& viewModel, const QString& componentId) {
63 ++viewModel.m_validationCompletedCount;
64 viewModel.m_validationQueue.complete(componentId);
65
66 if (viewModel.m_validationPendingCount > 0) {
67 --viewModel.m_validationPendingCount;
68 }
69
70 // BOM 导入完成后不触发额外的列表刷新，只继续消费验证队列。
71 if (viewModel.m_bomImportComplete) {
72 if (!viewModel.m_validationQueue.isEmpty()) {
73 processNext(viewModel);
74 } else if (viewModel.m_validationPendingCount == 0) {
75 viewModel.m_bomImportComplete = false;
76 }
77 return;
78 }
79
80 // BOM 导入过程中累积更新次数，普通模式下立即安排一次批量刷新。
81 if (!viewModel.m_bomImportMode) {
82 viewModel.scheduleListUpdate();
83 } else {
84 ++viewModel.m_bomImportPendingUpdates;
85 }
86
87 if (!viewModel.m_validationQueue.isEmpty()) {
88 processNext(viewModel);
89 } else {
90 // 验证期间可能有新元件加入，队列为空时重新扫描列表。
91 viewModel.startValidationQueue();
92 }
93}
