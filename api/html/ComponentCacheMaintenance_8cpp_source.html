30void ComponentCacheMaintenance::remove(const QString& componentId) {
31 const QString normalizedId = componentId.toUpper(); // 规范化后的元器件编号。
32 if (!BomParser::validateId(normalizedId)) {
33 qWarning() << "removeCache: invalid lcscId, ignoring:" << componentId;
34 return;
35 }
36
37 m_owner.m_cacheGeneration.fetch_add(1);
38 {
39 // 锁顺序保持为 disk 后 tombstone，与缓存写入策略一致。
40 QMutexLocker diskLocker(&m_owner.m_diskWriteMutex); // 保护磁盘缓存维护。
41 m_owner.m_tombstones.blockComponent(normalizedId);
42 const QString dirPath = m_owner.componentCacheDir(normalizedId); // 组件缓存目录。
43 if (dirPath.isEmpty())
44 return;
45 const QStringList files = CacheSafety::ownedComponentFiles(m_owner.cacheDir(), dirPath); // 可处理文件。
46 bool allFilesMoved = true; // 是否所有已验证文件均已移入回收站。
47 for (const QString& filePath : files) {
48 QString error; // 当前回收站操作的错误文本。
49 if (!CacheSafety::moveToTrash(filePath, &error, m_trash)) {
50 allFilesMoved = false;
51 emit m_owner.cacheMaintenanceWarning(error);
52 }
53 }
54 if (allFilesMoved && !files.isEmpty() &&
55 QDir(dirPath).entryInfoList(QDir::AllEntries | QDir::NoDotAndDotDot).isEmpty()) {
56 QString error; // 组件目录回收站操作的错误文本。
57 if (!CacheSafety::moveToTrash(dirPath, &error, m_trash))
58 emit m_owner.cacheMaintenanceWarning(error);
59 }
60 if (!files.isEmpty())
61 LOG_DEBUG(LogModule::Core, "Moved owned disk cache files to trash for: {}", normalizedId);
62 }
63
64 qint64 sizeAfterUpdate = 0; // 删除后的一级缓存大小。
65 {
66 QMutexLocker locker(&m_owner.m_mutex); // 保护一级缓存删除。
67 sizeAfterUpdate = m_owner.m_memoryCache.removeComponent(normalizedId);
68 }
69 emit m_owner.memoryCacheSizeChanged(sizeAfterUpdate);
70 emit m_owner.cacheSizeChanged(cacheSize(m_owner));
71}

129QStringList ComponentCacheMaintenance::cachedComponentIds(const ComponentCacheService& owner) {
130 QMutexLocker diskLocker(&owner.m_diskWriteMutex);
131 QStringList result; // 有效元器件编号结果。
132 QSet<QString> seenIds; // 已加入结果的规范化编号。
133 const QDir dir(owner.cacheDir()); // 缓存根目录。
134 if (!dir.exists())
135 return result;
136
137 for (const QString& entry : dir.entryList(QDir::Dirs)) {
138 if (entry == QStringLiteral(".") || entry == QStringLiteral("..") || entry == QStringLiteral("model3d") ||
139 entry == CacheSafety::ownershipMarkerName())
140 continue;
141 const QString normalizedId = entry.toUpper(); // 规范化后的目录编号。
142 const QJsonObject metadata = CacheMetadataStore::read(owner.metadataPath(entry)); // 组件元数据。
143 const QJsonValue metadataId = metadata.value(QStringLiteral("lcscId")); // 元数据中的编号。
144 const bool matchesEntry = // 元数据编号是否与目录名一致。
145 metadataId.isString() && metadataId.toString().compare(entry, Qt::CaseInsensitive) == 0;
146 if (CacheSafety::isOwnedComponentDirectory(owner.cacheDir(), dir.filePath(entry)) && matchesEntry &&
147 CacheMetadataStore::hasValidModel3D(metadata) && !seenIds.contains(normalizedId)) {
148 result.append(normalizedId);
149 seenIds.insert(normalizedId);
150 }
151 }
152 return result;
153}

160qint64 ComponentCacheMaintenance::cacheSize(const ComponentCacheService& owner) {

162 qint64 size = 0; // 已验证文件的累计大小。

170}
