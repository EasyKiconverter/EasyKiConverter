51bool ConfigService::loadConfig(const QString& path) {
52 // 解析调用方指定或默认的配置文件路径。
53 QString configPath = path.isEmpty() ? getDefaultConfigPath() : path;
54
55 QFile file(configPath);
56 if (!file.open(QIODevice::ReadOnly)) {
57 qWarning() << "Config file not found, using defaults:" << configPath;
58 saveConfig();
59 return false;
60 }
61
62 // 读取配置文件的原始 JSON 字节内容。
63 QByteArray fileData = file.readAll();
64 file.close();
65
66 // 保存 JSON 解析错误，供加载失败时诊断。
67 QJsonParseError parseError;
68 // 保存解析后的配置 JSON 文档。
69 QJsonDocument doc = QJsonDocument::fromJson(fileData, &parseError);
70
71 if (parseError.error != QJsonParseError::NoError) {
72 qWarning() << "Failed to parse config file:" << parseError.errorString();
73 return false;
74 }
75
76 QMutexLocker locker(&m_configMutex);
77
78 // 合并配置：只更新文件中存在的字段，保留文件中缺失字段的默认值
79 // 保存从文件解析出的配置字段，用于合并到默认配置。
80 QJsonObject loadedConfig = doc.object();
81 for (auto it = loadedConfig.begin(); it != loadedConfig.end(); ++it) {
82 m_config[it.key()] = it.value();
83 }
84
85 // 移除 debugMode 字段，因为它不应该从配置文件加载
86 // 调试模式完全由命令行参数和环境变量控制
87 m_config.remove("debugMode");
88
89 m_configPath = configPath;
90
91 qDebug() << "Config loaded from:" << configPath;
92 emit configChanged();
93
94 // 释放锁后保存，确保补全缺失的字段到文件中（但不包含 debugMode）
95 locker.unlock();
96 saveConfig();
97
98 return true;
99}

105bool ConfigService::saveConfig(const QString& path) {
106 // 解析调用方指定、已加载或默认的配置文件路径。
107 QString configPath = path.isEmpty() ? (m_configPath.isEmpty() ? getDefaultConfigPath() : m_configPath) : path;
108
109 QMutexLocker locker(&m_configMutex);
110
111 // 创建配置的副本，移除 debugMode 字段
112 QJsonObject configToSave = m_config;
113 configToSave.remove("debugMode");
114
115 QJsonDocument doc(configToSave);
116
117 QFile file(configPath);
118 if (!file.open(QIODevice::WriteOnly)) {
119 qWarning() << "Failed to open config file for writing:" << configPath;
120 return false;
121 }
122
123 file.write(doc.toJson());
124 file.close();
125
126 qDebug() << "Config saved to:" << configPath;
127 return true;
128}
